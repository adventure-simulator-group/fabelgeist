  69     /// Native scene-root translation advances by the existing controller step;
  70     /// admit its footprint before queries and its complete position before Bevy.

  72         &mut self,
  73         terrain: &SceneTerrain,
  74         current: Vec3,
  75         frame: &PlannedFrame,
  76         motion: &subject_motion::CaptureMotion,
  77         terrain_ik: TerrainIkMode,
  78     ) -> Option<Position<Scene>> {
  79         let horizontal = current.xz() + motion.world_velocity.xz() * motion.elapsed.seconds();
  80         let point = self.admit_terrain_point(horizontal)?;
  81         let vertical = if is_quickstep_scenario(&frame.scenario) {
  82             let ground = self.admit_terrain_height(terrain, point)?.metres()
  83                 + CAPTURE_ROOT_GROUND_OFFSET_METRES;
  84             ground
  85                 + quickstep_fixture_vertical_state(frame.scenario_frame)
  86                     .0
  87                     .max(0.0)
  88         } else if terrain_ik.is_enabled() {
  89             self.admit_terrain_height(terrain, point)?.metres() + CAPTURE_ROOT_GROUND_OFFSET_METRES

  91             current.y
  92         };

  98     pub(super) fn full_ragdoll_root(
  99         &mut self,
 100         terrain: &SceneTerrain,
 101         mut transform: Transform,
 102         frame: &PlannedFrame,
 103     ) -> Option<Transform> {
 104         let fall = ((frame.scenario_frame + 4) as f32 / 8.0).clamp(0.0, 1.0);
 105         transform.rotation = Quat::from_rotation_y(std::f32::consts::PI)
 106             * Quat::from_rotation_x(1.25 * smoothstep01(fall));
 107         let ground = self.admit_terrain_height(terrain, ScenePlanPoint::ORIGIN)?;
 108         transform.translation.y =
 109             ground.metres() + CAPTURE_ROOT_GROUND_OFFSET_METRES - 0.62 * smoothstep01(fall);
 110         transform.translation = self.admit_scene_position(transform.translation)?.metres();
 111         Some(transform)
 112     }

 116     pub(super) fn capture_ground(
 117         &mut self,
 118         terrain_ik: TerrainIkMode,
 119         subject: Vec3,
 120     ) -> Option<CaptureGround> {
 121         if terrain_ik.is_enabled() {
 122             return Some(CaptureGround::Terrain);
 123         }
 124         let height = subject.y - CAPTURE_ROOT_GROUND_OFFSET_METRES;
 125         if let Some(height) = SupportElevation::from_metres(height) {
 126             return Some(CaptureGround::AuthoredPlane(height));
 127         }
 128         self.admission = CaptureAdmission::Rejected(RejectedCaptureAdmission::Geometry(
 129             GeometryError::NonFinite {
 130                 role: GeometryRole::Elevation,
 131                 axis: CoordinateAxis::Y,

 133         ));
 134         None
