  39     /// Complete the reduction over native heightfield or compiled mesh samples.
  40     fn from_native_metres(mut heights: impl Iterator<Item = f32>) -> Option<Self> {
  41         let first = heights.next()?;
  42         if !first.is_finite() {
  43             return None;
  44         }
  45         let mut minimum = first;
  46         let mut maximum = first;
  47         for height in heights {
  48             if !height.is_finite() {
  49                 return None;
  50             }
  51             minimum = minimum.min(height);
  52             maximum = maximum.max(height);
  53         }
  54         Self::new(
  55             SupportElevation::from_metres(minimum)?,
  56             SupportElevation::from_metres(maximum)?,
  57         )
  58     }
