 568 /// Establish a journey-local clock without changing any participant's
 569 /// subjective age. The leader-selected time of day is placed on the canonical
 570 /// departure day; elapsed journey progress later wraps within that frozen day.
 571 pub(crate) fn synchronize_party_departure_time(
 572     ctx: &ReducerContext,
 573     member_ids: &[u64],
 574 ) -> Result<Option<StrategicMinute>, String> {
 575     if member_ids.is_empty() {
 576         return Err("Party has no living members".into());
 577     }
 578     for member_id in member_ids {
 579         ensure_character_time(ctx, *member_id)?;
 580     }
 581     let party_id = member_ids
 582         .iter()
 583         .find_map(|member_id| {
 584             ctx.db
 585                 .character()
 586                 .id()
 587                 .find(*member_id)
 588                 .and_then(|character| character.party_id)
 589         })
 590         .ok_or("Party members have no party")?;
 591     let mut party = ctx
 592         .db
 593         .party_authority()
 594         .id()
 595         .find(&party_id)
 596         .ok_or("Party not found")?;
 597     if party.wilderness_canonical_anchor_minute.is_none() {
 598         party.wilderness_canonical_anchor_minute = Some(refresh_clock(ctx)?);
 599         party.wilderness_elapsed_minutes = 0;
 600         ctx.db.party_authority().id().update(party.clone());
 601     }
 602     let anchor = party
 603         .wilderness_canonical_anchor_minute
 604         .ok_or("Party wilderness clock was not initialized")?;
 605     Ok(Some(anchor.with_wrapped_time_of_day(
 606         party.journey_start_minute_of_day,
 607         party.wilderness_elapsed_minutes,
 608     )))
 609 }
