 591 fn advance_party_movement_until_encounter(
 592     ctx: &ReducerContext,
 593     party_id: &str,
 594     traveler_ids: &[u64],
 595     proposed_leg_minutes: u64,
 596 ) -> Result<
 597     (
 598         u64,
 599         Option<StrategicEncounter>,
 600         Option<adventuresim_core::encounter::NarrativeSelection>,
 601         u64,
 602     ),
 603     String,
 604 > {
 605     let (requested_leg_minutes, mut encounter, mut narrative, mut next_roll) =
 606         maybe_interrupt_travel(ctx, party_id, proposed_leg_minutes)?;
 607     let (actual_minutes, _) =
 608         advance_party_movement(ctx, party_id, traveler_ids, requested_leg_minutes)?;
 609     if actual_minutes < requested_leg_minutes {
 610         let (rescanned_minutes, rescanned_encounter, rescanned_narrative, rescanned_next_roll) =
 611             maybe_interrupt_travel(ctx, party_id, actual_minutes)?;
 612         debug_assert_eq!(rescanned_minutes, actual_minutes);
 613         encounter = rescanned_encounter;
 614         narrative = rescanned_narrative;
 615         next_roll = rescanned_next_roll;
 616     }
 617     Ok((actual_minutes, encounter, narrative, next_roll))
 618 }
