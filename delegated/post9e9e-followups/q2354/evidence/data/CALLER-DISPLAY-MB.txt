1112 /// Like [`ensure_settlement_activity_inner`], but generates at most
1113 /// `quest_batch` new quests when `quest_batch > 0` (0 means "no cap", the
1114 /// behavior [`ensure_settlement_activity_inner`] relies on). The cheap
1115 /// idempotent ensures run unconditionally; only quest generation is throttled,
1116 /// letting a heavy settlement be filled across several transactions while the
1117 /// `active..target` loop still converges on the same final quest count.

1121     quest_batch: u32,
1122 ) -> Result<(), String> {

1130         |error| settlement_activity_stage_error(settlement_id, "settlement population", error),

1132     let official_minute = crate::time::refresh_clock(ctx).map_err(|error| {
1133         settlement_activity_stage_error(settlement_id, "official clock refresh", error)

1135     for mut contract in ctx
1136         .db
1137         .contract_authority()
1138         .settlement_id()
