  62 /// temporarily deferred player events do not consume capacity.

  71         .iter()

  74                 && row.status == CommitmentStatus::Reserved
  75                 && row.kind == CommitmentKind::Engagement
  76         })

  78             effective_minute: row.effective_minute,
  79             id: row.id,
  80             participant_id: row.first_character_id,
  81         })
  82         .chain(

  84                 .pregnancy()
  85                 .iter()
  86                 .filter(|row| row.due_minute <= now && row.status == PregnancyStatus::Active)

  89                     id: row.id,
  90                     mother_id: row.mother_id,
