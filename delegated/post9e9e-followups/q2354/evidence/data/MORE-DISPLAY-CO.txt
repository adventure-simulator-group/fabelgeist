  99 fn stable_npc_cohort(
 100     ctx: &ReducerContext,
 101     target_minute: adventuresim_world_schema::calendar::StrategicMinute,
 102 ) -> Vec<CharacterTime> {
 103     let mut candidates: Vec<_> = ctx
 104         .db
 105         .npc_policy()
 106         .iter()
 107         .filter_map(|policy| {
 108             let person = ctx.db.character().id().find(policy.character_id)?;
 109             let time = ctx
 110                 .db
 111                 .character_time()
 112                 .character_id()
 113                 .find(policy.character_id)?;
 114             (person.alive && time.minutes < target_minute).then_some(time)
 115         })
 116         .collect();
 117     candidates.sort_by_key(|time| (time.minutes, time.character_id));
 118     let Some(frontier) = candidates.first().map(|time| time.minutes) else {
 119         return candidates;
 120     };
 121     candidates.retain(|time| time.minutes == frontier);
 122     candidates.truncate(MAX_NPCS_PER_CAUSAL_TICK);
 123     candidates
 124 }
