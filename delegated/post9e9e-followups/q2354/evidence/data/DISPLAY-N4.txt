 448 /// Formal eligibility is directional even though the autonomous candidate
 449 /// encounter is not. Normalize an opposite-sex pair so scheduler character
 450 /// order cannot turn a father-approved relationship informal. Same-sex and
 451 /// otherwise non-formal pairs retain the deterministic actor/candidate order.
 452 fn npc_courtship_roles(
 453     actor_id: u64,
 454     actor_sex: Sex,
 455     candidate_id: u64,
 456     candidate_sex: Sex,
 457 ) -> (u64, u64) {
 458     match (actor_sex, candidate_sex) {
 459         (Sex::Female, Sex::Male) => (candidate_id, actor_id),
 460         _ => (actor_id, candidate_id),
 461     }
 462 }

 464 fn process_policy_decisions(ctx: &ReducerContext, time: &CharacterTime) -> Result<(), String> {
 465     let policy = ctx
 466         .db
 467         .npc_policy()
 468         .character_id()
 469         .find(time.character_id)
 470         .ok_or("NPC causal cohort contains a character without policy")?;
 471     let minute = time.minutes;
 472     initialize_saved_schedule_once(ctx, time.character_id, policy.policy_seed, minute)?;
 473     settle_housing_decision(ctx, time.character_id, &policy.home_settlement_id, minute)?;
 474     settle_romance_decision(ctx, time.character_id, policy.policy_seed, minute)?;
 475     Ok(())
 476 }
