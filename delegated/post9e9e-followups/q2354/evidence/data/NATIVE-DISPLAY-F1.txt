 127 fn seeded_household_plan(
 128     settlement_id: &str,
 129     family: &[u64],
 130 ) -> Result<Option<SeededHouseholdPlan>, String> {
 131     let Some(first_character_id) = family.first() else {
 132         return Ok(None);
 133     };
 134     let roles = [
 135         HouseholdRole::Head,
 136         HouseholdRole::Spouse,
 137         HouseholdRole::AdultChild,
 138         HouseholdRole::AdultChild,
 139     ];
 140     if family.len() > roles.len() {
 141         return Err("Seeded household exceeds the supported family shape".into());
 142     }
 143     let members = family.iter().copied().zip(roles).collect();
 144     let kinships = if family.len() == roles.len() {
 145         let mut edges = Vec::with_capacity(10);
 146         for child in [family[2], family[3]] {
 147             for parent in [family[0], family[1]] {
 148                 edges.push((child, parent, KinshipKind::Parent));
 149                 edges.push((parent, child, KinshipKind::Child));
 150             }
 151         }
 152         edges.push((family[2], family[3], KinshipKind::Sibling));
 153         edges.push((family[3], family[2], KinshipKind::Sibling));
 154         edges
 155     } else {
 156         Vec::new()
 157     };
 158     Ok(Some(SeededHouseholdPlan {
 159         household_id: format!("household:seeded:{settlement_id}:{first_character_id}"),
 160         family_key: format!("seeded:{settlement_id}:{first_character_id}"),
 161         members,
 162         kinships,
 163     }))
 164 }
