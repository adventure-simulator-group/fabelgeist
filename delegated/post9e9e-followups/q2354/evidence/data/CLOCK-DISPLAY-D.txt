 143 fn record_decision(
 144     ctx: &ReducerContext,
 145     character_id: u64,
 146     day: u64,
 147     phase: NpcPolicyDecisionPhase,
 148     outcome: NpcPolicyDecisionOutcome,
 149     target_character_id: Option<u64>,
 150     decided_minute: StrategicMinute,
 151 ) {
 152     let id = receipt_id(character_id, day, phase);
 153     if ctx
 154         .db
 155         .npc_policy_decision_receipt()
 156         .id()
 157         .find(&id)
 158         .is_none()
 159     {
 160         ctx.db
 161             .npc_policy_decision_receipt()
 162             .insert(NpcPolicyDecisionReceipt {
 163                 id,
 164                 character_id,
 165                 day,
 166                 phase,
 167                 outcome,
 168                 target_character_id,
 169                 decided_minute,
 170             });
 171     }
 172 }
