  54 /// The single lifecycle boundary for an authoritative personal-clock write.
  55 /// Callers finish injury/disease settlement first so wedding cancellation and
  56 /// widowhood observe the final alive state at this frontier.
  57 pub(crate) fn settle_lifecycle_after_character_time_write(
  58     ctx: &ReducerContext,
  59     character_id: u64,
  60     minute: StrategicMinute,
  61 ) -> Result<(), String> {
  62     crate::relationship::settle_character_age(ctx, character_id, minute);
  63     crate::continuity::settle_continuity_for_character(ctx, character_id, minute)?;
  64     crate::residence::settle_residence_billing(ctx, character_id)?;
  65     crate::relationship::settle_due_weddings(ctx, character_id, minute)?;
  66     crate::relationship::settle_due_births(ctx, character_id, minute)?;
  67     crate::relationship::settle_secret_courtship_discovery_for_character(
  68         ctx,
  69         character_id,
  70         minute,
  71     )?;
  72     crate::relationship::settle_marriage_lifecycle_for_character(ctx, character_id, minute);
  73     crate::outbreak::refresh_patient_context_after_time_write(ctx, character_id, minute);
  74     Ok(())
  75 }
