 101 fn require_tincture_vessel(
 102     ctx: &ReducerContext,
 103     object_id: u64,
 104 ) -> Result<crate::InventoryObject, String> {
 105     let object = ctx
 106         .db
 107         .inventory_object()
 108         .id()
 109         .find(object_id)
 110         .ok_or("Container not found")?;
 111     if !is_tincture_vessel(&object.item_id) {
 112         return Err("Tincturing requires a bottle or jar authored as a tincture vessel".into());
 113     }
 114     Ok(object)
 115 }
