  90 pub fn consume_personal(
  91     ctx: &ReducerContext,
  92     inventory_item_id: u64,
  93     requested_fraction: ConsumableFractionMicros,
  94 ) -> Result<ConsumableFractionMicros, String> {
  95     if requested_fraction.is_zero() {
  96         return Ok(ConsumableFractionMicros::ZERO);
  97     }
  98     let mut state = ctx
  99         .db
 100         .inventory_item_amount()
 101         .inventory_item_id()
 102         .find(inventory_item_id)
 103         .ok_or("Measured personal item state is missing")?;
 104     let available = persisted_fraction(state.remaining_fraction_micros);
 105     let consumed = requested_fraction.min(available);
 106     let remaining = available
 107         .checked_sub(consumed)
 108         .expect("consumed fraction cannot exceed available fraction");
 109     state.remaining_fraction_micros = remaining.get();
 110     if remaining.is_zero() {
 111         ctx.db
 112             .inventory_item_amount()
 113             .inventory_item_id()
 114             .delete(inventory_item_id);
 115         ctx.db.inventory_item().id().delete(inventory_item_id);
 116     } else {
 117         ctx.db
 118             .inventory_item_amount()
 119             .inventory_item_id()
 120             .update(state);
 121     }
 122     Ok(consumed)
 123 }
