  75 ) -> Result<Vec<ConsumableFractionMicros>, String> {
  76     fractions_micros
  77         .into_iter()
  78         .map(|value| {
  79             ConsumableFractionMicros::try_new(value)
  80                 .map_err(|_| "Ingredient fraction cannot exceed one whole".to_owned())
  81         })
  82         .collect()
  83 }

 135         return Err("This container is already cooking".into());
