 151         if !self.objects.contains_key(&root) {
 152             return Err("Unknown inventory object");
 153         }
 154         let mut result = vec![root];
 155         let mut cursor = 0;
 156         while cursor < result.len() {
 157             if cursor >= MAX_CONTAINER_DEPTH * self.objects.len().max(1) {
 158                 return Err("Containment traversal bound exceeded");
 159             }
 160             let parent = result[cursor];
 161             for child in self
 162                 .parent_by_child
 163                 .iter()
 164                 .filter_map(|(child, candidate)| (*candidate == parent).then_some(*child))
 165             {
 166                 if result.contains(&child) {
 167                     return Err("Containment cycle");
 168                 }
 169                 result.push(child);
 170             }
 171             cursor += 1;
 172         }
 173         Ok(result)
 174     }
 175 }
