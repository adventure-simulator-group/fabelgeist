 181     });
 182     Ok(())

 185 pub fn move_party_to_personal(
 186     ctx: &ReducerContext,
 187     party_inventory_item_id: u64,
 188     inventory_item_id: u64,
 189 ) -> Result<(), String> {
 190     let state = ctx
 191         .db
 192         .party_item_amount()
 193         .party_inventory_item_id()
 194         .find(party_inventory_item_id)
 195         .ok_or("Measured party item state is missing")?;
 196     ctx.db
 197         .party_item_amount()
 198         .party_inventory_item_id()
 199         .delete(party_inventory_item_id);
 200     ctx.db.inventory_item_amount().insert(InventoryItemAmount {
 201         inventory_item_id,
 202         remaining_fraction_micros: state.remaining_fraction_micros,
 203     });
 204     Ok(())
 205 }
