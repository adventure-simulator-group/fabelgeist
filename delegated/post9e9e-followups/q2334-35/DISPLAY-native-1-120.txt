  65 fn insert_carried_object(
  66     ctx: &ReducerContext,
  67     item_id: &str,
  68     location: InventoryLocation,
  69 ) -> Result<InventoryObject, String> {
  70     let scope = location
  71         .scope()
  72         .ok_or("Stable physical objects require carried custody at insertion")?;
  73     let row_id = location
  74         .row_id()
  75         .ok_or("Stable physical object has no backing inventory row")?;
  76     if object_for_row(ctx, scope, row_id)?.is_some() {
  77         return Err("Inventory row already has a stable physical object identity".into());
  78     }
  79     Ok(ctx.db.inventory_object().insert(InventoryObject {
  80         id: 0,
  81         item_id: item_id.into(),
  82         location,
  83     }))
  84 }

  86 pub(crate) fn insert_personal_object(
  87     ctx: &ReducerContext,
  88     row: &crate::InventoryItem,
  89 ) -> Result<InventoryObject, String> {
  90     if row.quantity != 1 {
  91         return Err("Stable personal inventory objects require quantity-one rows".into());
  92     }
  93     insert_carried_object(
  94         ctx,
  95         &row.item_id,
  96         InventoryLocation::personal(row.character_id, row.id),
  97     )
  98 }

 100 pub(crate) fn insert_party_object(
 101     ctx: &ReducerContext,
 102     row: &crate::strategic::PartyInventoryItem,
 103 ) -> Result<InventoryObject, String> {
 104     if row.quantity != 1 {
 105         return Err("Stable party inventory objects require quantity-one rows".into());
 106     }
 107     insert_carried_object(
 108         ctx,
 109         &row.item_id,
 110         InventoryLocation::party(row.party_id.clone(), row.id),
 111     )
 112 }
