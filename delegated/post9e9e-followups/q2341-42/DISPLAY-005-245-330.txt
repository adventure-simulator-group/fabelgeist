 245         let next = i128::from(birth_minute).checked_add(
 246             i128::try_from(completed.checked_add(1)?).ok()? * i128::from(MINUTES_PER_YEAR),
 247         )?;
 248         let next = Self::new(u64::try_from(next).ok()?);
 249         (self < next && next < end).then_some(next)
 250     }

 304     /// Advance the time of day while keeping this minute's calendar day.
 305     pub const fn wrapping_day_offset(self, elapsed_minutes: u64) -> Self {
 306         let minute_of_day =
 307             (self.minute_of_day() as u64 + elapsed_minutes % MINUTES_PER_DAY) % MINUTES_PER_DAY;
 308         self.day_start().saturating_add_minutes(minute_of_day)
 309     }

 311     /// Place a local clock reading on this minute's calendar day, wrapping
 312     /// elapsed journey time within that day.
 313     pub const fn with_wrapped_time_of_day(
 314         self,
 315         starting_minute_of_day: u16,
 316         elapsed_minutes: u64,
 317     ) -> Self {
 318         let minute_of_day = (starting_minute_of_day as u64 % MINUTES_PER_DAY
 319             + elapsed_minutes % MINUTES_PER_DAY)
 320             % MINUTES_PER_DAY;
 321         self.day_start().saturating_add_minutes(minute_of_day)
 322     }
