 350     refresh_limb_projection(ctx, character_id, limb);
 351     Ok(())
 352 }

 354 pub fn fracture_from_single_hit(blunt_damage: f32) -> f32 {
 355     (blunt_damage.max(0.0) - FRACTURE_SINGLE_HIT_THRESHOLD).max(0.0) * 0.65
 356 }

 358 fn has_projectile(ctx: &ReducerContext, character_id: u64, limb: BodyRegion) -> bool {
 359     ctx.db
 360         .retained_projectile()
 361         .character_id()
 362         .filter(character_id)
 363         .any(|projectile| projectile.limb == limb)
 364 }
