 508 impl FromStr for StrategicPlaceId {
 509     type Err = PlaceIdentityError;
 510 
 511     fn from_str(value: &str) -> Result<Self, Self::Err> {
 512         if value.len() > MAX_STRATEGIC_PLACE_ID_BYTES {
 513             return Err(PlaceIdentityError::MalformedEncoding);
 514         }
 515         let parts = value.split(':').collect::<Vec<_>>();
 516         if parts.first() != Some(&"place") {
 517             return Err(PlaceIdentityError::MalformedEncoding);
 518         }
 519         if parts.get(1) != Some(&FORMAT_VERSION) {
 520             return Err(PlaceIdentityError::UnsupportedVersion);
 521         }
 522         match parts.as_slice() {
 523             ["place", _, "settlement", settlement_id] => Self::settlement(decode(settlement_id)?),
 524             ["place", _, "venue", settlement_id, kind] => Self::settlement_venue(
 525                 decode(settlement_id)?,
 526                 SettlementVenueKind::from_id(kind).ok_or(PlaceIdentityError::UnknownVenue)?,
 527             ),
 528             [
 529                 "place",
 530                 _,
 531                 "chapter",
 532                 settlement_id,
 533                 organization_id,
 534                 authored_location_id,
 535             ] => Self::chapter_venue(
 536                 decode(settlement_id)?,
 537                 decode(organization_id)?,
 538                 decode(authored_location_id)?,
 539             ),
 540             ["place", _, "residence", settlement_id, holding_id] => {
 541                 Self::residence(decode(settlement_id)?, decode(holding_id)?)
 542             }
 543             ["place", _, "case-site", site_id] => Self::case_site(decode(site_id)?),
 544             ["place", _, "journey-camp", party_id, departure, movement] => Self::journey_camp(
 545                 decode(party_id)?,
 546                 StrategicMinute::new(parse_canonical_u64(departure)?),
 547                 parse_canonical_u64(movement)?,
 548             ),
 549             ["place", _, _, ..] => Err(PlaceIdentityError::UnknownPlaceKind),
 550             _ => Err(PlaceIdentityError::MalformedEncoding),
 551         }
 552     }
 553 }
