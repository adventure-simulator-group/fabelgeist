  74 /// Knowledge awarded by a trial must describe mechanics already consumed by
  75 /// combat. It never changes the threat profile or applies a hidden modifier.
  76 pub fn tactical_insight_for(threat_id: crate::bestiary::ThreatId) -> Option<TacticalInsight> {
  77     let profile = threat_id.profile();
  78     if profile.combat.ranged {
  79         return None;
  80     }
  81     let modeled = crate::bestiary::implemented_combat_lore(profile);
  82     if !modeled
  83         .weaknesses
  84         .iter()
  85         .any(|fact| fact == "Must close to melee range before attacking")
  86     {
  87         return None;
  88     }
  89     Some(TacticalInsight {
  90         kind: TacticalInsightKind::MustCloseToMelee,
  91         threat_id: threat_id.as_str().into(),
  92         finding: format!(
  93             "The {} carry no missile weapons and must close to melee range before striking.",
  94             profile.plural_name
  95         ),
  96         preparation: "Bring bows and arrows; archers can strike while these enemies close.".into(),
  97     })
  98 }

 100 /// A camp's stable identity within one journey. Elapsed time is deliberately
 101 /// absent: resting advances elapsed time without moving the camp.
 102 pub fn journey_camp_identity_matches(
 103     journey_departure_minute: StrategicMinute,
 104     completed_movement_minute: u64,
 105     camp_stop_minutes: &[u64],
 106     bound_departure_minute: StrategicMinute,
 107     bound_movement_minute: u64,
 108 ) -> bool {
 109     journey_departure_minute == bound_departure_minute
 110         && completed_movement_minute == bound_movement_minute
 111         && camp_stop_minutes.contains(&bound_movement_minute)
 112 }
 113 
 114 pub fn rested_road_trial_camp_matches(
 115     journey_departure_minute: StrategicMinute,
 116     completed_movement_minute: u64,
 117     completed_elapsed_minute: u64,
 118     camp_stop_minutes: &[u64],
 119     bound_departure_minute: StrategicMinute,
 120     bound_movement_minute: u64,
 121     available_at_elapsed_minute: u64,
 122 ) -> bool {
 123     journey_camp_identity_matches(
 124         journey_departure_minute,
 125         completed_movement_minute,
 126         camp_stop_minutes,
 127         bound_departure_minute,
 128         bound_movement_minute,
 129     ) && completed_elapsed_minute >= available_at_elapsed_minute
 130 }
