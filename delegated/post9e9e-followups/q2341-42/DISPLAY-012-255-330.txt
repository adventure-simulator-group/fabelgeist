 255     pub fn add_direct(&mut self, religion: OfficialReligion, hours: f32) {
 256         if hours.is_finite() && hours > 0.0 {
 257             let direct = self.direct_mut(religion);
 258             if !direct.is_finite() || *direct < 0.0 {
 259                 *direct = 0.0;
 260             }
 261             *direct += hours;
 262         }
 263     }
 264 }

 292     pub fn total(self) -> u64 {
 293         OfficialReligion::ALL
 294             .into_iter()
 295             .map(|r| u64::from(self.get(r)))
 296             .sum()
 297     }
 298 
 299     pub fn split_evenly(total: u16, targets: &[OfficialReligion]) -> Self {
 300         let mut result = Self::default();
 301         let mut targets = targets.to_vec();
 302         targets.sort_by_key(|religion| religion.index());
 303         targets.dedup();
 304         if targets.is_empty() {
 305             return result;
 306         }
 307         let count = targets.len() as u16;
 308         let base = total / count;
 309         let remainder = total % count;
 310         for (index, religion) in targets.into_iter().enumerate() {
 311             let value = base + u16::from(index < usize::from(remainder));
 312             match religion {
 313                 OfficialReligion::RomanCatholic => result.roman_catholic = value,
 314                 OfficialReligion::Lutheran => result.lutheran = value,
 315                 OfficialReligion::Reformed => result.reformed = value,
 316                 OfficialReligion::Anglican => result.anglican = value,
 317                 OfficialReligion::EasternOrthodox => result.eastern_orthodox = value,
 318                 OfficialReligion::Islamic => result.islamic = value,
 319                 OfficialReligion::Judaism => result.judaism = value,
 320             }
 321         }
 322         result
 323     }
