  22 fn authoritative_case_route_binding_digest(
  23     departure_minute: StrategicMinute,
  24     origin: &JourneyRoutePoint,
  25     destination: &JourneyRoutePoint,
  26     coordinates_are_geographic: bool,
  27     distance_m: u64,
  28     minutes: u64,
  29 ) -> String {
  30     use sha2::Digest as _;
  31 
  32     let mut hash = sha2::Sha256::new();
  33     for value in [
  34         b"adventuresim.authoritative-case-route".as_slice(),
  35         1u16.to_le_bytes().as_slice(),
  36         departure_minute.get().to_le_bytes().as_slice(),
  37         origin.latitude_e7.to_le_bytes().as_slice(),
  38         origin.longitude_e7.to_le_bytes().as_slice(),
  39         destination.latitude_e7.to_le_bytes().as_slice(),
  40         destination.longitude_e7.to_le_bytes().as_slice(),
  41         [u8::from(coordinates_are_geographic)].as_slice(),
  42         distance_m.to_le_bytes().as_slice(),
  43         minutes.to_le_bytes().as_slice(),
  44     ] {
  45         hash.update((value.len() as u64).to_le_bytes());
  46         hash.update(value);
  47     }
  48     format!("{:x}", hash.finalize())
  49 }
