  12 pub const MAX_STRATEGIC_ID_COMPONENT_BYTES: usize = 256;
  13 pub const MAX_STRATEGIC_PLACE_ID_BYTES: usize = 4_096;
  14 pub const MAX_STRATEGIC_FIXTURE_ID_BYTES: usize =
  15     MAX_STRATEGIC_PLACE_ID_BYTES * 2 + MAX_STRATEGIC_ID_COMPONENT_BYTES * 4 + 64;

  17 /// A validated opaque identifier used inside a canonical strategic identity.
  18 ///
  19 /// Domain IDs remain opaque: this type prevents delimiters, names, or prefix
  20 /// conventions from being reinterpreted as additional place structure.
  21 #[derive(Clone, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
  22 pub struct StrategicIdentityComponent(String);
  23 
  24 impl StrategicIdentityComponent {
  25     pub fn try_new(value: impl Into<String>) -> Result<Self, PlaceIdentityError> {
  26         let value = value.into();
  27         if value.is_empty()
  28             || value.len() > MAX_STRATEGIC_ID_COMPONENT_BYTES
  29             || !value.bytes().all(|byte| {
  30                 byte.is_ascii_alphanumeric() || matches!(byte, b':' | b'_' | b'-' | b'.')
  31             })
  32         {
  33             return Err(PlaceIdentityError::InvalidComponent);
  34         }
  35         Ok(Self(value))
  36     }
  37 
  38     pub fn as_str(&self) -> &str {
  39         &self.0
  40     }
  41 }

  43 /// Validated opaque identity for one investigation case site.
  44 ///
  45 /// This component is intentionally narrower than [`StrategicPlaceId`]: table
  46 /// fields that require a case site cannot accidentally accept a settlement or
  47 /// journey camp.
  48 #[derive(Clone, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
  49 #[cfg_attr(feature = "spacetimedb", derive(spacetimedb::SpacetimeType))]
  50 pub struct CaseSiteId {
  51     value: String,
  52 }
  53 
  54 impl CaseSiteId {
  55     pub fn try_new(value: impl Into<String>) -> Result<Self, PlaceIdentityError> {
  56         let component = StrategicIdentityComponent::try_new(value)?;
  57         Ok(Self { value: component.0 })
  58     }
  59 
  60     pub fn as_str(&self) -> &str {
  61         &self.value
  62     }
  63 
  64     pub fn into_string(self) -> String {
  65         self.value
  66     }
  67 
  68     pub fn to_place(&self) -> StrategicPlaceId {
  69         StrategicPlaceId::CaseSite {
  70             site_id: self.clone(),
  71         }
  72     }
  73 }
  74 
  75 impl From<String> for CaseSiteId {
  76     fn from(value: String) -> Self {
  77         Self::try_new(value).expect("server-authored case-site component must be canonical")
  78     }
  79 }
  80 
  81 impl std::ops::Deref for CaseSiteId {
  82     type Target = str;
  83 
  84     fn deref(&self) -> &Self::Target {
  85         self.as_str()
  86     }
  87 }

  89 impl fmt::Display for CaseSiteId {
  90     fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
  91         self.value.fmt(formatter)
  92     }
  93 }

  95 impl Serialize for CaseSiteId {
  96     fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
  97         #[derive(Serialize)]
  98         struct Wire<'a> {
  99             value: &'a str,
 100         }
 101 
 102         Wire {
 103             value: self.as_str(),
 104         }
 105         .serialize(serializer)
 106     }
 107 }

 109 impl<'de> Deserialize<'de> for CaseSiteId {
 110     fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
 111         #[derive(Deserialize)]
 112         #[serde(deny_unknown_fields)]
 113         struct Wire {
 114             value: String,
 115         }
 116 
 117         CaseSiteId::try_new(Wire::deserialize(deserializer)?.value).map_err(de::Error::custom)
 118     }
