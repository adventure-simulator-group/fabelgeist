  60     #[test]
  61     fn humanoid_contract_resolves_from_twenty_nine_semantic_variants() {
  62         assert_eq!(SemanticPose::HUMANOID_REQUIRED.len(), 30);
  63         let authored = SemanticPose::HUMANOID_REQUIRED
  64             .into_iter()
  65             .filter(|pose| {
  66                 pose.mirrored_counterpart()
  67                     .is_none_or(|other| pose.as_str() < other.as_str())
  68             })
  69             .collect::<BTreeSet<_>>();
  70         assert_eq!(authored.len(), 29);
  71         let mut library = AnimationPackLibrary::default();
  72         library
  73             .insert(AnimationPack {
  74                 id: "root".into(),
  75                 skeleton_family: SkeletonFamily::HUMANOID,
  76                 fallback: None,
  77                 clips: authored,
  78             })
  79             .unwrap();
  80         assert_eq!(
  81             library.validate_complete(&AnimationPackId::from("root")),
  82             Ok(())
  83         );
  84     }
