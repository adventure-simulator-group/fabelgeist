   1 //! Finite elapsed seconds shared by simulation, authority and presentation.
   2 
   3 use serde::{Deserialize, Serialize};
   4 use std::fmt;
   5 use std::time::Duration;
   6 
   7 pub type ElapsedResult<T> = Result<T, ElapsedError>;
   8 
   9 /// Elapsed seconds, including zero for an observation that does not advance.
  10 #[derive(Clone, Copy, Default, PartialEq, PartialOrd, Serialize, Deserialize)]
  11 #[serde(try_from = "f32", into = "f32")]
  12 pub struct AnimationElapsed(f32);
  13 
  14 #[derive(Clone, Copy, Debug, PartialEq, Eq)]
  15 pub enum ElapsedError {
  16     NonFinite,
  17     Negative,
  18     Overflow,
  19     NativeRange,
  20     NonFiniteContactOffset,
  21     ContactOffsetOverflow,
  22 }
  23 
  24 impl AnimationElapsed {
  25     pub const ZERO: Self = Self(0.0);
  26     pub const SECOND: Self = Self(1.0);
  27 
  28     pub const fn from_seconds(seconds: f32) -> ElapsedResult<Self> {
  29         if !seconds.is_finite() {
  30             return Err(ElapsedError::NonFinite);
  31         }
  32         if seconds < 0.0 {
  33             return Err(ElapsedError::Negative);
  34         }
  35         Ok(Self(seconds))
  36     }
  37 
  38     pub fn seconds(self) -> f32 {
  39         self.0
  40     }
  41 
  42     pub fn is_zero(self) -> bool {
  43         self.0 == 0.0
  44     }
  45 
  46     pub fn min(self, other: Self) -> Self {
  47         if self.0 < other.0 { self } else { other }
  48     }
  49 
  50     pub fn max(self, other: Self) -> Self {
  51         if self.0 > other.0 { self } else { other }
  52     }
  53 
  54     /// Bound an elapsed observation by a configured or authored seconds limit.
  55     pub fn capped_by_seconds(self, maximum: f32) -> ElapsedResult<Self> {
  56         Ok(self.min(Self::from_seconds(maximum)?))
  57     }
  58 
  59     /// Remaining duration, floored at zero after the same seconds subtraction.
  60     pub fn saturating_sub(self, other: Self) -> Self {
  61         Self((self.0 - other.0).max(0.0))
  62     }
  63 
  64     /// Separation of two finite, nonnegative elapsed observations. Their
  65     /// difference cannot exceed the larger operand, so it remains finite.
  66     pub fn abs_diff(self, other: Self) -> Self {
  67         Self((self.0 - other.0).abs())
  68     }
  69 
  70     pub fn checked_scale(self, factor: f32) -> ElapsedResult<Self> {
  71         let seconds = self.0 * factor;
  72         if !seconds.is_finite() {
  73             return Err(ElapsedError::Overflow);
  74         }
  75         Self::from_seconds(seconds)
  76     }
  77 
  78     /// Refine seconds at the native timer/authority boundary without a cap.
  79     pub fn native_duration(self) -> ElapsedResult<Duration> {
  80         Duration::try_from_secs_f32(self.0).map_err(|_| ElapsedError::NativeRange)
  81     }
  82 
  83     pub fn checked_add(self, other: Self) -> ElapsedResult<Self> {
  84         let seconds = self.0 + other.0;
  85         if !seconds.is_finite() {
  86             return Err(ElapsedError::Overflow);
  87         }
  88         Ok(Self(seconds))
  89     }
  90 }
  91 
  92 impl From<Duration> for AnimationElapsed {
  93     /// A native Duration is nonnegative and at most u64 seconds, whose f32
  94     /// seconds representation is finite. This is Bevy's delta-seconds boundary.
  95     fn from(duration: Duration) -> Self {
  96         Self(duration.as_secs_f32())
  97     }
  98 }
  99 
 100 impl TryFrom<f32> for AnimationElapsed {
 101     type Error = ElapsedError;
 102     fn try_from(value: f32) -> ElapsedResult<Self> {
 103         Self::from_seconds(value)
 104     }
 105 }
 106 
 107 impl From<AnimationElapsed> for f32 {
 108     fn from(elapsed: AnimationElapsed) -> Self {
 109         elapsed.0
 110     }
 111 }

 113 impl fmt::Debug for AnimationElapsed {
 114     fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
 115         fmt::Debug::fmt(&self.0, formatter)
 116     }
 117 }
 118 
 119 impl fmt::Display for ElapsedError {
 120     fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
 121         formatter.write_str(match self {
 122             Self::NonFinite => "elapsed animation seconds must be finite",
 123             Self::Negative => "elapsed animation seconds must be nonnegative",
 124             Self::Overflow => "elapsed animation seconds arithmetic overflowed",
 125             Self::NonFiniteContactOffset => "contact timing offset seconds must be finite",
 126             Self::ContactOffsetOverflow => "contact timing offset seconds arithmetic overflowed",
 127             Self::NativeRange => "elapsed seconds cannot be represented by the native clock",
 128         })
 129     }
 130 }
 131 impl std::error::Error for ElapsedError {}
