  10 /// Bevy's native clock requires a nonzero Duration. Preserve its f64 Hz
  11 /// reciprocal and native nanosecond rounding rather than the f32 pose interval.
  12 pub(crate) fn capture_fixed_clock(rate: AnimationSampleRate) -> AnimationValueResult<Time<Fixed>> {
  13     let seconds = 1.0 / f64::from(rate.hertz());
  14     let interval =
  15         Duration::try_from_secs_f64(seconds).map_err(|_| AnimationValueError::Overflow {
  16             quantity: AnimationQuantity::NativeSampleInterval,
  17         })?;
  18     if interval.is_zero() {
  19         return Err(AnimationValueError::NonPositive {
  20             quantity: AnimationQuantity::NativeSampleInterval,
  21         });
  22     }
  23     Ok(Time::from_duration(interval))
  24 }

  10 /// The original frequency is retained so scalar serialization preserves its bits.
  11 #[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
  12 #[serde(try_from = "f32", into = "f32")]
  13 pub struct AnimationSampleRate {
  14     frequency: AnimationFrequency,

  16 }

  18 /// A finite positive reference multiplier in samples per second. Calibration
  19 /// does not require a representable reciprocal sampling period.
  20 #[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize, derive_more::Into)]
  21 #[serde(try_from = "f32", into = "f32")]
  22 pub struct AnimationFrequency(f32);
  23 
  24 impl AnimationFrequency {
  25     /// Project at actual per-reference-sample multiplication or native math.
  26     pub fn hertz(self) -> f32 {
  27         self.0
  28     }
  29 }
  30 
  31 impl TryFrom<f32> for AnimationFrequency {
  32     type Error = AnimationValueError;
  33     fn try_from(hertz: f32) -> AnimationValueResult<Self> {
  34         let quantity = AnimationQuantity::SampleRate;
  35         if !hertz.is_finite() {
  36             return Err(AnimationValueError::NonFinite { quantity });
  37         }
  38         if hertz <= 0.0 {
  39             return Err(AnimationValueError::NonPositive { quantity });
  40         }
  41         Ok(Self(hertz))
  42     }
  43 }
  44 
  45 impl AnimationSampleRate {
  46     /// Samples per second at multiplication, ratio or framework clock boundaries.
  47     pub fn hertz(self) -> f32 {
  48         self.frequency.hertz()
  49     }
  50     /// Native reciprocal duration for fixed-step sampling.
  51     pub fn interval(self) -> PositiveAnimationDuration {
  52         self.interval
  53     }
  54     /// Convert a remaining sample span at its authoritative frequency. Preserve
  55     /// the original f32 count-before-division order, rather than reciprocal multiplication.
  56     pub fn elapsed_for_span(
  57         self,
  58         span: AnimationSampleSpan,
  59     ) -> AnimationValueResult<AnimationElapsed> {
  60         Ok(AnimationElapsed::from_seconds(
  61             span.samples() as f32 / self.hertz(),
  62         )?)
  63     }
  64 }
  65 
  66 impl TryFrom<f32> for AnimationSampleRate {
  67     type Error = AnimationValueError;
  68     fn try_from(hertz: f32) -> AnimationValueResult<Self> {
  69         let frequency = AnimationFrequency::try_from(hertz)?;
  70         let quantity = AnimationQuantity::SampleRate;
  71         let interval = PositiveAnimationDuration::from_elapsed(
  72             AnimationElapsed::from_seconds(hertz.recip())
  73                 .map_err(|error| AnimationValueError::from(error).for_quantity(quantity))?,
  74         )?;
  75         Ok(Self {
  76             frequency,
  77             interval,
  78         })
  79     }
  80 }

  83     fn from(rate: AnimationSampleRate) -> Self {
  84         rate.frequency.hertz()

  52         self.0.checked_add(span.0).map(Self)
