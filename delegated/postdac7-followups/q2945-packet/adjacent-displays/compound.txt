36: /// Side in the property's local frontage coordinates.
37: #[derive(Clone, Copy, Debug, Eq, PartialEq, Serialize, Deserialize, Reflect)]
38: #[serde(rename_all = "snake_case")]
39: pub enum PropertySide {
40:     Left,
41:     Right,
42: }
43: impl PropertySide {
44:     pub const fn sign(self) -> f32 {
45:         match self {
46:             Self::Left => -1.0,
47:             Self::Right => 1.0,
48:         }
49:     }
50:     pub const fn opposite(self) -> Self {
51:         match self {
52:             Self::Left => Self::Right,
53:             Self::Right => Self::Left,
54:         }
55:     }
56: }
