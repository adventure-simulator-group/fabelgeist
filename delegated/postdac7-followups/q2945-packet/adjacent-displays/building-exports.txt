33: impl<'de> Deserialize<'de> for BuildingOrientation {
34:     fn deserialize<D: serde::Deserializer<'de>>(d: D) -> Result<Self, D::Error> {
35:         let orientation = Self(f32::deserialize(d)?);
36:         if orientation.is_valid() {
37:             Ok(orientation)
38:         } else {
39:             Err(serde::de::Error::custom(
40:                 "building orientation must be finite and in [-pi, pi)",
41:             ))

43:     }
44: }
