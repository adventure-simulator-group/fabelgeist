27: /// Identity of one fixed enclosure member, retained through support projection.
28: #[derive(Clone, Copy, Debug, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
29: #[serde(rename_all = "snake_case")]
30: pub enum BoundarySupportElement {
31:     Owner,
32:     GateLanding,
33:     Wall(usize),
34:     WallCap(usize),
35:     GatePost(PropertySide),
36: }
37: 
38: /// The datum and support policy accompany each pose rather than being inferred
39: /// from an unframed vector or an independently enumerated member list.
40: #[derive(Clone, Copy, Debug, PartialEq)]
41: pub enum CityBoundaryPose {
42:     Wall {
43:         index: usize,
44:         centre: Position<GroundRelative>,
45:     },
46:     WallCap {
47:         index: usize,
48:         centre: Position<GroundRelative>,
49:     },
50:     GatePost {
51:         side: PropertySide,
52:         centre: Position<GateRelative>,
53:     },
54: }
55: impl CityBoundaryPose {
56:     pub fn element(self) -> BoundarySupportElement {
57:         match self {
58:             Self::Wall { index, .. } => BoundarySupportElement::Wall(index),
59:             Self::WallCap { index, .. } => BoundarySupportElement::WallCap(index),
60:             Self::GatePost { side, .. } => BoundarySupportElement::GatePost(side),
61:         }
62:     }
63:     /// Native arithmetic adapter: X/Z are already scene coordinates in all
64:     /// cases. The caller must match the pose before interpreting its Y datum.
65:     pub fn plan_metres(self) -> Vec2 {
66:         match self {
67:             Self::Wall { centre, .. } | Self::WallCap { centre, .. } => {
68:                 Vec2::new(centre.metres().x, centre.metres().z)
69:             }
70:             Self::GatePost { centre, .. } => Vec2::new(centre.metres().x, centre.metres().z),
71:         }
72:     }
73: }
74: #[derive(Clone, Copy, Debug, PartialEq)]
75: pub struct CityBoundaryMember {
76:     pub pose: CityBoundaryPose,
77:     pub size_metres: CuboidDimensions,
78:     pub orientation: BuildingOrientation,
79:     pub material: CityBoundaryMaterial,
80: }
