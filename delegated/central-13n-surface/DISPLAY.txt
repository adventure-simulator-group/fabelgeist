  41         normals: Option<Vec<Vec3>>,
  42         uvs: Option<Vec<[f32; 2]>>,
  43         triangles: Vec<[u32; 3]>,
  44     ) -> Self {
  45         let normals = normals.unwrap_or_else(|| vertex_normals(&positions, &triangles));
  46         let uvs = uvs.unwrap_or_else(|| vec![[0.0; 2]; positions.len()]);
  47         let mut cdf = Vec::with_capacity(triangles.len());
  48         let mut total = 0.0;
  49         for t in &triangles {
  50             let [a, b, c] = t.map(|i| positions[i as usize]);
  51             total += (b - a).cross(c - a).length() * 0.5;
  52             cdf.push(total);
  53         }

 170 pub fn read(bytes: &[u8]) -> Result<Ply> {
 171     let Header {
 172         ascii,
 173         body,
 174         elements,
 175     } = Header::read(bytes)?;
 176 
 177     let mut ply = Ply {
 178         vertex_properties: Vec::new(),
 179         vertices: Vec::new(),
 180         faces: Vec::new(),
 181     };
 182     let mut tokens = if ascii {
 183         Some(std::str::from_utf8(&bytes[body..])?.split_whitespace())
 184     } else {
 185         None
 186     };
 187     let mut cursor = body;
 188     let mut next = |kind: Scalar| -> Result<f64> {
 189         if let Some(tokens) = tokens.as_mut() {
 190             Ok(tokens
 191                 .next()
 192                 .ok_or(HairDataError::Truncated {
 193                     format: HairDataFormat::Ply,
 194                 })?
 195                 .parse()?)
 196         } else {
 197             let size = kind.size();
 198             let b = bytes
 199                 .get(cursor..cursor + size)
 200                 .ok_or(HairDataError::Truncated {
 201                     format: HairDataFormat::Ply,
 202                 })?;
 203             cursor += size;
 204             Ok(kind.read(b))
 205         }
 206     };
 207     for element in &elements {
 208         let is_vertex = element.name == "vertex";
 209         let is_face = element.name == "face";
 210         if is_vertex {
 211             ply.vertex_properties = element
 212                 .properties
 213                 .iter()
 214                 .map(|p| match p {
 215                     Property::Scalar(n, _) | Property::List(n, _, _) => n.clone(),
 216                 })
 217                 .collect();
 218         }
 219         for _ in 0..element.count {
 220             let mut row = Vec::new();
 221             let mut face = Vec::new();
 222             for property in &element.properties {
 223                 match property {
 224                     Property::Scalar(_, kind) => row.push(next(*kind)? as f32),
 225                     Property::List(name, count, item) => {
 226                         let n = next(*count)? as usize;
 227                         for _ in 0..n {
 228                             let v = next(*item)?;
 229                             if is_face && (name == "vertex_indices" || name == "vertex_index") {
 230                                 face.push(v as u32);
 231                             }
 232                         }
 233                         row.push(0.0);
 234                     }
 235                 }
 236             }
 237             if is_vertex {
 238                 ply.vertices.push(row);
 239             } else if is_face {
 240                 ply.faces.push(face);
 241             }
 242         }
 243     }
 244     Ok(ply)
 245 }
