  442 /// Finds the last occurrence of the 4-byte little-endian `NEEDLE` in `haystack`.
  443 ///
  444 /// Some benchmarks:
  445 ///
  446 /// - windows().rposition(): 2 GiB/s (complete data independence as it's just loop of u32 loads)
  447 /// - `memmem::rfind`: 8.4 GiB/s (586 MiB/s worst case)
  448 /// - This implementation: 16.0 GiB/s (4.63 GiB/s worst case)
  449 fn rfind<const NEEDLE: u32>(haystack: &[u8]) -> Option<usize> {
  450     const N: usize = core::mem::size_of::<u32>();
  451     if haystack.len() < N {
  452         return None;
  453     }
  454 
  455     // The fast path scans backwards for the needle's last byte as a candidate
  456     let search = &haystack[N - 1..];
  457 
  458     // SWAR lanes
  459     const LANES: usize = core::mem::size_of::<u64>();
  460     let lo = u64::from_ne_bytes([0x01; LANES]);
  461     let lo7 = u64::from_ne_bytes([0x7F; LANES]);
  462     let hi = u64::from_ne_bytes([0x80; LANES]);
  463 
  464     let haszero = |word: u64| word.wrapping_sub(lo) & !word & hi;
  465     let iszero = |word: u64| !(((word & lo7) + lo7) | word) & hi;
