  29             program,

  33             &mut self.wall_assemblies,
  34             &mut self.opening_assemblies,
  35             &mut self.resolved_geometry,
  36         )?;
  37         self.artillery_castle = resolve_artillery_castle(
  38             program,
  39             &self.towers,
  40             &mut self.wall_assemblies,
  41             &mut self.opening_assemblies,
  42             &mut self.resolved_geometry,

  44

  46             program,
  47             edits,

  49             &self.roof_dormers,
  50             &self.towers,

  58         self.timber_frame = timber_frame;

  61         wall_corner_bonds::resolve(&self.wall_assemblies, &mut self.resolved_geometry)?;
  62         if let Some(church) = &mut self.church {
  63             church.roof_assemblies = self.roof_assemblies.iter().map(|roof| roof.id).collect();
  64         }
  65 
  66         church_ground::resolve(crate::spiral_stairs::resolve(self)?)
  67     }
