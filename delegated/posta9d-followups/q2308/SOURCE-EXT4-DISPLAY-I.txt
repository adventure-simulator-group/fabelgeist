  21 fn grid_point(position: Vec2) -> GridPoint {
  22     let x = (position.x / GRID_UNIT_METRES).round() as i32;
  23     let z = (position.y / GRID_UNIT_METRES).round() as i32;
  24     debug_assert!((x as f32 * GRID_UNIT_METRES - position.x).abs() < 0.001);
  25     debug_assert!((z as f32 * GRID_UNIT_METRES - position.y).abs() < 0.001);
  26     GridPoint::new(x, z)
