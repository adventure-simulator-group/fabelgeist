 184 /// Dedicated projected-defense study tags change only the defense assembly,
 185 /// not the host castle's room/circulation randomization. This keeps isolated
 186 /// proofs comparable to the accepted seed-42 host instead of accidentally
 187 /// introducing an unrelated disconnected layout.
 188 fn layout_seed(program: &BuildingProgram) -> Seed {
 189     if program.archetype == BuildingArchetype::CastleGatehouse
 190         && matches!(program.seed.to_u64() % 1_000, 201..=203)
 191     {
 192         Seed::from_u64(42)
 193     } else {
 194         program.seed
 195     }
 196 }
