   5 /// An archive member's parsed filename, without path or existence validation.
   6 ///
   7 /// Stored bytes use the reader's lossy UTF-8 admission. Distinct invalid byte
   8 /// spellings can therefore identify the same parsed name. Text constructors
   9 /// preserve accepted spelling, including case, separators, NUL and emptiness.
  10 #[derive(Clone, Debug, Display, From, PartialEq, Eq)]
  11 #[debug("{_0:?}")]
  12 #[from(String, &str)]
  13 pub struct ArchiveMemberName(String);

  18     }
