30:             .get(&BoneRole::FootLeft)
31:             .and_then(|foot| globals.get(*foot).ok())
32:             .map(GlobalTransform::rotation);
33:         let right_last_rendered_foot_rotation_world = rig
34:             .get(&BoneRole::FootRight)
35:             .and_then(|foot| globals.get(*foot).ok())
36:             .map(GlobalTransform::rotation);
37:         if [
38:             left_last_rendered_world,
39:             right_last_rendered_world,
40:             left_last_rendered_toe_world,

42:         ]
43:         .into_iter()
44:         .flatten()
45:         .any(|point| !point.is_finite())
46:         {
47:             return Err(AnimationValueError::NonFinite {
48:                 quantity: AnimationQuantity::WorldPosition,
49:             });
50:         }
51:         if [
52:             left_last_rendered_foot_rotation_world,
53:             right_last_rendered_foot_rotation_world,
54:         ]
55:         .into_iter()
56:         .flatten()
57:         .any(|rotation| !rotation.is_finite())
58:         {
59:             return Err(AnimationValueError::NonFinite {
60:                 quantity: AnimationQuantity::BodyOrientation,
61:             });
62:         }
63:         self.left_last_rendered_world = left_last_rendered_world;
64:         self.right_last_rendered_world = right_last_rendered_world;
65:         self.left_last_rendered_toe_world = left_last_rendered_toe_world;
66:         self.right_last_rendered_toe_world = right_last_rendered_toe_world;
67:         self.left_last_rendered_foot_rotation_world = left_last_rendered_foot_rotation_world;
68:         self.right_last_rendered_foot_rotation_world = right_last_rendered_foot_rotation_world;
69:         Ok(())
70:     }
71: }
