 319     pub(crate) eocd_disk: u16,

 328 impl EndOfCentralDirectoryRecordFixed {
 329     pub(crate) const SIZE: usize = 22;
 330     pub fn parse(data: &[u8]) -> Result<EndOfCentralDirectoryRecordFixed, Error> {
 331         if data.len() < Self::SIZE {
 332             return Err(Error::from(ErrorKind::Eof));
 333         }
 334 
 335         let result = EndOfCentralDirectoryRecordFixed {
 336             signature: le_u32(&data[0..4]),
 337             disk_number: le_u16(&data[4..6]),
 338             eocd_disk: le_u16(&data[6..8]),
 339             num_entries: le_u16(&data[8..10]),
 340             total_entries: le_u16(&data[10..12]),
 341             central_dir_size: le_u32(&data[12..16]),
 342             central_dir_offset: le_u32(&data[16..20]),
 343             comment_len: le_u16(&data[20..22]),
 344         };
