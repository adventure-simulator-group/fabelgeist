 110         let comment_len = eocd.comment_len as usize;
 111         if comment_start + comment_len > data.len() {
 112             return Err(Error::from(ErrorKind::Eof));
 113         }
 114 
 115         // Always probe for a zip64 EOCD locator, which can be present without sentinels
 116         let zip64_locator = location

 121             return EndOfCentralDirectory::create(EndOfCentralDirectoryRecord::from_parts(

 125         };

 129             zip64_eocd,
 130             &eocd,
 131             &zip64_locator,
 132             location as u64 - Zip64EndOfCentralDirectoryLocatorRecord::SIZE as u64,
 133         ) {
 134             Some(record) => record,
 135             None => {
 136                 return EndOfCentralDirectory::create(EndOfCentralDirectoryRecord::from_parts(
 137                     location as u64,
 138                     &eocd,
