 234     fn validate(&self) -> Result<(), Error> {
 235         // It doesn't make sense if the start of the central directory is after
 236         // the end.
 237         if self.directory_offset() > self.head_eocd_offset() {
 238             return Err(Error::from(ErrorKind::InvalidEndOfCentralDirectory));
 239         }

 241         Ok(())
 242     }
