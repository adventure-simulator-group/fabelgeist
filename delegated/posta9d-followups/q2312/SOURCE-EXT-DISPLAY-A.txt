  20 /// Locates the End of Central Directory (EOCD) record in a ZIP archive.
  21 ///
  22 /// The `ZipLocator` is responsible for finding the EOCD record, which is
  23 /// crucial for reading the contents of a ZIP file.
  24 ///
  25 /// In the event, that the comment or tailing data contains the EOCD signature,
  26 /// causing the zip locator to fail to parse. One can reparse the data starting
  27 /// from the false EOCD offset using the reported offset
  28 /// [`Error::eocd_offset()`]
  29 #[derive(Debug)]
  30 pub struct ZipLocator {
  31     max_search_space: u64,
  32 }
  33

  35     fn default() -> Self {
  36         Self::new()
  37     }

  48     /// Sets the maximum number of bytes to search for the EOCD signature.
  49     ///
  50     /// The search is performed backwards from the end of the data source.
  51     ///
  52     /// ```rust
  53     /// use rawzip::ZipLocator;
  54     ///
  55     /// let locator = ZipLocator::new().max_search_space(1024 * 64); // 64 KiB
  56     /// ```
  57     pub fn max_search_space(mut self, max_search_space: u64) -> Self {

  62     fn locate_in_byte_slice(&self, data: &[u8]) -> Result<EndOfCentralDirectory, Error> {
  63         let max_search_space = self.max_search_space.min(data.len() as u64) as usize;
  64         let location = find_end_of_central_dir_signature(data, max_search_space)
  65             .ok_or(ErrorKind::MissingEndOfCentralDirectory)?;
  66 
  67         let mut eocd = self
  68             .locate_in_byte_slice_impl(data, location)
  69             .map_err(|e| e.with_eocd_offset(location as u64))?;

  71         // Transparently verify that the self reported central directory points
  72         // to a valid entry. If it is not a valid entry, we can attempt to
  73         // correct offsets when there is undeclared prelude data by testing if

  75         // marker, which should hold true in the vast majority of cases. If both
  76         // checks fail, defer returning an error until the user explicitly wants
  77         // to iterate through the central directory.
