 356 /// An iterator over the central directory file header records.
 357 ///
 358 /// Created from [`ZipSliceArchive::entries`].
 359 #[derive(Debug, Clone)]
 360 pub struct ZipSliceEntries<'data> {
 361     entry_data: &'data [u8],
 362     base_offset: u64,
 363     current_offset: u64,
 364 }
 365 
 366 impl<'data> ZipSliceEntries<'data> {
 367     /// Yield the next zip file entry in the central directory if there is any
 368     ///
 369     /// An error is terminal. After this method returns an error, all
 370     /// subsequent calls return `Ok(None)`.
 371     #[inline]
 372     pub fn next_entry(&mut self) -> Result<Option<ZipFileHeaderRecord<'data>>, Error> {
 373         let result = self.next_file_entry();
 374         if result.is_err() {
 375             self.entry_data = &[];
 376         }
 377         result
 378     }
 379 
 380     // Use inline always as this improved slice extraction of 100k entries by 14%.
 381     // The reason why is that the return value is large and if the caller only wants
 382     // the wayfinder then there is no need to materialize all the fields.
 383     #[inline(always)]
 384     fn next_file_entry(&mut self) -> Result<Option<ZipFileHeaderRecord<'data>>, Error> {
 385         if self.entry_data.is_empty() {
 386             return Ok(None);
 387         }
 388 
 389         let file_header = match ZipFileHeaderFixed::parse(self.entry_data) {
 390             Ok(file_header) => file_header,
 391             Err(err) => return self.next_entry_terminator(err),
 392         };
 393         let Some((file_name, extra_field, file_comment, entry_data)) =
 394             file_header.parse_variable_length(&self.entry_data[ZipFileHeaderFixed::SIZE..])
 395         else {
 396             return Err(Error::from(ErrorKind::Eof));
 397         };
 398 
 399         let mut entry = ZipFileHeaderRecord::from_parts(
 400             file_header,
 401             file_name,
 402             extra_field,
 403             file_comment,
 404             self.current_offset,
 405         );
 406         // An offset past u64::MAX is past the end of all possible inputs.
 407         entry.local_header_offset = entry
 408             .local_header_offset
 409             .checked_add(self.base_offset)
 410             .ok_or(ErrorKind::Eof)?;
