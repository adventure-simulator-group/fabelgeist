   4 /// An error that occurred while reading or writing a zip file
   5 #[derive(Debug)]
   6 pub struct Error {
   7     // When an allocator is available the inner payload is boxed to keep `Error`
   8     // pointer-sized, which measurably speeds up the common success path of
   9     // iterating entries (the boxed form was ~18-20% faster over 200k entries).
  10     // Without an allocator the payload is stored inline (40 bytes), which the
  11     // core-only tier accepts since it cannot box.
  12     #[cfg(feature = "alloc")]
  13     inner: Box<ErrorInner>,
  14     #[cfg(not(feature = "alloc"))]
  15     inner: ErrorInner,
  16 }
