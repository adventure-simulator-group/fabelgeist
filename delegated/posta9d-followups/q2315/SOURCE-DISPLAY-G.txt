 108 impl core::error::Error for Error {
 109     fn source(&self) -> Option<&(dyn core::error::Error + 'static)> {
 110         match &self.inner.kind {
 111             #[cfg(feature = "std")]
 112             ErrorKind::IO(e) => Some(e),
 113             #[cfg(feature = "alloc")]
 114             ErrorKind::InvalidUtf8(e) => Some(e),
 115             _ => None,
 116         }
 117     }
 118 }

 127 impl core::fmt::Display for ErrorKind {
 128     fn fmt(&self, f: &mut core::fmt::Formatter) -> core::fmt::Result {
 129         match *self {
 130             #[cfg(feature = "std")]
 131             ErrorKind::IO(ref err) => err.fmt(f),
 132             ErrorKind::MissingEndOfCentralDirectory => {
 133                 write!(f, "Missing end of central directory")
 134             }
 135             ErrorKind::BufferTooSmall { required } => {
 136                 write!(f, "Buffer size too small: required {required} bytes")
 137             }
 138             ErrorKind::Eof => {
