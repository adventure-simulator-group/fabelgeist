 171 impl From<ErrorKind> for Error {
 172     fn from(kind: ErrorKind) -> Error {
 173         let inner = ErrorInner {
 174             kind,
 175             eocd_offset: None,
 176         };
 177         Error {
 178             #[cfg(feature = "alloc")]
 179             inner: Box::new(inner),
 180             #[cfg(not(feature = "alloc"))]
 181             inner,
 182         }
 183     }
 184 }
