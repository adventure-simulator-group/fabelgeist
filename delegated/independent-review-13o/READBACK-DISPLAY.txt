   1 //! Allocate before taking a mapped view; initialize before exposing typed data.
   2 use anyhow::{Result, ensure};
   3 use bytemuck::AnyBitPattern;
   4 
   5 pub(super) struct Readback<T> {
   6     values: Vec<T>,
   7     byte_len: usize,
   8 }
   9 
  10 impl<T: AnyBitPattern> Readback<T> {
  11     pub(super) fn new(byte_len: u64) -> Result<Self> {
  12         let byte_len = usize::try_from(byte_len)?;
  13         let element_size = std::mem::size_of::<T>();
  14         ensure!(
  15             element_size != 0,
  16             "GPU readback requires nonzero-sized elements"
  17         );
  18         ensure!(
  19             byte_len.is_multiple_of(element_size),
  20             "GPU readback byte length is not a whole number of elements"
  21         );
  22         Ok(Self {
  23             values: Vec::with_capacity(byte_len / element_size),
  24             byte_len,
  25         })
  26     }
  27 
  28     pub(super) fn copy_from(mut self, bytes: &[u8]) -> Result<Vec<T>> {
  29         // GPU allocations may include trailing alignment padding.
  30         ensure!(bytes.len() >= self.byte_len, "GPU readback is truncated");
  31         // SAFETY: the allocation fits every byte and is aligned for T. The
  32         // buffers do not overlap, and AnyBitPattern permits all copied values.
  33         // Set the length only after every element has been initialized.
  34         unsafe {
  35             std::ptr::copy_nonoverlapping(
  36                 bytes.as_ptr(),
  37                 self.values.as_mut_ptr().cast::<u8>(),
  38                 self.byte_len,
  39             );
  40             self.values
  41                 .set_len(self.byte_len / std::mem::size_of::<T>());
  42         }
  43         Ok(self.values)
  44     }
  45 }

  51     #[test]
  52     fn readback_rejects_partial_and_zero_sized_elements() {
  53         assert!(Readback::<u32>::new(3).is_err());
  54         assert!(Readback::<()>::new(0).is_err());
  55         assert!(Readback::<u32>::new(4).unwrap().copy_from(&[0; 3]).is_err());
  56     }
  57 
  58     #[test]
  59     fn readback_copies_unaligned_bytes_and_preserves_bits() {
  60         let bytes = [0, 1, 2, 3, 4, 5, 6, 7, 8];
  61         let values = Readback::<u32>::new(8)
  62             .unwrap()
  63             .copy_from(&bytes[1..])
  64             .unwrap();
  65         assert_eq!(
  66             values,
  67             [
  68                 u32::from_ne_bytes([1, 2, 3, 4]),
  69                 u32::from_ne_bytes([5, 6, 7, 8])
  70             ]
  71         );
  72         assert!(
  73             Readback::<u32>::new(0)
  74                 .unwrap()
  75                 .copy_from(&[])
  76                 .unwrap()
  77                 .is_empty()
  78         );
  79     }

  82     fn readback_excludes_gpu_allocation_padding() {

  84             .unwrap()
  85             .copy_from(&[1, 2, 3, 0])
  86             .unwrap();
  87         assert_eq!(values, [1, 2, 3]);
  88     }

   1 //! Borrowed host element bytes admitted for a native buffer upload.

   3 use crate::globals::WgpuContext;
   4 
   5 use super::{BufferByteLength, BufferByteOffset};

   7 #[derive(Clone, Copy, Debug, PartialEq, Eq)]
   8 pub enum BufferUploadOccupancy {
   9     Empty,
  10     Populated,
  11 }
  12 
  13 /// Exact `NoUninit` element representation, retained through queue submission.
  14 ///
  15 /// ```compile_fail
  16 /// use crate::prelude::{BufferUpload, BufferByteLength};
  17 /// let data = BufferUpload::from_elements(&[7u32]);
  18 /// let _: BufferByteLength = data;
  19 /// ```
  20 #[derive(Clone, Copy, Debug)]
  21 pub struct BufferUpload<'a>(&'a [u8]);
  22 
  23 const EMPTY_BINDING_WORD: [u8; std::mem::size_of::<u32>()] = [0; std::mem::size_of::<u32>()];
  24 
  25 impl<'a> BufferUpload<'a> {
  26     pub fn from_elements<T: bytemuck::NoUninit>(elements: &'a [T]) -> Self {
  27         Self(bytemuck::cast_slice(elements))
  28     }
  29 
  30     pub fn length(self) -> BufferByteLength {
  31         BufferByteLength::from(self.0.len())
  32     }
  33 
  34     pub fn occupancy(self) -> BufferUploadOccupancy {
  35         if self.0.is_empty() {
  36             BufferUploadOccupancy::Empty
  37         } else {
  38             BufferUploadOccupancy::Populated
  39         }
  40     }
  41 
  42     /// Keep the established scalar-word binding for an empty armor input.
  43     pub fn with_empty_word(self) -> Self {
  44         match self.occupancy() {
  45             BufferUploadOccupancy::Empty => Self(&EMPTY_BINDING_WORD),
  46             BufferUploadOccupancy::Populated => self,
  47         }
  48     }
  49 
  50     pub(super) fn write_to(
  51         self,
  52         context: &WgpuContext,
  53         buffer: &wgpu::Buffer,
  54         at: BufferByteOffset,
  55     ) {
  56         context.queue.write_buffer(buffer, u64::from(at), self.0);
  57     }
  58 }
