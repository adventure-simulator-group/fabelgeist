  17 pub(super) fn forage_terminal_minute(
  18     ctx: &ReducerContext,
  19     character_id: u64,
  20     current_minute: StrategicMinute,
  21     duration: u64,
  22 ) -> Result<Option<StrategicMinute>, String> {
  23     let injury = crate::surgery::preview_injury_boundary(
  24         ctx,
  25         character_id,
  26         duration,
  27         crate::surgery::InjuryRecoveryMinutes::NONE,
  28     )?;
  29     let (disease_safe, disease_terminal) = crate::disease::preview_disease_terminal_boundary(
  30         ctx,
  31         character_id,
  32         injury.elapsed,
  33         false,
  34     )?;
  35     let safe = injury.elapsed.min(disease_safe);
  36     if safe < duration || injury.terminal || disease_terminal {
  37         Ok(Some(current_minute.checked_add_minutes(safe).ok_or(
  38             "Foraging terminal time exceeds the strategic clock",
  39         )?))
  40     } else {
  41         Ok(None)
  42     }
  43 }
