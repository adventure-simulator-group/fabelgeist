1880 /// Advance fatigue for strategic travel. The existing `calories_used` field is
1881 /// treated as a recoverable fatigue reservoir until food/day-boundary state is
1882 /// implemented.
1883 pub fn apply_travel_condition(
1884     ctx: &ReducerContext,
1885     character_id: u64,
1886     starting_minute: StrategicMinute,
1887     elapsed_minutes: u64,
1888     prayer_minutes: u16,
1889 ) -> Result<(), String> {
1890     if elapsed_minutes > adventuresim_core::alcohol::MAX_ALCOHOL_INTERVAL_MINUTES {
1891         return Err("Travel condition interval cannot exceed one year".into());
1892     }
1893     let interval_end = starting_minute
1894         .checked_add_minutes(elapsed_minutes)
1895         .ok_or("Travel condition interval overflow")?;
1896     for (segment_start, segment_end, history_minute) in
1897         adventuresim_core::alcohol::travel_evening_segments(starting_minute, interval_end)
1898             .map_err(str::to_string)?
1899     {
1900         apply_elapsed_needs(ctx, character_id, segment_end.elapsed_since(segment_start))?;
1901         // Movement alone may spend potable alcohol as emergency hydration.
1902         // Attribute each whole serving to the evening in which the deficit
1903         // arose; generic waits and camp downtime never invoke this fallback.
1904         if let Some(mut needs) = ctx.db.character_needs().character_id().find(character_id)
1905             && needs.water_balance_ml < 0.0
1906         {
1907             let supplied = crate::alcohol::consume_emergency_hydration(
1908                 ctx,
1909                 character_id,
1910                 -needs.water_balance_ml,
1911                 history_minute,
1912             );
1913             needs.water_balance_ml += supplied as f32;
1914             ctx.db.character_needs().character_id().update(needs);
1915         }
1916     }
1917     let mut stats = ctx
1918         .db
1919         .character_stats()
1920         .character_id()
1921         .find(character_id)
1922         .ok_or("Character stats not found")?;
1923     stats.calories_used += elapsed_minutes as f32 / (24.0 * 60.0) * STRATEGIC_TRAVEL_KCAL_PER_DAY;
1924     ctx.db.character_stats().character_id().update(stats);
1925 
1926     refuse_expired_holy_day_demands(ctx, character_id, true)?;
1927     let condition = refresh_character_strategic_condition_projection(ctx, character_id)?;
1928     let professes_religion = ctx
1929         .db
1930         .character_condition()
1931         .character_id()
1932         .find(character_id)
1933         .is_some_and(|row| row.religion_id.is_some());
1934     if professes_religion && condition.fervor > 0.0 {
1935         let command = party_command(ctx, character_id)?;
1936         let daily_penalty = religious_neglect_morale(condition.fervor, command);
1937         let missed_prayer = 1.0 - prayer_observance(condition.fervor, prayer_minutes);
1938         let elapsed_days = elapsed_minutes as f32 / MINUTES_PER_DAY as f32;
1939         let prayer_penalty = daily_penalty * missed_prayer * elapsed_days;
1940         if prayer_penalty > 0.0 {
1941             insert_morale_event_without_refresh(
1942                 ctx,
1943                 character_id,
1944                 MoraleEventKind::TravelPrayerNeglected,
1945                 -prayer_penalty,
1946                 format!(
1947                     "travel-prayer:{starting_minute}:{}",
1948                     starting_minute.saturating_add_minutes(elapsed_minutes)
1949                 ),
1950             );
