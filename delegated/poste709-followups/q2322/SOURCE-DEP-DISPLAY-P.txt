 262 /// Stationary but strenuous strategic time used by investigation actions.
 263 /// Unlike neutral waiting this applies ordinary needs and the same fatigue
 264 /// reservoir as travel, but it never records movement filth or terrain
 265 /// exposure.
 266 pub fn advance_investigation_time(
 267     ctx: &ReducerContext,
 268     character_id: u64,
 269     minutes: u64,
 270 ) -> Result<bool, String> {
 271     let mut remaining = minutes;
 272     while remaining > 0 {
 273         let safe = preview_travel_time(ctx, character_id, remaining)?;
 274         if safe == 0 {
 275             return settle_travel_boundary(ctx, character_id);
 276         }
 277         let before = ctx
 278             .db
 279             .character_time()
 280             .character_id()
 281             .find(character_id)
 282             .map_or(StrategicMinute::ZERO, |t| t.minutes);
 283         let alive = advance_character_time(ctx, character_id, safe)?;
 284         let after = ctx
 285             .db
 286             .character_time()
 287             .character_id()
 288             .find(character_id)
 289             .map_or(before, |t| t.minutes);
 290         let elapsed = after.elapsed_since(before);
 291         if !alive || elapsed < safe {
 292             return Ok(false);
 293         }
 294         remaining -= elapsed;
 295     }
 296     Ok(true)
 297 }
