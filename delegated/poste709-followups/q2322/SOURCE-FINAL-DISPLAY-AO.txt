 410 /// Foraging receipts bind every concrete harvested unit to an object and
 411 /// material lot. Preserve the shared grant helper's stacking semantics for
 412 /// every other caller while intentionally issuing one validated unit here.
 413 pub(crate) fn add_foraged_inventory_item_checked_rows(
 414     ctx: &ReducerContext,
 415     character_id: u64,
 416     item_id: &str,
 417     quantity: u32,
 418 ) -> Result<Vec<u64>, String> {
 419     let mut rows = Vec::with_capacity(quantity as usize);
 420     for _ in 0..quantity {
 421         rows.push(
 422             add_inventory_item_checked(ctx, character_id, item_id, 1)?
 423                 .ok_or("Foraged inventory insertion returned no row")?,
 424         );
 425     }
 426     Ok(rows)
 427 }

 440 pub fn is_currency(ctx: &ReducerContext, item_id: &str) -> bool {
 441     ctx.db
 442         .item()
 443         .id()
 444         .find(item_id.to_owned())
 445         .is_some_and(|item| item.kind == CatalogItemKind::Currency)
 446 }

 448 pub fn personal_currency_total(ctx: &ReducerContext, character_id: u64) -> u64 {
 449     ctx.db
 450         .inventory_item()
 451         .character_id()
 452         .filter(character_id)
 453         .filter(|stack| is_currency(ctx, &stack.item_id))
 454         .map(|stack| u64::from(stack.quantity))
 455         .sum()
 456 }
