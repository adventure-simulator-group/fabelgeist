 329 pub fn validate_duration(minutes: u64) -> Result<(), &'static str> {
 330     if (MIN_FORAGE_MINUTES..=MAX_FORAGE_MINUTES).contains(&minutes) && minutes.is_multiple_of(60) {
 331         Ok(())
 332     } else {
 333         Err("Foraging duration must use whole hours from one to 24 hours")
 334     }
