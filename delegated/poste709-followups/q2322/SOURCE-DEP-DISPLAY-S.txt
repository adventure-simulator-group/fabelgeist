 185 /// Commit a terminal injury or disease event that falls exactly on the
 186 /// character's current strategic minute. This intentionally grants no elapsed
 187 /// travel time, condition use, filth, or training.
 188 pub fn settle_travel_boundary(ctx: &ReducerContext, character_id: u64) -> Result<bool, String> {
 189     let character = ctx
 190         .db
 191         .character()
 192         .id()
 193         .find(character_id)
 194         .ok_or("Character not found")?;
 195     if !character.alive {
 196         return Ok(false);
 197     }
 198     advance_character_time(ctx, character_id, 0)
 199 }
