 484 /// Resolve the single completion/exposure check independently from yield.
 485 /// This accepts a partial elapsed duration so interrupted illegal work can
 486 /// still be noticed without granting partial forage yields.
 487 pub fn resolve_stealth(
 488     seed: Seed,
 489     environment: ForageEnvironment,
 490     elapsed_minutes: u64,
 491     stealth_check_millirank: u16,
 492 ) -> (Option<u16>, Option<bool>) {
 493     let dc = stealth_dc_millirank(environment, elapsed_minutes);
 494     let succeeded = dc.map(|dc| {
 495         let roll = ForageDraw::Stealth.below(seed, "", "", 1_001) as u16;
 496         stealth_check_millirank.saturating_add(roll) >= dc
 497     });
 498     (dc, succeeded)
 499 }
