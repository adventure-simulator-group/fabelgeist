 467 fn acting_checks(
 468     ctx: &ReducerContext,
 469     character_id: u64,
 470     mixture: LocalTerrainMixture,
 471 ) -> Result<(u16, u16), String> {
 472     let attributes = ctx
 473         .db
 474         .character_attributes()
 475         .character_id()
 476         .find(character_id)
 477         .ok_or("Character attributes not found")?;
 478     let limbs = ctx
 479         .db
 480         .character_limbs()
 481         .character_id()
 482         .find(character_id)
 483         .ok_or("Character limbs not found")?;
 484     let skills = ctx
 485         .db
 486         .character_skills()
 487         .character_id()
 488         .find(character_id)
 489         .ok_or("Character skills not found")?;
 490     let stats = ctx
 491         .db
 492         .character_stats()
 493         .character_id()
 494         .find(character_id)
 495         .ok_or("Character stats not found")?;
 496     let equipment = StrategicEquipment::load(ctx, character_id);
 497     let check = |skill| {
 498         skills.skill_check_by_parts(
 499             skill,
 500             &attributes,
 501             &limbs,
 502             &stats,
 503             &equipment,
 504             LimbWeights::all_equal(),
 505         )
 506     };
 507     let terrain_training = check(Skill::TerrainPlains) * f32::from(mixture.plains) / 1_000.0
 508         + check(Skill::TerrainForest) * f32::from(mixture.forest) / 1_000.0
 509         + check(Skill::TerrainHills) * f32::from(mixture.hills) / 1_000.0
 510         + check(Skill::TerrainWetlands) * f32::from(mixture.wetlands) / 1_000.0;
 511     let stealth_training = check(Skill::Stealth);
 512     Ok((
 513         (terrain_training.clamp(0.0, 5.0) * 1_000.0).round() as u16,
 514         (stealth_training.clamp(0.0, 5.0) * 1_000.0).round() as u16,
 515     ))
 516 }
