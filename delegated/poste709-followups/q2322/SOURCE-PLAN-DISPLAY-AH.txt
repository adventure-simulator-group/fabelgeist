 717 pub fn build_forage_plan(authority: ForagePlanAuthority) -> ForagePlanningOutcome {
 718     let sources_are_canonical = !authority.source_ids.is_empty()

 720             .source_ids
 721             .windows(2)
 722             .all(|pair| pair[0] < pair[1]);
 723     let actor = authority.coordinates.actor();
 724     let decisions_match_sources = authority.license_decisions.len() == authority.source_ids.len()
 725         && authority
 726             .license_decisions
 727             .iter()
 728             .zip(&authority.source_ids)
 729             .all(|(binding, source_id)| {
 730                 binding.actor == actor
 731                     && binding.source.id() == source_id
 732                     && matches!(
 733                         binding.decision.kind(),
 734                         RightsDecisionKind::Allowed | RightsDecisionKind::Denied
 735                     )
 736             });
 737     let any_denied = authority
 738         .license_decisions
 739         .iter()
 740         .any(|binding| binding.decision.kind() == RightsDecisionKind::Denied);
 741     let expected_legality = if authority.local_restriction || any_denied {
 742         ForageLegality::IllegalAttempt
 743     } else {
 744         ForageLegality::Legal
