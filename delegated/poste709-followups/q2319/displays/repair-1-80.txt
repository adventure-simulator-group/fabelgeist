  20 #[derive(Clone, Debug)]
  21 #[table(accessor = item_condition, public)]
  22 pub struct ItemCondition {
  23     #[primary_key]
  24     pub inventory_item_id: u64,
  25     pub tier_1: f32,
  26     pub tier_2: f32,
  27     pub tier_3: f32,
  28     pub tier_4: f32,
  29     pub tier_5: f32,
  30 }
  31 
  32 impl ItemCondition {
  33     pub fn bins(&self) -> DamageBins {
  34         DamageBins([
  35             self.tier_1,
  36             self.tier_2,
  37             self.tier_3,
  38             self.tier_4,
  39             self.tier_5,
  40         ])
  41         .normalized()
  42     }
  43     fn set_bins(&mut self, bins: DamageBins) {
  44         [
  45             self.tier_1,
  46             self.tier_2,
  47             self.tier_3,
  48             self.tier_4,
  49             self.tier_5,
  50         ] = bins.normalized().0;
  51     }
  52 }
