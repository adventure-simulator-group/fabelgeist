 914 pub(crate) fn require_object(
 915     ctx: &ReducerContext,
 916     character_id: u64,
 917     scope: CarriedInventoryScope,
 918     row_id: u64,
 919 ) -> Result<InventoryObject, String> {
 920     let object = object_for_row(ctx, scope, row_id)?
 921         .ok_or("Inventory row has no stable physical object identity")?;
 922     let actor = ctx
 923         .db
 924         .character()
 925         .id()
 926         .find(character_id)
 927         .ok_or("Character not found")?;
 928     crate::object_custody::require_actor_carried_object(ctx, &actor, &object)?;
 929     Ok(object)
 930 }
