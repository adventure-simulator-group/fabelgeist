 150         ctx.db
 151             .party_inventory_item()
 152             .id()
 153             .delete(party_inventory_item_id);
 154     } else {
 155         ctx.db
 156             .party_item_amount()
 157             .party_inventory_item_id()
 158             .update(state);
 159     }
 160     Ok(consumed)
 161 }

 163 pub fn move_personal_to_party(
 164     ctx: &ReducerContext,
 165     inventory_item_id: u64,
 166     party_inventory_item_id: u64,
 167 ) -> Result<(), String> {
 168     let state = ctx
 169         .db
 170         .inventory_item_amount()
 171         .inventory_item_id()
 172         .find(inventory_item_id)
 173         .ok_or("Measured personal item state is missing")?;
 174     ctx.db
 175         .inventory_item_amount()
 176         .inventory_item_id()
 177         .delete(inventory_item_id);
 178     ctx.db.party_item_amount().insert(PartyItemAmount {
 179         party_inventory_item_id,
 180         remaining_fraction_micros: state.remaining_fraction_micros,
 181     });
 182     Ok(())
 183 }
