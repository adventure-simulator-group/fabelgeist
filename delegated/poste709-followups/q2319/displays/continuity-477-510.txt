 477     let mut inventory = ctx
 478         .db
 479         .inventory_item()
 480         .character_id()
 481         .filter(decedent_id)
 482         .collect::<Vec<_>>();
 483     inventory.sort_by_key(|row| row.id);
 484     for mut row in inventory {
 485         row.character_id = heir_id;
 486         ctx.db.inventory_item().id().update(row);
 487     }
 488     crate::capability::refresh_character_capability(ctx, heir_id)?;
 489     Ok(())
 490 }
