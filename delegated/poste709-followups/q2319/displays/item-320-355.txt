 322 pub(crate) fn inventory_food_definition(
 323     kind: Option<CatalogItemKind>,
 324     item_id: &str,
 325 ) -> Result<Option<&'static adventuresim_core::food::FoodDefinition>, String> {
 326     let definition = adventuresim_core::food::definition(item_id);
 327     if kind == Some(CatalogItemKind::Food) || definition.is_some() {
 328         definition
 329             .map(Some)
 330             .ok_or_else(|| format!("Food definition not found for {item_id}"))
 331     } else {
 332         Ok(None)
 333     }
 334 }

 336 pub(crate) fn requires_stable_object(
 337     definition: Option<&Item>,
 338     food: bool,
 339     measured: bool,
 340 ) -> bool {
 341     food || measured
 342         || definition.is_some_and(|definition| {
 343             definition.repairable
 344                 || definition.kind == CatalogItemKind::Medication
 345                 || (definition.kind == CatalogItemKind::Weapon && definition.melee)
 346                 || definition.container_capacity_ml > 0
 347                 || !definition.attachment_points.is_empty()
 348         })
 349 }
