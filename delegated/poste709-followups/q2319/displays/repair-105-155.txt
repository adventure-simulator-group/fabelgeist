 106 fn service_matches(
 107     service: adventuresim_core::durability::RepairService,
 108     kind: CatalogItemKind,
 109 ) -> bool {
 110     repair_kind(kind).is_some_and(|kind| service.matches(kind))

 113 pub(crate) fn initialize_item_condition(ctx: &ReducerContext, inventory: &InventoryItem) {
 114     let Some(definition) = ctx.db.item().id().find(&inventory.item_id) else {
 115         return;
 116     };
 117     if !definition.repairable
 118         || ctx
 119             .db
 120             .item_condition()
 121             .inventory_item_id()
 122             .find(inventory.id)
 123             .is_some()
 124     {
 125         return;
 126     }
 127     ctx.db.item_condition().insert(ItemCondition {
 128         inventory_item_id: inventory.id,
 129         tier_1: 0.0,
 130         tier_2: 0.0,
 131         tier_3: 0.0,
 132         tier_4: 0.0,
 133         tier_5: 0.0,
 134     });
 135 }
