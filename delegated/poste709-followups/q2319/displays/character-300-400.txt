 302 /// Death authority is private because its timestamp may lie beyond another
 303 /// character's personal frontier. Strategic simulation consumes this trusted
 304 /// broad projection. Any player-facing presentation must additionally compare
 305 /// `strategic_minute` with the selected observer's CharacterTime.
 306 #[view(accessor = backend_character_deaths, public)]
 307 pub fn backend_character_deaths(ctx: &ViewContext) -> Vec<CharacterDeath> {
 308     if !character_view_is_gateway(ctx) {
 309         return Vec::new();
 310     }
 311     gateway_character_ids(ctx)
 312         .into_iter()
 313         .filter_map(|id| ctx.db.character_death().character_id().find(id))
 314         .collect()
 315 }

 334 /// and committed tactical/autoresolve outcomes. Repeated calls retain the first
 335 /// recorded cause, source, and strategic minute.
 336 pub fn transition_character_to_dead(
 337     ctx: &ReducerContext,
 338     character_id: u64,
 339     cause: DeathCause,
 340     source: DeathSource,
 341     source_id: Option<String>,
 342 ) -> Result<CharacterDeath, String> {

 344         .db
 345         .character_time()
 346         .character_id()
 347         .find(character_id)
 348         .map_or(StrategicMinute::ZERO, |t| t.minutes);
 349     transition_character_to_dead_at(
 350         ctx,
 351         character_id,
 352         cause,
 353         source,
 354         source_id,
 355         strategic_minute,
 356     )
 357 }

 371         return Ok(death);
 372     }

 374         .db
 375         .character()
 376         .id()
 377         .find(character_id)

 382         .character_id()
 383         .find(character_id)

 386         return Err("Character death cannot be after their personal clock".into());
