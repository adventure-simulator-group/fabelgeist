 274 pub(crate) fn subtree_object_ids(
 275     ctx: &ReducerContext,
 276     root_object_id: u64,
 277 ) -> Result<Vec<u64>, String> {

 286 /// Validates an entire custody move before its caller creates a replacement
 287 /// root row or changes an installed fixture. This keeps death cleanup and
 288 /// other exceptional returns fail-closed even before reducer rollback applies.
 289 pub(crate) fn prevalidate_rehome_subtree(
 290     ctx: &ReducerContext,
 291     root_object_id: u64,
 292     destination: &OperationalCustody,
 293 ) -> Result<(), String> {
 294     match destination {
 295         OperationalCustody::Character(character_id) => {
 296             if ctx.db.character().id().find(character_id.get()).is_none() {
 297                 return Err("Destination character custody is unavailable".into());
 298             }
 299         }
 300         OperationalCustody::Party(party_id) => {
 301             if ctx
 302                 .db
 303                 .party_authority()
 304                 .id()
 305                 .find(party_id.as_str().to_owned())
 306                 .is_none()
 307             {
 308                 return Err("Destination party custody is unavailable".into());
 309             }
 310         }
 311         _ => return Err("Invalid subtree destination".into()),
 312     }
 313 
 314     for id in subtree_object_ids(ctx, root_object_id)? {
 315         let object = ctx
 316             .db
 317             .inventory_object()
 318             .id()
 319             .find(id)
 320             .ok_or("Inventory subtree object is missing")?;
 321         crate::object_custody::resolve_object_custody(ctx, &object)?;
 322         match object.location {
 323             InventoryLocation::Personal(location) => {
 324                 let row = ctx
 325                     .db
 326                     .inventory_item()
 327                     .id()
 328                     .find(location.row_id)
 329                     .ok_or("Inventory subtree row is missing")?;
 330                 if row.quantity != 1 {
 331                     return Err("Stable inventory objects must be quantity one".into());
 332                 }
 333             }
 334             InventoryLocation::Party(location) => {
 335                 let row = ctx
 336                     .db
 337                     .party_inventory_item()
 338                     .id()
 339                     .find(location.row_id)
 340                     .ok_or("Party inventory subtree row is missing")?;
 341                 if row.quantity != 1 {
 342                     return Err("Stable inventory objects must be quantity one".into());
 343                 }
 344             }
 345             InventoryLocation::Fireplace(_) if id == root_object_id => {}
 346             _ => return Err("Inventory subtree has an unsupported location transition".into()),
 347         }
 348     }
 349     Ok(())
 350 }

 352 pub(crate) fn carried_location_for_row(
 353     destination: &OperationalCustody,
 354     row_id: u64,
 355 ) -> Result<InventoryLocation, String> {
 356     match destination {
 357         OperationalCustody::Character(character_id) => {
 358             Ok(InventoryLocation::personal(character_id.get(), row_id))
 359         }
 360         OperationalCustody::Party(party_id) => {
 361             Ok(InventoryLocation::party(party_id.as_str(), row_id))
 362         }
 363         OperationalCustody::Container(_)
 364         | OperationalCustody::Place(_)
 365         | OperationalCustody::Fixture(_) => {
 366             Err("Inventory row destination must be a character or party".into())
 367         }
 368     }
 369 }
