 751             merged_object_id =
 752                 object_for_row(ctx, CarriedInventoryScope::Party, target.id)?.map(|row| row.id);
 753             target.quantity = target
 754                 .quantity
 755                 .checked_add(source.quantity)
 756                 .ok_or("Container stack quantity overflow")?;
 757             ctx.db.party_inventory_item().id().update(target);
 758             ctx.db.party_inventory_item().id().delete(source.id);
 759         }
 760         _ => return Ok(()),
 761     }
 762     if let Some(other_id) = merged_object_id {
 763         ctx.db.inventory_object().id().delete(other_id);
 764     }
 765     ctx.db.inventory_object().id().delete(emptied.id);
 766     Ok(())
 767 }
