 322 pub fn contamination_at(anchor: f32, growth_per_hour: f32, elapsed_minutes: u64) -> f32 {
 323     if !anchor.is_finite() || !growth_per_hour.is_finite() {
 324         return MAX_CONTAMINATION;
 325     }
 326     let exponent = (growth_per_hour.max(0.0) as f64 * elapsed_minutes as f64 / 60.0).min(80.0);
 327     ((anchor.max(0.0) as f64 * exponent.exp()).min(MAX_CONTAMINATION as f64)) as f32
 328 }

 340 pub fn scale_contamination_contributions(
 341     raw_concentration: f32,
 342     surviving_concentration: f32,
 343     contribution_loads: &[f32],
 344 ) -> Vec<f32> {
 345     let survival = if raw_concentration > 0.0 {
 346         (surviving_concentration / raw_concentration).clamp(0.0, 1.0)
 347     } else {
 348         0.0
 349     };
 350     contribution_loads
 351         .iter()
 352         .map(|load| load * survival)
 353         .collect()
 354 }
