 204 #[expect(
 205     clippy::too_many_arguments,
 206     reason = "the fireplace row mirrors independent custody and installation form fields"
 207 )]
 208 pub(super) fn fireplace_inventory_row(
 209     action_base: &str,
 210     scope: &str,
 211     id: u64,
 212     item_id: &str,
 213     quantity: u32,
 214     measured_fraction_micros: Option<u32>,
 215     lot: Option<&FoodLot>,
 216     definitions: &[CatalogItemView],
 217     _installed: Option<&str>,
 218 ) -> Markup {
 219     let definition = definitions.iter().find(|d| d.id == item_id);
 220     let is_tool = matches!(item_id, "cooking_pan" | "cooking_pot" | "portable_oven");
 221     if !is_tool && !(lot.is_some() && adventuresim_core::food::is_cookable_ingredient(item_id)) {
 222         return html! {};
 223     }
 224     let display = lot.map_or_else(|| item_display_name(item_id), |l| l.display_name.clone());
 225     let measured_fraction = measured_fraction_micros.map(|value| {
 226         adventuresim_core::inventory_measurement::ConsumableFractionMicros::try_new(value)
 227             .expect("public consumable fraction must not exceed one whole")
 228     });
 229     let amount_micros = measured_fraction.map_or_else(
 230         || {
 231             quantity.saturating_mul(
 232                 adventuresim_core::inventory_measurement::ConsumableFractionMicros::MICROS_PER_WHOLE,
 233             )
 234         },
 235         adventuresim_core::inventory_measurement::ConsumableFractionMicros::get,
 236     );
 237     let display_units =
 238         measured_fraction.map_or(quantity as f32, |fraction| fraction.as_unit_f32());
 239     html! { tr class="trade-inventory-row trade-row-player" data-cooking-source=[lot.map(|_| id)] data-personal-inventory-id=[(scope == "personal").then_some(id)] data-party-inventory-id=[(scope == "party").then_some(id)] {
 240         td class="inventory-item-type" { (item_type_icon(item_id)) }
 241         td class="inventory-item-name" { (item_name_with_display(item_id, &display, definition))
 242             span class="inventory-row-actions" {
 243                 @if lot.is_some() && adventuresim_core::food::is_cookable_ingredient(item_id) {
 244                     button type="button" class="trade-transfer trade-transfer-left"
 245                         data-cooking-stage=(id) data-cooking-name=(&display) data-count=(amount_micros)
 246                         data-mass=(format!("{:.4}", lot.map_or(0.0, |l| l.mass_kg))) data-safety=(adventuresim_core::food::definition(item_id).map_or(5, |f| f.cooking_minutes))
 247                         data-culinary-fat=(adventuresim_core::food::definition(item_id).is_some_and(|f| f.culinary_fat))
 248                         data-salty=(lot.map_or(0.0, |l| l.salty_kg)) data-spicy=(lot.map_or(0.0, |l| l.spicy_kg))
 249                         data-sweet=(lot.map_or(0.0, |l| l.sweet_kg)) data-sour=(lot.map_or(0.0, |l| l.sour_kg)) data-savory=(lot.map_or(0.0, |l| l.savory_kg))
 250                         data-dynamic-transfer data-default-transfer-mode="one" data-transfer-mode="one"
 251                         data-label-one=(format!("Add 0.25 {display}")) data-label-target=(format!("Add {display}")) data-label-all=(format!("Add all {display}"))
 252                         aria-label=(format!("Add 0.25 {display}")) title=(format!("Add 0.25 {display}")) { (transfer_glyph(1)) }
 253                 } @else if is_tool {
 254                     form action=(format!("{action_base}/container/place")) method="post" {
 255                         input type="hidden" name="inventory_scope" value=(scope);
 256                         input type="hidden" name="inventory_item_id" value=(id);
 257                         button type="submit" class="btn btn-primary btn-small" { "Place over fire" }
 258                     }
 259                 }
 260             }
 261         }
 262         td class="inventory-count" { (format!("{display_units:.2}")) }
 263         td class="inventory-weight" { (definition.map_or(0.0, |d| d.weight)) }
 264     } }
 265 }
