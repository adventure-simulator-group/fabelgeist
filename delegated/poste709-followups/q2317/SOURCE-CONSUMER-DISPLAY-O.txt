 446 async fn post_fireplace_ingredients(
 447     state: AppState,
 448     actor: CharacterView,
 449     context: String,
 450     form: CookFoodForm,
 451     redirect: String,
 452 ) -> Response {
 453     let parse_ids = form
 454         .inventory_item_ids
 455         .split(',')
 456         .filter(|v| !v.is_empty())
 457         .map(str::parse)
 458         .collect::<Result<Vec<u64>, _>>();
 459     let parse_amounts = form
 460         .fractions_micros
 461         .split(',')
 462         .filter(|v| !v.is_empty())
 463         .map(str::parse)
 464         .collect::<Result<Vec<u32>, _>>();
 465     let (Ok(ids), Ok(amounts)) = (parse_ids, parse_amounts) else {
 466         return (StatusCode::BAD_REQUEST, "Invalid ingredient selection").into_response();
 467     };
 468     match state
 469         .db
 470         .call(
 471             "add_fireplace_ingredients",
 472             &[
 473                 json!(actor.id),
 474                 json!(context),
 475                 json!(form.inventory_scope),
 476                 json!(ids),
 477                 json!(amounts),
 478             ],
 479         )
 480         .await
 481     {
 482         Ok(()) => Redirect::to(&redirect).into_response(),
 483         Err(e) => (StatusCode::BAD_REQUEST, e.to_string()).into_response(),
 484     }
 485 }
