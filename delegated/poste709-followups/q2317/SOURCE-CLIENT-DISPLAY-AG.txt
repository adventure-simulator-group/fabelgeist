 166       const updatePotRow = (entry) => {
 167         if (entry.quantity === 0) {
 168           potBody?.querySelector(`[data-cooking-pot-id="${CSS.escape(entry.id)}"]`)?.remove();
 169           return;
 170         }
 171         const row = ensurePotRow(entry);
 172         if (!row) return;
 173         row.dataset.inventoryQuantity = String(entry.quantity);
 174         const count = row.querySelector(".inventory-count");
 175         if (count) {
 176           count.textContent = formatAmount(entry.quantity);
 177           count.dataset.base = String(entry.quantity);
 178           delete count.dataset.tradeDraftChange;
 179         }
 180         const weight = row.querySelector(".inventory-weight");
 181         if (weight) {
 182           const stagedMass = entry.mass * entry.quantity / entry.available;
 183           weight.textContent = formatNumber(stagedMass);
 184           weight.dataset.sortValue = String(stagedMass);
 185         }
 186         const value = row.querySelector(".inventory-gold");
 187         if (value) {
 188           const stagedValue = entry.value * entry.quantity / entry.available;
 189           value.textContent = formatNumber(stagedValue);
 190           value.dataset.sortValue = String(stagedValue);
 191         }
 192         const unstage = row.querySelector("[data-cooking-unstage]");
 193         if (unstage) unstage.dataset.count = String(entry.quantity);
 194       };
