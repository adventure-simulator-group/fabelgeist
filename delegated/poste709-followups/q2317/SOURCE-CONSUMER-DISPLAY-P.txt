 606 impl FireplaceContainerOperation {
 607     const fn reducer(self) -> &'static str {
 608         match self {
 609             Self::Place => "place_fireplace_container",
 610             Self::StartCooking => "start_fireplace_container_cooking",
 611             Self::Retrieve => "retrieve_fireplace_container",
 612         }
 613     }

 616 async fn post_fireplace_container(
 617     state: AppState,
 618     actor: CharacterView,
 619     context: String,
 620     operation: FireplaceContainerOperation,
 621     args: Vec<serde_json::Value>,
 622     redirect: String,
 623 ) -> Response {
 624     let mut reducer_args = vec![json!(actor.id), json!(context)];
 625     reducer_args.extend(args);
 626     match state.db.call(operation.reducer(), &reducer_args).await {
 627         Ok(()) => Redirect::to(&redirect).into_response(),
 628         Err(error) => (StatusCode::BAD_REQUEST, error.to_string()).into_response(),
 629     }
 630 }
