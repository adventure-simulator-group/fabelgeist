 371 pub fn cooking_duration_minutes(
 372     method: CookingMethod,
 373     safety_minutes: &[u32],
 374     total_mass_kg: f32,
 375 ) -> Option<u32> {
 376     if safety_minutes.is_empty()
 377         || !total_mass_kg.is_finite()
 378         || total_mass_kg <= 0.0
 379         || total_mass_kg > 25.0
 380     {
 381         return None;
 382     }
 383     let setup: u32 = match method {
 384         CookingMethod::PanFry => 5,
 385         CookingMethod::Stew => 12,
 386         CookingMethod::Roast => 7,
 387         CookingMethod::Bake => 30,
 388     };
 389     let slowest = safety_minutes.iter().copied().max()?;
 390     let batch = ((total_mass_kg - 0.5).max(0.0).sqrt() * 8.0).ceil() as u32;
 391     setup.checked_add(slowest)?.checked_add(batch)
 392 }

 394 /// Expertise shortens setup and batch handling, never the ingredient safety
 395 /// interval. Rank five removes at most thirty percent of overhead.
 396 pub fn cooking_duration_minutes_for_check(
 397     method: CookingMethod,
 398     safety_minutes: &[u32],
 399     total_mass_kg: f32,
 400     cooking_check: f32,
 401 ) -> Option<u32> {
 402     let baseline = cooking_duration_minutes(method, safety_minutes, total_mass_kg)?;
 403     let safety = safety_minutes.iter().copied().max()?;
 404     let overhead = baseline.saturating_sub(safety);
 405     let check = if cooking_check.is_finite() {
 406         cooking_check.clamp(0.0, 5.0)
 407     } else {
 408         0.0
 409     };
 410     safety.checked_add((overhead as f32 * (1.0 - 0.06 * check)).ceil() as u32)
