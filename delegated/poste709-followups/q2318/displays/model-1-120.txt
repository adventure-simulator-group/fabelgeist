   1 // Owns the cohesive food-lot, contamination, preparation-receipt, and fireplace schema.
   2 /// Public, inspectable description of one non-fungible inventory batch.
   3 /// Cooking/harvesting captures ingredient provenance and material totals;
   4 /// consumption and preparation owners conserve/update the remaining material.
   5 /// Current catalog recipes cannot reconstruct an already cooked batch.
   6 #[derive(Clone, Debug)]
   7 #[table(accessor = food_lot, public)]
   8 pub struct FoodLot {
   9     #[primary_key]
  10     #[auto_inc]
  11     pub id: u64,
  12     pub inventory_item_id: Option<u64>,
  13     pub party_inventory_item_id: Option<u64>,
  14     #[index(btree)]
  15     pub material_revision: u64,
  16     pub display_name: String,
  17     pub preparation: FoodPreparation,
  18     pub ingredient_item_ids: Vec<String>,
  19     /// Fractional source-unit provenance is conserved when a lot is partly eaten.
  20     pub ingredient_quantities: Vec<f32>,
  21     pub salty_kg: f32,
  22     pub spicy_kg: f32,
  23     pub sweet_kg: f32,
  24     pub sour_kg: f32,
  25     pub savory_kg: f32,
  26     /// Durable quality tier shared with item craftsmanship name colors.
  27     pub quality: u8,
  28     pub mass_kg: f32,
  29     pub nutrition_kcal: f32,
  30     pub total_value: f32,
  31     pub created_at_minute: StrategicMinute,
  32 }

  89 /// Private character-owned state for one exact physical fireplace context.
  90 /// The portrait is environmental/shared, but neither its tool nor dish leaks
  91 /// across player timelines.
  92 #[derive(Clone, Debug)]
  93 #[table(accessor = fireplace_station)]
  94 pub struct FireplaceStation {
  95     #[primary_key]
  96     pub key: String,
  97     #[index(btree)]
  98     pub character_id: u64,
  99     /// Canonical `StrategicFixtureId::Fireplace` encoding.
 100     pub fireplace_fixture_id: String,
 101     pub instrument_item_id: Option<String>,
 102     /// Stable root object for a placed cooking vessel. `None` is the loose
 103     /// spit-roast lane or an empty station.
 104     pub instrument_object_id: Option<u64>,
 105     /// Exact immutable carried custody to which removal returns the tool.
 106     pub instrument_return_custody: Option<crate::PersistedOperationalCustody>,
 107 }
