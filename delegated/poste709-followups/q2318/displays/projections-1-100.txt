  21 fn view_object_for_row(
  22     ctx: &ViewContext,
  23     scope: CarriedInventoryScope,
  24     row_id: u64,
  25 ) -> Option<crate::InventoryObject> {
  26     let mut matches = ctx
  27         .db
  28         .inventory_object()
  29         .item_id()
  30         .filter(""..)
  31         .filter(|object| match (&object.location, scope) {
  32             (InventoryLocation::Personal(location), CarriedInventoryScope::Personal) => {
  33                 location.row_id == row_id
  34             }
  35             (InventoryLocation::Party(location), CarriedInventoryScope::Party) => {
  36                 location.row_id == row_id
  37             }
  38             _ => false,
  39         });
  40     let object = matches.next()?;
  41     matches.next().is_none().then_some(object)
  42 }

  44 fn view_ancestry_reaches_fireplace(ctx: &ViewContext, object_id: u64) -> bool {
  45     let mut cursor = Some(object_id);
  46     for _ in 0..=adventuresim_core::inventory_containers::MAX_CONTAINER_DEPTH {
  47         let Some(id) = cursor else { return false };
  48         let Some(object) = ctx.db.inventory_object().id().find(id) else {
  49             return true;
  50         };
  51         if object.location.is_fireplace() {
  52             return true;
  53         }
  54         cursor = ctx
  55             .db
  56             .inventory_containment()
  57             .child_object_id()
  58             .find(id)
  59             .map(|edge| edge.parent_object_id);
  60     }
  61     true
  62 }
