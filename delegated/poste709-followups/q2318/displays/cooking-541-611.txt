 541                             inventory_item_id: id,
 542                             remaining_fraction_micros: remaining.get(),
 543                         },
 544                     );
 545                 }
 546                 CarriedInventoryScope::Party => {
 547                     ctx.db.party_item_amount().party_inventory_item_id().update(
 548                         crate::PartyItemAmount {
 549                             party_inventory_item_id: id,
 550                             remaining_fraction_micros: remaining.get(),

 552                     );
 553                 }
 554             };
 555         }
 556     }
 557     name_parts.sort();
 558     name_parts.dedup();
 559     let raw_contamination = food::microbial_concentration(loads.iter().sum(), mass);

 561     let ready_nutrition_retention =
 562         food::cooked_nutrition_retention(check) * food::method_nutrition_retention(method);
 563     ctx.db.fireplace_dish().insert(FireplaceDish {

 588         cooked_growth_per_hour: food::cooked_growth_per_hour(&growth, method),

 590             &contamination_contribution_ids,

 592         ),
 593         contamination_contribution_ids,
 594         contamination_contribution_loads,

 596         medicinal_profile_versions: vec![1; medicinal.len()],
 597         medicinal_potency_units: medicinal.values().copied().collect(),
 598     });
 599     if ctx
 600         .db
 601         .fireplace_station()
 602         .key()
 603         .find(station.key.clone())
 604         .is_none()
 605     {
 606         ctx.db.fireplace_station().insert(station);
 607     }
 608     Ok(())
 609 }
