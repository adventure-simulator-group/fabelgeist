  90     inventory_scope: String,

  93 ) -> Result<(), String> {

 105 /// Starts the independent dish lane belonging to one placed vessel. Every
 106 /// contained cookable food lot at any nesting depth is consumed in full;
 107 /// non-food solids and nested containers remain in place. Container water is used by the cooking evaluator and is
 108 /// mandatory for pots.
 109 #[reducer]
 110 pub fn start_fireplace_container_cooking(
 111     ctx: &ReducerContext,
 112     character_id: u64,
 113     fireplace_fixture_id: String,
 114     container_object_id: u64,
 115 ) -> Result<(), String> {
 116     crate::strategic::require_strategic_gateway(ctx)?;
 117     let actor = crate::character::require_living_character(ctx, character_id)?;
 118     validate_fireplace_fixture(ctx, &actor, &fireplace_fixture_id)?;
 119     let key = vessel_station_key(character_id, &fireplace_fixture_id, container_object_id);
 120     let station = ctx
 121         .db
 122         .fireplace_station()
 123         .key()
 124         .find(key.clone())
 125         .ok_or("Container is not over this fireplace")?;
 126     let fixture = validate_persisted_station_fixture(ctx, &station)?;
 127     let object = ctx
 128         .db
 129         .inventory_object()
 130         .id()
 131         .find(container_object_id)
 132         .ok_or("Container object is missing")?;
 133     crate::object_custody::require_object_at_fixture(ctx, &object, &fixture)?;
 134     if ctx.db.fireplace_dish().station_key().find(key).is_some() {
 135         return Err("This container is already cooking".into());
 136     }
 137     let return_custody = station
 138         .instrument_return_custody
 139         .as_ref()
 140         .ok_or("Container return custody is unknown")?;
 141     let destination = crate::object_custody::carried_destination(return_custody, character_id)?;
 142     let scope = match destination {
 143         OperationalCustody::Character(_) => CarriedInventoryScope::Personal,
 144         OperationalCustody::Party(_) => CarriedInventoryScope::Party,
 145         _ => return Err("Container return custody is not carried inventory".into()),
 146     };
 147     let mut ids = Vec::new();
 148     let mut amounts = Vec::new();
 149     let mut consumed_objects = Vec::new();
 150     // Cooking intentionally sees only direct contents. A hidden nested lot is
 151     // not a selected ingredient and remains inside its child container.
 152     for edge in ctx
 153         .db
 154         .inventory_containment()
 155         .parent_object_id()
 156         .filter(container_object_id)
 157     {
 158         let object_id = edge.child_object_id;
 159         let child = ctx
 160             .db
 161             .inventory_object()
 162             .id()
 163             .find(object_id)
 164             .ok_or("Contained object is missing")?;
 165         if !food::is_cookable_ingredient(&child.item_id) {
 166             continue;
 167         }
 168         let (row_id, lot, amount) = match (&child.location, scope) {
 169             (InventoryLocation::Personal(location), CarriedInventoryScope::Personal) => (
 170                 location.row_id,
 171                 personal_lot(ctx, location.row_id),
 172                 crate::inventory_amount::personal_fraction(ctx, location.row_id),
 173             ),
 174             (InventoryLocation::Party(location), CarriedInventoryScope::Party) => (
 175                 location.row_id,
 176                 party_lot(ctx, location.row_id),
 177                 crate::inventory_amount::party_fraction(ctx, location.row_id),
 178             ),
 179             _ => return Err("Contained food custody conflicts with its return inventory".into()),
 180         };
 181         let (Some(lot), Some(amount)) = (lot, amount) else {
 182             continue;
 183         };
 184         if !matches!(
 185             lot.preparation,
 186             FoodPreparation::Raw
 187                 | FoodPreparation::Cut
 188                 | FoodPreparation::Ground
 189                 | FoodPreparation::Preserved
