   1 //! Lookup of the stable physical object behind a carried inventory row.
   2 
   3 use adventuresim_core::physical_object::{CarriedInventoryScope, InventoryLocation};
   4 use spacetimedb::{ReducerContext, Table};
   5 
   6 use super::{InventoryObject, inventory_object};
   7 use crate::inventory_item;
   8 use crate::strategic::party_inventory_item;
   9 
  10 pub(crate) fn object_for_row(
  11     ctx: &ReducerContext,
  12     scope: CarriedInventoryScope,
  13     row_id: u64,
  14 ) -> Result<Option<InventoryObject>, String> {
  15     let is_row_object = |object: &InventoryObject| match (&object.location, scope) {
  16         (InventoryLocation::Personal(location), CarriedInventoryScope::Personal) => {
  17             location.row_id == row_id
  18         }
  19         (InventoryLocation::Party(location), CarriedInventoryScope::Party) => {
  20             location.row_id == row_id
  21         }
  22         _ => false,
  23     };
  24     let row_item_id = match scope {
  25         CarriedInventoryScope::Personal => ctx
  26             .db
  27             .inventory_item()
  28             .id()
  29             .find(row_id)
  30             .map(|row| row.item_id),
  31         CarriedInventoryScope::Party => ctx
  32             .db
  33             .party_inventory_item()
  34             .id()
  35             .find(row_id)
  36             .map(|row| row.item_id),
  37     };
  38     // A carried object always shares its backing row's item, so the item index
  39     // bounds the search. Only a row already consumed before reconciliation
  40     // requires scanning every object.
  41     let mut matches: Box<dyn Iterator<Item = InventoryObject>> = match &row_item_id {
  42         Some(item_id) => Box::new(
  43             ctx.db
  44                 .inventory_object()
  45                 .item_id()
  46                 .filter(item_id)
  47                 .filter(is_row_object),
  48         ),
  49         None => Box::new(ctx.db.inventory_object().iter().filter(is_row_object)),
  50     };
  51     let first = matches.next();
  52     if matches.next().is_some() {
  53         return Err("Inventory row has duplicate physical object identities".into());
  54     }
  55     Ok(first)
  56 }
