   435	            return Err("Ingredient preparation prerequisites changed before commit".into());
   436	        }
   437	    };

   446	            authority_binding: provenance.authority_binding,
   447	        },

   454	    )

   458	    let mut effect_commit = None;
   459	    for effect in planned.effects() {

   464	                    requested_minutes,
   465	                },

   467	            adventuresim_core::strategic_action::ActionEffect::Domain(
   468	                herbalism::PreparationPlanEffect::CommitPreparation {
   469	                    action,
   470	                    expected_revision,
   471	                    next_display_name,
   472	                },
   473	            ) => effect_commit = Some((*action, *expected_revision, next_display_name.clone())),
   474	            _ => return Err("Ingredient preparation planner emitted an unsupported effect".into()),
   475	        }
   476	    }

   478	        effect_duration.ok_or("Ingredient preparation planner omitted its wait effect")?;
