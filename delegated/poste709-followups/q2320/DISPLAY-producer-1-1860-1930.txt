  1860	    let terminal =
  1861	        first_private_terminal(ctx, character_id, &episodes, now, interval_end, immunity)?;

  1873	    )?;

  1886	    for event in events.iter().filter(|event| event.minute <= through) {
  1887	        match event.kind {
  1888	            DiseaseEventKind::SymptomOnset => notice(

  1898	                ctx,

  1913	        }
  1914	    }
  1915	    let Some(death_minute) = death_minute else {

  1918	    notice(
