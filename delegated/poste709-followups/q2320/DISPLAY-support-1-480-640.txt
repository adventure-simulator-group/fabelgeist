   589	#[derive(Clone, Debug, Eq, PartialEq)]
   590	pub enum PlanningOutcome<
   591	    T: DomainTarget,
   592	    R: DomainRequirement,
   593	    C: DomainCapability,
   594	    I: DomainInterruption,
   595	    E: DomainEffect,
   596	    P: PublicPreview,
   597	> {
   598	    Ready(Box<AuthoritativeActionPlan<T, R, C, I, E, P>>),
   599	    Rejected(PrivateRejection<R, C>),
   600	}
   601	
   602	pub fn build_plan<T, R, C, I, E, P>(
   603	    input: PlanInput<T, R, C, I>,
   604	    calculate: impl FnOnce(&ActionCoordinates<T>, &TimeResolution<I>) -> CalculatedAction<E, P>,
   605	) -> PlanningOutcome<T, R, C, I, E, P>
   606	where
   607	    T: DomainTarget,
   608	    R: DomainRequirement,
   609	    C: DomainCapability,
   610	    I: DomainInterruption,
   611	    E: DomainEffect,
   612	    P: PublicPreview,
   613	{
   614	    let failed = input
   615	        .requirements
   616	        .iter()
   617	        .filter(|check| !check.satisfied)
   618	        .map(|check| check.requirement.clone())
   619	        .collect::<Vec<_>>();
   620	    if !failed.is_empty() {
   621	        return PlanningOutcome::Rejected(PrivateRejection {
   622	            failed,
   623	            public: input.sanitized_rejection,
   624	        });
   625	    }
   626	    let time = resolve_time(input.current_minute, input.duration, &input.boundaries);
   627	    let calculation = calculate(&input.coordinates, &time);
   628	    PlanningOutcome::Ready(Box::new(AuthoritativeActionPlan {
   629	        coordinates: input.coordinates,
   630	        provenance: input.provenance,
   631	        snapshot: input.snapshot,
   632	        requirements: input.requirements,
   633	        time,
   634	        calculation,
   635	    }))
   636	}
