   393	/// Side-effect-free preview of the first injury terminal boundary in an
   394	/// interval. Recovery is explicit because only restorative minutes offset
   395	/// blood loss and heal wounds.
   396	pub(crate) fn preview_injury_boundary(
   397	    ctx: &ReducerContext,
   398	    character_id: u64,
   399	    requested: u64,
   400	    recovery: InjuryRecoveryMinutes,
   401	) -> Result<InjuryPreview, String> {
   402	    let blood = ctx
   403	        .db
   404	        .character_condition()
   405	        .character_id()
   406	        .find(character_id)
   407	        .map_or(1.0, |row| {
   408	            if row.maximum_blood_ml > 0.0 {
   409	                row.current_blood_ml / row.maximum_blood_ml
   410	            } else {
   411	                1.0
   412	            }
   413	        });
   414	    let cuts = BodyRegion::ALL
   415	        .into_iter()
   416	        .map(|limb| {
   417	            let injury = injury_for(ctx, character_id, limb);
   418	            if injury.bandaged {
   419	                0.0
   420	            } else {
   421	                injury.cut_damage
   422	            }
   423	        })
   424	        .collect::<Vec<_>>();
   425	    let interval = simulate_blood_interval(
   426	        blood,
   427	        &cuts,
   428	        requested,
   429	        adventuresim_core::morale::BLOOD_RECOVERY_FRACTION_PER_DAY
   430	            * recovery.get().min(requested) as f32
   431	            / requested.max(1) as f32,
   432	    );
   433	    Ok(InjuryPreview {
   434	        elapsed: interval.elapsed,
   435	        terminal: interval.terminal,
   436	    })
   437	}
