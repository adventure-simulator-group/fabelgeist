    20	/// Exact mass measured in whole milligrams.
    21	#[derive(Clone, Copy, Debug, Default, Eq, Hash, Ord, PartialEq, PartialOrd)]
    22	pub struct Milligrams(u64);
    23	
    24	impl Milligrams {
    25	    const MILLIGRAMS_PER_KILOGRAM: f64 = 1_000_000.0;
    26	
    27	    pub fn try_from_kilograms(kilograms: f64) -> Result<Self, MaterialError> {
    28	        if !kilograms.is_finite() || kilograms < 0.0 {
    29	            return Err(MaterialError::InvalidMeasure);
    30	        }
    31	        let milligrams = kilograms * Self::MILLIGRAMS_PER_KILOGRAM;
    32	        if milligrams > u64::MAX as f64 {
    33	            return Err(MaterialError::Overflow);
    34	        }
    35	        Ok(Self(milligrams.round() as u64))
    36	    }
    37	
    38	    pub const fn get(self) -> u64 {
    39	        self.0
    40	    }
    41	
    42	    pub fn as_kilograms(self) -> f64 {
    43	        self.0 as f64 / Self::MILLIGRAMS_PER_KILOGRAM
    44	    }
    45	}

    47	/// A positive material-component quantity in millionths of an authored unit.
    48	#[derive(Clone, Copy, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
    49	pub struct MaterialComponentMicrounits(NonZeroU64);
    50	
    51	impl MaterialComponentMicrounits {
    52	    const MICROUNITS_PER_UNIT: f64 = 1_000_000.0;
    53	
    54	    pub fn try_from_units(units: f32) -> Result<Option<Self>, MaterialError> {
    55	        if !units.is_finite() || units < 0.0 {
    56	            return Err(MaterialError::InvalidMeasure);
    57	        }
    58	        let microunits = f64::from(units) * Self::MICROUNITS_PER_UNIT;
    59	        if microunits > u64::MAX as f64 {
    60	            return Err(MaterialError::Overflow);
    61	        }
    62	        Ok(NonZeroU64::new(microunits.round() as u64).map(Self))
    63	    }
    64	
    65	    pub const fn get(self) -> NonZeroU64 {
    66	        self.0
    67	    }
    68	
    69	    pub fn as_units(self) -> f64 {
    70	        self.0.get() as f64 / Self::MICROUNITS_PER_UNIT
    71	    }
    72	}
