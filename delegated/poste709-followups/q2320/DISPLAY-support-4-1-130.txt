    34	/// A normalized position within the shared strategic day.
    35	#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
    36	pub struct StrategicMinuteOfDay(u16);
    37	
    38	impl StrategicMinuteOfDay {
    39	    /// Construct a validated position within one strategic day.
    40	    pub const fn new(minute_of_day: u16) -> Option<Self> {
    41	        if (minute_of_day as u64) < MINUTES_PER_DAY {
    42	            Some(Self(minute_of_day))
    43	        } else {
    44	            None
    45	        }
    46	    }
    47	
    48	    /// Construct a validated position from a clock hour and minute.
    49	    pub const fn from_hour_minute(hour: u16, minute: u16) -> Option<Self> {
    50	        if hour < HOURS_PER_DAY && minute < MINUTES_PER_HOUR {
    51	            Some(Self(hour * MINUTES_PER_HOUR + minute))
    52	        } else {
    53	            None
    54	        }
    55	    }
    56	
    57	    /// Normalize an absolute strategic minute to its position within the day.
    58	    pub const fn from_absolute(absolute_minute: StrategicMinute) -> Self {
    59	        Self(absolute_minute.minute_of_day())
    60	    }
    61	
    62	    pub const fn get(self) -> u16 {
    63	        self.0
    64	    }
    65	
    66	    /// Whether this minute lies outside the shared daylight window.
    67	    pub const fn is_night(self) -> bool {
    68	        self.0 < DAYLIGHT_START_MINUTE || self.0 >= DAYLIGHT_END_MINUTE
    69	    }
    70	
    71	    /// Minutes until night begins, or zero when it is already night.
    72	    pub const fn minutes_until_night(self) -> u16 {
    73	        if self.is_night() {
    74	            0
    75	        } else {
    76	            DAYLIGHT_END_MINUTE - self.0
    77	        }
    78	    }
    79	}
