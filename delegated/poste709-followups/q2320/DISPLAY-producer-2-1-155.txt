     3	fn current_minute(ctx: &ReducerContext, character_id: u64) -> StrategicMinute {
     4	    ctx.db
     5	        .character_time()
     6	        .character_id()
     7	        .find(character_id)
     8	        .map_or(StrategicMinute::ZERO, |t| t.minutes)
     9	}

    11	fn ensure_food_material_object(
    12	    ctx: &ReducerContext,
    13	    scope: CarriedInventoryScope,
    14	    row_id: u64,
    15	) -> Result<crate::InventoryObject, String> {
    16	    let (item_id, location, quantity) = match scope {
    17	        CarriedInventoryScope::Personal => {
    18	            let row = ctx
    19	                .db
    20	                .inventory_item()
    21	                .id()
    22	                .find(row_id)
    23	                .ok_or("Food inventory row is missing")?;
    24	            (
    25	                row.item_id,
    26	                InventoryLocation::personal(row.character_id, row.id),
    27	                row.quantity,
    28	            )
    29	        }
    30	        CarriedInventoryScope::Party => {
    31	            let row = ctx
    32	                .db
    33	                .party_inventory_item()
    34	                .id()
    35	                .find(row_id)
    36	                .ok_or("Party food inventory row is missing")?;
    37	            (
    38	                row.item_id,
    39	                InventoryLocation::party(row.party_id, row.id),
    40	                row.quantity,
    41	            )
    42	        }
    43	    };
    44	    if quantity != 1 {
    45	        return Err("Every food lot requires a quantity-one stable inventory object".into());
    46	    }
    47	    if let Some(object) = crate::inventory_container::object_for_row(ctx, scope, row_id)? {
    48	        if object.item_id != item_id || object.location != location {
    49	            return Err("Food inventory row has a mismatched stable object identity".into());
    50	        }
    51	        return Ok(object);
    52	    }
    53	    Ok(ctx.db.inventory_object().insert(crate::InventoryObject {
    54	        id: 0,
    55	        item_id,
    56	        location,
    57	    }))
    58	}

    60	pub fn create_personal_food_lot(
    61	    ctx: &ReducerContext,
    62	    character_id: u64,
    63	    inventory_item_id: u64,
    64	    item_id: &str,
    65	    quantity: u32,
    66	) -> Result<FoodLot, String> {
    67	    let definition = food::definition(item_id).ok_or("Food definition not found")?;
    68	    let minute = current_minute(ctx, character_id);
    69	    ensure_food_material_object(ctx, CarriedInventoryScope::Personal, inventory_item_id)?;
    70	    let lot = ctx.db.food_lot().insert(FoodLot {
    71	        id: 0,
    72	        inventory_item_id: Some(inventory_item_id),
    73	        party_inventory_item_id: None,
    74	        material_revision: 1,
    75	        display_name: definition.name.into(),
    76	        preparation: if definition.class == food::FoodClass::Ration {
    77	            FoodPreparation::Preserved
    78	        } else {
    79	            FoodPreparation::Raw
    80	        },
    81	        ingredient_item_ids: vec![item_id.into()],
    82	        ingredient_quantities: vec![quantity as f32],
    83	        salty_kg: definition.flavors_per_unit.salty * quantity as f32,
    84	        spicy_kg: definition.flavors_per_unit.spicy * quantity as f32,
    85	        sweet_kg: definition.flavors_per_unit.sweet * quantity as f32,
    86	        sour_kg: definition.flavors_per_unit.sour * quantity as f32,
    87	        savory_kg: definition.flavors_per_unit.savory * quantity as f32,
    88	        quality: definition.default_quality.clamp(1, 5),
    89	        mass_kg: definition.mass_kg_per_unit * quantity as f32,
    90	        nutrition_kcal: definition.kcal_per_unit * quantity as f32,
    91	        total_value: definition.value_per_unit * quantity as f32,
    92	        created_at_minute: minute,
    93	    });
    94	    ctx.db.food_contamination().insert(FoodContamination {
    95	        food_lot_id: lot.id,
    96	        concentration_anchor: food::deterministic_initial_contamination(
    97	            fabelgeist_determinism::StreamId::new("food.lot-consumption").rng(ctx.random::<fabelgeist_determinism::Seed>(), &[lot.id, character_id]).next_seed(),
    98	        ),
    99	        growth_per_hour: definition.growth_per_hour,
   100	        anchor_minute: minute,
   101	    });
   102	    Ok(lot)
   103	}
