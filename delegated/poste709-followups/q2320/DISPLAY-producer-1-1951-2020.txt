  1951	/// Side-effect-free party preflight. Acquisition and notice delivery happen
  1952	/// only in the subsequent committed interval.
  1953	fn preview_disease_boundary_planned(
  1954	    ctx: &ReducerContext,
  1955	    character_id: u64,
  1956	    requested: u64,
  1957	    allow_healing: bool,
  1958	    plan: Option<&PartyDiseaseIntervalPlan>,
  1959	) -> Result<(u64, bool), String> {
  1960	    let owned_plan;
  1961	    let plan = if let Some(plan) = plan {
  1962	        plan
  1963	    } else {
  1964	        owned_plan = plan_party_disease_interval(ctx, &[character_id], requested, allow_healing)?;
  1965	        &owned_plan
  1966	    };
  1967	    let now = ctx
  1968	        .db
  1969	        .character_time()
  1970	        .character_id()
  1971	        .find(character_id)
  1972	        .map_or(StrategicMinute::ZERO, |t| t.minutes);
  1973	    let immunity = ctx
  1974	        .db
  1975	        .character_attributes()
  1976	        .character_id()
  1977	        .find(character_id)
  1978	        .map_or(3.0, |a| a.immunity);
  1979	    let mut episodes = character_episodes(ctx, character_id)?;
  1980	    episodes.extend(plan.proposals_for(character_id, now, now.saturating_add_minutes(requested)));
  1981	    let terminal = first_private_terminal(
  1982	        ctx,
  1983	        character_id,
  1984	        &episodes,
  1985	        now,
  1986	        now.saturating_add_minutes(requested),
  1987	        immunity,
  1988	    )?;
  1989	    Ok((
  1990	        terminal.map_or(requested, |(minute, _)| minute.elapsed_since(now)),
  1991	        terminal.is_some(),
  1992	    ))
  1993	}
