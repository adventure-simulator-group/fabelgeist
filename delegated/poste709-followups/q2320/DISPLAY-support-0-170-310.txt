   170	    } else {
   171	        PrivateRightsDecision::denied(Vec::new(), provenance)
   172	    }
   173	}

   178	    digest.frame_resource(question.resource(), |_, resource| match *resource {});

   183	    );

   216	pub fn build_preparation_plan(authority: PreparationPlanAuthority) -> PreparationPlanningOutcome {
   217	    let requirements = [
   218	        (
   219	            PreparationPlanRequirement::StableObjectCustody,
   220	            authority.custody_matches,
   221	        ),
   222	        (
   223	            PreparationPlanRequirement::LotRevisionCurrent,
   224	            authority.revision_current,
   225	        ),
   226	        (
   227	            PreparationPlanRequirement::PreparationTransition,
   228	            authority.transition_allowed,
   229	        ),
   230	        (
   231	            PreparationPlanRequirement::RequiredTool,
   232	            authority.required_tool_available,
   233	        ),
   234	        (
   235	            PreparationPlanRequirement::RightsAllowed,
   236	            authority.rights.kind() == RightsDecisionKind::Allowed,
   237	        ),
   238	    ]
   239	    .into_iter()
   240	    .map(|(requirement, satisfied)| RequirementCheck {
   241	        requirement: ActionRequirement::Domain(requirement),
   242	        satisfied,
   243	    })
   244	    .collect();
   245	    let actor = authority.coordinates.actor();
   246	    let action = authority.action;
   247	    let revision = authority.expected_revision;
   248	    let duration = authority.duration.minutes();
   249	    let next_display_name = authority.next_display_name;
   250	    build_plan(
   251	        PlanInput {
   252	            coordinates: authority.coordinates,
   253	            provenance: authority.provenance,
   254	            snapshot: authority.snapshot,
   255	            current_minute: authority.current_minute,
   256	            duration: authority.duration,
   257	            boundaries: TimeBoundaries {
   258	                terminal_minute: authority.terminal_minute,
   259	                interruption: None,
   260	            },
   261	            requirements,
   262	            sanitized_rejection: PublicRejection::Unavailable,
   263	        },
   264	        move |_, time| {
   265	            let mut effects = vec![ActionEffect::Domain(PreparationPlanEffect::AttemptWait {
   266	                actor,
   267	                requested_minutes: duration,
   268	            })];
   269	            if time.permits_completion_effects() {
   270	                effects.push(ActionEffect::Domain(
   271	                    PreparationPlanEffect::CommitPreparation {
   272	                        action,
   273	                        expected_revision: revision,
   274	                        next_display_name,
   275	                    },
   276	                ));
   277	            }
   278	            CalculatedAction {
   279	                effects,
   280	                public_preview: PreparationPublicPreview {
   281	                    action,
   282	                    expected_revision: revision,
   283	                    duration_minutes: duration.min(u64::from(u32::MAX)) as u32,
   284	                },
   285	            }
   286	        },
   287	    )
   288	}

   290	pub fn physical_preparation_minutes(
   291	    preparation: IngredientPreparationAction,
   292	    governing_check: f32,
   293	    has_grinding_tool: bool,
   294	) -> u32 {
   295	    let base = match preparation {
   296	        IngredientPreparationAction::Cut => BASE_CUT_MINUTES,
   297	        IngredientPreparationAction::Grind => BASE_GRIND_MINUTES,
   298	    };
   299	    let check = if governing_check.is_finite() {
   300	        governing_check.clamp(0.0, 5.0)
   301	    } else {
   302	        0.0
   303	    };
   304	    let tool = if preparation == IngredientPreparationAction::Grind && has_grinding_tool {
   305	        GRINDING_TOOL_TIME_FACTOR
   306	    } else {
   307	        1.0
   308	    };
   309	    ((base as f32) * (1.0 - CHECK_TIME_REDUCTION_PER_RANK * check) * tool)
   310	        .ceil()
