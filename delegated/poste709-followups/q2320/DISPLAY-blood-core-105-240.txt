   105	    let starting_cut = starting_cut.clamp(0.0, 1.0);
   106	    let days = days.max(0.0);
   107	    let days_to_cap = (1.0 - starting_cut) / UNTREATED_CUT_DETERIORATION_PER_DAY;
   108	    let growing_days = days.min(days_to_cap.max(0.0));
   109	    let cut_days = starting_cut * growing_days
   110	        + 0.5 * UNTREATED_CUT_DETERIORATION_PER_DAY * growing_days * growing_days
   111	        + (days - growing_days).max(0.0);
   112	    (
   113	        (starting_cut + UNTREATED_CUT_DETERIORATION_PER_DAY * days).min(1.0),
   114	        cut_days,
   115	    )
   116	}

   118	#[derive(Clone, Debug)]
   119	pub struct BloodInterval {
   120	    pub elapsed: u64,
   121	    pub blood_fraction: f32,
   122	    pub open_cuts: Vec<f32>,
   123	    pub cut_days: Vec<f32>,
   124	    pub terminal: bool,
   125	}
   126	
   127	pub fn simulate_blood_interval(
   128	    starting_blood: f32,
   129	    open_cuts: &[f32],
   130	    requested: u64,
   131	    recovery_per_day: f32,
   132	) -> BloodInterval {
   133	    let mut blood = starting_blood.clamp(0.0, 1.0);
   134	    let mut cuts = open_cuts.to_vec();
   135	    let mut cut_days = vec![0.0; cuts.len()];
   136	    let minute_days = 1.0 / MINUTES_PER_DAY as f32;
   137	    let mut elapsed = 0;
   138	    for _ in 0..requested {
   139	        let mut loss = 0.0;
   140	        for (index, cut) in cuts.iter_mut().enumerate() {
   141	            if *cut <= 0.0 {
   142	                continue;
   143	            }
   144	            let (next, exposure) = untreated_cut_progress(*cut, minute_days);
   145	            *cut = next;
   146	            cut_days[index] += exposure;
   147	            loss += UNTREATED_CUT_BLOOD_LOSS_PER_DAY * exposure;
   148	        }
   149	        blood = (blood + recovery_per_day.max(0.0) * minute_days - loss).clamp(0.0, 1.0);
   150	        elapsed += 1;
   151	        if blood <= 0.10 {
   152	            return BloodInterval {
   153	                elapsed,
   154	                blood_fraction: blood,
   155	                open_cuts: cuts,
   156	                cut_days,
   157	                terminal: true,
   158	            };
   159	        }
   160	    }
   161	    BloodInterval {
   162	        elapsed,
   163	        blood_fraction: blood,
   164	        open_cuts: cuts,
   165	        cut_days,
   166	        terminal: false,
   167	    }
   168	}
