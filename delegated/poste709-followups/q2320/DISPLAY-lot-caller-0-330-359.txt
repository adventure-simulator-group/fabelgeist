   337	    definition: Option<&Item>,

   341	    food || measured
   342	        || definition.is_some_and(|definition| {

   344	                || definition.kind == CatalogItemKind::Medication
   345	                || (definition.kind == CatalogItemKind::Weapon && definition.melee)
   346	                || definition.container_capacity_ml > 0
   347	                || !definition.attachment_points.is_empty()
   348	        })
   349	}
