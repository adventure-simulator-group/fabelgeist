   358	#[derive(Clone, Debug, Eq, PartialEq)]

   360	    pub start_minute: StrategicMinute,
   361	    pub requested_minutes: u64,

   363	    pub end_minute: StrategicMinute,

   365	}

   367	impl<I: DomainInterruption> TimeResolution<I> {
   368	    /// Only this outcome permits a domain to emit completion-only effects.
   369	    pub const fn permits_completion_effects(&self) -> bool {
   370	        matches!(self.outcome, TimeOutcome::Completed)
   371	    }

   374	pub fn resolve_time<I: DomainInterruption>(
   375	    current_minute: StrategicMinute,
   376	    duration: RequestedDuration,
   377	    boundaries: &TimeBoundaries<I>,
   378	) -> TimeResolution<I> {
   379	    let requested_end = current_minute.checked_add_minutes(duration.minutes());
   380	    let latest_representable_end = requested_end.unwrap_or(StrategicMinute::MAX);
   381	    let terminal = boundaries
   382	        .terminal_minute
   383	        .map(|minute| minute.max(current_minute));
   384	    let interruption = boundaries
   385	        .interruption
   386	        .as_ref()
   387	        .map(|value| (value.at_minute.max(current_minute), value.cause.clone()));
   388	    let (end_minute, outcome) = match (terminal, interruption) {
   389	        (Some(terminal), Some((at, cause))) if at < terminal && at <= latest_representable_end => {
   390	            (at, TimeOutcome::Interrupted(cause))
   391	        }
   392	        (Some(terminal), _) if terminal <= latest_representable_end => {
   393	            (terminal, TimeOutcome::TerminalBoundary)
   394	        }
   395	        (None, Some((at, cause))) if at <= latest_representable_end => {
   396	            (at, TimeOutcome::Interrupted(cause))
   397	        }
   398	        _ => match requested_end {
   399	            Some(end) if end == StrategicMinute::MAX => (end, TimeOutcome::ClockExhausted),
   400	            Some(end) => (end, TimeOutcome::Completed),
   401	            None => (StrategicMinute::MAX, TimeOutcome::ClockExhausted),
   402	        },
   403	    };
   404	    TimeResolution {
   405	        start_minute: current_minute,
   406	        requested_minutes: duration.minutes(),
   407	        elapsed_minutes: end_minute.elapsed_since(current_minute),
   408	        end_minute,
   409	        outcome,
   410	    }
   411	}
