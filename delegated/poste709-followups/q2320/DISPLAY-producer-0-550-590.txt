   550	    refresh_all_limb_projections(ctx, character_id);
   551	    // The strategic time owner commits the elapsed frontier exactly once.
   552	    // Surgery settles injury state only; writing CharacterTime here used to
   553	    // double-advance terminal intervals when the caller committed the same
   554	    // elapsed span.
   555	    crate::condition::set_blood_fraction(ctx, character_id, interval.blood_fraction)?;
   556	    let alive = ctx
   557	        .db
   558	        .character()
   559	        .id()
   560	        .find(character_id)
   561	        .is_some_and(|row| row.alive);
   562	    Ok(InjurySettlement {
   563	        elapsed: interval.elapsed,
   564	        alive,
   565	    })
   566	}
