  13 /// One realization of a wearer's body, on the host.
  14 #[derive(Clone, Copy, Debug)]
  15 pub struct BodySurface<'a> {
  16     pub positions: &'a [[f32; 3]],
  17     pub normals: &'a [[f32; 3]],
  18     pub faces: &'a [[u32; 3]],
  19     /// Each vertex's anatomical surface coordinate.
  20     pub texcoords: &'a [[f32; 2]],
  21     pub joint_indices: &'a fabelgeist_rig::PackedRigSkin<8>,
  22     pub joint_weights: &'a [[f32; 8]],
  23     /// Global joint states: position, then rotation as a quaternion, then
  24     /// scale.
  25     pub joints: &'a fabelgeist_rig::RigJointTable<[f32; 8]>,
  26 }

  28 /// A wearer's body uploaded for fitting.
  29 #[derive(Clone, Debug)]
  30 pub struct GpuBody {
  31     pub positions: Buffer,
  32     pub normals: Buffer,
  33     pub faces: Buffer,
  34     pub texcoords: Buffer,
  35     pub joint_indices: Buffer,
  36     pub joint_weights: Buffer,
  37     pub joints: Buffer,
  38     pub vertex_count: u32,
  39     pub face_count: u32,
  40     pub joint_count: fabelgeist_rig::RigJointCount,
  41 }
  42 
  43 impl GpuBody {
  44     pub fn new(gpu: &ArmorGpu, surface: BodySurface) -> Result<Self, GenerateError> {
  45         let vertices = surface.positions.len();
  46         if vertices == 0
  47             || surface.normals.len() != vertices
  48             || surface.texcoords.len() != vertices
  49             || surface.joint_indices.len() != vertices
  50             || surface.joint_weights.len() != vertices
  51             || surface.joint_indices.joint_count() != surface.joints.joint_count()
  52             || surface
  53                 .faces
  54                 .iter()
  55                 .flatten()
  56                 .any(|i| *i as usize >= vertices)
  57         {
  58             return Err(GenerateError::InvalidSurface);
  59         }
  60         Ok(Self {
  61             positions: gpu.upload(BufferUpload::from_elements(surface.positions))?,
  62             normals: gpu.upload(BufferUpload::from_elements(surface.normals))?,
  63             faces: gpu.upload(BufferUpload::from_elements(surface.faces))?,
  64             texcoords: gpu.upload(BufferUpload::from_elements(surface.texcoords))?,
  65             joint_indices: gpu.upload(BufferUpload::from_elements(
  66                 surface.joint_indices.native_words(),
  67             ))?,
  68             joint_weights: gpu.upload(BufferUpload::from_elements(surface.joint_weights))?,
  69             joints: gpu.upload(BufferUpload::from_elements(
  70                 surface.joints.as_native_slice(),
  71             ))?,
  72             vertex_count: u32::try_from(vertices).map_err(|_| GenerateError::InvalidSurface)?,
  73             face_count: u32::try_from(surface.faces.len())
  74                 .map_err(|_| GenerateError::InvalidSurface)?,
  75             joint_count: surface.joints.joint_count(),
  76         })
  77     }
  78 }
