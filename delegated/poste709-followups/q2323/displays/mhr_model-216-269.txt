 216             None
 217         };
 218 
 219         Self::from_loaded_assets(character, definition, correctives, device)
 220     }

 222     fn from_loaded_assets(
 223         character: Character,
 224         definition: &str,
 225         correctives: Option<PoseCorrectives>,
 226         device: &Device,
 227     ) -> Result<Self> {
 228         let mut parameter_transform = parse_model_definition(definition, &character.skeleton)
 229             .context("parsing the MHR model definition")?;
 230         let num_model_parameters = parameter_transform.num_parameters();
 231         // momentum appends one model parameter per identity blend shape when a
 232         // blend shape is attached to the character.
 233         append_blend_shape_parameters(
 234             &mut parameter_transform,
 235             NpyDimension::from(NUM_IDENTITY_BLEND_SHAPES),
 236         )?;
 237         Self::new(
 238             character,
 239             parameter_transform,
 240             num_model_parameters,
 241             correctives,
 242             device,
 243         )
 244     }
