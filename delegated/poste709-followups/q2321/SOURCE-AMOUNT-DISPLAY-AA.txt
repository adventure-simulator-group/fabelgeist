  10 pub const SMITHING_MATERIAL_IDS: [&str; 4] =
  11     ["steel_stock", "leather_stock", "brass_stock", "wood_stock"];
  12 
  13 /// Remaining material in one stable personal inventory row. Row quantity stays
  14 /// one; consumption owns this fraction and transfer moves it between holders.
  15 #[derive(Clone, Debug)]
  16 #[table(accessor = inventory_item_amount, public)]
  17 pub struct InventoryItemAmount {
  18     #[primary_key]
  19     pub inventory_item_id: u64,
  20     pub remaining_fraction_micros: u32,
  21 }
  22 
  23 /// Party-held counterpart of the same remaining material authority. Transfer
  24 /// deletes the source amount before inserting the destination in one transaction.
  25 #[derive(Clone, Debug)]
  26 #[table(accessor = party_item_amount, public)]
  27 pub struct PartyItemAmount {
  28     #[primary_key]
  29     pub party_inventory_item_id: u64,
  30     pub remaining_fraction_micros: u32,
  31 }

  33 pub fn is_measured_definition(definition: &crate::Item) -> bool {
  34     definition.kind == crate::CatalogItemKind::Food
  35         || definition.alcohol_serving_ml > 0
  36         || definition.id == adventuresim_core::item_references::SOFT_SOAP_ID
  37         || SMITHING_MATERIAL_IDS.contains(&definition.id.as_str())
  38 }
  39 
  40 pub fn is_measured_item(ctx: &ReducerContext, item_id: &str) -> bool {
  41     ctx.db
  42         .item()
  43         .id()
  44         .find(item_id.to_owned())
  45         .is_some_and(|definition| is_measured_definition(&definition))
  46         || adventuresim_core::food::definition(item_id).is_some()
  47 }

  49 pub fn initialize_personal(ctx: &ReducerContext, inventory_item_id: u64) {
  50     ctx.db.inventory_item_amount().insert(InventoryItemAmount {
  51         inventory_item_id,
  52         remaining_fraction_micros: ConsumableFractionMicros::WHOLE.get(),

  54 }

  56 pub fn initialize_party(ctx: &ReducerContext, party_inventory_item_id: u64) {

  58         party_inventory_item_id,
  59         remaining_fraction_micros: ConsumableFractionMicros::WHOLE.get(),

  63 fn persisted_fraction(fraction_micros: u32) -> ConsumableFractionMicros {
  64     ConsumableFractionMicros::try_new(fraction_micros)
  65         .expect("persisted consumable fraction must not exceed one whole")
  66 }
  67 
  68 pub fn personal_fraction(
  69     ctx: &ReducerContext,
  70     inventory_item_id: u64,
  71 ) -> Option<ConsumableFractionMicros> {
  72     ctx.db
  73         .inventory_item_amount()
  74         .inventory_item_id()
  75         .find(inventory_item_id)
  76         .map(|row| persisted_fraction(row.remaining_fraction_micros))
  77 }
