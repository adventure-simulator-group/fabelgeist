 532 pub fn travel_consumption(deficit_kcal: f32, available_kcal: f32) -> f32 {
 533     if !deficit_kcal.is_finite() || !available_kcal.is_finite() {
 534         return 0.0;
 535     }
 536     (-deficit_kcal.min(0.0)).min(available_kcal.max(0.0))
 537 }

 539 pub fn explicit_meal_consumption(balance_kcal: f32, available_kcal: f32) -> f32 {
 540     if !balance_kcal.is_finite() || !available_kcal.is_finite() {
 541         return 0.0;
 542     }
 543     (MAX_MEAL_FULLNESS_KCAL - balance_kcal)
 544         .max(0.0)
 545         .min(available_kcal.max(0.0))
 546 }

 548 /// Scale a non-negative lot component while keeping malformed state from
 549 /// creating mass, nutrition, value, or provenance.
 550 pub fn retained_component(value: f32, retained_fraction: f32) -> f32 {
 551     if !value.is_finite() || !retained_fraction.is_finite() {
 552         return 0.0;
 553     }
 554     value.max(0.0) * retained_fraction.clamp(0.0, 1.0)
 555 }
