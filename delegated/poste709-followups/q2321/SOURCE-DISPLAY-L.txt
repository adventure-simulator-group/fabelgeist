  11 fn ensure_food_material_object(
  12     ctx: &ReducerContext,
  13     scope: CarriedInventoryScope,
  14     row_id: u64,
  15 ) -> Result<crate::InventoryObject, String> {
  16     let (item_id, location, quantity) = match scope {
  17         CarriedInventoryScope::Personal => {
  18             let row = ctx
  19                 .db
  20                 .inventory_item()
  21                 .id()
  22                 .find(row_id)
  23                 .ok_or("Food inventory row is missing")?;
  24             (
  25                 row.item_id,
  26                 InventoryLocation::personal(row.character_id, row.id),
  27                 row.quantity,
  28             )
  29         }
  30         CarriedInventoryScope::Party => {
  31             let row = ctx
  32                 .db
  33                 .party_inventory_item()
  34                 .id()
  35                 .find(row_id)
  36                 .ok_or("Party food inventory row is missing")?;
  37             (
  38                 row.item_id,
  39                 InventoryLocation::party(row.party_id, row.id),
  40                 row.quantity,
  41             )
  42         }
  43     };
  44     if quantity != 1 {
  45         return Err("Every food lot requires a quantity-one stable inventory object".into());
  46     }
  47     if let Some(object) = crate::inventory_container::object_for_row(ctx, scope, row_id)? {
  48         if object.item_id != item_id || object.location != location {
  49             return Err("Food inventory row has a mismatched stable object identity".into());
  50         }
  51         return Ok(object);
  52     }
  53     Ok(ctx.db.inventory_object().insert(crate::InventoryObject {
  54         id: 0,
  55         item_id,
  56         location,
  57     }))
  58 }
