  16 pub(super) async fn prepare_ingredient_lot(
  17     State(state): State<AppState>,
  18     session: Session,
  19     Form(form): Form<PrepareIngredientForm>,
  20 ) -> Response {
  21     let Some(character_id) = session.character_id_u64() else {
  22         return (StatusCode::UNAUTHORIZED, "Select a character first").into_response();
  23     };
  24     let Some(preparation_action) = parse_ingredient_preparation_action(&form.preparation_action) else {
  25         return (StatusCode::BAD_REQUEST, "Invalid ingredient preparation").into_response();
  26     };
  27     let action = match preparation_action {
  28         IngredientPreparationAction::Cut => json!({ "cut": {} }),
  29         IngredientPreparationAction::Grind => json!({ "grind": {} }),
  30     };
  31     if let Err(error) = state
  32         .db
  33         .call(
  34             "prepare_ingredient_lot",
  35             &[
  36                 json!(character_id),
  37                 json!(form.inventory_scope),
  38                 json!(form.inventory_item_id),
  39                 json!(form.food_lot_id),
  40                 json!(form.material_object_id),
  41                 json!(form.request_id),
  42                 json!(form.expected_revision),
  43                 json!(form.attempt_generation),
  44                 action,
  45             ],
  46         )
  47         .await
  48     {
  49         return (StatusCode::BAD_REQUEST, error.to_string()).into_response();
  50     }
  51     redirect_to_local(form.return_to.as_deref().unwrap_or(""), "/").into_response()
  52 }
