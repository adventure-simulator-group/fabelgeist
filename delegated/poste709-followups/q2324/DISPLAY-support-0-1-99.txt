     1	//! The studio's underlayers: cut from the wearer on the device and corrected
     2	//! for skeletal fit.
     3	use super::*;
     4	mod proportions;
     5	use adventuresim_character_creator::{
     6	    armor_frames::Wearer,
     7	    device_underlayer::{BodyShape, SurfaceDomain},
     8	    underlayer::UnderlayerDesign,
     9	    underlayer_armor::{self, UnderlayerBody},
    10	};
    11	use fabelgeist_armor::TrunkHoseDesign;
    12	
    13	use adventuresim_character_creator::armor_frames::FittingPlacement;
    14	/// Cut an underlayer from the wearer and fit it to every morph sample.
    15	pub(super) fn fitted(
    16	    model: &BodyModel,
    17	    generated: &GeneratedCharacter,
    18	    design: &UnderlayerDesign,
    19	    placement: FittingPlacement,
    20	    morphs: &[ForearmMorphSample],
    21	) -> Result<GeneratedArmor> {
    22	    let armor = with_body(model, generated, morphs, |body, shapes| {
    23	        underlayer_armor::fit_underlayer(
    24	            adventuresim_character_creator::armor_gpu()?,
    25	            design,
    26	            placement,
    27	            body,
    28	            shapes,
    29	        )
    30	    })?;
    31	    Ok(character_morphs::correct_armor_fit(
    32	        armor, generated, morphs,
    33	    ))
    34	}

    57	
    58	fn with_body(
    59	    model: &BodyModel,
    60	    generated: &GeneratedCharacter,
    61	    morphs: &[ForearmMorphSample],
    62	    fit: impl FnOnce(&UnderlayerBody<'_>, &[(&str, BodyShape<'_>)]) -> Result<GeneratedArmor>,
    63	) -> Result<GeneratedArmor> {
    64	    let character = &model.mhr.character;
    65	    let wearer = Wearer {
    66	        faces: &character.mesh.faces,
    67	        positions: &generated.positions,
    68	        normals: &generated.normals,
    69	        joints: &generated.global_joint_states,
    70	        joint_indices: character.skin_weights.indices(),
    71	        joint_weights: character.skin_weights.weights().as_native_slice(),
    72	        joint_names: &character.skeleton.names,
    73	    };
    74	    let static_samples = proportions::samples(model, generated)?;
    75	    let proportions = static_samples.iter().map(shape).collect::<Vec<_>>();
    76	    let shapes = morphs
    77	        .iter()
    78	        .map(|sample| (sample.name.as_str(), shape(sample)))
    79	        .collect::<Vec<_>>();
    80	    fit(
    81	        &UnderlayerBody {
    82	            wearer: &wearer,
    83	            domain: SurfaceDomain {
    84	                uv_faces: &character.mesh.texcoord_faces,
    85	                texcoords: &character.mesh.texcoords,
    86	            },
    87	            surface_domain: MHR_ANATOMICAL_UV_DOMAIN,
    88	            proportions: &proportions,
    89	        },
    90	        &shapes,
    91	    )
    92	}
    93	
    94	fn shape(sample: &ForearmMorphSample) -> BodyShape<'_> {
    95	    BodyShape {
    96	        positions: &sample.positions,
    97	        normals: &sample.normals,
    98	    }
    99	}
