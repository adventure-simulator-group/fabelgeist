  93     /// Variable-arity graphs may contain empty records and repeated addresses.
  94     /// Incidence identifies conflicts; it does not validate particle membership.
  95     pub fn for_constraints(constraints: &[&[ParticleIndex]]) -> Self {
  96         if constraints.is_empty() {
  97             return Self {
  98                 order: Vec::new(),
  99                 ranges: vec![ConstraintSlot(0)],
 100             };
 101         }
 102         let mut taken: HashMap<ParticleIndex, Vec<bool>> = HashMap::new();
 103         let mut colors = Vec::with_capacity(constraints.len());
 104         let mut color_count = ColorCount(0);
 105         for particles in constraints {
 106             let mut candidate = ConstraintColor(0);
 107             loop {
 108                 let mut free = true;
 109                 for particle in *particles {
 110                     if let Some(used) = taken.get(particle)
 111                         && used.get(candidate.0).is_some_and(|&occupied| occupied)
 112                     {
 113                         free = false;
 114                         break;
 115                     }
 116                 }
 117                 if free {
 118                     break;
 119                 }
 120                 candidate.0 += 1;
 121             }
 122             for &particle in *particles {
 123                 let used = taken.entry(particle).or_default();
 124                 if used.len() <= candidate.0 {
 125                     used.resize(candidate.0 + 1, false);
 126                 }
 127                 used[candidate.0] = true;
 128             }
 129             // Preserve the original native u32 color word narrowing.
 130             colors.push(ConstraintColor(candidate.0 as u32 as usize));
 131             color_count.0 = color_count.0.max(candidate.0 + 1);
 132         }
 133         let mut counts = vec![ConstraintSlot(0); color_count.0 + 1];
 134         for color in &colors {
 135             counts[color.0 + 1].0 += 1;
 136         }
 137         for i in 1..counts.len() {
 138             counts[i].0 += counts[i - 1].0;
 139         }
 140         let ranges = counts.clone();
 141         let mut cursor = counts;
 142         let mut order = vec![ConstraintIndex::from(0); constraints.len()];
 143         for (index, color) in colors.iter().enumerate() {
 144             let slot = &mut cursor[color.0];
 145             order[slot.0 as usize] = ConstraintIndex::from(index as u32);
 146             slot.0 += 1;
 147         }
 148         Self { order, ranges }
 149     }

  79         mesh: &ShellMesh,
  80         material: ShellMaterial,
  81     ) -> Result<Self> {
  82         mesh.validate()?;
  83         material.validate()?;
  84         if mesh.positions.is_empty() {
  85             return Err(anyhow!("Shell: the garment has no particles"));
  86         }
  87 
  88         let inverse_masses = mesh.inverse_masses();
  89         let particles = Particles::from_positions(context, &mesh.positions, &inverse_masses)?;
  90 
  91         let stretch = ConstraintSet::distance(
  92             context,
  93             cache,
  94             "stretch",
  95             &ConstraintEdges::from(mesh.edges.as_slice()),
  96             &mesh.rest_lengths,
  97             material.stretch_compliance,
  98         )?;
  99 
 100         // Seams start at zero rest length -- the two edges are meant to become
 101         // one -- and the solver closes the gap over the first few frames. That
 102         // pulling-together is the whole drape: it is what turns flat panels
 103         // into a garment shaped round a body.
 104         let seam_rest = vec![0.0f32; mesh.seams.len()];
 105         let seams = ConstraintSet::distance(
 106             context,
 107             cache,
 108             "seams",
 109             &ConstraintEdges::from(mesh.seams.as_slice()),
 110             &seam_rest,
 111             material.seam_compliance,
 112         )?;
 113 
 114         let bend_particles: Vec<[ParticleIndex; 4]> = mesh
 115             .bends
 116             .iter()
 117             .map(|bend| bend.particles().map(ParticleIndex::from))
 118             .collect();
 119         let bend_incidence = ConstraintIncidence::from_records(&bend_particles)?;
 120         let bend_kernel = cache.get(
 121             context,
 122             &fabelgeist_xpbd::wgsl::constraint_kernel(wgsl::BEND),
 123         )?;
 124         let mut bending = ConstraintSet::new(
 125             context,
 126             "bending",
 127             bend_kernel,
 128             cache,
 129             &bend_incidence,
 130             material.bend_compliance,
 131         )?;
 132         // The weights are per constraint and the set is stored in colour
 133         // order, so they have to be permuted the same way -- otherwise every
 134         // hinge reads another hinge's weights.
 135         let ordered_weights = bending.reorder(&mesh.bend_weights);
 136         let flat_weights: Vec<f32> = ordered_weights.into_iter().flatten().collect();
 137         bending.attach_raw(context, "weights", &flat_weights)?;
 138 
 139         let self_collision = SelfCollision::new(
 140             context,
 141             cache,
 142             mesh.positions.len() as u32,
 143             &mesh.adjacency(),
 144             material.particle_radius(),
 145         )?;
 146
