 129 | impl<'a> From<&'a NpyArray> for StoredValues<'a> {
 130 |     fn from(array: &'a NpyArray) -> Self {
 131 |         // The current dtype selects native fixed-width words. As in
 132 |         // chunks_exact, an incomplete final word is not decoded.
 133 |         match array.dtype {
 134 |             Dtype::F32 => Self::Float32(array.bytes.as_chunks::<4>().0.iter()),
 135 |             Dtype::F64 => Self::Float64(array.bytes.as_chunks::<8>().0.iter()),
 136 |             Dtype::I32 => Self::Integer32(array.bytes.as_chunks::<4>().0.iter()),
 137 |             Dtype::I64 => Self::Integer64(array.bytes.as_chunks::<8>().0.iter()),
 138 |             Dtype::U8 | Dtype::Bool => Self::Byte(array.bytes.iter()),
 139 |         }
 140 |     }
