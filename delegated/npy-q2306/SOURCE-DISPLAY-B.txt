  40 | /// The NumPy element types this crate decodes.
  41 | #[derive(Debug, Clone, Copy, PartialEq, Eq)]
  42 | pub enum Dtype {
  43 |     F32,
  44 |     F64,
  45 |     I32,
  46 |     I64,
  47 |     U8,
  48 |     Bool,
  49 | }
  50 | 
  51 | impl Dtype {
  52 |     fn parse(descr: &str) -> Result<Self> {
  53 |         Ok(match descr {
  54 |             "<f4" => Dtype::F32,
  55 |             "=f4" | "f4" if cfg!(target_endian = "little") => Dtype::F32,
  56 |             "<f8" => Dtype::F64,
  57 |             "=f8" | "f8" if cfg!(target_endian = "little") => Dtype::F64,
  58 |             "<i4" => Dtype::I32,
  59 |             "=i4" | "i4" if cfg!(target_endian = "little") => Dtype::I32,
  60 |             "<i8" => Dtype::I64,
  61 |             "=i8" | "i8" if cfg!(target_endian = "little") => Dtype::I64,
  62 |             "|u1" | "u1" => Dtype::U8,
  63 |             "|b1" | "b1" => Dtype::Bool,
  64 |             other => bail!("unsupported NumPy dtype {other:?}"),
  65 |         })
  66 |     }
  67 | 
  68 |     pub fn storage_width(self) -> NpyElementWidth {
  69 |         match self {
  70 |             Dtype::U8 | Dtype::Bool => NpyElementWidth::Byte,
  71 |             Dtype::F32 | Dtype::I32 => NpyElementWidth::Word32,
  72 |             Dtype::F64 | Dtype::I64 => NpyElementWidth::Word64,
  73 |         }
  74 |     }
  75 | }
  76 |
