#!/usr/bin/env python3
"""Bounded static inventory of root evidence-verifier progress assertions."""
from pathlib import Path
import hashlib,json,re,subprocess
root=Path.cwd();rev='8608c88ad0cd75ca188ce03908c7662c1aa436e8'
assert subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()==rev
raw=subprocess.check_output(['git','ls-tree','-z',rev]);tracked={}
for entry in raw.split(b'\0'):
 if not entry:continue
 meta,name=entry.split(b'\t',1);mode,kind,oid=meta.decode().split();name=name.decode()
 if name.startswith('verify_') and name.endswith('.py'):
  assert kind=='blob';tracked[name]=oid
files=sorted(root.glob('verify_*.py'));assert set(p.name for p in files)==set(tracked)
patterns={'lenses_zero':r'lenses_complete[^\n]{0,80}==\s*0','queue_zero':r'queue_records_semantically_reconciled[^\n]{0,80}==\s*0','partial_status':r'current_reconciliation[^\n]{0,120}startswith\([\"\']partial','candidate_count':r'new_current_ledger_candidates[^\n]{0,80}len\(ledger\)'}
rows=[];counts={k:0 for k in patterns}
for f in files:
 b=f.read_bytes();blob=hashlib.sha1(b'blob '+str(len(b)).encode()+b'\0'+b).hexdigest();assert blob==tracked[f.name]
 text=b.decode(errors='replace');matches=[k for k,p in patterns.items() if re.search(p,text)]
 for k in matches:counts[k]+=1
 rows.append({'path':f.name,'git_blob':blob,'sha256':hashlib.sha256(b).hexdigest(),'matched_heuristics':matches})
out={'schema':'726-root-verifier-static-inventory-v1','revision':rev,'patterns':patterns,'counts':counts,'files':rows,'qualification':'Exact root verifier file bytes match the recorded Git tree. Pattern matches are one-line lexical heuristics, not a proof that each assertion executes, is unsatisfiable, covers all equivalent assertions, or directly blocks a particular future change. No product code is read or executed.'}
p=root/'delegated/evidence-verifier-evolution/SOURCE-INVENTORY.json'
with p.open('x') as f:json.dump(out,f,indent=2);f.write('\n')
print('Inspected',len(rows),'root verifier files; heuristic match counts',counts)
