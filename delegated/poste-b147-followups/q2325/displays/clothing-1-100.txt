  59 impl GarmentSpecification {
  60     pub fn from_catalog(
  61         name: ClothingPieceName,
  62         selection: CatalogPlacement<'_>,
  63     ) -> Result<Self, ClothingError> {
  64         let material = selection
  65             .item()
  66             .equipment
  67             .as_ref()
  68             .and_then(|equipment| equipment.material)
  69             .ok_or_else(|| ClothingError::MissingMaterial {
  70                 garment: name.clone(),
  71             })?;
  72         Self::for_surface(name, selection.definition(), material)
  73     }
  74 
  75     fn for_surface(
  76         name: ClothingPieceName,
  77         placement: &EquipmentPlacement,
  78         material: EquipmentMaterial,
  79     ) -> Result<Self, ClothingError> {
  80         let channel = placement
  81             .outermost_channel()
  82             .unwrap_or(EquipmentChannel::BaseClothing);
  83         let normal_offset_metres = match channel {
  84             EquipmentChannel::BaseClothing | EquipmentChannel::Outerwear => 0.007,
  85             EquipmentChannel::Padding => 0.012,
  86             EquipmentChannel::FlexibleArmor => 0.017,
  87             EquipmentChannel::RigidArmor => 0.024,
  88             _ => 0.008,
  89         };
  90         let normal_offset_metres =
  91             SignedLength::from_metres(normal_offset_metres).map_err(ClothingError::NormalOffset)?;

  93         Ok(Self {
  94             name,
  95             material,
  96             placement: placement.clone(),
  97             weight_threshold: 0.35,
  98             expansion_rings: 1,
