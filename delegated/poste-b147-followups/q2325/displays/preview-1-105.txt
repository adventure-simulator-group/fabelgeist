   6 /// How a piece is shaded: its plate, and the band along its edges.
   7 pub(super) struct ArmorShading<'a> {
   8     pub plate: StandardMaterial,
   9     pub trim: Option<TrimPreview<'a>>,
  10 }
  11 
  12 /// The band along a piece's edges, as the preview shades it.
  13 pub(super) struct TrimPreview<'a> {
  14     pub texcoords: &'a [[f32; 2]],
  15     pub material: StandardMaterial,
  16 }

  18 pub(super) fn spawn_clothing(
  19     commands: &mut Commands,
  20     meshes: &mut Assets<Mesh>,
  21     materials: &mut Assets<StandardMaterial>,
  22     walk: &WalkPreview,
  23     model: &BodyModel,
  24     shells: Vec<ClothingShell>,
  25 ) -> Result<()> {
  26     for shell in shells {
  27         let specification = shell.specification;
  28         let indices = shell
  29             .faces
  30             .iter()
  31             .flat_map(|face| face.iter().copied())
  32             .collect::<Vec<_>>();
  33         let mut mesh = Mesh::new(
  34             PrimitiveTopology::TriangleList,
  35             RenderAssetUsages::default(),
  36         )
  37         .with_inserted_attribute(Mesh::ATTRIBUTE_POSITION, shell.positions)
  38         .with_inserted_attribute(Mesh::ATTRIBUTE_NORMAL, shell.normals)
  39         .with_inserted_indices(Indices::U32(indices));
  40         animation_preview::skin_mesh(
  41             &mut mesh,
  42             model.mhr.character.skin_weights.indices(),
  43             model.mhr.character.skin_weights.weights().as_native_slice(),
  44         )?;
  45         let [red, green, blue, alpha] = specification.base_color;
  46         commands.spawn((
  47             CharacterMesh,
  48             animation_preview::skin(walk).context("animation preview has no rebuilt rig")?,
  49             Name::new(specification.name.as_ref().to_owned()),
  50             Mesh3d(meshes.add(mesh)),
  51             MeshMaterial3d(materials.add(StandardMaterial {
  52                 base_color: Color::srgba(red, green, blue, alpha),
  53                 metallic: specification.metallic,
  54                 perceptual_roughness: specification.roughness,
  55                 ..default()
  56             })),
  57         ));
  58     }
  59     Ok(())
  60 }
  61 
  62 /// The cord lacing a piece's small plates, as the preview shades it.
  63 pub(super) fn lacing_material(cord: &fabelgeist_armor::Lacing) -> StandardMaterial {
  64     let [red, green, blue] = cord.color;
  65     StandardMaterial {
  66         base_color: Color::srgb(red, green, blue),
  67         metallic: 0.0,
  68         perceptual_roughness: cord.roughness,
  69         ..default()
  70     }
  71 }
