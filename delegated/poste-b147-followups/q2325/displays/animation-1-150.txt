 101 impl WalkPreview {
 102     pub fn ready(&self) -> bool {
 103         self.player.is_some() && self.graph.is_some() && self.node.is_some()
 104     }
 105 
 106     pub fn start_or_resume(&self, player: &mut AnimationPlayer) {
 107         if player.playing_animations().next().is_none() {
 108             player
 109                 .play(self.node.expect("ready walk preview has a graph node"))
 110                 .repeat();
 111         } else {
 112             player.resume_all();
 113         }
 114     }
 115 }

 117 impl ClothSkin {
 118     pub fn new(
 119         form: GarmentForm,
 120         positions: Vec<[f32; 3]>,
 121         _normals: Vec<[f32; 3]>,
 122         faces: Vec<[u32; 3]>,
 123         binding: ClothBinding,
 124     ) -> Result<Self> {
 125         if let ClothBinding::Weighted { indices, weights } = &binding {
 126             anyhow::ensure!(
 127                 indices.len() == positions.len() && weights.len() == positions.len(),
 128                 "cloth skin rows and positions have different vertex counts"
 129             );
 130         }
 131         let mut edges = std::collections::BTreeSet::new();
 132         for face in &faces {
 133             for [a, b] in [[face[0], face[1]], [face[1], face[2]], [face[2], face[0]]] {
 134                 edges.insert(if a < b { [a, b] } else { [b, a] });
 135             }
 136         }
 137         let current: Vec<Vec3> = positions.iter().copied().map(Vec3::from_array).collect();
 138         Ok(Self {
 139             render_vertices: None,
 140             form,
 141             positions,
 142             faces,
 143             edges: edges.into_iter().collect(),
 144             binding,
 145             previous: current.clone(),
 146             current,
 147             motion: ClothMotion::Skinned,
 148         })
 149     }
 150 }
