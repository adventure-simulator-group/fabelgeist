 311     let rig = preview.joints.first().copied();
 312     let body = bodies.iter().next().map(|body| {
 313         let (rest, end, faces) = colliders::posed_colliders(body, &armor, &matrices);
 314         let start = if body_motion.rig == rig && body_motion.positions.len() == end.len() {
 315             body_motion.positions.clone()
 316         } else {
 317             rest
 318         };
 319         (start, end, faces)
 320     });
 321     let targets = cloth
 322         .iter()
 323         .map(|(_, skin)| {
 324             if preview.physics && preview.ignore_cloth_weights {
 325                 return Some(None);
 326             }
 327             let ClothBinding::Weighted { indices, weights } = &skin.binding else {
 328                 return None;
 329             };
 330             if indices.joint_count() != matrices.joint_count() {
 331                 return None;
 332             }
 333             Some(Some(skinned_positions(
 334                 &skin.positions,
 335                 indices,
 336                 weights,
 337                 &matrices,
 338             )))
 339         })
 340         .collect::<Option<Vec<_>>>();
 341     let Some(targets) = targets else {
 342         // Weighted playback cannot evaluate unbound cloth. Keep its last
 343         // geometry until a binding or unweighted physics is selected.
 344         return;
 345     };
 346 
 347     if preview.physics {
 348         cloth_motion::simulate_outfit(
 349             &mut cloth,
 350             &body,
 351             &targets,
 352             time.delta_secs(),
 353             &preview.simulation,
 354         );
 355     } else {
 356         for (index, (_, mut skin)) in cloth.iter_mut().enumerate() {
 357             let Some(target) = targets[index].as_ref() else {
 358                 return;
 359             };
 360             skin.current.clone_from(target);
 361             skin.previous.clone_from(target);
 362             skin.motion = ClothMotion::Skinned;
 363         }
 364     }
 365     if let Some((_, end, _)) = body {
 366         body_motion.rig = rig;
 367         body_motion.positions = end;
 368     } else {
 369         body_motion.positions.clear();
 370         body_motion.rig = None;
 371     }
 372     for (handle, skin) in &cloth {
 373         if let Some(mut mesh) = meshes.get_mut(handle) {
 374             skin.update_render_mesh(&mut mesh);
 375         }
 376     }
 377 }
