 250     if args.export_only {
 251         for warning in export_character(&args.glb, &model, &recipe, &catalog, None)? {
 252             eprintln!("warning: {warning}");
 253         }
 254         println!("Exported {}", args.glb.display());
 255         return Ok(());
 256     }
 257 
 258     let libraries = Libraries {
 259         decorations: adventuresim_character_creator::decoration::DecorationLibrary::load(
 260             &args.decorations,
 261         )?,
 262         wardrobe: adventuresim_character_creator::wardrobe::Wardrobe::load(&args.wardrobe)?,
 263     };
 264     run_studio(args, model, catalog, recipe, libraries);
 265     Ok(())
 266 }

 296             DefaultPlugins

 298                     file_path: "../../assets".into(),

 300                 })

 303                         title: "Fabelgeist · Character Studio".into(),
 304                         resolution: (1440, 900).into(),
 305                         ..default()
 306                     }),
 307                     ..default()
 308                 }),
 309         )

 311             EguiPlugin::default(),
 312             hair_preview::HairShadingPlugin,
 313             hair_preview::HairGpuPlugin,
 314         ))

 318         .init_resource::<armory::Armory>()
 319         .init_resource::<studio_cache::StudioCache>()
 320         .init_resource::<wardrobe_tab::WardrobeTab>()
 321         .init_resource::<hair_preview::HairPreview>()
 322         .add_systems(Startup, (setup, armory::setup, animation_preview::request))
 323         .add_systems(
 324             EguiPrimaryContextPass,
 325             (studio_theme::install, studio_ui::show, armory_ui::labels).chain(),
 326         )
 327         .add_systems(
 328             Update,
 329             (
 330                 reload_model.before(regenerate_mesh),
 331                 animation_preview::prepare,
 332                 regenerate_mesh,
 333                 drape_preview::poll.after(regenerate_mesh),
 334                 hair_preview::refresh.after(regenerate_mesh),
 335                 hair_preview::show_layers.after(hair_preview::refresh),
 336                 drape_preview::refresh_mail.after(drape_preview::poll),
 337                 (
 338                     armory::enter_or_leave,
 339                     armory::refit,
 340                     armory::display,
 341                     armory::frame_camera,
 342                 )
 343                     .chain()
 344                     .after(reload_model)
 345                     .before(regenerate_mesh),
 346                 (wardrobe_tab::update, wardrobe_tab::display)
 347                     .chain()
 348                     .after(reload_model)
 349                     .before(regenerate_mesh),
