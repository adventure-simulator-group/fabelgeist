 165 pub fn source_for_item(id: &str) -> Option<&'static ItemSourceRef> {
 166     let sources = SOURCE_MAP.get_or_init(|| {
 167         let mut sources: Vec<ItemSourceRef> = serde_json::from_str(ITEM_CATALOG_SOURCE_MAP_JSON)
 168             .expect("validated embedded item source map");
 169         sources.sort_by(|a, b| a.id.cmp(&b.id));
 170         sources
 171     });
 172     sources
 173         .binary_search_by_key(&id, |source| source.id.as_str())
 174         .ok()
 175         .map(|index| &sources[index])
 176 }

 189 pub fn validate_references<'a>(
 190     references: impl IntoIterator<Item = &'a str>,
 191 ) -> Result<(), Vec<String>> {
 192     let missing: Vec<_> = references
 193         .into_iter()
 194         .filter(|id| definition(id).is_none())
 195         .map(str::to_owned)
 196         .collect();
 197     if missing.is_empty() {
 198         Ok(())
 199     } else {
 200         Err(missing)
 201     }
 202 }
