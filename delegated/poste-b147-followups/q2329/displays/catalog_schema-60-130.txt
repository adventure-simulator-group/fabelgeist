  88 #[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
  89 #[serde(deny_unknown_fields)]
  90 pub struct EquipmentPlacement {
  91     /// Stable within the item definition; persisted instead of an array index.
  92     pub id: String,
  93     /// A placement may combine physical character anchors with any number of
  94     /// required item attachment points. The selected targets are runtime
  95     /// graph edges and are not inferred from item kind.
  96     #[serde(default)]
  97     pub occupancy: Vec<OccupancyRequirement>,
  98     #[serde(default)]
  99     pub parents: Vec<ParentRequirement>,
 100     /// Local geometric layering exceptions. The ordinary inner-to-outer order
 101     /// follows equipment channels; these entries express constructions such as
 102     /// a boot enclosing padded hose even though footwear is base clothing.
 103     #[serde(default)]
 104     pub layers_over: Vec<EquipmentLayerPrecedence>,
 105     /// Explicit many-to-many combat projection. Locations never imply
 106     /// protection.
 107     #[serde(default)]
 108     pub protection: Vec<EquipmentBodyPart>,
 109     /// Fundamental visible surface coverage used to generate fitted MHR
 110     /// geometry. Each span is measured along its ordered anatomical regions.
 111     #[serde(default)]
 112     pub surface: Vec<EquipmentSurfaceSpan>,
 113 }

 115 impl EquipmentPlacement {
 116     /// Physical layer order also applies to garments carried by attachment points.
 117     pub fn outermost_channel(&self) -> Option<EquipmentChannel> {
 118         self.occupancy
 119             .iter()
 120             .map(|entry| entry.channel)
 121             .chain(self.parents.iter().map(|entry| entry.channel))
 122             .max_by_key(|channel| channel.order())
 123     }
 124 }
