   151	            })
   152	            .collect()
   153	    }

   156	pub(super) fn correct_armor_fit(
   157	    mut armor: GeneratedArmor,
   158	    reference: &GeneratedCharacter,
   159	    samples: &[ForearmMorphSample],
   160	) -> GeneratedArmor {
   161	    for target in &mut armor.morphs {
   162	        if let Some(sample) = samples.iter().find(|sample| sample.name == target.name) {
   163	            remove_skeletal_translation(
   164	                &mut target.position_deltas,
   165	                sample,
   166	                &reference.global_joint_states,
   167	                &armor.joint_indices,
   168	                armor
   169	                    .joint_weights
   170	                    .iter()
   171	                    .enumerate()
   172	                    .map(|(vertex, weights)| {
   173	                        (MeshVertexIndex::from_native_vertex_index(vertex), *weights)
   174	                    }),
   175	            );
   176	        }
   177	    }
   178	    armor
   179	}

   193	/// Hardware has its own surface, regardless of the catalog's plate material.
   194	pub(super) fn component_materials(
   195	    armor: &GeneratedArmor,
   196	    shells: &mut [RiggedShell<'_>],
   197	) -> Result<()> {
   198	    for (component, shell) in armor.components.iter().zip(shells) {
   199	        if let Some(material) = component.material {
   200	            shell.base_color = material.base_color;
   201	            shell.metallic = material.metallic;
   202	            shell.roughness = material.roughness;
   203	            shell.textures = None;
   204	            shell.surface = None;
   205	            shell.plate_edges = &[];
   206	        }
   207	        if let Some(textile) = &component.textile {
   208	            shell.textile_finish = Some(textile.finish);
   209	            shell.textures =
   210	                adventuresim_character_creator::textile_surface::textures(textile)?.map(Into::into);
   211	        }
   212	    }
   213	    Ok(())
   214	}

   241	/// The shells of a piece, one per component, or `name` for the whole piece
   242	/// and `trim_name` for its band.
   243	pub(super) fn rigged_armor<'a>(
   244	    name: &'a str,
   245	    trim_name: &'a str,
   246	    armor: &'a GeneratedArmor,
   247	    faces: &'a [[u32; 3]],
   248	    targets: &'a [RiggedMorphTarget<'a>],
   249	) -> RiggedArmor<'a> {
   250	    let shell = |name, faces, hinge| RiggedShell {
   251	        textile_finish: None,
   252	        plate_edges: &[],
   253	        surface: None,
   254	        textures: None,
   255	        texcoords: Some(&armor.texcoords),
   256	        name,
   257	        hinge,
   258	        positions: &armor.positions,
   259	        normals: &armor.normals,
   260	        faces,
   261	        joint_indices: Some(&armor.joint_indices),
   262	        joint_weights: Some(&armor.joint_weights),
   263	        morph_targets: targets,
   264	        base_color: [0.769, 0.776, 0.776, 1.0],
   265	        metallic: 1.0,
   266	        roughness: 0.20,
   267	    };
   268	    let triangles = |range: std::ops::Range<usize>| &faces[range.start / 3..range.end / 3];
   269	    let mut rigged = RiggedArmor {
   270	        plate: Vec::new(),
   271	        trim: Vec::new(),
   272	    };
   273	    for surface in armor.surfaces() {
   274	        let component = surface.component.map(|index| &armor.components[index]);
   275	        let hinge = component.and_then(|component| component.hinge);
   276	        rigged.plate.push(shell(
   277	            component.map_or(name, |component| component.role.name()),
   278	            triangles(surface.plate),
   279	            hinge,
   280	        ));
   281	        if !surface.trim.is_empty() {
   282	            rigged.trim.push(shell(
   283	                component.map_or(trim_name, |component| component.role.trim_name()),
   284	                triangles(surface.trim),
   285	                hinge,
   286	            ));
   287	        }
   288	    }
   289	    rigged
   290	}
