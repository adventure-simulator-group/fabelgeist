   225	    if let Some(output) = &args.armor_review_dir {
   226	        return review_export::export(
   227	            output,
   228	            &model,
   229	            &recipe,
   230	            &catalog,
   231	            args.armor_review_selection,
   232	        );
   233	    }
   234	
   235	    if args.generate_equipment {
   236	        let selection = EquipmentItemSelection::admit(&catalog, &args.equipment_item)?;
   237	        generate_equipment_assets(
   238	            &args.equipment_output,
   239	            &model,
   240	            &recipe,
   241	            &catalog,
   242	            &selection,
   243	        )?;
   244	        println!(
   245	            "Generated equipment under {}",
   246	            args.equipment_output.display()
   247	        );
   248	        return Ok(());
   249	    }
   250	    if args.export_only {
   251	        for warning in export_character(&args.glb, &model, &recipe, &catalog, None)? {
   252	            eprintln!("warning: {warning}");
   253	        }
   254	        println!("Exported {}", args.glb.display());
   255	        return Ok(());
   256	    }
   257	
   258	    let libraries = Libraries {
   259	        decorations: adventuresim_character_creator::decoration::DecorationLibrary::load(
   260	            &args.decorations,
   261	        )?,
   262	        wardrobe: adventuresim_character_creator::wardrobe::Wardrobe::load(&args.wardrobe)?,
   263	    };
   264	    run_studio(args, model, catalog, recipe, libraries);
   265	    Ok(())
   266	}
