 229 /// Take the running drape's progress, and show it when the design is shown.
 230 fn collect(wardrobe: &mut WardrobeTab, studio: &mut Studio) {
 231     let progress = wardrobe.worker.poll();
 232     let Some(drape) = wardrobe.drape.as_mut() else {
 233         return;
 234     };
 235     if let Some(garment) = progress
 236         .preview
 237         .and_then(|garments| garments.into_iter().next())
 238     {
 239         drape.garment = Some(garment);
 240         wardrobe.refresh |= wardrobe.showing == Showing::Design;
 241     }
 242     if let Some(outcome) = progress.finished {
 243         drape.finished = true;
 244         drape.problems = outcome.problems();
 245         studio.status = match &drape.problems {
 246             None => format!(
 247                 "{} settled; ready to save to the wardrobe",
 248                 drape.selection.name
 249             ),
 250             Some(problems) => format!("{} draped with problems: {problems}", drape.selection.name),
 251         };
 252     }
 253 }

 255 /// Drape the designed garment on the wardrobe's body.
 256 fn request(wardrobe: &mut WardrobeTab, model: &BodyModel, studio: &mut Studio) {
 257     let Some(input) = wardrobe.input(model, wardrobe.garment.clone()) else {
 258         return;
 259     };
 260     wardrobe.drape = Some(Drape {
 261         selection: wardrobe.garment.clone(),
 262         garment: None,
 263         finished: false,
 264         problems: None,
 265     });
 266     wardrobe.worker.request(vec![((), input)]);
 267     studio.status = format!("Draping {} in the wardrobe…", wardrobe.garment.name);
 268 }
