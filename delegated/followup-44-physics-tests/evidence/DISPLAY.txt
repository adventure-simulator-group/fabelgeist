  25 /// Native capacity selected by allocation policy, without proving a live buffer.
  26 /// Growth retains host rounding followed by unsigned-word truncation.
  27 #[derive(Clone, Copy, Debug, PartialEq, Eq)]
  28 pub(super) struct ColliderCapacity(u32);
  29 impl ColliderCapacity {
  30     // Reserve 16 collider slots (1 KiB). Larger lists grow the buffer
  31     // to the next power of two.
  32     pub(super) const INITIAL: Self = Self(16);
  33     pub(super) fn for_count(count: ColliderCount) -> Self {
  34         Self(count.0.next_power_of_two() as u32)
  35     }
  36     pub(super) fn fit(self, count: ColliderCount) -> ColliderCapacityFit {
  37         if count.0 as u32 > self.0 {
  38             ColliderCapacityFit::GrowthRequired
  39         } else {
  40             ColliderCapacityFit::Retained
  41         }
  42     }
  43     pub(super) fn byte_length(self) -> BufferByteLength {
  44         (u64::from(self.0) * COLLIDER_BYTES as u64).into()
  45     }
  46 }
