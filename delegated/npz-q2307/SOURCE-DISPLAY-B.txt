   7 | const NPY_MEMBER_SUFFIX: &str = ".npy";

   9 | /// A NumPy archive query key, distinct from a full archive member name.
  10 | ///
  11 | /// A candidate matches when its full parsed name or that name with one trailing
  12 | /// `.npy` removed equals this key. Queries are never normalized. The archive's
  13 | /// first matching member wins, even when another is an exact filename match.
  14 | /// This key validates neither spelling nor membership.
  15 | ///
  16 | /// ```
  17 | /// use fabelgeist_numpy_storage::{ArchiveMemberName, NpzArrayName};
  18 | /// let member = ArchiveMemberName::from("weights.npy");
  19 | /// let key = NpzArrayName::from(&member);
  20 | /// assert_eq!(key, NpzArrayName::new("weights"));
  21 | /// ```
  22 | ///
  23 | /// A full member name cannot be used as an array key:
  24 | ///
  25 | /// ```compile_fail
  26 | /// use fabelgeist_numpy_storage::{ArchiveMemberName, Npz};
  27 | /// fn read(archive: &Npz, member: &ArchiveMemberName) {
  28 | ///     let _ = archive.array(member);
  29 | /// }
  30 | /// ```
  31 | ///
  32 | /// An array key cannot be used as an exact member lookup:
  33 | ///
  34 | /// ```compile_fail
  35 | /// use fabelgeist_numpy_storage::{NpzArrayName, ZipArchive};
  36 | /// fn read(archive: &ZipArchive, key: &NpzArrayName) {
  37 | ///     let _ = archive.bytes(key);
  38 | /// }
  39 | /// ```
  40 | #[derive(Clone, Debug, Display, From, PartialEq, Eq)]
  41 | #[debug("{_0:?}")]
  42 | #[from(String, &str)]
  43 | pub struct NpzArrayName(String);
  44 | 
  45 | impl NpzArrayName {
  46 |     /// Constructs an owned query without removing a suffix or changing spelling.
  47 |     pub fn new(name: &str) -> Self {
  48 |         Self::from(name)
  49 |     }
  50 | 
  51 |     pub(crate) fn matches(&self, member: &ArchiveMemberName) -> bool {
  52 |         let member = <&str>::from(member);
  53 |         member == self.0 || member.strip_suffix(NPY_MEMBER_SUFFIX) == Some(self.0.as_str())
  54 |     }
  55 | }
  56 | 
  57 | impl From<&ArchiveMemberName> for NpzArrayName {
  58 |     fn from(member: &ArchiveMemberName) -> Self {
  59 |         let member = <&str>::from(member);
  60 |         Self::new(member.strip_suffix(NPY_MEMBER_SUFFIX).unwrap_or(member))
  61 |     }
  62 | }
  63 | 
  64 | impl<'a> From<&'a NpzArrayName> for &'a str {
  65 |     fn from(name: &'a NpzArrayName) -> Self {
  66 |         &name.0
  67 |     }
  68 | }
