   478	    let height = 0.012;

   480	        let coordinate = step as f32 / SUBDIVISIONS_PER_METRE as f32;

   482	        let color = if step == 0 {

   484	        } else if whole_metre {

   488	        };
   489	        gizmos.line(
   490	            Vec3::new(coordinate, height, -HALF_EXTENT_METRES as f32),
   491	            Vec3::new(coordinate, height, HALF_EXTENT_METRES as f32),
   492	            color,

   494	        gizmos.line(
   495	            Vec3::new(-HALF_EXTENT_METRES as f32, height, coordinate),

   263	pub(super) fn update_readiness(
   264	    mut capture: ResMut<ArmorCapture>,
   265	    armor: Query<(Entity, &ItemProperties), With<CapturedArmor>>,
   266	    visuals: EquipmentVisualStatus,
   267	    equipment: Res<crate::equipment::RuntimeEquipmentBodyCache>,
   268	    mut exit: MessageWriter<AppExit>,
   269	) {
   270	    let Some(harness) = capture.harness else {
   271	        return;
   272	    };
   273	    if capture.failed {
   274	        return;
   275	    }
   276	    let was_ready = capture.ready;
   277	    capture.ready = armor.iter().count() == harness.placements().count();
   278	    for (entity, item) in &armor {
   279	        match visuals.state(entity, harness.visual_requirements()) {
   280	            EquipmentVisualState::Ready => {}
   281	            EquipmentVisualState::Failed => {
   282	                fs::write(
   283	                    capture.output.join("armor-readiness-failure.json"),
   284	                    serde_json::to_vec_pretty(&serde_json::json!({
   285	                        "item_id": item.id, "parts": visuals.summary(entity),
   286	                    }))
   287	                    .expect("serialize failed armor readiness"),
   288	                )
   289	                .expect("write failed armor readiness");
   290	                capture.fail(
   291	                    &format!(
   292	                        "Procedural equipment asset or primary skin weights failed for {}",
   293	                        item.id
   294	                    ),
   295	                    &mut exit,
   296	                );
   297	                return;
   298	            }
   299	            EquipmentVisualState::Missing | EquipmentVisualState::Loading => capture.ready = false,
   300	        }
   301	    }
   302	    if capture.ready && !was_ready {
   303	        let parts = armor
   304	            .iter()
   305	            .map(|(entity, item)| {
   306	                serde_json::json!({
   307	                    "item_id": item.id, "parts": visuals.summary(entity),
   308	                })
   309	            })
   310	            .collect::<Vec<_>>();
   311	        fs::write(
   312	            capture.output.join("armor-readiness.json"),
   313	            serde_json::to_vec_pretty(&parts).expect("serialize armor readiness"),
   314	        )
   315	        .expect("write armor readiness");
   316	    }
   317	    // A requested fit includes first-use device compilation. Count unresolved
   318	    // presentation frames after that asynchronous task completes.
   319	    if !capture.ready && !equipment.is_preparing() {
   320	        capture.waited += 1;
   321	        if capture.waited >= CAPTURE_LOAD_FRAME_LIMIT {
   322	            capture.fail("Timed out waiting for every armor mesh, material, morph, and wearer skin to resolve", &mut exit);
   323	        }
   324	    }
   325	}
