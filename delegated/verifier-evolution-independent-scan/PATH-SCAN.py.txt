#!/usr/bin/env python3
"""Inspect selected patterns in paths from one NUL-delimited Git tree listing."""
from pathlib import Path
import collections,gzip,hashlib,json,os,pwd,re,sys
src=Path(sys.argv[1]);stored=src.read_bytes();rows=gzip.decompress(stored).split(b'\0');assert rows[-1]==b'';rows=rows[:-1];assert len(rows)==23030
home=pwd.getpwuid(os.getuid()).pw_dir.encode();tmp=b'/'+b'tmp'+b'/726-'
patterns={'local_home':re.compile(re.escape(home)+rb'(?:/|["\'\s]|$)'),'tmp_task':re.compile(re.escape(tmp)+rb'[A-Za-z0-9_-]{3,}'),'windows_user':re.compile(rb'[A-Za-z]:[\\/]+Users[\\/]+[A-Za-z0-9._-]+',re.I),'email':re.compile(rb'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b'),'credential':re.compile(rb'gh[pousr]_[A-Za-z0-9_]{25,}|github_pat_[A-Za-z0-9_]{30,}|xox[baprs]-[A-Za-z0-9-]{20,}|AKIA[0-9A-Z]{16}|sk-(?:proj-)?[A-Za-z0-9_-]{32,}',re.I)}
counts=collections.Counter()
for row in rows:
 meta,name=row.split(b'\t',1);parts=meta.split();assert len(parts)==4 and len(parts[2])==40
 for k,p in patterns.items():counts[k]+=len(p.findall(name))
out={'schema':'726-tree-listing-path-pattern-scan-v1','stored_sha256':hashlib.sha256(stored).hexdigest(),'entries':len(rows),'counts':dict(counts),'qualification':'Checks only selected local-home, task-temp, Windows-user, email and credential patterns within pathnames in this listing. It does not establish general privacy, inspect contents or clear Git history. The earlier manual attempt assumed three metadata fields and stopped at an assertion; this version accepts the actual four fields including size.'}
print(json.dumps(out,indent=2));assert not any(counts.values())
