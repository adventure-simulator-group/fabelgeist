  76     #[test]
  77     fn zip64_range_retains_high_bits_and_exclusive_end_despite_capacity_cap() {
  78         for wire in [4096u64, 4097, 1 << 32, (1 << 32) + 1, 1 << 63, u64::MAX] {
  79             let count = ArchiveMemberCount::from(wire);
  80             let mut range = count.native_iteration_range();
  81             assert_eq!(range.next(), Some(0));
  82             assert_eq!(range.end, wire);
  83             assert_eq!(range.next_back(), Some(wire - 1));
  84             assert_eq!(count.initial_capacity(), 4096);
  85         }
  86         let empty = ArchiveMemberCount::from(0u64);
  87         assert_eq!(empty.native_iteration_range().next(), None);
  88         let single = ArchiveMemberCount::from(1u64);
  89         assert_eq!(single.native_iteration_range().collect::<Vec<_>>(), [0]);
  90         assert_eq!(single.initial_capacity(), 1);
