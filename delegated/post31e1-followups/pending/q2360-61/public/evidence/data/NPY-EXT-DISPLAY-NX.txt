 118 impl NpyArray {
 119     /// Parses a `.npy` buffer: header dictionary first, then the element payload.
 120     pub fn from_bytes(bytes: &[u8]) -> Result<Self> {
 121         let NpyHeader {
 122             dtype,
 123             shape,
 124             payload,
 125         } = NpyHeader::from_bytes(bytes)?;
 126         Ok(Self {
 127             shape,
 128             dtype,
 129             bytes: payload.to_vec(),
 130         })
 131     }
 132 
 133     /// Reads a standalone `.npy` file.
 134     pub fn read(path: impl AsRef<Path>) -> Result<Self> {
 135         let path = path.as_ref();
 136         let bytes = std::fs::read(path).with_context(|| format!("reading {}", path.display()))?;
 137         Self::from_bytes(&bytes).with_context(|| format!("parsing {}", path.display()))
 138     }
 139 }

 145     pub(crate) fn encode(dtype: &str, shape: &str, payload: &[u8]) -> Vec<u8> {
 146         let header =
 147             format!("{{'descr': '{dtype}', 'fortran_order': False, 'shape': ({shape}), }}");
 148         let mut padded = header.into_bytes();
 149         while (10 + padded.len()) % 64 != 63 {
 150             padded.push(b' ');
 151         }
 152         padded.push(b'\n');
 153         let mut out = b"\x93NUMPY\x01\x00".to_vec();
 154         out.extend_from_slice(&(padded.len() as u16).to_le_bytes());
 155         out.extend_from_slice(&padded);
 156         out.extend_from_slice(payload);
 157         out
 158     }
