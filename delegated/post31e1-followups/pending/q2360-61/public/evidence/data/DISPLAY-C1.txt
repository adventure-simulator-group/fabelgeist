   7 /// The count selected by rawzip, retained without native-width narrowing.
   8 #[derive(Debug, Clone, Copy, From, PartialEq, Eq)]
   9 #[from(u16, u64)]
  10 pub(super) struct ArchiveMemberCount(u64);
  11 
  12 impl ArchiveMemberCount {
  13     // The inherited allocation cap is not a directory cardinality limit.
  14     const INITIAL_CAPACITY_LIMIT: Self = Self(4096);
  15 
  16     /// Projects only the capped count into native `Vec` allocation capacity.
  17     pub(super) fn initial_capacity(self) -> usize {
  18         self.0.min(Self::INITIAL_CAPACITY_LIMIT.0) as usize
  19     }
  20 
  21     /// Projects into the original native u64 range; callers ignore its ordinal.
  22     ///
  23     /// The count is never narrowed to usize for directory iteration.
  24     pub(super) fn native_iteration_range(self) -> Range<u64> {
  25         0..self.0
  26     }
  27 }
