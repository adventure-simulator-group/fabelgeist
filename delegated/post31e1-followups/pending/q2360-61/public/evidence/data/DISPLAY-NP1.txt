  17 /// An `.npz` archive backed by a read-only mapping or owned bytes.

  22 /// A selected NPZ member tied to the archive that owns its bytes.
  23 ///
  24 /// Metadata and decoding use this exact first match. The original query is
  25 /// retained for parsing diagnostics; a present member may still fail to decode.
  26 pub struct NpzMember<'a> {
  27     member: ZipMember<'a>,
  28     query: NpzArrayName,
  29 }
  30 
  31 impl Npz {
  32     pub fn open(path: impl AsRef<Path>) -> Result<Self> {
  33         Ok(Self {
  34             archive: ZipArchive::open(path)?,
  35         })
  36     }
  37 
  38     /// Opens an archive already loaded in memory, for browser and streamed assets.
  39     pub fn from_bytes(bytes: Vec<u8>) -> Result<Self> {
  40         Ok(Self {
  41             archive: ZipArchive::from_bytes(bytes)?,
  42         })
  43     }
  44 
  45     /// Parsed member names in central-directory order, retaining their suffix.
  46     pub fn names(&self) -> impl Iterator<Item = &ArchiveMemberName> {
  47         self.archive.names()
  48     }
  49 
  50     /// Array keys in member order, each with one trailing `.npy` removed.
  51     pub fn keys(&self) -> impl Iterator<Item = NpzArrayName> {
  52         self.names().map(NpzArrayName::from)
  53     }
  54 
  55     /// Selects the first full-name or single-suffix match in directory order.
  56     pub fn get(&self, name: &NpzArrayName) -> Option<NpzMember<'_>> {
  57         self.archive
  58             .find(|member| name.matches(member))
  59             .map(|member| NpzMember {
  60                 member,
  61                 query: name.clone(),
  62             })
  63     }
  64 
  65     /// Requires a member and decodes it, reporting absence as a lookup error.
  66     pub fn array(&self, name: &NpzArrayName) -> Result<NpyArray> {
  67         self.get(name)
  68             .with_context(|| format!("archive has no member {name:?}"))?
  69             .array()
  70     }
  71 }
