  39 pub(in crate::tactical_scene_viewer) fn ready(
  40     state: Option<Res<SceneCaptureState>>,
  41     requirements: Option<Res<ReviewRequirements>>,
  42     readiness: Option<Res<Readiness>>,
  43     exposure: Option<Res<crate::presentation::interior_lighting::InteriorExposure>>,
  44 ) -> bool {
  45     requirements.is_none()
  46         || state.is_none_or(|state| state.phase == CapturePhase::Configure)
  47         || (readiness.is_some_and(|readiness| readiness.ready)
  48             && exposure.is_none_or(|exposure| exposure.is_settled()))
  49 }

 112 impl Observation<'_, '_> {
 113     fn assets_ready(
 114         &self,
 115         mesh: &Mesh3d,
 116         handle: &crate::presentation::interior_lighting::InteriorMaterialSource,
 117     ) -> bool {
 118         self.materials.get(&handle.0).is_some_and(|material| {
 119             self.gpu.contains(
 120                 &mesh.0,
 121                 [
 122                     &material.base_color_texture,
 123                     &material.normal_map_texture,
 124                     &material.metallic_roughness_texture,
 125                 ]
 126                 .into_iter()
 127                 .flatten()
 128                 .map(Handle::id),
 129             )
 130         })
 131     }
 132 
 133     fn check(&self, requirements: &ReviewRequirements) -> bool {
 134         if self.gardens.check(&self.gpu).is_err()
 135             || self.buildings.iter().count() != requirements.buildings
 136             || self.boundaries.iter().count() != requirements.boundaries
 137             || self.doors.iter().count() != requirements.doors
 138             || self.windows.iter().count() != requirements.windows
 139         {
 140             return false;
 141         }
 142         if !self
 143             .batches
 144             .iter()
 145             .all(|(_, _, mesh, material)| self.assets_ready(mesh, material))
 146             || !self
 147                 .doors
 148                 .iter()
 149                 .all(|(_, mesh, material)| self.assets_ready(mesh, material))
 150             || !self
 151                 .windows
 152                 .iter()
 153                 .all(|(_, mesh, material)| self.assets_ready(mesh, material))
 154         {
 155             return false;
 156         }
 157         for (mesh, material, parent) in &self.parts {
 158             let is_sign = self.signs.contains(parent.parent())
 159                 || self
 160                     .parents
