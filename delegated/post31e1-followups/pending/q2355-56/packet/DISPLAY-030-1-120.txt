  89 impl Plugin for TacticalCameraPlugin {
  90     fn build(&self, app: &mut App) {
  91         app.init_resource::<CameraMode>()
  92             .init_resource::<CameraRigConfig>()
  93             .init_resource::<CameraRigState>()
  94             .init_resource::<CameraRigDebugState>()
  95             .init_resource::<CameraAimState>()
  96             .init_resource::<CameraDebugEnabled>()
  97             .add_systems(Update, toggle_camera_mode)
  98             .add_systems(
  99                 PostUpdate,
 100                 update_camera_rig
 101                     .in_set(TacticalCameraSet::Offset)
 102                     .before(TransformSystems::Propagate),
 103             )
 104             .add_systems(
 105                 PostUpdate,
 106                 update_camera_aim
 107                     .in_set(TacticalCameraSet::Aim)
 108                     .after(TacticalCameraSet::Offset)
 109                     .before(TransformSystems::Propagate),
 110             );
 111         #[cfg(feature = "debug")]
 112         app.add_systems(Update, toggle_camera_debug);
 113     }
 114 }
