  66 #[derive(Clone)]
  67 struct CompiledBuildingBatch {
  68     material: BuildingLodMaterial,
  69     mesh: Handle<Mesh>,
  70     triangles: usize,
  71     transform: Mat4,
  72     uv_offset: Vec2,
  73 }
  74 
  75 impl CompiledBuildingBatch {
  76     fn from_recipe(
  77         batch: &LodMesh,
  78         origin: Vec3,
  79         detail: BuildingDetail,
  80         meshes: &mut Assets<Mesh>,
  81     ) -> Self {
  82         let mut mesh = recipe_mesh(batch, origin);
  83         if detail != BuildingDetail::Dynamic {
  84             // GPU assembly owns storage-buffer uploads for these vertices.
  85             mesh.asset_usage = RenderAssetUsages::MAIN_WORLD;
  86         }
  87         Self {
  88             material: batch.material,
  89             mesh: meshes.add(mesh),
  90             triangles: batch.indices.len() / 3,
  91             transform: Mat4::IDENTITY,
  92             uv_offset: Vec2::ZERO,
  93         }
  94     }
  95 }
