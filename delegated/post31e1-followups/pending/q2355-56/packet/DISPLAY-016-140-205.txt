 140             camera_origin,
 141             camera_forward,
 142             actor_origin,
 143             maximum_distance,
 144             choices
 145                 .iter()
 146                 .filter(|choice| choice.body == body)
 147                 .filter_map(|choice| {
 148                     choice
 149                         .resolved
 150                         .map(|resolved| (resolved, resolved.1, choice.stable_key))
 151                 }),
 152         ) {
 153             return Some(resolved);
 154         }
 155         exhausted.push(body);
 156     }
 157 }
