  51 #[derive(Clone, Copy, Debug, Serialize, Deserialize)]
  52 pub enum PixelEncoding {
  53     R8,
  54     Rg8,
  55     Rgba8,
  56     Srgb8,
  57 }
  58 
  59 impl PixelEncoding {
  60     pub const fn channels(self) -> usize {
  61         match self {
  62             Self::R8 => 1,
  63             Self::Rg8 => 2,
  64             Self::Rgba8 | Self::Srgb8 => 4,
  65         }
  66     }
  67 
  68     pub const fn texture_format(self) -> TextureFormat {
  69         match self {
  70             Self::R8 => TextureFormat::R8Unorm,
  71             Self::Rg8 => TextureFormat::Rg8Unorm,
  72             Self::Rgba8 => TextureFormat::Rgba8Unorm,
  73             Self::Srgb8 => TextureFormat::Rgba8UnormSrgb,
  74         }
  75     }
  76 }

  89 #[derive(Clone)]
  90 pub struct BakedRecipe {
  91     pub recipe: TextureRecipeId,
  92     pub tile_metres: f32,
  93     pub height_range_metres: f32,
  94     pub maps: Vec<BakedMap>,
  95 }
  96 
  97 impl BakedRecipe {
  98     pub fn generate(recipe: TextureRecipeId, params: &TextureParameters) -> Self {
  99         let mut images = Assets::<Image>::default();
 100         let handles = generate_maps(recipe, params, &mut images);
 101         let maps = handles
 102             .into_iter()
 103             .map(|(channel, handle)| {
 104                 let mut image = images.get_mut(&handle).expect("generated image exists");
 105                 let encoding = match image.texture_descriptor.format {
 106                     TextureFormat::R8Unorm => PixelEncoding::R8,
 107                     TextureFormat::Rg8Unorm => PixelEncoding::Rg8,
 108                     TextureFormat::Rgba8Unorm => PixelEncoding::Rgba8,
 109                     TextureFormat::Rgba8UnormSrgb => PixelEncoding::Srgb8,
 110                     format => panic!("unsupported procedural encoding: {format:?}"),
 111                 };
 112                 BakedMap {
 113                     channel,
 114                     size: image.width(),
 115                     mip_levels: image.texture_descriptor.mip_level_count,
 116                     encoding,
 117                     sampler: image.sampler.clone(),
 118                     bytes: image.data.take().expect("CPU recipe produces pixels"),
 119                 }
 120             })
 121             .collect();
 122         let (tile_metres, height_range_metres) = physical_scale(recipe, params);
 123         Self {
 124             recipe,
 125             tile_metres,
 126             height_range_metres,
 127             maps,
 128         }
 129     }
 130 
 131     pub fn map(&self, channel: MapChannel) -> Option<&BakedMap> {
 132         self.maps.iter().find(|map| map.channel == channel)
 133     }
 134 }
