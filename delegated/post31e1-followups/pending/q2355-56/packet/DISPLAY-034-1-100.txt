  23 /// Bounds constraint refinement when close framing reveals another obstacle.
  24 const MAX_FRAMING_SWEEPS: usize = 8;

  26 /// Close framing keeps the subject beside the view without changing manual aim.
  27 #[derive(Debug, Clone, Copy)]
  28 pub struct CloseCameraProfile {
  29     pub start_distance: SignedLength,
  30     pub full_distance: SignedLength,
  31     /// Camera-local right and up offsets from the subject focus.
  32     pub offset: FramingOffset,
  33 }
  34 
  35 /// Render view coordinates; positive Z points behind the eye.
  36 #[derive(Clone, Copy, Debug, PartialEq, Reflect)]
  37 enum CameraView {}
  38 impl GeometryFrame for CameraView {}
  39 
  40 /// A conservative box enclosing the eye and all four near-plane corners.
  41 /// Its orientation and dimensions follow the actual render projection.
  42 pub(crate) struct CameraVolume {
  43     pub(crate) shape: Collider,
  44     local_center: Displacement<CameraView>,
  45     size: CuboidDimensions,
  46 }
  47 
  48 #[derive(Clone, Copy, Debug)]
  49 pub(crate) struct CameraVolumeDebug {
  50     centre: Position<Scene>,
  51     rotation: RigidRotation,
  52     size: CuboidDimensions,
  53 }
  54 
  55 #[derive(Clone, Copy, Debug)]
  56 pub(crate) struct CameraHit {
  57     pub(crate) entity: Entity,
  58     pub(crate) normal: SpatialDirection<Scene>,
  59 }
  60 
  61 struct CameraSweep {
  62     fraction: UnitInterval,
  63     hit: Option<CameraHit>,
  64 }
  65 
  66 pub(crate) struct CameraFrame {
  67     /// Controller capsule center: a collision origin independent of focus lag.
  68     pub(crate) origin: Position<Scene>,
  69     pub(crate) anchor: Position<Scene>,
  70     pub(crate) focus: Position<Scene>,
  71     pub(crate) rotation: RigidRotation,
  72     pub(crate) shoulder_offset: SignedLength,
  73     pub(crate) distance: SignedLength,
  74 }

  76 pub(crate) struct CameraPlacement {

  78     pub(crate) desired: Position<Scene>,
