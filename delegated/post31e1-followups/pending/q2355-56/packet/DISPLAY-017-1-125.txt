   1 //! Orbit, proportional zoom and camera-relative panning for every exhibit.
   2 
   3 use crate::presentation::TacticalGameplayCamera;
   4 use bevy::prelude::*;
   5 use bevy::{camera::Exposure, light::AmbientLight};
   6 
   7 pub(super) const VERTICAL_FOV_DEGREES: f32 = 45.0;
   8 const ORBIT_RADIANS_PER_PIXEL: f32 = 0.006;
   9 const ZOOM_LOG_SCALE_PER_PIXEL: f32 = 0.001;
  10 const MAX_PITCH_RADIANS: f32 = 1.35;
  11 const STUDIO_EXPOSURE_EV100: f32 = 12.5;
  12 const STUDIO_AMBIENT_BRIGHTNESS: f32 = 155.0;
  13 const PAN_DISTANCE_PER_SECOND: f32 = 0.6;
  14 const MAX_INPUT_SECONDS: f32 = 0.05;
  15 
  16 #[derive(Clone, Copy, PartialEq, Eq)]
  17 pub(super) enum CameraSpace {
  18     Studio,
  19     Landscape,
  20 }
  21 
  22 #[derive(Clone, Resource)]
  23 pub(super) struct OrbitView {
  24     pub focus: Vec3,
  25     pub distance: f32,
  26     pub yaw: f32,
  27     pub pitch: f32,
  28     pub limits: std::ops::RangeInclusive<f32>,
  29     pub space: CameraSpace,
  30 }
  31 
  32 impl Default for OrbitView {
  33     fn default() -> Self {
  34         Self::new(Vec3::ZERO, 1.0, CameraSpace::Studio)
  35     }
  36 }
  37 
  38 impl OrbitView {
  39     pub fn new(focus: Vec3, distance: f32, space: CameraSpace) -> Self {
  40         Self {
  41             focus,
  42             distance,
  43             yaw: 0.45,
  44             pitch: 0.16,
  45             limits: distance * 0.45..=distance * 2.5,
  46             space,
  47         }
  48     }
  49 
  50     pub fn orbit(&mut self, x: f32, y: f32) {
  51         self.yaw = (self.yaw - x * ORBIT_RADIANS_PER_PIXEL).rem_euclid(std::f32::consts::TAU);
  52         let minimum = if self.space == CameraSpace::Landscape {
  53             0.05
  54         } else {
  55             -MAX_PITCH_RADIANS
  56         };
  57         self.pitch = (self.pitch + y * ORBIT_RADIANS_PER_PIXEL).clamp(minimum, MAX_PITCH_RADIANS);
  58     }
  59 
  60     pub fn zoom(&mut self, delta: f32) {
  61         self.distance = (self.distance * (delta * ZOOM_LOG_SCALE_PER_PIXEL).exp())
  62             .clamp(*self.limits.start(), *self.limits.end());
  63     }
  64 
  65     pub fn pan(&mut self, right: f32, forward: f32, seconds: f32) {
  66         let input = Vec2::new(right, forward).clamp_length_max(1.0);
  67         let right = Vec3::new(self.yaw.cos(), 0.0, -self.yaw.sin());
  68         let forward = if self.space == CameraSpace::Landscape {
  69             Vec3::new(-self.yaw.sin(), 0.0, -self.yaw.cos())
  70         } else {
  71             Vec3::new(
  72                 -self.yaw.sin() * self.pitch.sin(),
  73                 self.pitch.cos(),
  74                 -self.yaw.cos() * self.pitch.sin(),
  75             )
  76         };
  77         self.focus += (right * input.x + forward * input.y)
  78             * self.distance
  79             * PAN_DISTANCE_PER_SECOND
  80             * seconds.clamp(0.0, MAX_INPUT_SECONDS);
  81     }
  82 }
  83 
  84 pub(super) fn apply(
  85     mut commands: Commands,
  86     view: Res<OrbitView>,
  87     mut cameras: Query<(Entity, &mut Transform), With<TacticalGameplayCamera>>,
  88 ) {
  89     for (entity, mut camera) in &mut cameras {
  90         if view.space == CameraSpace::Studio {
  91             if view.is_changed() {
  92                 commands.entity(entity).insert(AmbientLight {
  93                     color: Color::srgb(0.78, 0.84, 0.94),
  94                     brightness: STUDIO_AMBIENT_BRIGHTNESS,
  95                     ..default()
  96                 });
  97             }
  98         } else if view.is_changed() {
  99             commands.entity(entity).remove::<AmbientLight>();
 100         }
 101         let direction = Vec3::new(
 102             view.yaw.sin() * view.pitch.cos(),
 103             view.pitch.sin(),
 104             view.yaw.cos() * view.pitch.cos(),
 105         );
 106         *camera = Transform::from_translation(view.focus + direction * view.distance)
 107             .looking_at(view.focus, Vec3::Y);
 108     }
 109 }
 110 
 111 // Production interior adaptation runs after transform propagation. Apply the
 112 // studio exposure afterward, leaving outdoor adaptation in charge of landscapes.
 113 pub(super) fn studio_exposure(
 114     view: Res<OrbitView>,
 115     mut cameras: Query<&mut Exposure, With<TacticalGameplayCamera>>,
 116 ) {
 117     if view.space == CameraSpace::Studio {
 118         for mut exposure in &mut cameras {
 119             exposure.ev100 = STUDIO_EXPOSURE_EV100;
 120         }
 121     }
 122 }
