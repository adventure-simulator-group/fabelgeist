 121     pub base_elevation_metres: crate::city_layout::grounding::SupportElevation,

 123 }

 125 impl DistantBuildingPlacement {
 126     /// The occupied recipe is retained for promotion into a playable venue.
 127     pub fn occupied_program(self) -> BuildingProgram {
 128         let mut program = match self.usage {
 129             Some(usage) => BuildingProgram::settlement(self.archetype, Some(usage), self.seed),
 130             None => BuildingProgram::fixture(self.archetype, self.seed),
 131         };
 132         if let Some(size) = self.service_size {
 133             program = program.with_service_size(size);
 134         }
 135         program
 136     }
 137 }
