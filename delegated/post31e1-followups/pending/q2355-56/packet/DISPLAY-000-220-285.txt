 220         }
 221         CrenellationMask => vec![(
 222             MapChannel::Opacity,
 223             crate::generate_crenellation_mask(params, images),
 224         )],
 225         _ => unreachable!("surface and leaf recipes are dispatched before specialized maps"),
 226     }
 227 }
