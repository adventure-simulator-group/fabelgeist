  12 macro_rules! parameter_block {
  13     (pub struct $name:ident { $($field:ident: $ty:ty = $default:expr;)* }) => {
  14         #[derive(Clone, serde::Serialize, serde::Deserialize)]
  15         #[serde(deny_unknown_fields)]
  16         pub struct $name { $(pub $field: $ty,)* }
  17         impl Default for $name {
  18             fn default() -> Self { Self { $($field: $default,)* } }
  19         }
  20     };
  21 }
  22 pub(crate) use parameter_block;

  54 /// A cap changes sampling density, never the physical size or authored feature count.
  55 #[derive(Clone, Copy, Debug, PartialEq, Eq, Default, Serialize, Deserialize)]
  56 pub enum BakeResolution {
  57     Draft,
  58     Medium,
  59     #[default]
  60     Full,
  61 }
  62 impl TextureParameters {
  63     pub(crate) fn size(&self, native: u32) -> u32 {
  64         match self.resolution {
  65             BakeResolution::Draft => native.min(128),
  66             BakeResolution::Medium => native.min(256),
  67             BakeResolution::Full => native,
  68         }
  69     }
  70 }
  71 impl TextureParameters {
  72     /// A recipe owns each purpose; context identifies authored features or cells.
  73     pub(crate) fn rng(
  74         &self,
  75         purpose: fabelgeist_determinism::StreamId,
  76         context: &[u64],
  77     ) -> fabelgeist_determinism::DeterministicRng {
  78         purpose.rng(self.seed, context)
  79     }
  80 
  81     /// Frame a derived surface-element seed as the sole context word while
  82     /// retaining this material's root and the caller-owned stream purpose.
  83     pub(crate) fn element_rng(
  84         &self,
  85         stream: fabelgeist_determinism::StreamId,
  86         element: fabelgeist_determinism::Seed,
  87     ) -> fabelgeist_determinism::DeterministicRng {
  88         self.rng(stream, &[element.to_u64()])
  89     }
  90 
  91     pub(crate) fn field_seed(
  92         &self,
  93         purpose: fabelgeist_determinism::StreamId,
  94         context: &[u64],
  95     ) -> Seed {
  96         purpose.seed(self.seed, context)
  97     }
  98 }
