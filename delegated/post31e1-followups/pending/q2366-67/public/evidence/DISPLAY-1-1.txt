  71     let cases = [
  72         DirectoryCase {
  73             label: "stored-deflated",

  75         },
  76         DirectoryCase {
  77             label: "nearest-end-is-empty",
  78             names: Some(&[]),
  79         },
  80         DirectoryCase {
  81             label: "nearest-end-selects-second",
  82             names: Some(&["second.npy"]),
  83         },
  84         DirectoryCase {
  85             label: "end-with-comment-candidate",
  86             names: Some(&[]),
  87         },
  88         DirectoryCase {
  89             label: "end-window-inclusive",
  90             names: Some(&["values.npy", "second.npy"]),
  91         },
  92         DirectoryCase {
  93             label: "unknown-central-3735928559",
  94             names: None,
  95         },
  96         DirectoryCase {
  97             label: "central-second-is-local",
  98             names: None,
  99         },
 100         DirectoryCase {
 101             label: "zip64-count-saturated",
 102             names: None,
 103         },
 104         DirectoryCase {
 105             label: "zip64-offset-saturated",
 106             names: None,
 107         },
 108         DirectoryCase {
 109             label: "zip64-locator-mismatch",
 110             names: None,
 111         },
 112         DirectoryCase {
 113             label: "zip64-end-mismatch",
 114             names: None,
 115         },
 116         DirectoryCase {
 117             label: "zip64-locator-known-wrong",
 118             names: None,
 119         },
 120         DirectoryCase {
 121             label: "zip64-end-known-wrong",
 122             names: None,
 123         },
 124         DirectoryCase {
 125             label: "zip64-record-outside",
 126             names: None,
 127         },
 128         DirectoryCase {
 129             label: "zip64-not-triggered",
 130             names: Some(&["values.npy", "second.npy"]),
 131         },
 132         DirectoryCase {
 133             label: "short-end-no-locator",
 134             names: Some(&[]),
 135         },
 136     ];
 137     for case in cases {
 138         let archive = ZipArchive::from_bytes(fixture(case.label).to_vec());

 140             archive
 141                 .map(|archive| archive
 142                     .names()
 143                     .map(|name| name.to_string())
 144                     .collect::<Vec<_>>())
 145                 .map_err(|error| matches!(
 146                     error,
 147                     fabelgeist_numpy_storage::zip::ZipReadError::Archive { .. }
 148                 )),
 149             case.names
 150                 .map(|names| names
 151                     .iter()
 152                     .map(|name| name.to_string())
 153                     .collect::<Vec<_>>())
 154                 .ok_or(true),
 155             "{}",
 156             case.label
 157         );
 158     }
 159     let error = ZipArchive::from_bytes(fixture("end-window-excluded").to_vec())
 160         .err()
 161         .unwrap();
 162     assert_eq!(
 163         error.to_string(),
 164         "not a zip archive: no end-of-central-directory record"
 165     );

 191 fn assert_member_storage_and_values(archive: &ZipArchive, npz: &Npz) {
 192     let stored = archive
 193         .bytes(&ArchiveMemberName::from("values.npy"))
 194         .unwrap();
 195     assert!(matches!(&stored, Cow::Borrowed(_)));
 196     assert_eq!(
 197         stored.as_ptr(),
 198         archive
 199             .bytes(&ArchiveMemberName::from("values.npy"))
 200             .unwrap()
 201             .as_ptr()
 202     );
 203     assert!(matches!(
 204         archive
 205             .bytes(&ArchiveMemberName::from("second.npy"))
 206             .unwrap(),
 207         Cow::Owned(_)
 208     ));
 209     assert_eq!(
 210         npz.keys().map(|name| name.to_string()).collect::<Vec<_>>(),
 211         ["values", "second"]
 212     );
 213     assert_eq!(
 214         npz.array(&NpzArrayName::new("values"))
 215             .unwrap()
 216             .shape()
 217             .dimensions(),
 218         [fabelgeist_numpy_storage::NpyDimension::from(4)]
 219     );
 220     assert_eq!(
 221         Vec::<f32>::from(fabelgeist_numpy_storage::NpyFloat32Values::from(
 222             &npz.array(&NpzArrayName::new("values")).unwrap()
 223         ))
 224         .iter()
 225         .map(|v| v.to_bits())
 226         .collect::<Vec<_>>(),
 227         [0x3f800000, 0x80000000, 0x7fc01234, 0x40200000]
 228     );
 229     assert_eq!(
 230         Vec::<f32>::from(fabelgeist_numpy_storage::NpyFloat32Values::from(
 231             &npz.array(&NpzArrayName::new("second")).unwrap()
 232         )),
 233         [5.0, 6.0, 7.0, 8.0]
 234     );
 235 }

 260 #[test]
 261 fn duplicate_and_disagreeing_headers_keep_member_semantics() {
 262     let bytes = fixture("duplicate-members");
 263     let archive = ZipArchive::from_bytes(bytes.to_vec()).unwrap();
 264     assert_eq!(
 265         archive
 266             .names()
 267             .map(|name| name.to_string())
 268             .collect::<Vec<_>>(),
 269         ["values.npy", "values.npy"]
 270     );
 271     assert!(matches!(
 272         archive
 273             .bytes(&ArchiveMemberName::from("values.npy"))
 274             .unwrap(),
 275         Cow::Borrowed(_)
 276     ));
 277     let npz = Npz::from_bytes(bytes.to_vec()).unwrap();
 278     assert_eq!(
 279         Vec::<f32>::from(fabelgeist_numpy_storage::NpyFloat32Values::from(
 280             &npz.array(&NpzArrayName::new("values")).unwrap()
 281         ))[0],
 282         1.0
 283     );
 284     for label in [
 285         "stored-central-deflate-local",
 286         "deflate-central-stored-local",
 287         "different-local-name-extra",
 288         "zip64-entry-extra",
 289     ] {
 290         let archive = ZipArchive::from_bytes(fixture(label).to_vec()).unwrap();
 291         assert_eq!(
 292             archive.uncompressed_size(&ArchiveMemberName::from("values.npy")),
 293             Some(DeclaredUncompressedByteLength::from_bytes(96))
 294         );
 295         assert_eq!(
 296             matches!(
 297                 archive
 298                     .bytes(&ArchiveMemberName::from("values.npy"))
 299                     .unwrap(),
 300                 Cow::Owned(_)
 301             ),
 302             label == "deflate-central-stored-local"
 303         );
 304         let npz = Npz::from_bytes(fixture(label).to_vec()).unwrap();
 305         assert_eq!(
 306             Vec::<f32>::from(fabelgeist_numpy_storage::NpyFloat32Values::from(
 307                 &npz.array(&NpzArrayName::new("values")).unwrap()
 308             ))[0],
 309             1.0,
 310             "{label}"
 311         );
 312     }
