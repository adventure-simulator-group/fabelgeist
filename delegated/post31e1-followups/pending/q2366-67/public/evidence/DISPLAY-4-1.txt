  65     pub fn open(path: impl AsRef<Path>) -> Result<Self> {
  66         let path = path.as_ref();
  67         let file = std::fs::File::open(path).map_err(|source| ZipReadError::Native {
  68             operation: ZipFileOperation::Open,

  75             operation: ZipFileOperation::Map,

  78         })?;

  86     fn from_data(data: ArchiveData) -> Result<Self> {
  87         let archive = ZipLocator::new()
  88             .max_search_space(END_DIRECTORY_SEARCH_WINDOW_BYTES)
  89             .locate_in_slice(data)
  90             .map_err(|(_, source)| ZipReadError::from(source))?;
  91         let member_count = ArchiveMemberCount::from(archive.entries_hint());
  92         let mut entries = Vec::with_capacity(member_count.initial_capacity());
  93         let mut directory = archive.entries();
  94         // The declaration remains our upper bound; rawzip bounds each record
  95         // by the selected directory span. No partial archive is published on error.
  96         for _ in member_count.native_iteration_range() {
  97             let Some(header) = directory.next_entry().map_err(ZipReadError::from)? else {
  98                 break;
  99             };
 100             entries.push(Entry {
 101                 name: ArchiveMemberName::from(header.file_path().as_ref()),
 102                 compression: ZipCompression::from(header.compression_method()),
 103                 uncompressed_size: DeclaredUncompressedByteLength::from_bytes(
 104                     header.uncompressed_size_hint(),
 105                 ),
 106                 location: header.wayfinder(),
 107             });
 108         }
 109         Ok(Self { archive, entries })
 110     }

 151 impl<'archive> ZipMember<'archive> {
 152     pub(crate) fn name(&self) -> &ArchiveMemberName {
 153         &self.entry.name
 154     }
 155 
 156     pub(crate) fn uncompressed_size(&self) -> DeclaredUncompressedByteLength {
 157         self.entry.uncompressed_size
 158     }

 160     pub(crate) fn bytes(&self) -> Result<Cow<'archive, [u8]>> {
 161         let entry = self.entry;
 162         let member = self
 163             .archive
 164             .archive
 165             .get_entry(entry.location)
 166             .map_err(|source| ZipReadError::member(entry.name.clone(), source))?;
 167         let raw = member.data();
 168         match entry.compression {
 169             ZipCompression::Stored => Ok(Cow::Borrowed(raw)),
 170             ZipCompression::Deflated => {
 171                 let mut out =
 172                     Vec::with_capacity(entry.uncompressed_size.native_allocation_capacity());
 173                 DeflateDecoder::new(raw)
 174                     .read_to_end(&mut out)
 175                     .map_err(|source| ZipReadError::Inflate {
 176                         name: entry.name.clone(),
 177                         source,
 178                     })?;
 179                 Ok(Cow::Owned(out))
 180             }
 181             ZipCompression::Unsupported(code) => Err(ZipReadError::UnsupportedCompression { code }),
 182         }
 183     }
