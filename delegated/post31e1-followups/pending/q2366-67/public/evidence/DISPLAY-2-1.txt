  44     // Native fixture encoding port: the end record names the authored central
  45     // header; only its compression word is replaced with a rejected wire code.
  46     // Earlier local-header/extent checks must still win over that method.
  47     let mut corrupt = include_bytes!("fixtures/zip-errors/corrupt-local.zip").to_vec();

  49         corrupt[corrupt.len() - 6..corrupt.len() - 2]
  50             .try_into()
  51             .unwrap(),

  54     let archive = ZipArchive::from_bytes(corrupt).unwrap();
  55     let error = rejected(archive.bytes(&ArchiveMemberName::from("values.npy")));

  57         matches!(&error, ZipReadError::CorruptMember { name, .. } if name == &ArchiveMemberName::from("values.npy"))

  60     assert!(
  61         error

 104 #[test]
 105 fn inflate_failure_retains_the_native_cause_through_npz_anyhow() {
 106     let input = include_bytes!("fixtures/zip-errors/bad-deflate.zip");
 107     let archive = ZipArchive::from_bytes(input.to_vec()).unwrap();
 108     let error = rejected(archive.bytes(&ArchiveMemberName::from("values.npy")));
 109     let ZipReadError::Inflate { name, source } = &error else {
 110         panic!("unexpected classification: {error:?}")
 111     };
 112     assert_eq!(name, &ArchiveMemberName::from("values.npy"));
 113     assert_eq!(source.kind(), io::ErrorKind::InvalidInput);
 114     assert_eq!(source.to_string(), "corrupt deflate stream");
 115     let retained = error.source().unwrap().downcast_ref::<io::Error>().unwrap();
 116     assert!(std::ptr::eq(source, retained));
 117     assert_eq!(error.to_string(), "inflating zip entry values.npy");
 118 
 119     let npz = Npz::from_bytes(input.to_vec()).unwrap();
 120     let outer = npz.array(&NpzArrayName::new("values")).err().unwrap();
 121     assert!(matches!(
 122         outer.downcast_ref::<ZipReadError>(),
 123         Some(ZipReadError::Inflate { .. })
 124     ));
 125     assert!(outer.downcast_ref::<io::Error>().is_none());
 126     assert_eq!(
 127         outer.chain().map(ToString::to_string).collect::<Vec<_>>(),
 128         ["inflating zip entry values.npy", "corrupt deflate stream"]
 129     );
 130     assert_eq!(
 131         outer
 132             .chain()
 133             .find_map(|cause| cause.downcast_ref::<io::Error>())
 134             .unwrap()
 135             .kind(),
 136         io::ErrorKind::InvalidInput
 137     );
 138 }

 189 #[test]
 190 fn successful_stored_and_deflated_reads_preserve_borrowing_and_float_bits() {
 191     let stored = ZipArchive::from_bytes(STORED.to_vec()).unwrap();
 192     let deflated = ZipArchive::from_bytes(DEFLATED.to_vec()).unwrap();
 193     let borrowed = stored
 194         .bytes(&ArchiveMemberName::from("values.npy"))
 195         .unwrap();
 196     let repeated = stored
 197         .bytes(&ArchiveMemberName::from("values.npy"))
 198         .unwrap();
 199     let decoded = deflated
 200         .bytes(&ArchiveMemberName::from("values.npy"))
 201         .unwrap();
 202     assert!(matches!(borrowed, Cow::Borrowed(_)));
 203     assert!(matches!(decoded, Cow::Owned(_)));
 204     assert!(std::ptr::eq(borrowed.as_ptr(), repeated.as_ptr()));
 205     assert_eq!(borrowed, decoded);
 206     for input in [STORED, DEFLATED] {
 207         let npz = Npz::from_bytes(input.to_vec()).unwrap();
 208         let array = npz.array(&NpzArrayName::new("values")).unwrap();
 209         assert_eq!(
 210             Vec::<f32>::from(NpyFloat32Values::from(&array))
 211                 .iter()
 212                 .map(|value| value.to_bits())
 213                 .collect::<Vec<_>>(),
 214             [0, 0x8000_0000, 1, 0x7f80_0000, 0xff80_0000, 0x7fc0_1234]
 215         );
 216     }
