  196     /// Signed birth coordinate for an age at this instant, capped to storage.
  197     pub fn signed_birth_minute_for_age(self, age_years: u16) -> i64 {
  198         let birth = i128::from(self.minutes)
  199             .saturating_sub(i128::from(age_years) * i128::from(MINUTES_PER_YEAR));
  200         birth.clamp(i128::from(i64::MIN), i128::from(i64::MAX)) as i64
  201     }

  203     pub fn elapsed_minutes_since_signed_birth(self, birth_minute: i64) -> u128 {
  204         (i128::from(self.minutes) - i128::from(birth_minute)).max(0) as u128
  205     }

  207     /// Progress toward an age threshold in caller-selected integer units.
  208     pub fn scaled_progress_toward_signed_birth_age(
  209         self,
  210         birth_minute: i64,
  211         target_years: u16,
  212         whole: u16,
  213     ) -> u16 {
  214         if target_years == 0 {
  215             return whole;
  216         }
  217         let elapsed = self.elapsed_minutes_since_signed_birth(birth_minute);
  218         let target = u128::from(target_years) * u128::from(MINUTES_PER_YEAR);
  219         elapsed
  220             .saturating_mul(u128::from(whole))
  221             .div_euclid(target)
  222             .min(u128::from(whole)) as u16
  223     }
