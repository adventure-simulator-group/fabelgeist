   1 /// Blood exposure replay over a bounded absolute interval.
   2 #[expect(
   3     clippy::too_many_arguments,
   4     reason = "exposure replay keeps each bounded cursor and authority input explicit"
   5 )]
   6 pub fn blood_exposure_attempts_through(
   7     ctx: &ReducerContext,
   8     character_id: u64,
   9     from: StrategicMinute,
  10     to: StrategicMinute,
  11     persist_checkpoint: bool,
  12     allow_healing: bool,
  13     plan: Option<&crate::disease::PartyDiseaseIntervalPlan>,
  14     max_work: u64,
  15 ) -> Result<Vec<adventuresim_core::disease::AcquisitionAttempt>, String> {
  16     if to <= from {
  17         return Ok(Vec::new());
  18     }
  19     let all_deposits = deposits(ctx, character_id)?;
  20     if !exposure_windows::has_active_compatible_foreign_blood(&all_deposits, from, to) {
  21         return Ok(Vec::new());
  22     }
  23     let routes = predicted_wound_routes(ctx, character_id, allow_healing)?;
  24     let mut attempts = Vec::new();
  25     let mut work = 0;
  26     for disease_id in adventuresim_core::disease::STARTER_DISEASES
  27         .iter()
  28         .filter(|definition| {
  29             definition.supports(adventuresim_core::disease::TransmissionVector::Blood)
  30         })
  31         .map(|definition| definition.id)
  32     {
  33         let key = format!("{character_id}:{}", disease_id.stable_id());
  34         let checkpoint = ctx.db.blood_exposure_checkpoint().id().find(&key);
  35         let start = from.saturating_add_minutes(1).max(
  36             checkpoint.as_ref().map_or(StrategicMinute::ZERO, |row| row.evaluated_through.saturating_add_minutes(1)),
  37         );
  38         let relevant = relevant_blood_deposits(&all_deposits, disease_id);
  39         let windows = exposure_windows::blood_infectious_windows(
  40             &relevant,
  41             disease_id,
  42             start.saturating_sub_minutes(1),
  43             to,
  44         );
  45         for minute in windows
  46             .into_iter()
  47             .flat_map(|(window_start, window_end)| window_start.iter_through(window_end))
  48         {
  49             adventuresim_core::disease::add_bounded_work(&mut work, 1, max_work)
  50                 .map_err(str::to_string)?;
  51             let route = filth::timed_cut_exposure(
  52                 &routes,
  53                 minute.elapsed_since(from),
  54             );
  55             let check = plan.map_or_else(
  56                 || crate::disease::party_physiology_check_at(ctx, character_id, minute),
  57                 |plan| {
  58                     plan.check_at(
  59                         character_id,
  60                         minute,
  61                     )
  62                 },
  63             );
  64             // Any infectious deposit still present here survived or preceded
  65             // explicit washing, so clean handling is not available for this
  66             // dose. Bandaging/stitching still raises the physical affordance
  67             // through the predicted cut-route state.
  68             let affordance = adventuresim_core::disease::blood_caregiving_affordance(false, route);
  69             let exposure = adventuresim_core::disease::residual_exposure_with_affordance(
  70                 filth::blood_exposure(&relevant, disease_id, minute, route)
  71                     / MINUTES_PER_DAY as f32,
  72                 adventuresim_core::disease::TransmissionVector::Blood,
  73                 check,
  74                 affordance,
  75             );
  76             if exposure <= 0.0 {
  77                 continue;
  78             }
  79             let seed = adventuresim_core::disease::outbreak_exposure_seed(
  80                 character_id,
  81                 &format!("blood:{}:{minute}", disease_id.stable_id()),
  82             );
  83             attempts.push(adventuresim_core::disease::AcquisitionAttempt::exposure(
  84                 adventuresim_core::disease::InfectionEpisode {
  85                     id: seed.to_u64(),
  86                     character_id,
  87                     disease_id,
  88                     contracted_at: minute,
  89                     ruleset_version: adventuresim_core::physiology::PHYSIOLOGY_RULESET_VERSION,
  90                     phenotype_key_version: adventuresim_core::physiology::PHENOTYPE_KEY_VERSION,
  91                 },
  92                 adventuresim_core::disease::definition(disease_id).base_acquisition,
  93                 exposure,
  94             ));
  95         }
  96         if persist_checkpoint {
  97             persist_blood_checkpoint(ctx, key, character_id, disease_id, to, checkpoint.is_some());
  98         }
  99     }
 100     Ok(attempts)
 101 }
 102 
 103 fn relevant_blood_deposits(
 104     all_deposits: &[Deposit],
 105     disease_id: adventuresim_core::disease::DiseaseId,
 106 ) -> Vec<Deposit> {
 107     all_deposits
 108         .iter()
 109         .filter(|deposit| {
 110             deposit.foreign_blood()
 111                 && deposit
 112                     .diseases
 113                     .iter()
 114                     .any(|snapshot| snapshot.disease_id == disease_id)
 115         })
 116         .cloned()
 117         .collect()
 118 }
 119 
 120 fn persist_blood_checkpoint(
 121     ctx: &ReducerContext,
 122     key: String,
 123     character_id: u64,
 124     disease_id: adventuresim_core::disease::DiseaseId,
 125     to: StrategicMinute,
 126     existed: bool,
 127 ) {
 128     let row = BloodExposureCheckpoint {
 129         id: key,
 130         character_id,
 131         disease_id: disease_id.stable_id().into(),
 132         evaluated_through: to,
 133     };
 134     if existed {
 135         ctx.db.blood_exposure_checkpoint().id().update(row);
 136     } else {
 137         ctx.db.blood_exposure_checkpoint().insert(row);
 138     }
 139 }

 141 #[cfg(test)]
 142 mod source_tests {
 143     #[test]
 144     fn blood_route_receives_partial_physician_protection_after_physical_controls() {
 145         let source = crate::production_source(include_str!("blood_attempts.rs"));
 146         let exposure = source
 147             .split("pub fn blood_exposure_attempts_through")
 148             .nth(1)
 149             .expect("blood exposure source");
 150         let prevention = exposure.find("residual_exposure").unwrap();
 151         let physical = exposure.find("filth::blood_exposure").unwrap();
 152         assert!(prevention < physical);
 153         assert!(exposure.contains("TransmissionVector::Blood"));
 154         assert!(exposure.contains("timed_cut_exposure"));
 155         assert!(exposure.contains("blood_caregiving_affordance(false, route)"));
 156     }
 157 }
