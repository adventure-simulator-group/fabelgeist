1: //! NumPy array storage for Burn.
2: //!
3: //! Reads `.npy` files and `.npz` archives (stored or deflated, zip64 included)
4: //! and retains decoded values in bespoke collections until Burn tensor adapters.
5: //! Pure Rust: no NumPy, no C zlib.
6: //!
7: //! ```no_run
8: //! # fn main() -> anyhow::Result<()> {
9: //! use burn::tensor::Device;
10: //! use fabelgeist_numpy_storage::{Npz, NpzArrayName};
11: //!
12: //! let archive = Npz::open("weights.npz")?;
13: //! let weights = archive.array(&NpzArrayName::new("layer0"))?.to_tensor::<2>(&Device::default())?;
14: //! # let _ = weights;
15: //! # Ok(())
16: //! # }
17: //! ```
18:
