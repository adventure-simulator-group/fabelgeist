  64 | impl InflateBackend for Inflate {
  65 |     fn make(zlib_header: bool, _window_bits: u8) -> Self {
  66 |         let format = format_from_bool(zlib_header);
  67 | 
  68 |         Inflate {
  69 |             inner: InflateState::new_boxed(format),
  70 |             total_in: 0,
  71 |             total_out: 0,
  72 |         }
  73 |     }
  74 | 
  75 |     fn decompress(
  76 |         &mut self,
  77 |         input: &[u8],
  78 |         output: &mut [u8],
  79 |         flush: FlushDecompress,
  80 |     ) -> Result<Status, DecompressError> {
  81 |         let mz_flush = flush.into();
  82 |         let res = inflate::stream::inflate(&mut self.inner, input, output, mz_flush);
  83 |         self.total_in += res.bytes_consumed as u64;
  84 |         self.total_out += res.bytes_written as u64;
  85 | 
  86 |         match res.status {
  87 |             Ok(status) => match status {
  88 |                 MZStatus::Ok => Ok(Status::Ok),
  89 |                 MZStatus::StreamEnd => Ok(Status::StreamEnd),
  90 |                 MZStatus::NeedDict => {
  91 |                     mem::decompress_need_dict(self.inner.decompressor().adler32().unwrap_or(0))
  92 |                 }
  93 |             },
  94 |             Err(status) => match status {
  95 |                 MZError::Buf => Ok(Status::BufError),
  96 |                 _ => mem::decompress_failed(ErrorMessage),
  97 |             },
  98 |         }
  99 |     }
 100 | 
 101 |     fn reset(&mut self, zlib_header: bool) {
 102 |         self.inner.reset(format_from_bool(zlib_header));
 103 |         self.total_in = 0;
 104 |         self.total_out = 0;
 105 |     }
 106 | }
