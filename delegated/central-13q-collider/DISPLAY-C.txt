 132     /// Write one host upload at a checked byte address within this buffer.
 133     pub fn write_at(
 134         &self,
 135         context: &WgpuContext,
 136         at: BufferByteOffset,
 137         data: BufferUpload<'_>,
 138     ) -> Result<()> {
 139         let end = at
 140             .checked_after(data.length())
 141             .ok_or_else(|| anyhow!("Buffer write overflows an offset"))?;
 142         if u64::from(end) > u64::from(self.size) {
 143             return Err(anyhow!(
 144                 "Writing {} bytes at {at} runs past the end of a {}-byte buffer",
 145                 data.length(),
 146                 self.size
 147             ));
 148         }
 149         data.write_to(context, &self.buffer, at);
 150         Ok(())
