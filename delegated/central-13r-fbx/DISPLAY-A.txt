 425     raw: &[u8],
 426     count: usize,
 427     encoding: FbxArrayEncoding,
 428     parse: impl Fn(&[u8]) -> T,
 429     width: usize,
 430 ) -> Result<Vec<T>> {
 431     let bytes = match encoding {
 432         FbxArrayEncoding::Uncompressed => raw.to_vec(),
 433         FbxArrayEncoding::Zlib => {
 434             let mut out = Vec::with_capacity(count * width);
 435             ZlibDecoder::new(raw)
 436                 .read_to_end(&mut out)
 437                 .context("inflating FBX array property")?;
 438             out
 439         }
 440     };
 441     if bytes.len() < count * width {
 442         bail!(
 443             "FBX array property is short: {} bytes for {count} x {width}",
 444             bytes.len()
 445         );
 446     }
 447     Ok((0..count)
 448         .map(|i| parse(&bytes[i * width..(i + 1) * width]))
