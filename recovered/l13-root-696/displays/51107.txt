16: impl ReflectedInterface {
17:     pub(super) fn new(context: &WgpuContext, code: &str) -> Result<Self> {
18:         let module = parse_naga(code, wgpu::naga::ShaderStage::Compute)
19:             .map_err(|e| anyhow!("Compute Shader Parse Error for Reflection: {e}"))?;
20:         let entry = module
21:             .entry_points
22:             .iter()
23:             .find(|entry| entry.stage == wgpu::naga::ShaderStage::Compute)
24:             .ok_or_else(|| anyhow!("Compute Shader missing entry point"))?;
25:         let mut groups = BTreeMap::new();
26:         for (_, variable) in module.global_variables.iter() {
27:             let Some(binding) = &variable.binding else {
28:                 continue;
29:             };
30:             groups
31:                 .entry(binding.group)
32:                 .or_insert_with(|| GroupBuilder::new(binding.group))
33:                 .reflect(&module, variable, binding.binding);
34:         }
35:         let mut layouts = Vec::new();
36:         let mut reflected = Vec::new();
37:         for (index, mut group) in groups {
38:             group.entries.sort_by_key(|entry| entry.binding);
39:             layouts.push(Arc::new(context.device.create_bind_group_layout(
40:                 &wgpu::BindGroupLayoutDescriptor {
41:                     label: Some(&format!("ComputePipeline Bind Layout Group {index}")),
42:                     entries: &group.entries,
43:                 },
44:             )));
45:             reflected.push(group.reflection);
46:         }
47:         Ok(Self {
48:             reflection: Arc::new(ReflectionData {
49:                 bind_groups: reflected,
50:                 fragment_entry_point: String::new(),
51:                 vertex_entry_point: entry.name.clone(),
52:             }),
53:             layouts,
54:         })
55:     }
56: }

69:     fn reflect(
70:         &mut self,
71:         module: &wgpu::naga::Module,
72:         variable: &wgpu::naga::GlobalVariable,
73:         binding: u32,
74:     ) {
75:         let name = variable.name.clone().unwrap_or_default();
76:         match variable.space {
77:             wgpu::naga::AddressSpace::Uniform => {
78:                 self.uniform(module, variable.ty, binding);
79:             }
80:             wgpu::naga::AddressSpace::Storage { access } => {
81:                 let ty = wgpu::BufferBindingType::Storage {
82:                     read_only: !access.contains(wgpu::naga::StorageAccess::STORE),
83:                 };
84:                 self.reflection
85:                     .buffer_bindings
86:                     .push(BufferBinding { name, binding, ty });
87:                 self.entry(
88:                     binding,
89:                     wgpu::BindingType::Buffer {
90:                         ty,
91:                         has_dynamic_offset: false,
92:                         min_binding_size: None,
93:                     },
94:                 );
95:             }
96:             wgpu::naga::AddressSpace::Handle => match &module.types[variable.ty].inner {
97:                 wgpu::naga::TypeInner::Image { dim, class, .. } => {
98:                     self.image(name, binding, *dim, class);
99:                 }
100:                 wgpu::naga::TypeInner::Sampler { .. } => {
101:                     self.reflection.sampler_bindings.push((name, binding));
102:                     self.entry(
103:                         binding,
104:                         wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Filtering),
105:                     );
106:                 }
107:                 _ => {}
108:             },
109:             _ => {}
110:         }
111:     }

113:     fn uniform(
114:         &mut self,
115:         module: &wgpu::naga::Module,
116:         ty: wgpu::naga::Handle<wgpu::naga::Type>,
117:         binding: u32,
118:     ) {
119:         let ty = &module.types[ty];
120:         self.reflection.uniform_binding = Some(binding);
121:         if let wgpu::naga::TypeInner::Struct { members, span } = &ty.inner {
122:             self.reflection.uniform_buffer_size = *span;
123:             for member in members {
124:                 self.reflection.uniform_members.push(UniformMember {
125:                     name: member.name.clone().unwrap_or_default(),
126:                     offset: member.offset,
127:                     size: module.types[member.ty].inner.size(module.to_ctx()),
128:                 });
129:             }
130:         } else {
131:             self.reflection.uniform_buffer_size = ty.inner.size(module.to_ctx());
132:         }
133:         self.entry(
134:             binding,
135:             wgpu::BindingType::Buffer {
136:                 ty: wgpu::BufferBindingType::Uniform,
137:                 has_dynamic_offset: false,
138:                 min_binding_size: None,
139:             },
140:         );
141:     }

143:     fn image(
144:         &mut self,
145:         name: String,
146:         binding: u32,
147:         dimension: wgpu::naga::ImageDimension,
148:         class: &wgpu::naga::ImageClass,
149:     ) {
150:         let dimension = match dimension {
151:             wgpu::naga::ImageDimension::D1 => wgpu::TextureViewDimension::D1,
152:             wgpu::naga::ImageDimension::D2 => wgpu::TextureViewDimension::D2,
153:             wgpu::naga::ImageDimension::D3 => wgpu::TextureViewDimension::D3,
154:             wgpu::naga::ImageDimension::Cube => wgpu::TextureViewDimension::Cube,
155:         };
156:         let mut format = None;
157:         let ty = match class {
158:             wgpu::naga::ImageClass::Storage {
159:                 format: storage,
160:                 access,
161:             } => {
162:                 let storage = TextureFormat::naga_to_wgpu_format(*storage);
163:                 format = Some(storage);
164:                 wgpu::BindingType::StorageTexture {
165:                     access: if access.contains(wgpu::naga::StorageAccess::STORE) {
166:                         wgpu::StorageTextureAccess::WriteOnly
167:                     } else {
168:                         wgpu::StorageTextureAccess::ReadOnly
169:                     },
170:                     format: storage,
171:                     view_dimension: dimension,
172:                 }
173:             }
174:             _ => wgpu::BindingType::Texture {
175:                 multisampled: false,
176:                 view_dimension: dimension,
177:                 sample_type: sample_type(class),
178:             },
179:         };
180:         self.entry(binding, ty);
181:         self.reflection.texture_bindings.push(TextureBinding {
182:             name,
183:             binding,
184:             format,
185:             dimension,
186:         });
187:     }
188: 
189:     fn entry(&mut self, binding: u32, ty: wgpu::BindingType) {
190:         self.entries.push(wgpu::BindGroupLayoutEntry {
191:             binding,
192:             visibility: wgpu::ShaderStages::COMPUTE,
193:             ty,
194:             count: None,
195:         });
196:     }
197: }
