295:     /// The cut, every length times `factor`, the hemline as far from the
296:     /// crown as the hair is long.
297:     pub fn scaled(mut self, factor: f32) -> Self {
298:         for length in [
299:             &mut self.top,
300:             &mut self.front,
301:             &mut self.sides,
302:             &mut self.back,
303:         ] {
304:             *length *= factor;
305:         }
306:         let crown = 0.115;
307:         self.hemline = self.hemline.map(|y| crown - (crown - y) * factor);
308:         self.dome = if self.dome > 0.0 {
309:             0.09 + (self.dome - 0.09) * factor
310:         } else {
311:             0.0
312:         };
313:         self
314:     }

333:     /// How long the hair rooted at `root` is, metres (before any hemline
334:     /// or dome).
335:     pub fn length(&self, head: &Head, root: Vec3) -> f32 {
336:         if let Some(fringe) = self.fringe
337:             && fringe.holds(root)
338:         {
339:             // Long enough to reach the fringe's line.
340:             return 0.1 + fringe.drop;
341:         }
342:         let e = head.elevation(root);
343:         let a = head.azimuth(root);
344:         // Round the head by where the root is near the hairline, the top's
345:         // length towards the crown.
346:         let ring = around(self.front, self.sides, self.back, a);
347:         let mut length = ring + (self.top - ring) * smoothstep(0.25, 0.85, e);
348:         if self.clip > 0.0 {
349:             let half = 0.5 * self.fade + 1e-3;
350:             let t = smoothstep(self.clip - half, self.clip + half, e);
351:             length = self.clipper + (length - self.clipper) * t;
352:         }
353:         if self.strip > 0.0 {
354:             // Clipped outside the strip, the full length a little inside.
355:             let t = 1.0 - smoothstep(self.strip - 0.006, self.strip, root.x.abs());
356:             length = self.clipper + (length - self.clipper) * t;
357:         }
358:         length.max(0.0005)
359:     }
