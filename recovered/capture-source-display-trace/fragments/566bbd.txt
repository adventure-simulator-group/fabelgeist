799: pub(super) fn visual_fingerprint(image: &Image) -> u64 {
800:     let Some(data) = image.data.as_deref() else {
801:         return 0;
802:     };
803:     let width = image.texture_descriptor.size.width as usize;
804:     let height = image.texture_descriptor.size.height as usize;
805:     let stride = width.saturating_mul(4);
806:     if stride == 0 || data.len() < stride.saturating_mul(height) {
807:         return 0;
808:     }
809:     // Ignore the top UI label and hash a regular sample of the rendered 3D
810:     // view. This catches accidentally identical camera outputs cheaply.
811:     let mut hash = 0xcbf29ce484222325_u64;
812:     for y in (96.min(height)..height).step_by(8) {
813:         for x in (0..width).step_by(8) {
814:             for channel in 0..3 {
815:                 hash ^= data[y * stride + x * 4 + channel] as u64;
816:                 hash = hash.wrapping_mul(0x100000001b3);
817:             }
818:         }
819:     }
820:     hash
821: }
