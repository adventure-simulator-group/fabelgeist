63: #[cfg(not(target_family = "wasm"))]
64: fn main() -> Result<(), Box<dyn std::error::Error>> {
65:     let args = Args::parse();
66:     if args.capture_output.is_some() && args.lod != LodChoice::Shell {
67:         eprintln!("--capture-output requires --lod shell");
68:         std::process::exit(2);
69:     }
70:     let plan = generate(&BuildingProgram::fixture(args.fixture, args.seed))?;
71:     let lod = compile_building_lod(&plan, args.lod.into())?;
72:     let dimensions = plan.dimensions_metres();
73:     let maximum_height = lod
74:         .meshes
75:         .iter()
76:         .flat_map(|mesh| mesh.vertices.iter())
77:         .map(|vertex| vertex.position.y)
78:         .fold(1.0_f32, f32::max);
79:     let title = format!("Fabelgeist building LOD: {:?} {:?}", args.fixture, args.lod);
80:     let capture_output = args.capture_output.clone();
81:     let capture_enabled = capture_output.is_some();
82:     let mut app = App::new();
83:     app.add_plugins((
84:         DefaultPlugins.set(WindowPlugin {
85:             primary_window: Some(Window {
86:                 title,
87:                 resolution: WindowResolution::new(1440, 900),
88:                 present_mode: PresentMode::AutoNoVsync,
89:                 ..default()
90:             }),
91:             ..default()
92:         }),
93:         PanOrbitCameraPlugin,
94:     ))
95:     .insert_resource(ClearColor(Color::srgb(0.72, 0.80, 0.86)))
96:     .add_systems(Startup, move |world: &mut World| {
97:         setup_lod_scene(world, &lod.meshes, dimensions, capture_enabled)
98:     })
99:     .add_systems(Update, capture::drive_shell_capture);
100:     if let Some(output) = capture_output {
101:         let config = capture::ShellCaptureConfig::new(
102:             output,
103:             args.capture_settle_frames,
104:             args.fixture.slug().to_owned(),
105:             args.seed,
106:             dimensions,
107:             maximum_height,
108:         )
109:         .unwrap_or_else(|error| {
110:             eprintln!("failed to initialize Shell capture: {error}");
111:             std::process::exit(2);
112:         });
113:         app.insert_resource(config);
114:     }
115:     let exit = app.run();
116:     if exit != AppExit::Success {
117:         std::process::exit(1);
118:     }
119:     Ok(())
120: }

122: #[cfg(not(target_family = "wasm"))]
123: fn setup_lod_scene(
124:     world: &mut World,
125:     batches: &[LodMesh],
126:     dimensions: Vec2,
127:     capture_enabled: bool,
128: ) {
129:     let textures = create_lod_textures(world);
130:     let translation = Vec3::new(-dimensions.x * 0.5, 0.0, -dimensions.y * 0.5);
131:     let mut maximum_height = 1.0_f32;
132:     for batch in batches {
133:         maximum_height = maximum_height.max(
134:             batch
135:                 .vertices
136:                 .iter()
137:                 .map(|vertex| vertex.position.y)
138:                 .fold(0.0, f32::max),
139:         );
140:         let mesh = world.resource_mut::<Assets<Mesh>>().add(bevy_mesh(batch));
141:         if capture_enabled && batch.material == BuildingLodMaterial::CrownMask {
142:             capture::spawn_crown_diagnostic(
143:                 world,
144:                 mesh.clone(),
145:                 textures.crown_mask.clone(),
146:                 Transform::from_translation(translation),
147:             );
148:         }
149:         let material = lod_material(world, batch.material, &textures);
150:         world.spawn((
151:             Name::new(format!("building LOD {:?}", batch.material)),
152:             Mesh3d(mesh),
153:             MeshMaterial3d(material),
154:             Transform::from_translation(translation),
155:         ));
156:     }
157: 
158:     let ground_mesh = world
159:         .resource_mut::<Assets<Mesh>>()
160:         .add(Plane3d::default().mesh().size(100.0, 100.0));
161:     let ground_material = world
162:         .resource_mut::<Assets<StandardMaterial>>()
163:         .add(StandardMaterial {
164:             base_color: Color::srgb(0.25, 0.30, 0.21),
165:             perceptual_roughness: 1.0,
166:             ..default()
167:         });
168:     world.spawn((
169:         Name::new("LOD viewer ground"),
170:         Mesh3d(ground_mesh),
171:         MeshMaterial3d(ground_material),
172:     ));
173: 
174:     let focus = Vec3::new(0.0, maximum_height * 0.45, 0.0);
175:     let radius = dimensions.length().max(maximum_height) * 1.15;
176:     let camera_position = focus + Vec3::new(radius * 0.72, radius * 0.52, radius * 0.82);
177:     let mut camera = world.spawn((
178:         Camera3d::default(),
179:         Transform::from_translation(camera_position).looking_at(focus, Vec3::Y),
180:     ));
181:     if capture_enabled {
182:         camera.insert(capture::ShellCaptureCamera);
183:     } else {
184:         camera.insert(PanOrbitCamera {
185:             focus,
186:             radius: Some(camera_position.distance(focus)),
187:             ..default()
188:         });
189:     }
190:     world.spawn((
191:         DirectionalLight {
192:             illuminance: 18_000.0,
193:             shadow_maps_enabled: true,
194:             ..default()
195:         },
196:         Transform::from_rotation(Quat::from_euler(EulerRot::XYZ, -0.85, -0.65, 0.0)),
197:     ));
198:     world.spawn((
199:         PointLight {
200:             intensity: 500_000.0,
201:             range: radius * 2.0,
202:             shadow_maps_enabled: false,
203:             ..default()
204:         },
205:         Transform::from_translation(camera_position.lerp(focus, 0.25)),
206:     ));
207: }
