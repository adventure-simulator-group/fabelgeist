836: fn write_manifest(state: &ShellCaptureConfig, passed: bool, failures: Vec<String>) {
837:     let manifest = CaptureManifest {
838:         schema: 1,
839:         pipeline: CAPTURE_PIPELINE,
840:         status: if passed { "PASS" } else { "FAIL" },
841:         fixture: &state.fixture,
842:         seed: state.seed,
843:         lod: "shell",
844:         revision: &state.git_head,
845:         dirty_state: &state.dirty_state,
846:         width: VIEW_WIDTH,
847:         height: VIEW_HEIGHT,
848:         alpha_mode: "mask",
849:         alpha_cutoff: CRENELLATION_ALPHA_CUTOFF,
850:         mask_texture_size: CRENELLATION_MASK_TEXTURE_SIZE,
851:         mask_mip_levels: CRENELLATION_MASK_TEXTURE_SIZE.ilog2() + 1,
852:         camera_path_semantics: "32 strictly increasing fine-step distances with a bounded azimuth sweep; only the first pose uses the configured startup settle, then each pose advances after one rendered frame",
853:         temporal_readback_semantics: "two preserved same-pose production GPU readbacks plus one final production frame per pose require identical raw hashes; a second traversal requests all 32 production and CrownMask frames on consecutive Update frames without awaiting completion",
854:         crown_diagnostic_semantics: "offscreen render of the exact CrownMask mesh and shared alpha-tested texture on an isolated render layer; both settled and consecutive-motion validation require presence and bounded distance-normalized occupancy continuity",
855:         validation_passed: passed,
856:         validation_failures: failures,
857:         discarded_readbacks: &state.discarded_readbacks,
858:         temporal_readbacks: &state.temporal_readbacks,
859:         continuous_readbacks: &state.continuous_readbacks,
860:         continuous_crown_readbacks: &state.continuous_crown_readbacks,
861:         captures: &state.records,
862:     };
863:     fs::write(
864:         state.output.join("manifest.json"),
865:         serde_json::to_vec_pretty(&manifest).expect("serialize Shell capture manifest"),
866:     )
867:     .expect("write Shell capture manifest");
868: }
