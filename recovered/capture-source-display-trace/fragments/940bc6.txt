639: fn stable_hash(bytes: &[u8]) -> u64 {
640:     bytes.iter().fold(0xcbf2_9ce4_8422_2325, |hash, byte| {
641:         (hash ^ u64::from(*byte)).wrapping_mul(0x100_0000_01b3)

645: fn non_clear_pixel_bps(bytes: &[u8]) -> u32 {
646:     if bytes.len() < 4 {
647:         return 0;
648:     }
649:     let clear = [184_i16, 204, 219];
650:     let non_clear = bytes
651:         .as_chunks::<4>()
652:         .0
653:         .iter()
654:         .filter(|pixel| {
655:             (i16::from(pixel[0]) - clear[0]).abs() > 12
656:                 || (i16::from(pixel[1]) - clear[1]).abs() > 12
657:                 || (i16::from(pixel[2]) - clear[2]).abs() > 12
658:         })
659:         .count();
660:     (non_clear as u64 * 10_000 / (bytes.len() / 4) as u64) as u32
661: }

672: fn validation_failures(state: &ShellCaptureConfig) -> Vec<String> {
673:     let mut failures = Vec::new();
674:     if state.records.len() != state.poses.len() {
675:         failures.push(format!(
676:             "capture count {} != pose count {}",
677:             state.records.len(),
678:             state.poses.len()
679:         ));
680:     }
681:     if state.git_head.len() != 40 || !state.git_head.bytes().all(|byte| byte.is_ascii_hexdigit()) {
682:         failures.push(format!(
683:             "revision provenance is invalid: {:?}",
684:             state.git_head
685:         ));
686:     }
687:     if state
688:         .records
689:         .windows(2)
690:         .any(|pair| pair[1].distance_metres <= pair[0].distance_metres)
691:     {
692:         failures.push("camera distances are not strictly increasing".to_owned());
693:     }
694:     let expected_temporal_readbacks = state.poses.len() * usize::from(PRIME_READBACKS_PER_POSE);
695:     if state.temporal_readbacks.len() != expected_temporal_readbacks {
696:         failures.push(format!(
697:             "temporal readback count {} != expected {}",
698:             state.temporal_readbacks.len(),
699:             expected_temporal_readbacks
700:         ));
701:     }
702:     for record in &state.records {
703:         if record.pixel_hash == "cbf29ce484222325" {
704:             failures.push(format!(
705:                 "capture {} has an empty pixel hash",
706:                 record.sample_index
707:             ));
708:         }
709:         if record.non_clear_pixel_bps < 25 {
710:             failures.push(format!(
711:                 "capture {} has only {} bps non-clear pixels",
712:                 record.sample_index, record.non_clear_pixel_bps
713:             ));
714:         }
715:         if record.prime_readbacks != PRIME_READBACKS_PER_POSE {
716:             failures.push(format!(
717:                 "capture {} had {} prime readbacks",
718:                 record.sample_index, record.prime_readbacks
719:             ));
720:         }
721:         if record.crown_pixel_count == 0 {
722:             failures.push(format!(
723:                 "capture {} has no CrownMask diagnostic pixels",
724:                 record.sample_index
725:             ));
726:         }
727:         let repeats = state
728:             .temporal_readbacks
729:             .iter()
730:             .filter(|readback| readback.sample_index == record.sample_index)
731:             .collect::<Vec<_>>();
732:         if repeats.len() == usize::from(PRIME_READBACKS_PER_POSE) {
733:             if repeats
734:                 .iter()
735:                 .any(|readback| readback.pixel_hash != record.pixel_hash)
736:             {
737:                 failures.push(format!(
738:                     "capture {} changed across repeated same-pose GPU readbacks",
739:                     record.sample_index
740:                 ));
741:             }
742:             if repeats
743:                 .iter()
744:                 .any(|readback| readback.non_clear_pixel_bps < 25)
745:             {
746:                 failures.push(format!(
747:                     "capture {} has a blank repeated GPU readback",
748:                     record.sample_index
749:                 ));
750:             }
751:         }
752:     }
753:     for pair in state.records.windows(2) {
754:         let first =
755:             f64::from(pair[0].crown_pixel_count) * f64::from(pair[0].distance_metres).powi(2);
756:         let second =
757:             f64::from(pair[1].crown_pixel_count) * f64::from(pair[1].distance_metres).powi(2);
758:         if first > 0.0 {
759:             let ratio = second / first;
760:             if !(0.55..=1.80).contains(&ratio) {
761:                 failures.push(format!(
762:                     "CrownMask normalized occupancy jumped by ratio {ratio:.3} between samples {} and {}",
763:                     pair[0].sample_index, pair[1].sample_index
764:                 ));
765:             }
766:         }
767:     }
768:     if state.continuous_readbacks.len() != state.poses.len() {
769:         failures.push(format!(
770:             "continuous production readback count {} != pose count {}",
771:             state.continuous_readbacks.len(),
772:             state.poses.len()
773:         ));
774:     }
775:     if state.continuous_crown_readbacks.len() != state.poses.len() {
776:         failures.push(format!(
777:             "continuous CrownMask readback count {} != pose count {}",
778:             state.continuous_crown_readbacks.len(),
779:             state.poses.len()
780:         ));
781:     }
782:     let mut continuous = state.continuous_readbacks.iter().collect::<Vec<_>>();
783:     continuous.sort_by_key(|record| record.sample_index);
784:     if continuous
785:         .windows(2)
786:         .any(|pair| pair[1].requested_frame_index != pair[0].requested_frame_index + 1)
787:     {
788:         failures.push(
789:             "continuous production readbacks were not requested on consecutive frames".to_owned(),
790:         );
791:     }
792:     for record in &continuous {
793:         if record.non_clear_pixel_bps < 25 {
794:             failures.push(format!(
795:                 "continuous production sample {} is blank",
796:                 record.sample_index
797:             ));
798:         }
799:     }
800:     let mut continuous_crowns = state.continuous_crown_readbacks.iter().collect::<Vec<_>>();
801:     continuous_crowns.sort_by_key(|record| record.sample_index);
802:     if continuous_crowns
803:         .windows(2)
804:         .any(|pair| pair[1].requested_frame_index != pair[0].requested_frame_index + 1)
805:     {
806:         failures.push(
807:             "continuous CrownMask readbacks were not requested on consecutive frames".to_owned(),
808:         );
809:     }
810:     for record in &continuous_crowns {
811:         if record.crown_pixel_count == 0 {
812:             failures.push(format!(
813:                 "continuous CrownMask sample {} disappeared",

816:         }
817:     }

819:         let first =
