86: #[command(version, about = "Capture and validate a tactical environment fixture")]
87: struct Args {
88:     /// Fixture name from assets/tactical-scenes, without the .json suffix.
89:     #[arg(long, conflicts_with = "scene_input")]
90:     fixture: Option<String>,
91: 
92:     /// TacticalSceneInput fixture stem or explicit JSON path.
93:     #[arg(long, conflicts_with = "fixture", value_parser = resolve_scene_fixture)]
94:     scene_input: Option<PathBuf>,
95: 
96:     /// Fresh output directory. A timestamped directory is chosen when omitted.
97:     #[arg(long)]
98:     output: Option<PathBuf>,
99: 
100:     /// Render frames allowed to settle between fixed camera views.
101:     #[arg(long, default_value_t = 12)]
102:     settle_frames: u32,
103: 
104:     /// Test-only world canopy override in basis points (0..=10000).
105:     #[arg(long, value_parser = clap::value_parser!(u16).range(0..=10_000))]
106:     canopy_bps: Option<u16>,
107: 
108:     /// Test-only celestial time override, in absolute world minutes.
109:     #[arg(long)]
110:     absolute_minute: Option<u64>,
111: 
112:     /// Benchmark each leaf representation for this many frames in dense woodland.
113:     #[arg(long, value_parser = clap::value_parser!(u32).range(30..))]
114:     leaf_benchmark_frames: Option<u32>,
115: 
116:     /// Benchmark baseline, canopy AO, shadows, and combined tree lighting.
117:     #[arg(long, value_parser = clap::value_parser!(u32).range(30..), conflicts_with = "leaf_benchmark_frames")]
118:     tree_lighting_benchmark_frames: Option<u32>,
119: 
120:     /// Benchmark natural and isolated tree/LOD costs for this many frames per mode.
121:     #[arg(long, value_parser = clap::value_parser!(u32).range(30..), conflicts_with_all = ["leaf_benchmark_frames", "tree_lighting_benchmark_frames"])]
122:     scene_performance_benchmark_frames: Option<u32>,
123: 
124:     /// Collect GPU timestamps and render diagnostics during the scene benchmark.
125:     #[arg(long, requires = "scene_performance_benchmark_frames")]
126:     scene_performance_render_diagnostics: bool,
127: 
128:     /// Capture only terrain as color-coded wireframes and count visible triangles by LOD.

132:     /// Count resident and submitted terrain, building, and grass triangles without timing frames.

135: 
136:     /// Azimuth around the review tree for locked leaf-LOD comparison views.

153: #[cfg(not(target_family = "wasm"))]
154: fn main() {
155:     let args = Args::parse();
156:     if args.fixture.is_none() && args.scene_input.is_none() {
157:         clap::Error::raw(
158:             clap::error::ErrorKind::MissingRequiredArgument,
159:             "one of --fixture or --scene-input is required",
160:         )
161:         .exit();
162:     }
163:     tactical_scene_viewer::run(
164:         args.fixture,
165:         args.scene_input,
166:         args.output,
167:         args.settle_frames,
168:         args.canopy_bps,
169:         args.absolute_minute,
170:         args.leaf_benchmark_frames,
171:         args.tree_lighting_benchmark_frames,
172:         args.scene_performance_benchmark_frames,
173:         args.scene_performance_render_diagnostics,
174:         args.terrain_wireframe,
175:         args.triangle_census,
176:         args.tree_review_azimuth_degrees,
177:         match args.profile {
178:             CaptureProfile::PlantReview => "plant-review",
179:             CaptureProfile::FungusReview => "fungus-review",
180:             CaptureProfile::PlantLodReview => "plant-lod-review",
181:             CaptureProfile::PlantLodIsolated => "plant-lod-isolated",
182:             CaptureProfile::Semantic => "semantic",
183:             CaptureProfile::EnvironmentReview => "environment-review",
184:             CaptureProfile::LandformReview => tactical_scene_viewer::LANDFORM_REVIEW_PROFILE,
185:             CaptureProfile::InteriorReview => "interior-review",
186:             CaptureProfile::CityReview => "city-review",
187:             CaptureProfile::TerrainGrounding => tactical_scene_viewer::fixed_city_cameras::PROFILE,
188:             CaptureProfile::FurnitureReview => "furniture-review",
189:             CaptureProfile::InteriorFurnitureCatalog => "interior-furniture-catalog",
190:             CaptureProfile::FurnishedRoomReview => "furnished-room-review",
191:             CaptureProfile::ShopSignReview => "shop-sign-review",
192:             CaptureProfile::WorkplaceReview => "workplace-review",
193:             CaptureProfile::ParishReview => "parish-review",
194:             CaptureProfile::CompoundReview => "compound-review",
195:             CaptureProfile::HeatingReview => "heating-review",
196:             CaptureProfile::FacadeReview => "facade-review",
197:             CaptureProfile::GardenReview => "garden-review",
198:             CaptureProfile::GableReview => "gable-review",
199:             CaptureProfile::AnimationPlay => "animation-play",
200:             CaptureProfile::TreeColdTraversal => "tree-cold-traversal",
201:             CaptureProfile::BeechLeafMotion => "beech-leaf-motion",
202:         },
203:         args.views,
204:         args.city_cameras,
205:     );
206: }

213:     fn authored_building_review_profiles_are_available_from_the_cli() {
214:         let fixtures =
215:             PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("../../assets/tactical-scenes");
216:         let mut reviews = 0;
217:         for entry in std::fs::read_dir(fixtures).unwrap() {
218:             let name = entry.unwrap().file_name();
219:             let name = name.to_str().unwrap();
220:             let Some(profile) = name.strip_suffix(".review.json") else {
221:                 continue;
222:             };
223:             Args::try_parse_from([
224:                 "tactical-scene-viewer",
225:                 "--scene-input",
226:                 profile,
227:                 "--profile",
228:                 profile,
229:             ])
230:             .unwrap_or_else(|error| panic!("authored review {profile} is inaccessible: {error}"));
231:             reviews += 1;
232:         }
233:         assert!(reviews > 0, "building review fixtures must be present");
234:     }
