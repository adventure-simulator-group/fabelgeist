33: /// Stable observer-facing result of evaluating whether an investigation action
34: /// can begin now.
35: #[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
36: #[cfg_attr(feature = "spacetimedb", derive(spacetimedb::SpacetimeType))]
37: #[serde(rename_all = "snake_case")]
38: pub enum InvestigationActionUnavailableReason {
39:     PartyNotReady,
40:     TravelRequired,
41:     NightWindow,
42:     TargetChanged,
43:     ContactScheduleWindow,
44:     ContactNotPresent,
45:     CharacterUnavailable,
46:     PartyRequired,
47: }
48: 
49: #[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
50: #[serde(rename_all = "snake_case")]
51: pub enum InvestigationActionAvailability {
52:     Available,
53:     Unavailable {
54:         reason: InvestigationActionUnavailableReason,
55:         can_travel_to_required_site: bool,
56:         wait_minutes: u32,
57:     },
58: }

60: #[cfg(feature = "spacetimedb")]
61: #[derive(Clone, Copy, Debug, PartialEq, Eq, spacetimedb::SpacetimeType)]
62: #[sats(name = "InvestigationActionUnavailableFields")]
63: struct InvestigationActionUnavailableFieldsSats {
64:     reason: InvestigationActionUnavailableReason,
65:     can_travel_to_required_site: bool,
66:     wait_minutes: u32,
67: }

69: #[cfg(feature = "spacetimedb")]
70: #[derive(Clone, Copy, Debug, PartialEq, Eq, spacetimedb::SpacetimeType)]
71: #[sats(name = "InvestigationActionAvailability")]
72: enum InvestigationActionAvailabilitySats {
73:     Available,
74:     Unavailable(InvestigationActionUnavailableFieldsSats),
75: }
76: 
77: #[cfg(feature = "spacetimedb")]
78: impl From<InvestigationActionAvailability> for InvestigationActionAvailabilitySats {
79:     fn from(availability: InvestigationActionAvailability) -> Self {
80:         match availability {
81:             InvestigationActionAvailability::Available => Self::Available,
82:             InvestigationActionAvailability::Unavailable {
83:                 reason,
84:                 can_travel_to_required_site,
85:                 wait_minutes,
86:             } => Self::Unavailable(InvestigationActionUnavailableFieldsSats {
87:                 reason,
88:                 can_travel_to_required_site,
89:                 wait_minutes,
90:             }),
91:         }
92:     }
93: }
94: 
95: #[cfg(feature = "spacetimedb")]
96: impl From<InvestigationActionAvailabilitySats> for InvestigationActionAvailability {
97:     fn from(availability: InvestigationActionAvailabilitySats) -> Self {
98:         match availability {
99:             InvestigationActionAvailabilitySats::Available => Self::Available,
100:             InvestigationActionAvailabilitySats::Unavailable(
101:                 InvestigationActionUnavailableFieldsSats {
102:                     reason,
103:                     can_travel_to_required_site,
104:                     wait_minutes,
105:                 },
106:             ) => Self::Unavailable {
107:                 reason,
108:                 can_travel_to_required_site,
109:                 wait_minutes,
110:             },
111:         }
112:     }
113: }
