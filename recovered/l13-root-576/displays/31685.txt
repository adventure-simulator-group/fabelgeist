501:     generated: &GeneratedWeapon,
502:     focus: &mut Vec<[f32; 3]>,
503:     principal_head: &mut Vec<[f32; 3]>,
504: ) {
505:     let span = generated.bounds.max[1] - generated.bounds.min[1];
506:     let focus_cutoff = generated.bounds.max[1] - span * 0.2;
507:     let head_cutoff = generated.bounds.max[1] - SYNTHETIC_HEAD_LENGTH_M.min(span * 0.2);
508:     for point in generated
509:         .parts
510:         .iter()
511:         .flat_map(|part| part.positions.iter().copied())
512:         .filter(|point| point[1] >= focus_cutoff)
513:     {
514:         focus.push(point);
515:         focus.push([point[0], focus_cutoff, point[2]]);
516:         if point[1] >= head_cutoff {
517:             principal_head.push(point);
518:             principal_head.push([point[0], head_cutoff, point[2]]);
519:         }
520:     }
521: }
