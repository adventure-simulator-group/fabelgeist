30: #[derive(Clone, Copy, Debug, PartialEq)]
31: pub struct IconBounds {
32:     pub min: [f32; 2],
33:     pub max: [f32; 2],
34: }
35: 
36: impl IconBounds {
37:     fn empty() -> Self {
38:         Self {
39:             min: [f32::INFINITY; 2],
40:             max: [f32::NEG_INFINITY; 2],
41:         }
42:     }
43: 
44:     fn include(&mut self, point: [f32; 2]) {
45:         for (axis, value) in point.into_iter().enumerate() {
46:             self.min[axis] = self.min[axis].min(value);
47:             self.max[axis] = self.max[axis].max(value);
48:         }
49:     }
50: 
51:     pub fn center(self) -> [f32; 2] {
52:         [
53:             (self.min[0] + self.max[0]) * 0.5,
54:             (self.min[1] + self.max[1]) * 0.5,
55:         ]
56:     }
57: 
58:     fn is_finite(self) -> bool {
59:         self.min.into_iter().chain(self.max).all(f32::is_finite)
60:     }
61: }
