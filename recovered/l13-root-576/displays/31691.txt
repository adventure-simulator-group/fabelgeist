1: //! Supersampled orthographic color rasterizer with interpolated normals and depth.
2: 
3: use super::{IconError, Projection, WeaponIconSpec, studio};
4: use crate::{Material as RecipeMaterial, MeshPart};
5: use glam::{Vec2, Vec3};
6: 
7: const CAMERA_YAW_DEGREES: f32 = 12.0;
8: pub(super) const CAMERA_YAW: f32 = CAMERA_YAW_DEGREES.to_radians();
9: const MIN_TRIANGLE_AREA: f32 = 1.0e-8;
10: 
11: pub(super) fn render(
12:     parts: &[MeshPart],
13:     spec: WeaponIconSpec,
14:     projection: &Projection,
15: ) -> Result<(Vec<u8>, Vec<u8>), IconError> {
16:     if !(16..=512).contains(&spec.size) || !(1..=8).contains(&spec.supersampling) {
17:         return Err(IconError::InvalidSpec);
18:     }
19:     let width = usize::from(spec.size) * usize::from(spec.supersampling);
20:     let mut target = Target {
21:         width,
22:         colors: vec![Vec3::ZERO; width * width],
23:         depth: vec![f32::NEG_INFINITY; width * width],
24:     };
25:     let view = Vec3::new(-CAMERA_YAW.sin(), 0.0, CAMERA_YAW.cos());
26:     for part in parts {
27:         let material = Material::from(part.material);
28:         for indices in part.indices.as_chunks::<3>().0 {
29:             let vertices = indices.map(|index| {
30:                 let position = part.positions[index as usize];
31:                 Vertex {
32:                     screen: Vec2::from_array(projection.point(position)) * width as f32,
33:                     depth: Vec3::from_array(position).dot(view),
34:                     normal: Vec3::from_array(part.normals[index as usize]),
35:                 }
36:             });
37:             target.triangle(vertices, &material, view);
38:         }
39:     }
40:     Ok(target.resolve(spec))
41: }
42: 
43: struct Vertex {
44:     screen: Vec2,
45:     depth: f32,
46:     normal: Vec3,
47: }
48: struct Material {
49:     albedo: Vec3,
50:     metallic: f32,
51:     roughness: f32,
52: }
53: 
54: impl From<RecipeMaterial> for Material {
55:     fn from(value: RecipeMaterial) -> Self {
56:         let (color, metallic, roughness) = match value {
57:             RecipeMaterial::Steel => ([0.62, 0.66, 0.7], 1.0, 0.24),
58:             RecipeMaterial::DarkSteel => ([0.16, 0.19, 0.22], 1.0, 0.36),
59:             RecipeMaterial::Brass => ([0.72, 0.43, 0.13], 1.0, 0.28),
60:             RecipeMaterial::Wood => ([0.26, 0.11, 0.035], 0.0, 0.7),
61:             RecipeMaterial::Leather => ([0.18, 0.065, 0.025], 0.0, 0.65),
62:             RecipeMaterial::DarkLeather => ([0.045, 0.022, 0.014], 0.0, 0.7),
63:             material => (
64:                 material.color().map(|n| n as f32),
65:                 if material.is_metal() { 1.0 } else { 0.0 },
66:                 0.5,
67:             ),
68:         };
69:         Self {
70:             albedo: Vec3::from_array(color),
71:             metallic,
72:             roughness,
73:         }
74:     }
75: }
76: 
77: struct Target {
78:     width: usize,
79:     colors: Vec<Vec3>,
80:     depth: Vec<f32>,
81: }
82: 
83: impl Target {
84:     fn triangle(&mut self, vertices: [Vertex; 3], material: &Material, view: Vec3) {
85:         let [a, b, c] = vertices.map(|v| (v.screen, v.depth, v.normal));
86:         let area = (b.0 - a.0).perp_dot(c.0 - a.0);
87:         if area.abs() < MIN_TRIANGLE_AREA {
88:             return;
89:         }
90:         let min = a.0.min(b.0).min(c.0).floor().max(Vec2::ZERO);
91:         let max =
92:             a.0.max(b.0)
93:                 .max(c.0)
94:                 .ceil()
95:                 .min(Vec2::splat(self.width as f32 - 1.0));

97:             for x in min.x as usize..=max.x.max(0.0) as usize {
