5: /// Declares a fieldless enum and the WGSL `u32` constants naming its
6: /// variants' discriminants, in `SCREAMING_SNAKE_CASE` with a prefix.
7: macro_rules! shared_codes {
8:     ($(#[$meta:meta])* $name:ident { $($(#[$variant_meta:meta])* $variant:ident),* $(,)? }) => {
9:         $(#[$meta])*
10:         #[derive(Clone, Copy, Debug, PartialEq, Eq)]
11:         pub(super) enum $name {
12:             $($(#[$variant_meta])* $variant),*
13:         }
14: 
15:         impl $name {
16:             #[allow(dead_code, reason = "not every code set is counted")]
17:             pub(super) const COUNT: usize = [$(stringify!($variant)),*].len();
18: 
19:             /// WGSL constants for every variant.
20:             pub(super) fn wgsl(prefix: &str) -> String {
21:                 [$((stringify!($variant), $name::$variant as u32)),*]
22:                     .iter()
23:                     .map(|(variant, code)| {
24:                         format!("const {prefix}{}: u32 = {code}u;\n", screaming(variant))
25:                     })
26:                     .collect()
27:             }
28:         }
29:     };
30: }

41: /// All disconnected coverage of one item over one coarse combat body part.
42: /// Serialization emits only occupied entries and rejects oversized input.
43: #[derive(Debug, Clone, Copy, Default, PartialEq, Serialize, Deserialize)]
44: #[serde(
45:     try_from = "Vec<ArmorCoverageSegment>",
46:     into = "Vec<ArmorCoverageSegment>"
47: )]
48: pub struct AuthoredArmorCoverage {
49:     segments: [Option<ArmorCoverageSegment>; MAX_EQUIPMENT_SURFACE_SEGMENTS],
50: }
51: 
52: impl AuthoredArmorCoverage {
53:     /// Explicit interval for innate protection or a synthetic combat fixture.
54:     pub fn from_span(span: ArmorCoverageSpan) -> Self {
55:         let mut result = Self::default();
56:         result.segments[0] = Some(ArmorCoverageSegment {
57:             span,
58:             region: None,
59:             anchor: SurfaceAnchor::Center,
60:             laterality: None,
61:             surface_index: None,
62:         });
63:         result
64:     }
65: 
66:     /// Projects every authored region; absent regions give no protection.
67:     pub fn from_placement(placement: &EquipmentPlacement, part: BodyPart) -> Self {
68:         let mut segments = Vec::new();
69:         for (surface_index, surface) in placement.surface.iter().enumerate() {
70:             let count = surface.regions.len().max(1) as f32;
71:             let coverage = surface.coverage.clamp(0.0, 1.0);
72:             let retained = match surface.anchor {
73:                 SurfaceAnchor::Proximal => (0.0, coverage),
74:                 SurfaceAnchor::Distal => (1.0 - coverage, 1.0),
75:                 SurfaceAnchor::Center => ((1.0 - coverage) * 0.5, (1.0 + coverage) * 0.5),
76:             };
77:             for (index, region) in surface.regions.iter().copied().enumerate() {
78:                 let (region_part, body_start, body_end, laterality) = region_body_interval(region);
79:                 if region_part != part {
80:                     continue;
81:                 }
82:                 let start = retained.0.max(index as f32 / count);
83:                 let end = retained.1.min((index as f32 + 1.0) / count);
84:                 if start >= end {
85:                     continue;
86:                 }
87:                 let local_start = start * count - index as f32;
88:                 let local_end = end * count - index as f32;
89:                 segments.push(ArmorCoverageSegment {
90:                     span: ArmorCoverageSpan {
91:                         start: body_start + (body_end - body_start) * local_start,
92:                         end: body_start + (body_end - body_start) * local_end,
93:                     },
94:                     region: Some(region),
95:                     anchor: surface.anchor,
96:                     laterality,
97:                     surface_index: Some(surface_index),
98:                 });
99:             }
100:         }
101:         Self::try_from(segments).expect("catalog validates anatomical segment capacity")
102:     }
103: 
104:     pub fn segments(&self) -> impl Iterator<Item = &ArmorCoverageSegment> {
105:         self.segments.iter().flatten()
106:     }
107: 
108:     pub fn contains(self, coordinate: f32) -> bool {
109:         self.segments()
110:             .any(|segment| segment.span.contains(coordinate))
111:     }
112: }

114: impl TryFrom<Vec<ArmorCoverageSegment>> for AuthoredArmorCoverage {
115:     type Error = &'static str;
116: 
117:     fn try_from(segments: Vec<ArmorCoverageSegment>) -> Result<Self, Self::Error> {
118:         if segments.len() > MAX_EQUIPMENT_SURFACE_SEGMENTS {
119:             return Err("too many anatomical coverage segments");
120:         }
121:         if segments.iter().any(|segment| {
122:             !segment.span.start.is_finite()
123:                 || !segment.span.end.is_finite()
124:                 || segment.span.start < 0.0
125:                 || segment.span.end > 1.0
126:                 || segment.span.start > segment.span.end
127:         }) {
128:             return Err("invalid anatomical coverage interval");
129:         }
130:         let mut result = Self::default();
131:         for (entry, segment) in result.segments.iter_mut().zip(segments) {
132:             *entry = Some(segment);
133:         }
134:         Ok(result)
135:     }
136: }
