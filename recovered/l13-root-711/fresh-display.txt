35: pub(super) fn extract_surface_nets(
36:     grid: SurfaceNetsGrid,
37:     field: impl Fn(Vec3) -> f32,
38: ) -> Option<ExtractedSurface> {
39:     let [nx, ny, nz] = grid.sample_counts;
40:     if nx < 2
41:         || ny < 2
42:         || nz < 2
43:         || !grid.minimum.is_finite()
44:         || !grid.maximum.is_finite()
45:         || !grid.minimum.cmplt(grid.maximum).all()
46:     {
47:         return None;
48:     }
49:     let sample_count = nx.checked_mul(ny)?.checked_mul(nz)?;
50:     let cell_counts = [nx - 1, ny - 1, nz - 1];
51:     let cell_count = cell_counts[0]
52:         .checked_mul(cell_counts[1])?
53:         .checked_mul(cell_counts[2])?;
54:     let spacing = (grid.maximum - grid.minimum)
55:         / Vec3::new((nx - 1) as f32, (ny - 1) as f32, (nz - 1) as f32);
56:     let sample_position = |x: usize, y: usize, z: usize| {
57:         grid.minimum + Vec3::new(x as f32, y as f32, z as f32) * spacing
58:     };
59:     let sample_index = |x: usize, y: usize, z: usize| (z * ny + y) * nx + x;
60:     let cell_index = |x: usize, y: usize, z: usize| (z * cell_counts[1] + y) * cell_counts[0] + x;
61:     let mut samples = Vec::with_capacity(sample_count);
62:     for z in 0..nz {
63:         for y in 0..ny {
64:             for x in 0..nx {
65:                 let value = field(sample_position(x, y, z));
66:                 if !value.is_finite() {
67:                     return None;
68:                 }
69:                 samples.push(value);
70:             }
71:         }
72:     }
73: 
74:     const CORNERS: [[usize; 3]; 8] = [
75:         [0, 0, 0],
76:         [1, 0, 0],
77:         [0, 1, 0],
78:         [1, 1, 0],
79:         [0, 0, 1],
80:         [1, 0, 1],
81:         [0, 1, 1],
82:         [1, 1, 1],
83:     ];
84:     const EDGES: [(usize, usize); 12] = [
85:         (0, 1),
86:         (2, 3),
87:         (4, 5),
88:         (6, 7),
89:         (0, 2),
90:         (1, 3),
91:         (4, 6),
92:         (5, 7),
93:         (0, 4),
94:         (1, 5),
95:         (2, 6),
96:         (3, 7),
97:     ];
98:     let mut vertices = vec![None; cell_count];
99:     let mut surface = ExtractedSurface::default();
100:     let normal_step = spacing.min_element() * 0.35;
101:     for z in 0..cell_counts[2] {
102:         for y in 0..cell_counts[1] {
103:             for x in 0..cell_counts[0] {
104:                 let corner_values =
105:                     CORNERS.map(|[dx, dy, dz]| samples[sample_index(x + dx, y + dy, z + dz)]);
106:                 let inside = corner_values.map(|value| value <= 0.0);
107:                 if inside.iter().all(|value| *value) || inside.iter().all(|value| !*value) {
108:                     continue;
109:                 }
110:                 let mut point = Vec3::ZERO;
111:                 let mut intersections = 0_u32;
112:                 for (start, end) in EDGES {
113:                     if inside[start] == inside[end] {
114:                         continue;
115:                     }
116:                     let [sx, sy, sz] = CORNERS[start];
117:                     let [ex, ey, ez] = CORNERS[end];
118:                     let start_position = sample_position(x + sx, y + sy, z + sz);
119:                     let end_position = sample_position(x + ex, y + ey, z + ez);
120:                     let start_value = corner_values[start];
121:                     let end_value = corner_values[end];
122:                     let amount = (start_value / (start_value - end_value)).clamp(0.0, 1.0);
123:                     point += start_position.lerp(end_position, amount);
124:                     intersections += 1;
125:                 }
126:                 if intersections == 0 {
127:                     continue;
128:                 }
129:                 point /= intersections as f32;
130:                 let step = Vec3::splat(normal_step);
131:                 let gradient = Vec3::new(
132:                     field(point + Vec3::X * step.x) - field(point - Vec3::X * step.x),
133:                     field(point + Vec3::Y * step.y) - field(point - Vec3::Y * step.y),
134:                     field(point + Vec3::Z * step.z) - field(point - Vec3::Z * step.z),
135:                 );
136:                 let normal = gradient.try_normalize().unwrap_or(Vec3::Y);
137:                 let index = surface.positions.len() as u32;
138:                 surface.positions.push(point.to_array());
139:                 surface.normals.push(normal.to_array());
140:                 vertices[cell_index(x, y, z)] = Some(index);
141:             }
142:         }
143:     }
144: 
145:     let mut add_quad = |cells: [[usize; 3]; 4]| {
146:         let [Some(a), Some(b), Some(c), Some(d)] =
147:             cells.map(|[x, y, z]| vertices[cell_index(x, y, z)])
148:         else {
149:             return;
150:         };
151:         let indices = [a, b, c, d];
152:         let positions = indices.map(|index| Vec3::from_array(surface.positions[index as usize]));
153:         let face_normal = (positions[1] - positions[0]).cross(positions[2] - positions[0]);
154:         let expected_normal = indices
155:             .iter()
156:             .map(|index| Vec3::from_array(surface.normals[*index as usize]))
157:             .sum::<Vec3>();
158:         if face_normal.dot(expected_normal) >= 0.0 {
159:             surface.indices.extend_from_slice(&[
160:                 indices[0], indices[1], indices[2], indices[0], indices[2], indices[3],
161:             ]);
162:         } else {
163:             surface.indices.extend_from_slice(&[
164:                 indices[0], indices[3], indices[2], indices[0], indices[2], indices[1],
165:             ]);
166:         }
167:     };
168:     let changes_sign = |left: f32, right: f32| (left <= 0.0) != (right <= 0.0);
169:     for z in 1..nz - 1 {
170:         for y in 1..ny - 1 {
171:             for x in 0..nx - 1 {
172:                 if changes_sign(
173:                     samples[sample_index(x, y, z)],
174:                     samples[sample_index(x + 1, y, z)],
175:                 ) {
176:                     add_quad([[x, y - 1, z - 1], [x, y, z - 1], [x, y, z], [x, y - 1, z]]);
177:                 }
178:             }
179:         }
180:     }
181:     for z in 1..nz - 1 {
182:         for y in 0..ny - 1 {
183:             for x in 1..nx - 1 {
184:                 if changes_sign(
185:                     samples[sample_index(x, y, z)],
186:                     samples[sample_index(x, y + 1, z)],
187:                 ) {
188:                     add_quad([[x - 1, y, z - 1], [x - 1, y, z], [x, y, z], [x, y, z - 1]]);
189:                 }
190:             }
191:         }
192:     }
193:     for z in 0..nz - 1 {
194:         for y in 1..ny - 1 {
195:             for x in 1..nx - 1 {
196:                 if changes_sign(
197:                     samples[sample_index(x, y, z)],
198:                     samples[sample_index(x, y, z + 1)],
199:                 ) {
200:                     add_quad([[x - 1, y - 1, z], [x, y - 1, z], [x, y, z], [x - 1, y, z]]);
201:                 }
202:             }
203:         }
204:     }
205:     Some(surface)
206: }

21: impl MappedGeologicWindow {
22:     pub fn is_valid(&self) -> bool {
23:         let [west, south, east, north] = self.bounds_metres.map(i64::from);
24:         !self.id.is_empty()
25:             && self.id.len() <= 256
26:             && east > west
27:             && north > south
28:             && east - west <= i64::from(MAX_GEOLOGIC_WINDOW_SIDE_METRES)
29:             && north - south <= i64::from(MAX_GEOLOGIC_WINDOW_SIDE_METRES)
30:     }
31: 
32:     pub fn contains(&self, point: [f64; 2]) -> bool {

34:         point[0] >= west && point[0] <= east && point[1] >= south && point[1] <= north
35:     }
