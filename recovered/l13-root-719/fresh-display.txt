106:     /// Uploads the dirty ranges to the GPU.
107:     pub fn flush(
108:         &mut self,
109:         render_queue: &RenderQueue,
110:         indirect_buffer: &Buffer,
111:         uniform_buffer: &Buffer,
112:         metadata_buffer: &Buffer,
113:     ) {
114:         if self.dirty_indices.is_empty() {
115:             return;
116:         }

118:         self.dirty_indices.sort_unstable();

128:         while i < self.dirty_indices.len() {

130:             let mut end = start;

132:             // find contiguous dirty indices
133:             while let Some(&next) = self.dirty_indices.get(i + 1) {

6: /// Ensures a GPU buffer has sufficient capacity, resizing and copying data if necessary.
7: pub(crate) fn ensure_buffer_capacity(
8:     device: &RenderDevice,
9:     queue: &RenderQueue,
10:     buffer_opt: &mut Option<Buffer>,
11:     capacity: u64,
12:     usage: BufferUsages,
13:     label: &str,
14:     copy: bool,
15: ) {
16:     let aligned_size = (capacity + 3) & !3; // 4-byte align
17:     let current_size = buffer_opt.as_ref().map(|b| b.size()).unwrap_or(0);
18:     if aligned_size <= current_size {
19:         return;
20:     }
21: 
22:     let size = aligned_size.max(1024);
23: 
24:     #[cfg(feature = "trace")]
25:     trace!(
26:         "Resizing Buffer [{}]: {} bytes -> {} bytes (Copy: {})",
27:         label, current_size, size, copy
28:     );
29: 
30:     let buffer = device.create_buffer(&BufferDescriptor {
31:         label: Some(label),
32:         size,
33:         usage,
34:         mapped_at_creation: false,
35:     });
36: 
37:     if copy && let Some(old) = buffer_opt {
38:         let mut encoder = device.create_command_encoder(&default());
39:         let copy_size = old.size().min(size);
40: 
41:         encoder.copy_buffer_to_buffer(old, 0, &buffer, 0, copy_size);
42: 
43:         queue.submit(Some(encoder.finish()));
44:     }
45: 
46:     *buffer_opt = Some(buffer);
47: }

3: pub(super) fn clip(
4:     polygon: Vec<bevy::math::DVec2>,
5:     side: impl Fn(bevy::math::DVec2) -> f64,
6: ) -> Vec<bevy::math::DVec2> {
7:     let mut result = Vec::new();
8:     let Some(mut previous) = polygon.last().copied() else {
9:         return result;
10:     };
11:     let mut previous_side = side(previous);
12:     for point in polygon {
13:         let current_side = side(point);
14:         if (current_side >= 0.0) != (previous_side >= 0.0) {
15:             result.push(previous.lerp(point, previous_side / (previous_side - current_side)));
16:         }
17:         if current_side >= 0.0 {
18:             result.push(point);
19:         }
20:         previous = point;
21:         previous_side = current_side;
22:     }
23:     result
24: }

26: pub(super) fn subtract(
27:     mut inside: Vec<bevy::math::DVec2>,
28:     corners: &[bevy::math::DVec2],
29: ) -> Vec<Vec<bevy::math::DVec2>> {
30:     // Half-plane clipping can partition a wholly outside triangle along the
31:     // infinite extensions of property edges. Test the complete intersection
32:     // first: unrelated source triangles must keep their original topology.
33:     let mut intersection = inside.clone();
34:     for index in 0..corners.len() {
35:         let a = corners[index];
36:         let edge = corners[(index + 1) % corners.len()] - a;
37:         intersection = clip(intersection, |p| edge.perp_dot(p - a));
38:     }
39:     if intersection.len() < 3 || signed_area(&intersection) == 0.0 {
40:         return vec![inside];
41:     }
42:     let mut outside = Vec::new();
43:     for index in 0..corners.len() {
44:         let a = corners[index];
45:         let edge = corners[(index + 1) % corners.len()] - a;
46:         let remainder = clip(inside.clone(), |p| -edge.perp_dot(p - a));
47:         if remainder.len() >= 3 {
48:             outside.push(remainder);
49:         }
50:         inside = clip(inside, |p| edge.perp_dot(p - a));
51:     }
52:     outside
53: }

56:     let Some(origin) = polygon.first() else {
57:         return 0.0;
58:     };
59:     polygon
60:         .iter()
61:         .zip(polygon.iter().cycle().skip(1))
62:         .take(polygon.len())
