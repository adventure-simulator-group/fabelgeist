68:     pub(crate) fn validate(
69:         &self,
70:         heights: std::ops::RangeInclusive<u16>,
71:     ) -> Result<(), DesignError> {
72:         let b = self;
73:         if b.count_per_row > 8
74:             || !(1..=4).contains(&b.rows)
75:             || !(2..=6).contains(&b.width.0)
76:             || !(8..=20).contains(&b.length.0)
77:             || !(20..=60).contains(&b.span.0)
78:             || !(14..=25).contains(&b.row_spacing.0)
79:             || !(10..=100).contains(&b.center_offset.0)
80:             || !heights.contains(&b.height.0)
81:             || b.rounding.0 > 1000
82:             || !(-90..=90).contains(&b.inclination.0)
83:         {
84:             return Err(DesignError::ParametricParameters);
85:         }
86:         if b.count_per_row == 0 {
87:             return Ok(());
88:         }
89:         let angle = f32::from(b.inclination.0).to_radians();
90:         let width = f32::from(b.width.0) * angle.cos() + f32::from(b.length.0) * angle.sin().abs();
91:         let height = f32::from(b.length.0) * angle.cos() + f32::from(b.width.0) * angle.sin().abs();
92:         let web = crate::pierced_plate_domain::MINIMUM_WEB_MM as f32;
93:         let required_span =
94:             width * f32::from(b.count_per_row) + web * f32::from(b.count_per_row - 1);
95:         if required_span > f32::from(b.span.0)
96:             || (b.sides == VentSides::Both
97:                 && 2.0 * f32::from(b.center_offset.0) - f32::from(b.span.0) < web)
98:             || (b.rows > 1 && height + web > f32::from(b.row_spacing.0))
99:         {
100:             return Err(DesignError::VisorOpeningSpacing);
101:         }
102:         Ok(())
103:     }

105:     pub(crate) fn openings(
106:         &self,
107:         chart_height: f64,
108:         detail: crate::ArmorDetail,
109:     ) -> Vec<Vec<[f64; 2]>> {
110:         let b = self;
111:         let mut holes = Vec::new();
112:         let angle = f64::from(b.inclination.0).to_radians();
113:         let width = f64::from(b.width.0);
114:         let length = f64::from(b.length.0);
115:         let projected_width = width * angle.cos() + length * angle.sin().abs();
116:         for side in [-1.0, 1.0] {
117:             if matches!(b.sides, VentSides::Left) && side < 0.0
118:                 || matches!(b.sides, VentSides::Right) && side > 0.0
119:             {
120:                 continue;
121:             }
122:             for row in 0..b.rows {
123:                 for slot in 0..b.count_per_row {
124:                     let offset = if b.count_per_row == 1 {
125:                         0.0
126:                     } else {
127:                         (f64::from(slot) / f64::from(b.count_per_row - 1) - 0.5)
128:                             * (f64::from(b.span.0) - projected_width)
129:                     };
130:                     let center = [
131:                         side * (f64::from(b.center_offset.0) + offset),
132:                         f64::from(b.height.0) * chart_height / 1000.0
133:                             + (f64::from(row) - f64::from(b.rows - 1) * 0.5)
134:                                 * f64::from(b.row_spacing.0),
135:                     ];
136:                     holes.push(crate::pierced_plate_domain::rounded_slot(
137:                         center,
138:                         width,
139:                         length,
140:                         angle * side,
141:                         f64::from(b.rounding.0) / 1000.0,
142:                         detail,
143:                     ));
144:                 }
145:             }
146:         }
147:         holes
148:     }
