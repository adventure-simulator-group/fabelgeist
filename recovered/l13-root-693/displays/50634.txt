70:     /// Across-chart samples include every flute crest, slope and land edge.
71:     pub fn columns(&self, base_segments: usize) -> Vec<f32> {
72:         const PROFILE_SEGMENTS: usize = 8;
73:         const MERGE_TOLERANCE: f32 = 1e-5;
74:         let pitch = self.spread.unit() / f32::from(self.count.0);
75:         let left = (1.0 - self.spread.unit()) * 0.5;
76:         let right = 1.0 - left;
77:         let minimum_land = pitch * (1.0 - self.width.unit()) * 0.5;
78:         // The feature grid owns the fluted field. Interleaving an unrelated
79:         // regular grid creates tiny strips whose offset walls can reverse.
80:         let mut samples = (0..=base_segments)
81:             .map(|i| i as f32 / base_segments as f32)
82:             .filter(|u| *u < left - minimum_land || *u > right + minimum_land)
83:             .collect::<Vec<_>>();
84:         samples.extend([left, right]);
85:         for flute in 0..self.count.0 {
86:             let center = (1.0 - self.spread.unit()) * 0.5 + (f32::from(flute) + 0.5) * pitch;
87:             for sample in 0..=PROFILE_SEGMENTS {
88:                 samples.push(
89:                     center
90:                         + (sample as f32 / PROFILE_SEGMENTS as f32 - 0.5)
91:                             * pitch
92:                             * self.width.unit(),
93:                 );
94:             }
95:         }
96:         samples.sort_by(f32::total_cmp);
97:         samples.dedup_by(|a, b| (*a - *b).abs() < MERGE_TOLERANCE);
98:         samples
99:     }

102:     pub fn fan_coordinate(&self, u: f32, v: f32) -> f32 {
103:         let lateral = 2.0 * u - 1.0;
104:         let span = self.spread.unit();
105:         let fan = self.lower_spread.unit() + (1.0 - self.lower_spread.unit()) * smooth(v);
106:         let absolute = lateral.abs();
107:         let mapped = if absolute <= span {
108:             absolute * fan
109:         } else {
110:             span * fan + (absolute - span) * (1.0 - span * fan) / (1.0 - span)
111:         };
112:         (1.0 + lateral.signum() * mapped) * 0.5
113:     }
114: 
115:     pub fn relief(&self, u: f32, v: f32) -> f32 {
116:         let pitch = self.spread.unit() / f32::from(self.count.0);
117:         let start = (1.0 - self.spread.unit()) * 0.5;
118:         let slot = ((u - start) / pitch).floor();
119:         if slot < 0.0 || slot >= f32::from(self.count.0) {
120:             return 0.0;
121:         }
122:         let center = start + (slot + 0.5) * pitch;
123:         let distance = (u - center).abs() / (pitch * self.width.unit() * 0.5);
124:         if distance >= 1.0 {
125:             return 0.0;
126:         }
127:         let fade = smooth((v - self.start.unit()) / self.fade.unit())
128:             * smooth((self.end.unit() - v) / self.fade.unit());
129:         self.depth.metres() * fade * (1.0 + (std::f32::consts::PI * distance).cos()) * 0.5
130:     }
131: 
147: fn smooth(t: f32) -> f32 {
148:     let t = t.clamp(0.0, 1.0);
149:     t * t * (3.0 - 2.0 * t)
150: }
