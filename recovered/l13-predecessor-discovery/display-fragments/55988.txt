4: const BOND_DOMAIN: u64 = 14_u64 << 60;
5: const BOND_TOLERANCE_METRES: f32 = 0.005;
6: 
7: pub(super) fn resolve(geometry: &mut ResolvedGeometry) -> Result<(), crate::GenerationError> {
8:     let solids = &geometry.solids;
9:     let _: () = for (index, a) in solids.iter().enumerate() {
10:         for b in &solids[index + 1..] {
11:             if a.owner == b.owner || !bearing_pair(a, b) {
12:                 continue;
13:             }
14:             let (amin, amax) = bounds(a);
15:             let (bmin, bmax) = bounds(b);
16:             let min = amin.max(bmin);
17:             let max = amax.min(bmax);
18:             let overlap = max - min;
19:             if overlap.min_element() <= 0.0
20:                 || overlap.y > flight::DECK_THICKNESS_METRES + BOND_TOLERANCE_METRES
21:             {
22:                 continue;
23:             }
24:             let mut axes = [overlap.x, overlap.y, overlap.z];
25:             axes.sort_by(f32::total_cmp);
26:             geometry.junction_bonds.push(crate::JunctionBond {
27:                 id: ResolvedItemId(BOND_DOMAIN | geometry.junction_bonds.len() as u64),
28:                 owners: [a.owner, b.owner],
29:                 bounds: SpatialBounds::<Architectural>::from_metres(min, max)?,
30:                 minimum_interface_area_square_metres: axes[1] * axes[2] * 0.9,
31:                 maximum_penetration_metres: overlap.x.min(overlap.z) + BOND_TOLERANCE_METRES,
32:             });
33:         }
34:     };
35:     Ok(())
36: }
