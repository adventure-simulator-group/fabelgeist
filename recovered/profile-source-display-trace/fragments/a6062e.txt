48: fn partition_review_camera(building: &GeneratedBuilding) -> Result<BuildingReviewCamera> {
49:     let storey = &building.plan.storeys[0];
50:     let room = storey
51:         .rooms
52:         .iter()
53:         .max_by_key(|room| room.cells.len())
54:         .expect("partition review ground storey has a room");
55:     let wall = storey
56:         .walls
57:         .iter()
58:         .enumerate()
59:         .filter(|(_, wall)| wall.inside_room == room.id && !wall.exterior())
60:         .max_by_key(|(index, _)| {
61:             usize::from(storey.openings.iter().any(|opening| opening.wall == *index))
62:         })
63:         .map(|(_, wall)| *wall)
64:         .expect("partition review building has an internal wall facing its largest room");
65:     let centroid =
66:         room.cells.iter().map(|cell| cell.centre()).sum::<Vec2>() / room.cells.len() as f32;
67:     let local_origin = building.collision.bounds.centre()?.metres();
68:     let transform = building.transform()?;
69:     Ok(BuildingReviewCamera {
70:         position: Position::from_metres(
71:             transform.transform_point(Vec3::new(centroid.x, 1.48, centroid.y) - local_origin),
72:         )?,
73:         target: Position::from_metres(
74:             transform
75:                 .transform_point(Vec3::new(wall.centre().x, 1.38, wall.centre().y) - local_origin),
76:         )?,
77:         plaster_raking_light: None,
78:     })
