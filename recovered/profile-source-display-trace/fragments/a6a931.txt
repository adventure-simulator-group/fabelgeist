9:     STANDARD_CAPTURE_PIXELS.x as f32 / STANDARD_CAPTURE_PIXELS.y as f32;
10: // Match the frozen terrain comparison images. This is a capture setting, not

14: pub(super) fn physical_pixels(profile: &str) -> UVec2 {
15:     if profile == super::fixed_city_cameras::PROFILE {
16:         TERRAIN_COMPARISON_PIXELS
17:     } else {
18:         STANDARD_CAPTURE_PIXELS
19:     }
20: }

22: /// Start with the requested physical dimensions before native window creation.
23: pub(super) fn window_resolution(profile: &str) -> WindowResolution {
24:     let pixels = physical_pixels(profile);
25:     WindowResolution::new(pixels.x, pixels.y).with_scale_factor_override(1.0)
26: }
27: 
28: fn restore_resolution(profile: &str, resolution: &mut WindowResolution) {
29:     let pixels = physical_pixels(profile);
30:     resolution.set_physical_resolution(pixels.x, pixels.y);
31: }
32: 
33: /// Native window creation applies the host scale to physical dimensions even
34: /// with an override. Restore the capture size after creation, before readbacks.
35: pub(super) fn apply(
36:     state: Res<super::capture_state::SceneCaptureState>,
37:     mut window: Single<&mut Window, With<PrimaryWindow>>,
38: ) {
39:     restore_resolution(&state.profile, &mut window.resolution);
40: }

67:     ground_materials: Res<'w, Assets<CityGroundMaterial>>,

73: impl Observation<'_, '_> {
74:     fn check(&self, expected: &ExpectedFurniture) -> bool {
75:         !self.ground.is_empty()
76:             && self.ground.iter().all(|(mesh, handle)| {
77:                 self.ground_materials
78:                     .get(&handle.0)
79:                     .is_some_and(|material| {
80:                         self.gpu.contains(&mesh.0, material.extension.texture_ids())
81:                     })
82:             })
83:             && self.furniture.iter().count() == expected.instances
84:             && self.batches.iter().count() == expected.batches
85:             && self.batches.iter().all(|(batch, parent, mesh, handle)| {
86:                 let Ok(instance) = self.furniture.get(parent.parent()) else {
87:                     return false;
88:                 };
89:                 if handle.0 != self.palette.for_building(instance.id.0).get(batch.material) {
90:                     return false;
91:                 }
92:                 self.materials.get(&handle.0).is_some_and(|material| {
93:                     self.gpu.contains(
94:                         &mesh.0,
95:                         [
96:                             &material.base_color_texture,
97:                             &material.normal_map_texture,
98:                             &material.metallic_roughness_texture,
99:                         ]
100:                         .into_iter()
101:                         .flatten()
102:                         .map(Handle::id),
103:                     )
104:                 })
105:             })
106:     }
107: }

109: fn observe(
110:     mut readiness: ResMut<Readiness>,
111:     expected: Option<Res<ExpectedFurniture>>,
112:     state: Option<Res<SceneCaptureState>>,
113:     time: Res<Time<Real>>,
114:     observation: Observation,
115: ) {
116:     let (Some(expected), Some(state)) = (expected, state) else {
117:         return;
118:     };
119:     if state.phase == CapturePhase::Configure {
120:         readiness.ready = false;
121:         readiness.pending_since = None;
122:         return;
123:     }
124:     readiness.ready = observation.check(&expected);
125:     if !readiness.ready {
126:         let since = *readiness
127:             .pending_since
128:             .get_or_insert(time.elapsed_secs_f64());
129:         assert!(
130:             time.elapsed_secs_f64() - since < MAX_ASSET_WAIT_SECONDS,
131:             "furniture production meshes or textures did not become GPU resident"
132:         );
133:         return;
134:     }
135:     if readiness.recorded.insert(state.view) {
136:         let evidence = serde_json::json!({
137:             "renderer": "TacticalPresentationPlugin", "backend": format!("{:?}", observation.adapter.backend),
138:             "adapter": observation.adapter.name, "instances": expected.instances, "batches": expected.batches,
139:             "material_bindings_verified": true, "ground_traffic_masks_gpu_resident": true, "captured_views_ready": readiness.recorded,
140:         });
141:         std::fs::write(
142:             state.output.join("furniture-presentation.json"),
143:             serde_json::to_vec_pretty(&evidence).unwrap(),
144:         )
145:         .expect("write production furniture evidence");
146:     }
147: }

20: impl Camera {

22:         let position = Vec3::from_array(self.position_metres);
23:         let target = Vec3::from_array(self.target_metres);
24:         position.is_finite()
25:             && target.is_finite()

27:     }

32:             target: Position::from_metres(Vec3::from_array(self.target_metres))?,
33:             plaster_raking_light: None,

35:     }

79: pub(super) fn apply(contract: Res<Contract>, mut state: ResMut<SceneCaptureState>) -> Result {
80:     let Some(document) = &contract.0 else {
81:         return Ok(());
82:     };
83:     let exterior = document
84:         .exterior
85:         .iter()
86:         .map(Camera::review)
87:         .collect::<Result<Vec<_>>>()?;
88:     let benchmark = document
89:         .benchmark
90:         .as_ref()
91:         .map(Camera::review)
92:         .transpose()?;
93:     state.city_exterior_cameras = exterior;
94:     if let Some(camera) = benchmark {
95:         state.ground_eye_position = camera.position;
96:         state.ground_eye_target = camera.target;
97:     }
98:     std::fs::write(
99:         state.output.join("fixed-camera-contract.json"),
100:         serde_json::to_vec_pretty(document)?,
101:     )?;
102:     Ok(())
103: }
