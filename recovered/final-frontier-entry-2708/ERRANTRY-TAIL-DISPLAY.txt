738:     fn subset_style_score(
739:         pool: &[OrderedSigilClue],
740:         indices: &[usize],
741:     ) -> (usize, usize, Vec<usize>) {
742:         let exact = indices
743:             .iter()
744:             .filter(|index| matches!(pool[**index], OrderedSigilClue::Exact { .. }))
745:             .count();
746:         let not_at = indices
747:             .iter()
748:             .filter(|index| matches!(pool[**index], OrderedSigilClue::NotAt { .. }))
749:             .count();
750:         (exact, not_at, indices.to_vec())
751:     }

753:     fn best_unique_subset(
754:         pool: &[OrderedSigilClue],
755:         solution: &[Sigil; ORDERED_SIGIL_COUNT],
756:         cardinality: usize,
757:     ) -> Option<(usize, usize, Vec<usize>)> {
758:         let orderings = all_orderings();
759:         let target = 1_u128
760:             << orderings
761:                 .iter()
762:                 .position(|candidate| candidate == solution)
763:                 .unwrap();
764:         let masks = pool
765:             .iter()
766:             .map(|clue| {
767:                 orderings
768:                     .iter()
769:                     .enumerate()
770:                     .fold(0_u128, |mask, (index, ordering)| {
771:                         mask | (u128::from(clue_holds(clue, ordering)) << index)
772:                     })
773:             })
774:             .collect::<Vec<_>>();
775:         let mut best = None;
776:         visit_mask_subsets(
777:             pool,
778:             &masks,
779:             cardinality,
780:             0,
781:             (1_u128 << orderings.len()) - 1,
782:             target,
783:             &mut Vec::new(),
784:             &mut best,
785:         );
786:         best
787:     }
788: 
789:     #[expect(
790:         clippy::too_many_arguments,
791:         reason = "this domain boundary names each independent input explicitly"
792:     )]
793:     fn visit_mask_subsets(
794:         pool: &[OrderedSigilClue],
795:         masks: &[u128],
796:         remaining: usize,
797:         start: usize,
798:         intersection: u128,
799:         target: u128,
800:         chosen: &mut Vec<usize>,
801:         best: &mut Option<(usize, usize, Vec<usize>)>,
802:     ) {
803:         if remaining == 0 {
804:             if intersection == target {
805:                 let candidate = subset_style_score(pool, chosen);
806:                 if best.as_ref().is_none_or(|current| candidate < *current) {
807:                     *best = Some(candidate);
808:                 }
809:             }
810:             return;
811:         }
812:         for index in start..=pool.len() - remaining {
813:             chosen.push(index);
814:             visit_mask_subsets(
815:                 pool,
816:                 masks,
817:                 remaining - 1,
818:                 index + 1,
819:                 intersection & masks[index],
820:                 target,
821:                 chosen,
822:                 best,
823:             );
824:             chosen.pop();
825:         }
826:     }
