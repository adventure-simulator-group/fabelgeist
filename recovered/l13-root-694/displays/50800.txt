90:     pub fn validate(&self) -> Result<(), GenerateError> {
91:         self.plate_shape.validate(self.kind)?;
92:         if let Some(pattern) = &self.fluting {
93:             if self.plate_shape == GarmentPlateShape::None {
94:                 return Err(DesignError::ParametricParameters.into());
95:             }
96:             pattern.validate()?;
97:         }
98:         if !(1..=40).contains(&self.clearance.0)
99:             || !(1..=16).contains(&self.wall_thickness.0)
100:             || !(if self.kind == GarmentArmorKind::Fauld {
101:                 100
102:             } else {
103:                 500
104:             }..=1_300)
105:                 .contains(&self.length.0)
106:             || self.flare.0 > 500
107:             || !(800..=1_100).contains(&self.waist.0)
108:             || !(1..=if self.kind == GarmentArmorKind::Tassets {
109:                 12
110:             } else {
111:                 8
112:             })
113:                 .contains(&self.lame_count)
114:         {
115:             return Err(DesignError::ParametricParameters.into());
116:         }
117:         if let GarmentPlateShape::WrappedTassets(shape) = self.plate_shape
118:             && shape.section_break >= self.lame_count
119:         {
120:             return Err(DesignError::ParametricParameters.into());
121:         }
122:         if let GarmentPlateShape::Fauld { front_arch, .. } = self.plate_shape
123:             && front_arch.0 >= self.length.0
124:         {
125:             return Err(DesignError::ParametricParameters.into());
126:         }
127:         Ok(())
128:     }
