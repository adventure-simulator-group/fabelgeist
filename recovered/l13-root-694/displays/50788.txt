1: //! Parametric sampling for shipped armor and its separate bake source.
2: //!
3: //! Detail levels change where a recipe is evaluated. They never collapse edges
4: //! or weld vertices from an already triangulated armor mesh.
5: 
6: #[derive(Clone, Copy, Debug, Eq, PartialEq)]
7: pub enum ArmorLod {
8:     Lod4,
9:     Lod5,
10:     Lod6,
11: }
12: 
13: impl TryFrom<u8> for ArmorLod {
14:     type Error = &'static str;
15: 
16:     fn try_from(value: u8) -> Result<Self, Self::Error> {
17:         match value {
18:             4 => Ok(Self::Lod4),
19:             5 => Ok(Self::Lod5),
20:             6 => Ok(Self::Lod6),
21:             _ => Err("armor LOD must be 4, 5 or 6"),
22:         }
23:     }
24: }
25: 
26: #[derive(Clone, Copy, Debug, Eq, PartialEq)]
27: pub enum ArmorDetail {
28:     Runtime(ArmorLod),
29:     BakeSource,
30: }
31: 
32: impl ArmorDetail {
33:     /// Uniform intervals between structural boundaries. Callers retain explicit
34:     /// openings, plate overlaps, tips and other authored feature coordinates.
35:     pub fn segments(self, source: usize, minimum: usize) -> usize {
36:         let divisor = match self {
37:             Self::BakeSource => 1,
38:             Self::Runtime(ArmorLod::Lod4) => 12,
39:             Self::Runtime(ArmorLod::Lod5) => 16,
40:             Self::Runtime(ArmorLod::Lod6) => 24,
41:         };
42:         source.div_ceil(divisor).max(minimum).min(source)
43:     }
