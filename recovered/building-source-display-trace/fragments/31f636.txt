32:         let side_length = ((length - gate_width) * 0.5).max(0.1);
33:         let midpoint = (start + end) * 0.5;
34:         for sign in [-1.0, 1.0] {
35:             let centre = midpoint + tangent * (gate_width + side_length) * 0.5 * sign;

38:                 centre,

47:             midpoint,

51:         );

54:                 world,
55:                 &palette.void,
56:                 if horizontal {
57:                     Vec3::new(gate_width * 0.9, wall.gate_height_metres * 0.94, 0.08)
58:                 } else {
59:                     Vec3::new(0.08, wall.gate_height_metres * 0.94, gate_width * 0.9)
60:                 },
61:                 Vec3::new(midpoint.x, wall.gate_height_metres * 0.47, midpoint.y),
62:                 Quat::IDENTITY,
63:                 "open curtain-wall gate passage",
64:             );
65:         }

68:             Direction::East => -Vec2::X,

105:                 }
106:             }
107:         }

109:         wall_box(
110:             world,

112:             length,

114:             wall.height_metres * 0.5,
115:         );
116:     }

119: fn spawn_gatehouse_curtain(
120:     world: &mut World,
121:     palette: &RenderPalette,
122:     wall: CurtainWallRun,
123:     defense: &GateDefense,
124:     towers: &[RoundTower],
125:     origin: Vec2,
126: ) {
127:     let GatehouseLoadPath::BondedTowerBearing {
128:         left_tower_index,
129:         right_tower_index,
130:         arch_centre,
131:         arch_spring_elevation_metres,
132:         arch_ring_depth,
133:         arch_rise,
134:         curtain_return_bond,
135:         ..
136:     } = defense.guard_chamber.load_path;
137:     let (Some(left), Some(right)) = (
138:         towers.get(left_tower_index).copied(),
139:         towers.get(right_tower_index).copied(),
140:     ) else {
141:         return;
142:     };
143:     let tangent = (wall.end - wall.start).normalize_or_zero();
144:     let return_bond = curtain_return_bond.metres();
145:     let left_outer = left.centre_metres() - tangent * (left.radius_metres() - return_bond);
146:     let right_outer = right.centre_metres() + tangent * (right.radius_metres() - return_bond);
147:     for (start, end) in [(wall.start, left_outer), (right_outer, wall.end)] {
148:         if (end - start).length() > 0.05 {
149:             spawn_curtain_wall(
150:                 world,
151:                 palette,
152:                 CurtainWallRun {
153:                     start,
154:                     end,
155:                     gate_width_metres: None,
156:                     ..wall
157:                 },
158:                 origin,
159:                 &[],
160:             );
161:         }
162:     }
163: 
164:     let chamber = &defense.guard_chamber;
165:     let horizontal = tangent.x.abs() >= tangent.y.abs();
166:     let arch_depth = chamber.size.dot(direction_vector_2d(wall.outward).abs());
167:     let ring = arch_ring_depth.metres();
168:     let rise = arch_rise.metres();
169:     let half_span = wall.gate_width_metres.unwrap_or(3.2) * 0.5;
170:     let segments = 15;
171:     let block_width = half_span * 2.0 / segments as f32;
172:     for segment in 0..segments {
173:         let along = -half_span + (segment as f32 + 0.5) * block_width;
174:         let normalized = along / half_span;
175:         let elevation =
176:             arch_spring_elevation_metres + ring * 0.5 + rise * (1.0 - normalized * normalized);
177:         let slope = -2.0 * rise * normalized / half_span;
178:         let angle = slope.atan();
179:         let position = arch_centre + origin + tangent * along;
180:         spawn_box(
181:             world,
182:             &palette.stone,
183:             if horizontal {
184:                 Vec3::new(block_width * 1.12, ring, arch_depth)
185:             } else {
186:                 Vec3::new(arch_depth, ring, block_width * 1.12)
187:             },
188:             Vec3::new(position.x, elevation, position.y),
189:             if horizontal {
190:                 Quat::from_rotation_z(angle)
191:             } else {
192:                 Quat::from_rotation_x(-angle * tangent.y.signum())
193:             },
194:             "bonded segmental gate arch voussoir",
195:         );
196:     }
197:     let chamber_along = chamber.size.dot(tangent.abs());
198:     let shoulder_width = ((chamber_along - half_span * 2.0) * 0.5).max(0.0);
199:     let spandrel_height = ring + rise;
200:     for sign in [-1.0, 1.0] {
201:         spawn_wall_local_box(
202:             world,
203:             &palette.stone,
204:             chamber.centre + origin,
205:             tangent,
206:             direction_vector_2d(wall.outward),
207:             sign * (half_span + shoulder_width * 0.5),
208:             0.0,
209:             shoulder_width,
210:             arch_depth,
211:             spandrel_height,
212:             arch_spring_elevation_metres + spandrel_height * 0.5,
213:             "tower-bonded gate arch spandrel bearing",
214:         );
215:     }
216:     spawn_gate_closures(world, palette, wall, defense, origin);
217: }
