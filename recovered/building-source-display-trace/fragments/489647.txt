219: fn validate_projected_suite_records(
220:     records: &[(&str, ProjectedSuiteManifest)],
221: ) -> Result<(), String> {
222:     if records.len() != PROJECTED_PROOF_SUITE.len() {
223:         return Err(format!(
224:             "expected {} projected-defense manifests, found {}",
225:             PROJECTED_PROOF_SUITE.len(),
226:             records.len()
227:         ));
228:     }
229:     let first = &records[0].1;
230:     let mut fixtures = std::collections::HashMap::<(&str, Seed), (&str, &str)>::new();
231:     for ((actual_name, manifest), expected) in records.iter().zip(PROJECTED_PROOF_SUITE) {
232:         if *actual_name != expected.basename
233:             || manifest.fixture != expected.fixture
234:             || manifest.view != expected.view
235:             || manifest.seed != expected.seed
236:             || manifest.projected_defense_kind.as_deref() != expected.kind
237:             || manifest.projected_defense_deployment.as_deref() != expected.deployment
238:         {
239:             return Err(format!(
240:                 "projected proof {actual_name} does not match its expected fixture/view/seed/state"
241:             ));
242:         }
243:         if !manifest.validation_passed || manifest.resolver_schema_version != 2 {
244:             return Err(format!(
245:                 "projected proof {actual_name} is invalid or not resolver schema 2"
246:             ));
247:         }
248:         if manifest.source_revision != first.source_revision
249:             || manifest.source_dirty_fingerprint != first.source_dirty_fingerprint
250:         {
251:             return Err(format!(
252:                 "projected proof {actual_name} comes from a mixed source build"
253:             ));
254:         }
255:         if expected.kind.is_some()
256:             && (manifest.focus_kind.as_deref() != Some("resolved_projected")
257:                 || manifest.focused_resolved_item_ids.is_empty()
258:                 || manifest.focused_resolved_void_ids.is_empty()
259:                     && expected.deployment != Some("sockets_only")
260:                 || manifest.focused_projected_ray_count == 0
261:                     && expected.deployment != Some("sockets_only")
262:                 || manifest.projected_tactical_target.is_none())
263:         {
264:             return Err(format!(
265:                 "projected proof {actual_name} lacks exact assembly IDs, voids, rays, or tactical target"
266:             ));
267:         }
268:         let fixture_key = (expected.fixture, expected.seed);
269:         if let Some((plan_hash, resolved_hash)) = fixtures.get(&fixture_key) {
270:             if *plan_hash != manifest.plan_hash || *resolved_hash != manifest.resolved_geometry_hash
271:             {
272:                 return Err(format!(
273:                     "projected proof {actual_name} disagrees with its fixture/seed hashes"
274:                 ));
275:             }
276:         } else {
277:             fixtures.insert(
278:                 fixture_key,
279:                 (&manifest.plan_hash, &manifest.resolved_geometry_hash),
280:             );
281:         }
282:     }
283:     Ok(())
284: }

287:     let mut owned = Vec::with_capacity(PROJECTED_PROOF_SUITE.len());
288:     for expected in PROJECTED_PROOF_SUITE {

292:             .map_err(|error| format!("parse {}: {error}", path.display()))?;
293:         owned.push((expected.basename, manifest));
294:     }
295:     validate_projected_suite_records(&owned)
