113: pub(crate) fn validate_timber_suite(directory: &std::path::Path) -> Result<(), String> {
114:     let specs = timber_proof_specs();
115:     let actual_count = fs::read_dir(directory)
116:         .map_err(|error| format!("read {}: {error}", directory.display()))?
117:         .filter_map(Result::ok)
118:         .filter(|entry| {
119:             entry
120:                 .file_name()
121:                 .to_string_lossy()
122:                 .ends_with(".capture.json")
123:         })
124:         .count();
125:     if actual_count != specs.len() {
126:         return Err(format!(
127:             "expected exactly {} timber manifests, found {actual_count}",
128:             specs.len()
129:         ));
130:     }
131:     let mut records = Vec::new();
132:     for (slug, archetype, view) in specs {
133:         let path = directory.join(format!("{slug}.capture.json"));
134:         let manifest: TimberSuiteManifest = serde_json::from_slice(
135:             &fs::read(&path).map_err(|error| format!("read {}: {error}", path.display()))?,
136:         )
137:         .map_err(|error| format!("parse {}: {error}", path.display()))?;
138:         records.push((slug, archetype, view, manifest));
139:     }
140:     validate_timber_suite_records(&records)
141: }
