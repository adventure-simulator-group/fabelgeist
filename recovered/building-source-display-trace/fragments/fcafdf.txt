357: fn validate_timber_suite_records(
358:     records: &[(String, BuildingArchetype, ViewerView, TimberSuiteManifest)],
359: ) -> Result<(), String> {
360:     if records.len() != 35 {
361:         return Err(format!(
362:             "expected 35 timber proofs, found {}",
363:             records.len()
364:         ));
365:     }
366:     let first = &records[0].3;
367:     let mut fixture_hashes = std::collections::HashMap::<String, (String, String, String)>::new();
368:     let mut pixels = std::collections::HashSet::new();
369:     for (slug, archetype, view, manifest) in records {
370:         let expected_view = timber_proof_suffix(*view).expect("timber proof suffix");
371:         if manifest.fixture != archetype.slug()
372:             || manifest.view != expected_view
373:             || manifest.seed != fabelgeist_determinism::Seed::from_u64(47)
374:             || manifest.resolver_schema_version != 2
375:             || manifest.plan_audit_issue_count != 0
376:             || !manifest.validation_passed
377:         {
378:             return Err(format!(
379:                 "{slug} violates its fixture/view validation contract"
380:             ));
381:         }
382:         if manifest.source_revision != first.source_revision
383:             || manifest.source_dirty_fingerprint != first.source_dirty_fingerprint
384:         {
385:             return Err(format!("{slug} comes from mixed or stale source authority"));
386:         }
387:         if let Some((plan_hash, geometry_hash, frame_hash)) = fixture_hashes.get(&manifest.fixture)
388:         {
389:             if plan_hash != &manifest.plan_hash
390:                 || geometry_hash != &manifest.resolved_geometry_hash
391:                 || frame_hash != &manifest.timber_program_hash
392:             {
393:                 return Err(format!("{slug} is fixture-inconsistent"));
394:             }
395:         } else {
396:             fixture_hashes.insert(
397:                 manifest.fixture.clone(),
398:                 (
399:                     manifest.plan_hash.clone(),
400:                     manifest.resolved_geometry_hash.clone(),
401:                     manifest.timber_program_hash.clone(),
402:                 ),
403:             );
404:         }
405:         if manifest.timber_program.is_none()
406:             || manifest.timber_assembly_id.is_none()
407:             || manifest.timber_member_ids.len() < 20
408:             || manifest.timber_joint_ids.len() < 12
409:             || manifest.timber_node_ids.len() < 12
410:             || manifest.focused_resolved_item_ids.is_empty()
411:             || manifest.visible_focused_resolved_item_count
412:                 + manifest.timber_removed_target_item_ids.len()
413:                 != manifest.focused_resolved_item_ids.len()
414:             || manifest.timber_target_component_ids.len() != 1
415:             || !manifest.timber_target_component_ids[0].starts_with("timber:")
416:             || !manifest.timber_target_component_ids[0].contains('/')
417:             || manifest.timber_focus_interface_ids.is_empty()
418:             || !manifest
419:                 .timber_removed_target_item_ids
420:                 .iter()
421:                 .all(|id| manifest.section_removed_item_ids.contains(id))
422:             || !manifest.timber_legend_visible
423:             || manifest.timber_required_roles.is_empty()
424:             || manifest.timber_focused_roles.is_empty()
425:             || !manifest.section_annotation_visible
426:             || manifest.timber_required_roles.iter().any(|role| {
427:                 !manifest
428:                     .timber_focused_roles
429:                     .iter()
430:                     .any(|found| found == role)
431:                     || manifest
432:                         .timber_role_item_ids
433:                         .get(role)
434:                         .is_none_or(Vec::is_empty)
435:                     || manifest
436:                         .timber_role_bounds_fraction
437:                         .get(role)
438:                         .is_none_or(|bounds| {
439:                             bounds[0] < 0.0
440:                                 || bounds[1] < 0.0
441:                                 || bounds[2] > 1.0
442:                                 || bounds[3] > 1.0
443:                                 || (bounds[2] - bounds[0]) * (bounds[3] - bounds[1]) < 0.0004
444:                         })
445:             })
446:         {
447:             return Err(format!(
448:                 "{slug} lacks exact frame IDs, roles, focus, or legend"
449:             ));
450:         }
451:         if matches!(
452:             view,
453:             ViewerView::TimberOpeningBayExterior
454:                 | ViewerView::TimberOpeningBayInterior
455:                 | ViewerView::TimberOpeningBaySection
456:         ) && (manifest.focused_resolved_void_ids.len() != 1
457:             || manifest
458:                 .timber_role_item_ids
459:                 .get("WallHost")
460:                 .is_none_or(Vec::is_empty))
461:         {
462:             return Err(format!(
463:                 "{slug} does not prove both exact opening void and Gefach cells"
464:             ));
465:         }
466:         if *view == ViewerView::TimberJointClose
467:             && (manifest
468:                 .timber_focused_roles
469:                 .iter()
470:                 .any(|role| role == "WallHost")
471:                 || manifest.timber_focus_interface_ids.len() < 2)
472:         {
473:             return Err(format!(
474:                 "{slug} hides its participant contact behind enclosure geometry"
475:             ));
476:         }
477:         if *view == ViewerView::TimberGableRoofBearing && manifest.focused_roof_item_ids.is_empty()
478:         {
479:             return Err(format!("{slug} omits the exact Stage 4 roof face"));
480:         }
481:         let expects_cut = timber_section_proof(*view);
482:         if expects_cut != manifest.section_cut_applied
483:             || expects_cut != manifest.timber_cut_plane.is_some()
484:         {
485:             return Err(format!("{slug} lacks its exact declared cut state"));
486:         }
487:         let bounds = manifest.focused_bounds_fraction;
488:         if bounds[0] < 0.0
489:             || bounds[1] < 0.0
490:             || bounds[2] > 1.0
491:             || bounds[3] > 1.0
492:             || bounds[2] - bounds[0] < 0.12
493:             || bounds[3] - bounds[1] < 0.20
494:         {
495:             return Err(format!("{slug} target is clipped or too small"));
496:         }
497:         if manifest.pixel_hash.is_empty() || !pixels.insert(manifest.pixel_hash.clone()) {
498:             return Err(format!("{slug} lacks unique pixel evidence"));
499:         }
500:     }
501:     Ok(())
502: }
