32: fn spawn_square_tower(
33:     world: &mut World,
34:     palette: &RenderPalette,
35:     tower: SquareTower,
36:     origin: Vec2,
37:     view: ViewerView,
38: ) {
39:     if view == ViewerView::Cutaway {
40:         return;
41:     }
42:     let centre = tower.centre + origin;
43:     // Bell towers hand authority to resolved SquareTowerFace bays at 8 m so
44:     // their roof junction and bell openings can own real subtractions.  Keep
45:     // only the monolithic grounded base here; spawning the old full-height box
46:     // would duplicate those resolved walls and conceal the abutment contour.
47:     let lower_height = if tower.bell_openings {
48:         8.0
49:     } else {
50:         tower.wall_height_metres
51:     };
52:     spawn_box(
53:         world,
54:         &palette.stone,
55:         Vec3::new(tower.size.x, lower_height, tower.size.y),
56:         Vec3::new(centre.x, lower_height * 0.5, centre.y),
57:         Quat::IDENTITY,
58:         "square bell-tower lower mass",
59:     );
60:     // The bell stage itself is rendered exclusively from its resolved
61:     // WallAssembly/OpeningAssembly bays. Keeping a second viewer-owned stage
62:     // here would conceal voids and duplicate the authoritative masonry.
63:     // The roof is rendered once from the authoritative RoofAssembly graph.
64: }
