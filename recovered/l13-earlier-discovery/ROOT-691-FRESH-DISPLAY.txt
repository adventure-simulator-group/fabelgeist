0051                 <serde_json::Error as serde::de::Error>::custom(format!(

0061                         .placements
0062                         .iter()
0063                         .find(|placement| placement.id == asset.placement_id)
0064                 })
0065                 .ok_or_else(|| {
0066                     <serde_json::Error as serde::de::Error>::custom(format!(
0067                         "unknown placement {} for {}",
0068                         asset.placement_id, item.id
0069                     ))
0070                 })?;
0071             if entries.iter().any(|entry| {
0072                 std::ptr::eq(entry.item, item) && std::ptr::eq(entry.placement, placement)
0073             }) {
0074                 return Err(<serde_json::Error as serde::de::Error>::custom(
0075                     "procedural equipment manifest keys must be unique",
0076                 ));
0077             }
0078             entries.push(EquipmentAssetEntry {
0079                 item,
0080                 placement,
0081                 file: asset.file,
0082             });
0083         }
0084         // Preserve the previous nested ordered-map default-placement selection.
0085         entries.sort_by(|left, right| {
0086             (&left.item.id, &left.placement.id).cmp(&(&right.item.id, &right.placement.id))
0087         });

0089     }
