12: pub(super) fn resolved_energy_log(result: AttackResult) -> ResolvedEnergyLog {
13:     match result {
14:         AttackResult::ToAttacker {
15:             contact_force,
16:             physical_contact,
17:             ..
18:         } => ResolvedEnergyLog {
19:             outcome: if physical_contact {
20:                 TacticalContactOutcome::Defended
21:             } else {
22:                 TacticalContactOutcome::Avoided
23:             },
24:             coverage_contact: TacticalCoverageContact::None,
25:             armor_impact: None,
26:             contact: contact_force,
27:             cut: 0.0,
28:             blunt: 0.0,
29:         },
30:         AttackResult::ToDefender {
31:             cut_damage,
32:             blunt_damage,
33:             contact_force,
34:             armor_impact,
35:             ..
36:         } => ResolvedEnergyLog {
37:             outcome: TacticalContactOutcome::Hit,
38:             coverage_contact: if armor_impact.is_some() {
39:                 TacticalCoverageContact::ArmorSurface
40:             } else {
41:                 TacticalCoverageContact::Gap
42:             },
43:             armor_impact,
44:             contact: contact_force,
45:             cut: cut_damage,
46:             blunt: blunt_damage,
47:         },
48:     }
49: }
