132: fn defensive_implement_kind(
133:     event: &MeleeAttackResolved,
134:     viewer: &TacticalPlayerViewer<'_, '_>,
135: ) -> Option<TacticalDefenseImplement> {
136:     let slot = event.defender_blocking_slot?;
137:     let defender = viewer.get(event.target).ok()?;
138:     let side_slot = |side| match side {
139:         BodySide::Left => Some(EquipSlot::HoldingLeft),
140:         BodySide::Right => Some(EquipSlot::HoldingRight),
141:         BodySide::Both => None,
142:     };
143:     if defender.shield_holding_side().and_then(side_slot) == Some(slot) {
144:         Some(TacticalDefenseImplement::Shield)
145:     } else if defender.weapon_holding_side().and_then(side_slot) == Some(slot) {
146:         Some(TacticalDefenseImplement::Weapon)
147:     } else {
148:         Some(TacticalDefenseImplement::Other)
149:     }
150: }

156:         DefenderResponse::Parry { .. } => TacticalDecision::Parry,
157:         DefenderResponse::Dodge { .. } => TacticalDecision::Dodge,
