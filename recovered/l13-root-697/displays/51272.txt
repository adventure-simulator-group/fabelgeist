6: /// Where a tied style's strands meet, and the path the tie hangs along.
7: pub(crate) struct Tied {
8:     kind: Tie,
9:     /// The tie, off the scalp.
10:     pub(crate) point: Vec3,
11:     /// Out of the head at the tie.
12:     pub(crate) out: Vec3,
13:     /// The tail's axis from the tie, a polyline two millimetres a step.
14:     pub(crate) axis: Vec<Vec3>,

17: impl Tied {
18:     pub(crate) fn new(head: &Head, hairdo: &Hairdo) -> Option<Self> {
19:         let Flow::Tie(kind) = hairdo.flow() else {
20:             return None;
21:         };
22:         let c = head.cranium.center;
23:         let r = head.cranium.radii;
24:         // Up the back of the head: high for a tail, on the crown's back for
25:         // a bun, low at the nape for a braid.
26:         let a = hairdo.hairstyle.tie;
27:         let on_scalp = c + Vec3::new(0.0, r.y * a.sin(), -r.z * a.cos());
28:         let out = head.gradient(on_scalp);
29:         let point = on_scalp + out * (0.012 - head.distance(on_scalp));
30:         // The tail leaves the tie out of the head and bends down as a
31:         // cantilever -- stiffer than one strand: it is a bundle.
32:         let mut axis = vec![point];
33:         if !matches!(kind, Tie::Bun) {
34:             // As long as the longest hair: clippers leave the back short.
35:             let length = hairdo.haircut.longest();
36:             let stiffness = hairdo.hairstyle.hold * 2.0;
37:             let mut p = point;
38:             let mut d = safe_normalize(out + Vec3::down() * 0.4, Vec3::down());
39:             let ds = 0.002;
40:             let mut s = 0.0;
41:             while s < length {
42:                 d = cantilever(d, Vec3::down(), length - s, stiffness, ds);
43:                 let mut next = p + d * ds;
44:                 let distance = head.distance(next);
45:                 if distance < 0.02 {
46:                     next += head.gradient(next) * (0.02 - distance);
47:                     next = p + safe_normalize(next - p, d) * ds;
48:                 }
49:                 d = safe_normalize(next - p, d);
50:                 p = next;
51:                 axis.push(p);
52:                 s += ds;
53:             }
54:         }
55:         if axis.len() < 2 {
56:             axis.push(point + out * 0.002);
57:         }
58:         Some(Self {
59:             kind,
60:             point,
61:             out,
62:             axis,
63:         })
64:     }

66:     /// The axis at arc length `s` from the tie, its direction and a frame
67:     /// across it (the second axis points out of the head).
68:     fn frame(&self, s: f32) -> (Vec3, Vec3, Vec3, Vec3) {
69:         let f = (s / 0.002).max(0.0);
70:         let i = (f as usize).min(self.axis.len().saturating_sub(2));
71:         let t = (f - i as f32).clamp(0.0, 1.0);
72:         let p = self.axis[i].lerp(self.axis[i + 1], t);
73:         let d = safe_normalize(self.axis[i + 1] - self.axis[i], Vec3::down());
74:         let across = safe_normalize(d.cross(self.out), Vec3::right());
75:         let depth = across.cross(d);
76:         (p, d, across, depth)
77:     }
78: 
79:     /// Where a strand is `along` metres past the tie: in the tail, coiled
80:     /// in the bun or plaited in its lane of the braid, `bundle` off the
81:     /// middle.
82:     pub(super) fn bundled(&self, along: f32, bundle: Vec3, lane: f32, phase: f32) -> Vec3 {
83:         use std::f32::consts::TAU;
84:         match self.kind {
85:             Tie::Tail => {
86:                 let (c, _, across, depth) = self.frame(along);
87:                 let spread = 1.0 + 0.6 * smoothstep(0.1, 0.4, along);
88:                 c + (across * bundle.x + depth * bundle.y) * spread
89:             }
90:             Tie::Braid => {
91:                 let (c, _, across, depth) = self.frame(along);
92:                 let phase = along / 0.035 * TAU + lane * TAU / 3.0;
93:                 let width = 0.016 * (1.0 - 0.45 * smoothstep(0.0, 0.45, along));
94:                 let taper = 1.0 - smoothstep(0.38, 0.5, along);
95:                 c + across * (phase.sin() * width * taper + bundle.x * 0.6)
96:                     + depth * ((2.0 * phase).sin() * width * 0.35 * taper + bundle.y * 0.6)
97:             }
98:             Tie::Bun => {
99:                 // Coiled round the tie, out from the head as it winds.
100:                 let turns = along / 0.11;
101:                 let a = phase + turns * TAU;
102:                 let r = 0.018 + 0.01 * (turns * 0.5).min(1.0) + bundle.x;
103:                 let u = safe_normalize(tangent(Vec3::up(), self.out), Vec3::up());
104:                 let v = self.out.cross(u);
105:                 self.point
106:                     + (u * a.cos() + v * a.sin()) * r
107:                     + self.out * (0.004 + 0.012 * (turns * 0.4).min(1.0) + bundle.y)
108:             }
109:         }
110:     }
111: }
