4: //! DiffLocks grows hair on the scalp of an SMPL-X body, a 513-vertex patch

16: /// A triangle mesh with normals and a UV map.
17: pub struct Surface {
18:     pub positions: Vec<Vec3>,
19:     pub normals: Vec<Vec3>,
20:     pub uvs: Vec<[f32; 2]>,

27: /// A point on a surface.
28: #[derive(Clone, Copy, Debug)]
29: pub struct SurfacePoint {
30:     pub position: Vec3,
31:     pub normal: Vec3,
32:     pub uv: [f32; 2],
33:     pub triangle: u32,
34: }

36: impl Surface {
37:     /// A surface from its parts. Missing normals are computed from the
38:     /// triangles, area-weighted; missing UVs are zero.
39:     pub fn new(
40:         positions: Vec<Vec3>,
41:         normals: Option<Vec<Vec3>>,
42:         uvs: Option<Vec<[f32; 2]>>,
43:         triangles: Vec<[u32; 3]>,
44:     ) -> Self {
45:         let normals = normals.unwrap_or_else(|| vertex_normals(&positions, &triangles));
46:         let uvs = uvs.unwrap_or_else(|| vec![[0.0; 2]; positions.len()]);
47:         let mut cdf = Vec::with_capacity(triangles.len());
48:         let mut total = 0.0;
49:         for t in &triangles {
50:             let [a, b, c] = t.map(|i| positions[i as usize]);
51:             total += (b - a).cross(c - a).length() * 0.5;
52:             cdf.push(total);
53:         }
54:         let bvh = TriangleBvh::new(positions.clone(), triangles.clone());
55:         Self {
56:             positions,
57:             normals,
58:             uvs,
59:             triangles,
60:             cdf,
61:             bvh,
62:         }
63:     }
64: 
65:     pub fn area(&self) -> f32 {
66:         *self.cdf.last().unwrap_or(&0.0)
67:     }
68: 
69:     /// A point uniformly distributed by area.
70:     pub fn sample(&self, rng: &mut Rng) -> SurfacePoint {
71:         let share = rng.next_f32();
72:         self.sample_at(share, rng.next_f32(), rng.next_f32())
73:     }
74: 
75:     /// The point at `share` of the area (which triangle) and `u`, `v` across
76:     /// it (where in it), each in `[0, 1)`: [`Surface::sample`] from numbers
77:     /// drawn elsewhere.
78:     pub fn sample_at(&self, share: f32, u: f32, v: f32) -> SurfacePoint {
79:         let target = share * self.area();
80:         let t = self
81:             .cdf
82:             .partition_point(|&c| c < target)
83:             .min(self.cdf.len() - 1);
84:         let (mut u, mut v) = (u, v);
85:         if u + v > 1.0 {
86:             u = 1.0 - u;
87:             v = 1.0 - v;
88:         }
89:         self.at_barycentric(t as u32, [1.0 - u - v, u, v])
90:     }

92:     fn at_barycentric(&self, triangle: u32, w: [f32; 3]) -> SurfacePoint {
93:         let [a, b, c] = self.triangles[triangle as usize].map(|i| i as usize);
94:         let position =
95:             self.positions[a] * w[0] + self.positions[b] * w[1] + self.positions[c] * w[2];
96:         let normal =
97:             (self.normals[a] * w[0] + self.normals[b] * w[1] + self.normals[c] * w[2]).normalize();
98:         let uv =
99:             [0, 1].map(|k| self.uvs[a][k] * w[0] + self.uvs[b][k] * w[1] + self.uvs[c][k] * w[2]);
100:         SurfacePoint {
101:             position,
102:             normal,
103:             uv,
104:             triangle,
105:         }
106:     }

108:     /// The closest point on the surface within `max_distance`.
109:     /// Points along the surface's open edges -- for a scalp, its hairline --
110:     /// no further than `step` apart.
111:     pub fn boundary(&self, step: f32) -> Vec<Vec3> {
112:         let mut edges = std::collections::HashMap::new();
113:         for t in &self.triangles {
114:             for k in 0..3 {
115:                 let (a, b) = (t[k], t[(k + 1) % 3]);
116:                 *edges.entry((a.min(b), a.max(b))).or_insert(0u32) += 1;
117:             }
118:         }
119:         let mut open: Vec<(u32, u32)> = edges
120:             .into_iter()
121:             .filter(|&(_, count)| count == 1)
122:             .map(|(edge, _)| edge)
123:             .collect();
124:         // In a fixed order: the map's is random.
125:         open.sort_unstable();
126:         let mut points = Vec::new();
127:         for (a, b) in open {
128:             let (pa, pb) = (self.positions[a as usize], self.positions[b as usize]);
129:             let steps = ((pb - pa).length() / step.max(1e-5)).ceil().max(1.0) as usize;
130:             for i in 0..steps {
131:                 points.push(pa.lerp(pb, i as f32 / steps as f32));
132:             }
133:         }
134:         points
135:     }

137:     pub fn closest(&self, point: Vec3, max_distance: f32) -> Option<SurfacePoint> {
138:         let (triangle, closest, _) = self.bvh.closest_point(point, max_distance)?;
139:         let [a, b, c] = self.triangles[triangle as usize].map(|i| self.positions[i as usize]);
140:         Some(self.at_barycentric(triangle, barycentric(closest, a, b, c)))
141:     }

144:     /// `x y z`, optional `nx ny nz` and `s t` (or `u v`, `texture_u
145:     /// texture_v`), and triangle or quad faces -- the layout of DiffLocks'
146:     /// `scalp.ply`.
215: /// Area-weighted vertex normals.
216: pub fn vertex_normals(positions: &[Vec3], triangles: &[[u32; 3]]) -> Vec<Vec3> {
217:     let mut normals = vec![Vec3::default(); positions.len()];
218:     for t in triangles {
219:         let [a, b, c] = t.map(|i| positions[i as usize]);
220:         let n = (b - a).cross(c - a);
221:         for &i in t {
222:             normals[i as usize] += n;
223:         }
224:     }
225:     normals.iter().map(|n| n.normalize()).collect()
226: }

228: /// Barycentric weights of `p` in triangle `a b c`.
229: pub fn barycentric(p: Vec3, a: Vec3, b: Vec3, c: Vec3) -> [f32; 3] {
230:     let (v0, v1, v2) = (b - a, c - a, p - a);
231:     let d00 = v0.dot(v0);
232:     let d01 = v0.dot(v1);
233:     let d11 = v1.dot(v1);
234:     let d20 = v2.dot(v0);
235:     let d21 = v2.dot(v1);
236:     let denom = d00 * d11 - d01 * d01;
237:     if denom.abs() < 1e-20 {
238:         return [1.0, 0.0, 0.0];
239:     }
240:     let v = (d11 * d20 - d01 * d21) / denom;
241:     let w = (d00 * d21 - d01 * d20) / denom;
242:     [1.0 - v - w, v, w]
243: }

1: //! A hash grid over points, for the nearest-root queries the hair nodes
2: //! make: the guides nearest a new strand's root, the clump guide nearest a
3: //! strand, whether a root is too close to one already placed.
4: //!
5: //! Points carry a group; a query sees only the groups its filter accepts,
6: //! which is how parting keeps strands from interpolating across a part
7: //! (Blender offsets each group's points far apart for the same effect).

20: impl PointGrid {
21:     pub fn new(cell: f32) -> Self {
22:         Self {
23:             cell: cell.max(1e-6),
24:             cells: HashMap::new(),
25:             points: Vec::new(),

29:     /// A grid over `points`, sized so a cell holds a few of them.
30:     pub fn build(points: &[(Vec3, PartingGroup)], cell: f32) -> Self {
31:         let mut grid = Self::new(cell);
32:         for &(p, g) in points {
33:             grid.insert(p, g);

63:     /// Visit every point in the cells within `ring` cells of `p`'s.
64:     fn visit_shell(&self, p: Vec3, ring: i32, mut visit: impl FnMut(u32)) {
65:         let c = self.key(p);
66:         for dz in -ring..=ring {
67:             for dy in -ring..=ring {
68:                 for dx in -ring..=ring {
69:                     if dx.abs().max(dy.abs()).max(dz.abs()) != ring {
70:                         continue;
71:                     }
72:                     if let Some(list) = self.cells.get(&[c[0] + dx, c[1] + dy, c[2] + dz]) {
73:                         list.iter().for_each(|&i| visit(i));
74:                     }
75:                 }
76:             }
77:         }
78:     }

80:     /// Whether any accepted point lies within `radius` of `p`.
81:     pub fn any_within(&self, p: Vec3, radius: f32, accept: impl Fn(PartingGroup) -> bool) -> bool {
82:         let rings = (radius / self.cell).ceil() as i32;
83:         let r2 = radius * radius;
84:         for ring in 0..=rings {
85:             let mut hit = false;
86:             self.visit_shell(p, ring, |i| {
87:                 let (q, g) = self.points[i as usize];
88:                 if !hit && accept(g) && (q - p).length_squared() < r2 {
89:                     hit = true;
90:                 }
91:             });
92:             if hit {
93:                 return true;
94:             }
95:         }
96:         false
97:     }

99:     /// How many points lie within `radius` of `p`.
100:     pub fn count_within(&self, p: Vec3, radius: f32) -> usize {
101:         let rings = (radius / self.cell).ceil() as i32;
102:         let mut count = 0;
103:         for ring in 0..=rings {
104:             self.visit_shell(p, ring, |i| {
105:                 if (self.points[i as usize].0 - p).length() <= radius {
106:                     count += 1;
107:                 }
108:             });
109:         }
110:         count
111:     }

113:     /// The `k` accepted points nearest `p`, nearest first, searching no
114:     /// further than `max_radius`. Distances are Euclidean.
115:     pub fn nearest_k(
116:         &self,
117:         p: Vec3,
118:         k: usize,
119:         max_radius: f32,
120:         accept: impl Fn(PartingGroup) -> bool,
121:     ) -> Vec<(u32, f32)> {
122:         let mut found: Vec<(u32, f32)> = Vec::with_capacity(k + 8);
123:         let max_rings = (max_radius / self.cell).ceil().min(1e6) as i32;
124:         for ring in 0..=max_rings {
125:             self.visit_shell(p, ring, |i| {
126:                 let (q, g) = self.points[i as usize];
127:                 if accept(g) {
128:                     found.push((i, (q - p).length()));
129:                 }
130:             });
131:             // Every point outside the searched rings is at least `ring *
132:             // cell` away; once k are nearer than that, the search is done.
133:             if found.len() >= k {
134:                 found.sort_by(|a, b| a.1.total_cmp(&b.1));
135:                 if found[k - 1].1 <= ring as f32 * self.cell {
136:                     found.truncate(k);
137:                     return found;
138:                 }
139:             }
140:         }
141:         found.sort_by(|a, b| a.1.total_cmp(&b.1));
142:         found.retain(|f| f.1 <= max_radius);
143:         found.truncate(k);
144:         found
145:     }

147:     pub fn nearest(
148:         &self,
149:         p: Vec3,
150:         max_radius: f32,
151:         accept: impl Fn(PartingGroup) -> bool,
152:     ) -> Option<(u32, f32)> {
153:         self.nearest_k(p, 1, max_radius, accept).first().copied()
154:     }
