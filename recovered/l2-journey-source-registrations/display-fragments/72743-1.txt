1812: struct ErrantryChallengeId<'a> {
1813:     puzzle_kind: &'a str,
1814:     namespace: ErrantryChallengeNamespace,

1818: impl<'a> ErrantryChallengeId<'a> {
1819:     fn parse(value: &'a str) -> Option<Self> {
1820:         let mut parts = value.split(':');
1821:         let domain = parts.next()?;
1822:         let puzzle_kind = parts.next()?;
1823:         let namespace = match parts.next()? {
1824:             "demo" => ErrantryChallengeNamespace::Demo,
1825:             "order" => ErrantryChallengeNamespace::Order,
1826:             _ => return None,
1827:         };
1828:         let character_id = parts.next()?.parse().ok()?;
1829:         let _ordinal = parts.next()?.parse::<u64>().ok()?;
1830:         if domain != "challenge" || puzzle_kind.is_empty() || parts.next().is_some() {
1831:             return None;
1832:         }
1833:         Some(Self {
1834:             puzzle_kind,
1835:             namespace,
1836:             character_id,
1837:         })
1838:     }
1839: }
