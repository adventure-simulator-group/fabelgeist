20: impl StructuralNode {
21:     /// Admit a node emitted by an architectural set-out kernel, preserving its
22:     /// identity if the computed position is nonfinite.
23:     pub fn from_metres(
24:         id: StructuralNodeId,
25:         owner: GeometryOwnerId,
26:         kind: StructuralNodeKind,
27:         position: Vec3,
28:         supported_by: Vec<StructuralNodeId>,
29:         grounded: bool,
30:     ) -> Result<Self, StructuralNodeError> {
31:         let position = crate::spatial_geometry::Position::from_metres(position)
32:             .map_err(|cause| StructuralNodeError { node: id, cause })?;
33:         Ok(Self::new(id, owner, kind, position, supported_by, grounded))
34:     }
35:     pub fn new(
36:         id: StructuralNodeId,
37:         owner: GeometryOwnerId,
38:         kind: StructuralNodeKind,
39:         position: crate::spatial_geometry::Position<Architectural>,
40:         supported_by: Vec<StructuralNodeId>,
41:         grounded: bool,
42:     ) -> Self {
43:         Self {
44:             id,
45:             owner,
46:             kind,
47:             position,
48:             supported_by,
49:             grounded,
50:         }
51:     }
52: }
53: impl<'de> Deserialize<'de> for StructuralNode {
54:     fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
55:         #[derive(Deserialize)]
56:         struct SerializedNode {
57:             id: StructuralNodeId,
58:             owner: GeometryOwnerId,
59:             kind: StructuralNodeKind,
60:             position: Vec3,
61:             supported_by: Vec<StructuralNodeId>,
62:             grounded: bool,
63:         }
64:         let value = SerializedNode::deserialize(deserializer)?;
65:         Self::from_metres(
66:             value.id,
67:             value.owner,
68:             value.kind,
69:             value.position,
70:             value.supported_by,
71:             value.grounded,
72:         )
73:         .map_err(serde::de::Error::custom)
74:     }
75: }
