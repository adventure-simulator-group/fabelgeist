1: //! The topology of a thickened plate, decided on the host from its carrier's
2: //! triangles alone.
3: //!
4: //! A carrier becomes a closed shell of three parts: the carrier itself as the
5: //! outer wall, a copy offset inward as the inner wall, and a return wall
6: //! closing every boundary edge between them. Which vertices exist, and which
7: //! triangles join them, depends only on the carrier's connectivity; where
8: //! they end up is the device's business. Each final vertex is therefore
9: //! described by its [`VertexSource`]: the carrier vertex it derives from and
10: //! the wall it lies on.
11: 
12: use std::collections::BTreeMap;
13: 
14: use crate::{BoundaryNormals, GenerateError, PlateFace};
15: 
16: /// The wall a final vertex lies on, and the carrier vertex it derives from.
17: #[derive(Clone, Copy, Debug, PartialEq, Eq)]
18: pub(crate) enum VertexSource {
19:     Outer(u32),
20:     Inner(u32),
21: }
22: 
23: impl VertexSource {
24:     /// Packed for the device: the carrier index, with the top bit set for the
25:     /// inner wall.
26:     pub(crate) fn packed(self) -> u32 {
27:         match self {
28:             Self::Outer(i) => i,
29:             Self::Inner(i) => i | INNER_BIT,
30:         }
31:     }
32: }
33: 
34: pub(crate) const INNER_BIT: u32 = 1 << 31;
35: 
36: /// The final vertices and triangles of one thickened carrier, indexed from
37: /// zero within the shell.
38: #[derive(Clone, Debug)]
39: pub(crate) struct ShellPlan {
40:     pub sources: Vec<VertexSource>,
41:     /// The carrier's own triangles first, as the outer wall, then the inner
42:     /// and return walls.
43:     pub indices: Vec<u32>,
44:     pub carrier_triangles: usize,
45: }
46: 
47: impl ShellPlan {
48:     /// Thicken a carrier of `count` vertices wound outward, closing every
49:     /// boundary edge. Seams must share indices; an edge used more than twice,
50:     /// or twice in the same direction, is not a surface.
51:     pub(crate) fn new(
52:         count: u32,
53:         carrier: &[u32],
54:         boundary_normals: BoundaryNormals,
55:     ) -> Result<Self, GenerateError> {
56:         if count == 0
57:             || carrier.is_empty()
58:             || !carrier.len().is_multiple_of(3)
59:             || carrier.iter().any(|i| *i >= count)
60:         {
61:             return Err(GenerateError::InvalidSurface);
62:         }
63:         let mut edges = BTreeMap::<(u32, u32), Vec<(u32, u32)>>::new();
64:         for triangle in carrier.as_chunks::<3>().0 {
65:             for (a, b) in [
66:                 (triangle[0], triangle[1]),
67:                 (triangle[1], triangle[2]),
68:                 (triangle[2], triangle[0]),
69:             ] {
70:                 edges.entry((a.min(b), a.max(b))).or_default().push((a, b));
71:             }
72:         }
73:         if edges
74:             .values()
75:             .any(|e| e.len() > 2 || (e.len() == 2 && e[0] != (e[1].1, e[1].0)))
76:         {
77:             return Err(GenerateError::InvalidSurface);
78:         }
79:         let mut sources = (0..count)
80:             .map(VertexSource::Outer)
81:             .chain((0..count).map(VertexSource::Inner))
82:             .collect::<Vec<_>>();
83:         let mut indices = carrier.to_vec();
84:         indices.extend(
85:             carrier
86:                 .as_chunks::<3>()
87:                 .0
88:                 .iter()
89:                 .flat_map(|t| [t[0] + count, t[2] + count, t[1] + count]),
90:         );
91:         for edge in edges.values().filter(|e| e.len() == 1) {
92:             let (a, b) = edge[0];
93:             // Keep alias ordering independent of winding, including reflected fits.
94:             let reversed = a > b;
95:             let (a, b) = if reversed { (b, a) } else { (a, b) };
96:             let [a, b, c, d] = match boundary_normals {
97:                 BoundaryNormals::Smooth => [a, b, a + count, b + count],
98:                 BoundaryNormals::Separate => {
99:                     let first = sources.len() as u32;
100:                     sources.extend([
101:                         VertexSource::Outer(a),
102:                         VertexSource::Outer(b),
103:                         VertexSource::Inner(a),
104:                         VertexSource::Inner(b),
105:                     ]);
106:                     [first, first + 1, first + 2, first + 3]
107:                 }
108:             };
109:             indices.extend(if reversed {
110:                 [b, c, a, b, d, c]
111:             } else {
112:                 [b, a, c, b, c, d]
113:             });
114:         }
115:         Ok(Self {
116:             sources,
117:             indices,
118:             carrier_triangles: carrier.len() / 3,
119:         })
120:     }
121: 
122:     /// The plate face of each triangle, in [`ShellPlan::indices`] order.
123:     pub(crate) fn faces(&self) -> impl Iterator<Item = PlateFace> + '_ {
124:         let walls = self.indices.len() / 3 - 2 * self.carrier_triangles;
125:         std::iter::repeat_n(PlateFace::Outer, self.carrier_triangles)
126:             .chain(std::iter::repeat_n(
127:                 PlateFace::Inner,
128:                 self.carrier_triangles,
129:             ))
130:             .chain(std::iter::repeat_n(PlateFace::Edge, walls))
131:     }
132: }
