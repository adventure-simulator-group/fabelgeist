"""Publication successor that requires and publishes the human progress report.

The original backend stays byte-identical for historical replay. New adoption
callers use this successor so a checkpoint cannot finish without its report.
"""
from datetime import datetime, timezone
import json
from pathlib import Path
import runpy
from audit_l2_publication_kernel_v2 import Publication as EvidencePublication
from audit_progress_report_v1 import check_context, render, update, verify

class Publication(EvidencePublication):
    def publish(self,**kwargs):
        number=kwargs["checkpoint"]
        context=check_context(number)
        super().publish(**kwargs)
        if self.mode=="--preflight":
            # Render the prospective checkpoint without writing any report files.
            render(self.new["current"],number,context,datetime.now(timezone.utc).isoformat())
        elif self.mode=="--apply":
            state=update()
            assert state["checkpoint_number"]==number
            intake=Path(__file__).resolve().parents[4] / "726-intake-20261030"
            runpy.run_path(str(intake / "publish_progress.py"))["export"]()
            print("Remote publication pending: run publish_progress.py --push before checkpoint completion.")
        else:
            state=verify()
            assert state["checkpoint_number"]==number
            intake=Path(__file__).resolve().parents[4] / "726-intake-20261030"
            remote=json.loads((intake / "REMOTE-PROGRESS-STATE.json").read_text())
            assert remote["checkpoint_number"]==number and remote["remote_verified"]
