35: impl LogicGridSpec {
36:     pub fn validate(self) -> Result<Self, &'static str> {
37:         if !(3..=MAX_GRID_SIZE as u8).contains(&self.size) {
38:             return Err("logic-grid size must be three or four");
39:         }
40:         if !self.allow_positive_clues && !self.allow_negative_clues {
41:             return Err("logic grid must enable at least one clue polarity");
42:         }
43:         if !(1..=12).contains(&self.max_clues) {
44:             return Err("logic-grid clue limit must be between one and twelve");
45:         }
46:         Ok(self)
47:     }
48: }

65: impl LogicGridClue {
66:     pub fn text(self) -> String {
67:         let relation = match self.relation {
68:             GridRelation::TravelerToken => {
69:                 format!("{} carried the {}", traveler(self.left), token(self.right))
70:             }
71:             GridRelation::TravelerRoad => {
72:                 format!("{} took the {}", traveler(self.left), road(self.right))
73:             }
74:             GridRelation::TokenRoad => format!(
75:                 "the traveler bearing the {} took the {}",
76:                 token(self.left),
77:                 road(self.right)
78:             ),
79:         };
80:         if self.matches {
81:             format!("{relation}.")
82:         } else {
83:             format!("It is not true that {relation}.")
84:         }
85:     }
86: }

113: impl LogicGridPuzzle {
114:     pub fn generate(seed: Seed) -> Self {
115:         Self::generate_with_spec(seed, LogicGridSpec::default())
116:             .expect("standard logic-grid specification is valid")
117:     }
118: 
119:     pub fn generate_with_spec(seed: Seed, spec: LogicGridSpec) -> Result<Self, &'static str> {
120:         let spec = spec.validate()?;
121:         let size = usize::from(spec.size);
122:         let mut rng = LOGIC_GRID_GENERATION_DOMAIN.rng(seed, &[]);
123:         let mut token_order = (0..spec.size).collect::<Vec<_>>();
124:         let mut road_order = (0..spec.size).collect::<Vec<_>>();
125:         shuffle(&mut token_order, &mut rng);
126:         shuffle(&mut road_order, &mut rng);
127:         let solution = (0..spec.size)
128:             .map(|traveler| LogicGridAssignment {
129:                 traveler,
130:                 token: token_order[usize::from(traveler)],
131:                 road: road_order[usize::from(traveler)],
132:             })
133:             .collect::<Vec<_>>();
134: 
135:         let mut pool = true_grid_clues(&solution, spec);
136:         shuffle(&mut pool, &mut rng);
137:         let candidates = all_grid_solutions(spec.size);
138:         let mut retained = Vec::new();
139:         let mut legal = candidates.clone();
140:         while legal.len() > 1 && retained.len() < usize::from(spec.max_clues) {
141:             let best = pool
142:                 .iter()
143:                 .copied()
144:                 .enumerate()
145:                 .filter(|(_, clue)| !retained.contains(clue))
146:                 .map(|(index, clue)| {
147:                     let remaining = legal
148:                         .iter()
149:                         .filter(|candidate| grid_clue_holds(&clue, candidate))
150:                         .count();
151:                     (legal.len().saturating_sub(remaining), index, clue)
152:                 })
153:                 .filter(|(eliminated, _, _)| *eliminated > 0)
154:                 .max_by_key(|(eliminated, index, _)| (*eliminated, std::cmp::Reverse(*index)))
155:                 .map(|(_, _, clue)| clue)
156:                 .ok_or("logic-grid clue grammar cannot isolate the solution")?;
157:             retained.push(best);
158:             legal.retain(|candidate| grid_clue_holds(&best, candidate));
159:         }
160:         if legal.len() != 1 {
161:             return Err("logic-grid clue limit cannot prove one solution");
162:         }
163:         let mut index = retained.len();
164:         while index > 0 {
165:             index -= 1;
166:             let mut reduced = retained.clone();
167:             reduced.remove(index);
168:             if grid_solutions(spec.size, &reduced, 2).len() == 1 {
169:                 retained = reduced;
170:             }
171:         }
172:         let puzzle = Self {
173:             rules_version: LOGIC_GRID_RULES_VERSION,
174:             seed,
175:             spec,
176:             solution,
177:             clues: retained,
178:         };
179:         puzzle.validate()?;
180:         debug_assert_eq!(size, puzzle.solution.len());
181:         Ok(puzzle)
182:     }

193:             roads: ROADS[..size].iter().map(|value| (*value).into()).collect(),
194:             clues: self.clues.clone(),
195:         }
196:     }
