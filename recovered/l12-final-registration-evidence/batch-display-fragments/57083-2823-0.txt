201: impl TacticalCausalSummary {
202:     pub(super) fn finish_means(&mut self) -> MeasurementResult<()> {
203:         self.john.finish_means()?;
204:         self.opponent.finish_means()?;
205:         Ok(())
206:     }

209: impl TacticalCombatantCausal {
210:     pub(super) fn finish_means(&mut self) -> MeasurementResult<()> {
211:         self.mean_attack_start_interval_seconds = self
212:             .mean_attack_start_interval_seconds
213:             .mean(self.attack_interval_samples)?;
214:         self.mean_contact_energy_joules = self
215:             .mean_contact_energy_joules
216:             .mean(self.contact_energy_samples)?;
217:         Ok(())
218:     }
219: }

221: impl AutoresolveCombatantCausal {
222:     pub(super) fn record_final(
223:         &mut self,
224:         outcome: &adventuresim_core::autoresolve::CombatantOutcome,
225:     ) -> MeasurementResult<()> {
226:         self.final_pain_incapacitation.checked_add(
227:             Measurement::<PainIncapacitation>::from_native(outcome.pain_incapacitation)?,
228:         )?;
229:         self.final_blood_loss_fraction.checked_add(
230:             Measurement::<BloodLossFraction>::from_native(outcome.blood_loss_fraction)?,
231:         )?;
232:         self.final_acute_trauma.checked_add(
233:             Measurement::<AcuteTrauma>::from_native(outcome.acute_trauma)?,
234:         )?;
235:         self.final_fatigue.checked_add(
236:             Measurement::<Fatigue>::from_native(outcome.fatigue)?,
237:         )?;
238:         self.final_imbalance.checked_add(
239:             Measurement::<Imbalance>::from_native(outcome.imbalance)?,
240:         )?;
241:         self.final_open_wounds += outcome.open_wound_count as u64;
242:         self.final_internal_wounds += outcome.internal_wound_count as u64;
243:         self.incapacitated_runs += u64::from(outcome.incapacitated);
244:         self.yielded_runs += u64::from(outcome.yielded);
245:         if let Some(cause) = outcome.terminal_cause {
246:             *self
247:                 .terminal_causes
248:                 .entry(format!("{cause:?}"))
249:                 .or_default() += 1;
250:         }
251:         Ok(())
252:     }
253: 
254:     pub(super) fn finish_means(&mut self) -> MeasurementResult<()> {
255:         self.mean_attack_interval_seconds = self
256:             .mean_attack_interval_seconds
257:             .mean(self.attack_samples)?;
258:         self.mean_attack_power_multiplier = self
259:             .mean_attack_power_multiplier
260:             .mean(self.attack_samples)?;
261:         self.mean_scheduled_contact_measure_metres = self
262:             .mean_scheduled_contact_measure_metres
263:             .mean(self.contact_measure_samples)?;
264:         self.mean_actual_contact_measure_metres = self
265:             .mean_actual_contact_measure_metres
266:             .mean(self.contact_measure_samples)?;
267:         Ok(())
268:     }
