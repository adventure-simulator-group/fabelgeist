280:     pub fn check(&self, submission: &PuzzleSubmission) -> Result<bool, PuzzleSubmissionError> {
281:         match (self, submission) {
282:             (Self::OrderedSigils(puzzle), PuzzleSubmission::OrderedSigils { ordering }) => puzzle
283:                 .check(&OrderedSigilSubmission {
284:                     expected_revision: 0,
285:                     ordering: *ordering,
286:                 })
287:                 .map_err(|_| PuzzleSubmissionError::Malformed),
288:             (
289:                 Self::TruthfulWitnesses(puzzle),
290:                 PuzzleSubmission::TruthfulWitnesses { safe_path },
291:             ) => Ok(*safe_path == puzzle.solution_path),
292:             (Self::RuneTransformation(puzzle), PuzzleSubmission::RuneTransformation { result }) => {
293:                 Ok(*result == puzzle.solution)
294:             }
295:             (Self::LogicGrid(puzzle), PuzzleSubmission::LogicGrid { assignments }) => puzzle
296:                 .check(assignments)
297:                 .map_err(|_| PuzzleSubmissionError::Malformed),
298:             (
299:                 Self::ResourceAllocation(puzzle),
300:                 PuzzleSubmission::ResourceAllocation { provisions },
301:             ) => puzzle
302:                 .check(provisions)
303:                 .map_err(|_| PuzzleSubmissionError::Malformed),
304:             _ => Err(PuzzleSubmissionError::WrongKind),
305:         }
306:     }

361: impl TruthfulWitnessSpec {
362:     pub fn validate(self) -> Result<Self, &'static str> {
363:         if !(self.allow_path_claims || self.allow_witness_claims) {
364:             return Err("truthful-witness spec must enable a claim family");
365:         }
366:         Ok(self)
367:     }
368: 
369:     fn allows(self, claim: WitnessClaim) -> bool {
370:         match claim {
371:             WitnessClaim::PathIsSafe(_) | WitnessClaim::PathIsUnsafe(_) => self.allow_path_claims,
372:             WitnessClaim::WitnessLies(_) | WitnessClaim::WitnessSpeaksTruth(_) => {
373:                 self.allow_witness_claims
374:             }
375:         }
376:     }
377: }

381:         let claim = match self.claim {

383:             WitnessClaim::PathIsUnsafe(path) => format!("the {} is unsafe", path.label()),
384:             WitnessClaim::WitnessLies(witness) => {
385:                 format!("the {} witness is the liar", witness.label())
386:             }
387:             WitnessClaim::WitnessSpeaksTruth(witness) => {
388:                 format!("the {} witness speaks truth", witness.label())
389:             }
390:         };
391:         format!("The {} witness says that {}.", self.speaker.label(), claim)
392:     }
