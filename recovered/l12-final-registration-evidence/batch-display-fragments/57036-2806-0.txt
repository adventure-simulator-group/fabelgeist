1915:             ..
1916:         }) = &mut self.action
1917:         {
1918:             let recovery_progress = curve.follow_through_fraction.clamp(0.0, 1.0);
1919:             let transition_tick =
1920:                 timeline
1921:                     .contact_tick()
1922:                     .saturating_add(AnimationSampleSpan::from(
1923:                         (timeline.recovery_ticks.samples() as f64 * recovery_progress as f64).ceil()
1924:                             as u64,
1925:                     ));
1926:             let may_follow = queued.is_none()
1927:                 && spec.hand == AttackHand::Main

1933:                     ActionTimeline::with_recovery(start_tick, contact_tick, end_tick);

1938:                 let transition_phase = 0.5
