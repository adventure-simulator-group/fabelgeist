131: impl ResourceAllocationSpec {
132:     pub fn validate(self) -> Result<Self, &'static str> {
133:         if !(4..=ProvisionId::ALL.len() as u8).contains(&self.provision_count) {
134:             return Err("resource puzzle must offer between four and seven provisions");
135:         }
136:         if !(2..=4).contains(&self.hazard_count) {
137:             return Err("resource puzzle must present between two and four hazards");
138:         }
139:         if !(3..=14).contains(&self.capacity) {
140:             return Err("resource-puzzle capacity must be between three and fourteen");
141:         }
142:         Ok(self)
143:     }
144: }
