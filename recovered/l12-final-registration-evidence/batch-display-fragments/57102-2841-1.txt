1: //! Declare bounded deck/wall and tread/landing contacts, never stair obstructions.
2: use super::*;
3: 
4: const BOND_DOMAIN: u64 = 14_u64 << 60;
5: const BOND_TOLERANCE_METRES: f32 = 0.005;
6: 
7: pub(super) fn resolve(geometry: &mut ResolvedGeometry) -> Result<(), crate::GenerationError> {
8:     let solids = &geometry.solids;
9:     let _: () = for (index, a) in solids.iter().enumerate() {
10:         for b in &solids[index + 1..] {
11:             if a.owner == b.owner || !bearing_pair(a, b) {
12:                 continue;
13:             }
14:             let (amin, amax) = bounds(a);
15:             let (bmin, bmax) = bounds(b);
16:             let min = amin.max(bmin);
17:             let max = amax.min(bmax);
18:             let overlap = max - min;
19:             if overlap.min_element() <= 0.0
20:                 || overlap.y > flight::DECK_THICKNESS_METRES + BOND_TOLERANCE_METRES
21:             {
22:                 continue;
23:             }
24:             let mut axes = [overlap.x, overlap.y, overlap.z];
25:             axes.sort_by(f32::total_cmp);
26:             geometry.junction_bonds.push(crate::JunctionBond {
27:                 id: ResolvedItemId(BOND_DOMAIN | geometry.junction_bonds.len() as u64),
28:                 owners: [a.owner, b.owner],
29:                 bounds: SpatialBounds::<Architectural>::from_metres(min, max)?,
30:                 minimum_interface_area_square_metres: axes[1] * axes[2] * 0.9,
31:                 maximum_penetration_metres: overlap.x.min(overlap.z) + BOND_TOLERANCE_METRES,
32:             });
33:         }
34:     };
35:     Ok(())
36: }
37: 
38: fn bearing_pair(a: &ResolvedSolid, b: &ResolvedSolid) -> bool {
39:     [(a, b), (b, a)].into_iter().any(|(deck, host)| {
40:         ((deck.role == SolidRole::InteriorFloor || owns_landing(deck))
41:             && matches!(
42:                 host.role,
43:                 SolidRole::WallHost
44:                     | SolidRole::DefenseHostWall
45:                     | SolidRole::OpeningJamb
46:                     | SolidRole::OpeningHead
47:                     | SolidRole::OpeningSpandrel
48:                     | SolidRole::OpeningSill
49:                     | SolidRole::Landing
50:             ))
51:             || (matches!(deck.role, SolidRole::StairTread | SolidRole::Landing)
52:                 && deck.owner.0 >= FIRST_FLIGHT_OWNER
53:                 && matches!(
54:                     host.role,
55:                     SolidRole::WalkSurface
56:                         | SolidRole::Landing
57:                         | SolidRole::CircuitWalk
58:                         | SolidRole::GalleryFloor
59:                 ))
60:     })
61: }
62: 
63: fn bounds(solid: &ResolvedSolid) -> (Vec3, Vec3) {
64:     let q = bevy::math::Quat::from_rotation_y(solid.yaw_radians.radians());
65:     let half = (q * Vec3::X).abs() * solid.size.metres().x * 0.5
66:         + Vec3::Y * solid.size.metres().y * 0.5
67:         + (q * Vec3::Z).abs() * solid.size.metres().z * 0.5;
68:     (solid.centre.metres() - half, solid.centre.metres() + half)
69: }
