4: fn spiral_flight_has_grounded_newel_and_exact_supported_storey_landings() {
5:     let stair = Stair::Spiral {
6:         centre: Vec2::new(8.0, 9.0),
7:         base_height_metres: 0.15,
8:         rise_metres: 10.05,
9:         inner_radius_metres: 0.25,
10:         outer_radius_metres: 1.25,
11:         turns: 2.8,
12:         clockwise: true,
13:         tread_count: required_treads(0.15, 10.05, 3.4),
14:     };
15:     let flight = compile_flight(stair, 3.4).unwrap();
16:     let Stair::Spiral { tread_count, .. } = stair else {
17:         unreachable!()
18:     };
19:     assert_eq!(
20:         flight
21:             .members
22:             .iter()
23:             .filter(|m| m.role == SolidRole::StairTread)
24:             .count(),
25:         usize::from(tread_count)
26:     );
27:     let newel = flight
28:         .members
29:         .iter()
30:         .find(|m| m.role == SolidRole::StairNewel)
31:         .unwrap();
32:     assert!((newel.centre.y - newel.size.y * 0.5).abs() < 0.0001);
33:     for landing in &flight.landings {
34:         let point = Vec3::new(
35:             landing.position_metres.x,
36:             landing.elevation_metres - 0.01,
37:             landing.position_metres.y,
38:         );
39:         assert!(
40:             flight
41:                 .members
42:                 .iter()
43:                 .any(|m| m.role == SolidRole::Landing && contains(m, point))

46:             Vec2::new(-0.3, -0.3),

61:                 flight
62:                     .members
63:                     .iter()

65:                         && (m.centre.y + m.size.y * 0.5 - landing.elevation_metres).abs() < 0.0001)

71: #[test]
72: fn spiral_authored_resolution_and_landing_headroom_are_preserved() {
73:     let count = required_treads(0.15, 10.05, 3.4);
74:     let make = |tread_count| Stair::Spiral {
75:         centre: Vec2::ZERO,
76:         base_height_metres: 0.15,
77:         rise_metres: 10.05,
78:         inner_radius_metres: 0.25,
79:         outer_radius_metres: 1.25,
80:         turns: 2.8,
81:         clockwise: true,
82:         tread_count,
83:     };
84:     assert!(compile_flight(make(count - 1), 3.4).is_none());
85:     for authored in [count, count + 7] {
86:         let flight = compile_flight(make(authored), 3.4).unwrap();
87:         let treads = flight
88:             .members
89:             .iter()
90:             .filter(|m| m.role == SolidRole::StairTread)
91:             .collect::<Vec<_>>();
92:         assert_eq!(treads.len(), usize::from(authored));
93:         let mut previous = 0.0;
94:         for tread in treads {
95:             let height = tread.centre.y + tread.size.y * 0.5;
96:             assert!(height - previous <= 0.1901);
97:             previous = height;
98:         }
99:         for landing in flight
100:             .members
101:             .iter()
102:             .filter(|m| m.role == SolidRole::Landing && m.centre.y > 1.0)
103:         {
104:             // The radial bridge joins only the tread outer end, leaving the
105:             // inner walking line beneath the arrival clear of an overhang.
106:             let radial = bevy::math::Quat::from_rotation_y(landing.yaw_radians) * Vec3::X;
107:             let point = radial * 0.75 + Vec3::Y * landing.centre.y;
108:             assert!(!contains(landing, point));
109:         }
110:     }
111: }

118: #[test]
119: fn fortified_stairs_and_floor_holes_are_shared_by_detail_and_collision() {
120:     let plan = crate::generate(&crate::BuildingProgram::fixture(
121:         BuildingArchetype::WalledKeep,
122:         fabelgeist_determinism::Seed::from_u64(42),
123:     ))
124:     .unwrap_or_else(|error| match error {
125:         crate::GenerationError::StructuralContract { issues, .. } => panic!(
126:             "{} issues; first: {:?}",
127:             issues.len(),
128:             &issues[..issues.len().min(8)]
129:         ),
130:         other => panic!("{other}"),
131:     });
132:     let collision = crate::compile_building_collision(&plan).unwrap();
133:     let detail = crate::compile_building_detail(&plan).unwrap();
134:     for solid in plan.resolved_geometry.solids.iter().filter(|s| {
135:         matches!(
136:             s.role,
137:             SolidRole::StairTread | SolidRole::StairNewel | SolidRole::InteriorFloor
138:         ) || owns_landing(s)
139:     }) {
140:         assert!(collision.cuboids.iter().any(|c| c.source == solid.id));
141:         assert!(detail.meshes.iter().flat_map(|m| &m.vertices).any(|v| {
142:             let local = bevy::math::Quat::from_rotation_y(-solid.yaw_radians.radians())
143:                 * (v.position - solid.centre.metres());
144:             (local.abs() - solid.size.metres() * 0.5)
145:                 .abs()
146:                 .max_element()
147:                 < 0.001
148:         }));
149:     }
150:     for (index, stair) in plan.stairs.iter().enumerate() {
151:         let Stair::Spiral {
152:             centre,
153:             outer_radius_metres,
154:             ..
155:         } = stair
156:         else {
157:             continue;
158:         };
159:         if centre.cmpge(Vec2::ZERO).all() && centre.cmple(plan.dimensions_metres()).all() {
160:             assert!(!landings(&plan, index).is_empty());
161:         }
162:         for solid in plan
163:             .resolved_geometry
164:             .solids
165:             .iter()
166:             .filter(|s| s.role == SolidRole::InteriorFloor && s.centre.metres().y > 1.0)
167:         {
168:             let separation =
169:                 (Vec2::new(solid.centre.metres().x, solid.centre.metres().z) - *centre).abs()
170:                     - Vec2::new(solid.size.metres().x, solid.size.metres().z) * 0.5;
171:             assert!(
172:                 separation.max_element()
173:                     >= outer_radius_metres + flight::WELL_MARGIN_METRES - 0.001
174:             );
175:         }
176:     }
177: }
