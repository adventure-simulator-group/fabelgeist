1: use bevy_render::render_resource::{Buffer, BufferDescriptor, BufferUsages};
2: use bevy_render::renderer::{RenderDevice, RenderQueue};
3: use bevy_utils::default;
4: use tracing::trace;
5: 
6: /// Ensures a GPU buffer has sufficient capacity, resizing and copying data if necessary.
7: pub(crate) fn ensure_buffer_capacity(
8:     device: &RenderDevice,
9:     queue: &RenderQueue,
10:     buffer_opt: &mut Option<Buffer>,
11:     capacity: u64,
12:     usage: BufferUsages,
13:     label: &str,
14:     copy: bool,
15: ) {
16:     let aligned_size = (capacity + 3) & !3; // 4-byte align
17:     let current_size = buffer_opt.as_ref().map(|b| b.size()).unwrap_or(0);
18:     if aligned_size <= current_size {
19:         return;
20:     }
21: 
22:     let size = aligned_size.max(1024);
23: 
24:     #[cfg(feature = "trace")]
25:     trace!(
26:         "Resizing Buffer [{}]: {} bytes -> {} bytes (Copy: {})",
27:         label, current_size, size, copy
28:     );
29: 
30:     let buffer = device.create_buffer(&BufferDescriptor {
31:         label: Some(label),
32:         size,
33:         usage,
34:         mapped_at_creation: false,
35:     });
36: 
37:     if copy && let Some(old) = buffer_opt {
38:         let mut encoder = device.create_command_encoder(&default());
39:         let copy_size = old.size().min(size);
40: 
41:         encoder.copy_buffer_to_buffer(old, 0, &buffer, 0, copy_size);
42: 
43:         queue.submit(Some(encoder.finish()));
44:     }
45: 
46:     *buffer_opt = Some(buffer);
47: }
