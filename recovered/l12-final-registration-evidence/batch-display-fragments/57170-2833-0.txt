622: impl RuneTransformationSpec {
623:     pub fn validate(self) -> Result<Self, &'static str> {
624:         if !(1..=RuneGate::ALL.len() as u8).contains(&self.gate_count) {
625:             return Err("rune gate count must be between one and three");
626:         }
627:         if !(1..=8).contains(&self.route_length) {
628:             return Err("rune route length must be between one and eight");
629:         }
630:         if !self.allow_repeated_route_gates && self.route_length > self.gate_count {
631:             return Err("non-repeating rune route cannot exceed the gate count");
632:         }
633:         if !(1..=2).contains(&self.examples_per_gate) {
634:             return Err("rune examples per gate must be one or two");
635:         }
636:         if !(1..=RuneOperation::ALL.len() as u8).contains(&self.minimum_single_example_candidates) {
637:             return Err("rune single-example candidate minimum is out of bounds");
638:         }
639:         if self.examples_per_gate == 1 && self.minimum_single_example_candidates > 1 {
640:             return Err("one rune example cannot both stay ambiguous and prove its gate law");
641:         }
642:         Ok(self)
643:     }
644: }

754:         if rules_version != RUNE_TRANSFORMATION_RULES_VERSION {

760:     pub fn generate_with_spec(
761:         seed: Seed,
762:         spec: RuneTransformationSpec,
763:     ) -> Result<Self, &'static str> {
764:         let spec = spec.validate()?;
765:         const RUNE_TRANSFORMATION_GENERATION_DOMAIN: fabelgeist_determinism::StreamId =
766:             fabelgeist_determinism::StreamId::new("puzzle.rune_transformation");
767:         let mut rng = RUNE_TRANSFORMATION_GENERATION_DOMAIN.rng(seed, &[]);
768:         let active_gates = RuneGate::ALL[..usize::from(spec.gate_count)].to_vec();
769:         let mut available_operations = RuneOperation::ALL.to_vec();
770:         shuffle(&mut available_operations, &mut rng);
771:         let gate_laws = active_gates
772:             .iter()
773:             .copied()
774:             .enumerate()
775:             .map(|(index, gate)| RuneGateLaw {
776:                 gate,
777:                 operation: if spec.allow_repeated_operations {
778:                     RuneOperation::ALL[rng.index(RuneOperation::ALL.len())]
779:                 } else {
780:                     available_operations[index]
781:                 },
782:             })
783:             .collect::<Vec<_>>();
784:         let query = Sigil::ALL[rng.index(Sigil::ALL.len())];
785:         let mut route = Vec::with_capacity(usize::from(spec.route_length));
786:         if spec.allow_repeated_route_gates {
787:             for _ in 0..spec.route_length {
788:                 route.push(active_gates[rng.index(active_gates.len())]);
789:             }
790:         } else {
791:             let mut shuffled = active_gates.clone();
792:             shuffle(&mut shuffled, &mut rng);
793:             route.extend(shuffled.into_iter().take(usize::from(spec.route_length)));
794:         }
795:         let mut examples =
796:             Vec::with_capacity(usize::from(spec.gate_count) * usize::from(spec.examples_per_gate));
797:         for law in &gate_laws {
798:             let gate = law.gate;
799:             let operation = law.operation;
800:             let mut inputs = Sigil::ALL;
