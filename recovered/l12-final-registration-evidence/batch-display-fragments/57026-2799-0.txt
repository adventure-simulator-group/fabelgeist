219:     /// Project the complete enclosure atomically from its exact accepted owner.
220:     /// No member floor, neighbouring support or sampled-terrain fallback exists.
221:     pub fn project(
222:         compound: &CityCompound,
223:         terrain: &SceneTerrain,
224:     ) -> Result<Self, SceneInputError> {

226:             BoundarySupportConstraint, BoundarySupportElement,
227:         };
228:         use crate::city_layout::grounding::{BoundarySupportError, BoundarySupportMesh};
229:         let foundation = terrain.property_foundation(compound.id).ok_or_else(|| {
230:             BoundarySupportError::new(

232:                 BoundarySupportElement::Owner,
233:                 BoundarySupportConstraint::OwnerBinding,

235:                 1.0,
236:                 0.0,
237:             )
238:         })?;
239:         let policy = crate::city_layout::CompoundGradingPolicy::bounded_settlement();
240:         let projection =
241:             BoundarySupportMesh::project(compound, foundation, policy.limits, policy.embedment)?;
242:         Ok(Self {
243:             scene: SceneBoundary {
244:                 property_id: compound.id,
245:                 front_building_id: compound.front_building_id,
246:                 boundary: compound.boundary.clone(),
247:                 fixed_support: projection.mesh,
