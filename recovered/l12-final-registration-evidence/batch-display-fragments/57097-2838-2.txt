1: //! Occupied floor bridges and auxiliary tower crown arrivals are distinct destinations.
2: use super::*;
3: 
4: pub(super) fn has_floor(plan: &BuildingPlan, landing: &SpiralLanding) -> bool {
5:     if !keep_core::owns_occupied_storeys(plan.archetype) {
6:         return false;
7:     }
8:     plan.storeys.iter().any(|storey| {
9:         storey.level == landing.storey
10:             && storey
11:                 .rooms
12:                 .iter()
13:                 .flat_map(|room| &room.cells)
14:                 .any(|cell| {
15:                     (landing.position_metres - cell.centre())
16:                         .abs()
17:                         .max_element()
18:                         <= crate::CELL_SIZE_METRES * 0.5
19:                 })
20:     })
21: }
22: 
23: pub(super) fn resolve(plan: &BuildingPlan, flight: &mut SpiralFlight) {
24:     let keep = flight
25:         .landings
26:         .iter()
27:         .filter(|landing| has_floor(plan, landing))
28:         .map(|landing| landing.elevation_metres)
29:         .collect::<Vec<_>>();
30:     flight.members.retain(|member| {
31:         member.role != SolidRole::Landing
32:             || keep
33:                 .iter()
34:                 .any(|height| (member.centre.y + member.size.y * 0.5 - height).abs() < 0.001)
35:     });
36:     flight.landings.retain(|landing| has_floor(plan, landing));
37:     if flight.landings.is_empty() {
38:         // Auxiliary tower flights terminate on the crown instead of inventing
39:         // intermediate floors through the hollow tower shell.
40:         let Some(last) = flight
41:             .members
42:             .iter()
43:             .rev()
44:             .find(|member| member.role == SolidRole::StairTread)
45:         else {
46:             return;
47:         };
48:         let radial = bevy::math::Quat::from_rotation_y(last.yaw_radians) * Vec3::X;
49:         let length = 0.45;
50:         let centre = Vec3::new(
51:             flight.centre.x,
52:             flight.top_height_metres - flight::DECK_THICKNESS_METRES * 0.5,
53:             flight.centre.y,
54:         ) + radial * (flight.outer_radius_metres + length * 0.5 - 0.12);
55:         flight.members.push(SpiralMember {
56:             centre,
57:             size: Vec3::new(length, flight::DECK_THICKNESS_METRES, 0.9),
58:             yaw_radians: last.yaw_radians,
59:             role: SolidRole::Landing,
60:         });
61:     }
62: }
