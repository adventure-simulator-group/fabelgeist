116: /// Parses a `.npy` buffer: header dictionary first, then the element payload.
117: pub fn parse(bytes: &[u8]) -> Result<NpyArray> {
118:     const MAGIC: &[u8] = b"\x93NUMPY";
119:     if bytes.len() < 10 || &bytes[..6] != MAGIC {
120:         bail!("not a .npy array");
121:     }
122:     let (header_len, header_start) = if bytes[6] == 1 {
123:         (
124:             u16::from_le_bytes(bytes[8..10].try_into().unwrap()) as usize,
125:             10,
126:         )
127:     } else {
128:         (
129:             u32::from_le_bytes(bytes[8..12].try_into().unwrap()) as usize,
130:             12,
131:         )
132:     };
133:     if header_start + header_len > bytes.len() {
134:         bail!("truncated .npy header");
135:     }
136:     let header = std::str::from_utf8(&bytes[header_start..header_start + header_len])
137:         .context("non-UTF-8 .npy header")?;
138: 
139:     let dtype =
140:         Dtype::parse(header_value(header, "'descr'").context("missing 'descr' in .npy header")?)?;
141:     if header_value(header, "'fortran_order'").is_some_and(|v| v.starts_with("True")) {
142:         bail!("Fortran-ordered .npy arrays are not supported");
143:     }
144: 
145:     let shape_text = header
146:         .split_once("'shape'")
147:         .and_then(|(_, rest)| rest.split_once('('))
148:         .and_then(|(_, rest)| rest.split_once(')'))
149:         .map(|(inside, _)| inside)
150:         .context("missing 'shape' in .npy header")?;
151:     let shape: Vec<usize> = shape_text
152:         .split(',')
153:         .map(str::trim)
154:         .filter(|token| !token.is_empty())
155:         .map(|token| token.parse::<usize>().context("bad .npy shape"))
156:         .collect::<Result<_>>()?;
157: 
158:     let count: usize = shape.iter().product();
159:     let start = header_start + header_len;
160:     let end = start + count * dtype.size();
161:     if end > bytes.len() {
162:         bail!("truncated .npy payload");
163:     }
164: 
165:     Ok(NpyArray {
166:         shape,
167:         dtype,
168:         bytes: bytes[start..end].to_vec(),
169:     })
170: }

1: //! A minimal read-only zip archive reader.
2: //!
3: //! Stored and deflated members, zip64 included. Lives here because `.npz` is a
4: //! zip of `.npy` members, and is public because PyTorch archives are zips too:
5: //! `burn-torch-storage` reads `.pt`/`.ckpt` through this same code rather than

41: const END_OF_CENTRAL_DIRECTORY: u32 = 0x0605_4b50;
42: const CENTRAL_FILE_HEADER: u32 = 0x0201_4b50;
43: const LOCAL_FILE_HEADER: u32 = 0x0403_4b50;
44: const ZIP64_END_OF_CENTRAL_DIRECTORY: u32 = 0x0606_4b50;
45: const ZIP64_LOCATOR: u32 = 0x0706_4b50;

47: pub(crate) struct Entry {
48:     pub name: String,
49:     compression: u16,
50:     compressed_size: u64,
51:     uncompressed_size: u64,
52:     local_header_offset: u64,
53: }

55: /// A memory-mapped zip archive.
56: pub struct ZipArchive {
57:     data: ArchiveData,
58:     entries: Vec<Entry>,
59: }

61: fn u16_at(data: &[u8], offset: usize) -> u16 {
62:     u16::from_le_bytes(data[offset..offset + 2].try_into().unwrap())
63: }
64: 
65: fn u32_at(data: &[u8], offset: usize) -> u32 {
66:     u32::from_le_bytes(data[offset..offset + 4].try_into().unwrap())
67: }
68: 
69: fn u64_at(data: &[u8], offset: usize) -> u64 {
70:     u64::from_le_bytes(data[offset..offset + 8].try_into().unwrap())
71: }

73: /// Replaces saturated 32-bit sizes/offsets with their zip64 values.
74: fn apply_zip64_extra(extra: &[u8], entry: &mut Entry) {
75:     let mut offset = 0;
76:     while offset + 4 <= extra.len() {
77:         let id = u16_at(extra, offset);
78:         let size = u16_at(extra, offset + 2) as usize;
79:         let body = offset + 4;
80:         if body + size > extra.len() {
81:             return;
82:         }
83:         if id == 0x0001 {
84:             let mut cursor = body;
85:             for slot in [
86:                 &mut entry.uncompressed_size,
87:                 &mut entry.compressed_size,
88:                 &mut entry.local_header_offset,
89:             ] {
90:                 if *slot == u32::MAX as u64 && cursor + 8 <= body + size {
91:                     *slot = u64_at(extra, cursor);
92:                     cursor += 8;
93:                 }
94:             }
95:             return;
96:         }
97:         offset = body + size;
98:     }
99: }

101: /// Locates the central directory, following the zip64 locator when present.
102: fn find_central_directory(data: &[u8]) -> Result<(u64, u64)> {
103:     // The end-of-central-directory record is within 64 KiB of the file end.
104:     let earliest = data.len().saturating_sub(66_000);
105:     let mut eocd = None;
106:     for offset in (earliest..data.len().saturating_sub(21)).rev() {
107:         if u32_at(data, offset) == END_OF_CENTRAL_DIRECTORY {
108:             eocd = Some(offset);
109:             break;
110:         }
111:     }
112:     let eocd = eocd.context("not a zip archive: no end-of-central-directory record")?;
113:     let mut count = u16_at(data, eocd + 10) as u64;
114:     let mut offset = u32_at(data, eocd + 16) as u64;
115: 
116:     if (count == u16::MAX as u64 || offset == u32::MAX as u64) && eocd >= 20 {
117:         let locator = eocd - 20;
118:         if u32_at(data, locator) == ZIP64_LOCATOR {
119:             let record = u64_at(data, locator + 8) as usize;
120:             if record + 56 <= data.len() && u32_at(data, record) == ZIP64_END_OF_CENTRAL_DIRECTORY {
121:                 count = u64_at(data, record + 32);
122:                 offset = u64_at(data, record + 48);
123:             }
124:         }
125:     }
126:     Ok((count, offset))
127: }

129: impl ZipArchive {
130:     pub fn open(path: impl AsRef<Path>) -> Result<Self> {
131:         let path = path.as_ref();
132:         let file =
133:             std::fs::File::open(path).with_context(|| format!("opening {}", path.display()))?;
134:         // SAFETY: model assets are read-only inputs; concurrent truncation by
135:         // another process is out of scope, as elsewhere in this workspace.
136:         let data = unsafe { Mmap::map(&file) }
137:             .with_context(|| format!("memory-mapping {}", path.display()))?;
138:         Self::from_data(ArchiveData::Mapped(data))
139:     }
140: 
141:     pub fn from_bytes(data: Vec<u8>) -> Result<Self> {
142:         Self::from_data(ArchiveData::Owned(data))
143:     }
144: 
145:     fn from_data(data: ArchiveData) -> Result<Self> {
146:         let (count, mut offset) = find_central_directory(data.as_ref())?;
147:         let mut entries = Vec::with_capacity(count.min(4096) as usize);
148:         for _ in 0..count {
149:             let base = offset as usize;
150:             if base + 46 > data.len() || u32_at(&data, base) != CENTRAL_FILE_HEADER {
151:                 break;
152:             }
153:             let name_len = u16_at(&data, base + 28) as usize;
154:             let extra_len = u16_at(&data, base + 30) as usize;
155:             let comment_len = u16_at(&data, base + 32) as usize;
156:             let mut entry = Entry {
157:                 name: String::from_utf8_lossy(&data[base + 46..base + 46 + name_len]).into_owned(),
158:                 compression: u16_at(&data, base + 10),
159:                 compressed_size: u32_at(&data, base + 20) as u64,
160:                 uncompressed_size: u32_at(&data, base + 24) as u64,
161:                 local_header_offset: u32_at(&data, base + 42) as u64,
162:             };
163:             apply_zip64_extra(
164:                 &data[base + 46 + name_len..base + 46 + name_len + extra_len],
165:                 &mut entry,
166:             );
167:             entries.push(entry);
168:             offset += (46 + name_len + extra_len + comment_len) as u64;
169:         }
170: 
171:         Ok(Self { data, entries })
172:     }

196:     pub fn bytes(&self, name: &str) -> Result<Cow<'_, [u8]>> {
197:         let entry = self
198:             .entries
199:             .iter()

203:     }

205:     pub(crate) fn entry_bytes(&self, entry: &Entry) -> Result<Cow<'_, [u8]>> {
206:         let base = entry.local_header_offset as usize;
207:         if base + 30 > self.data.len() || u32_at(&self.data, base) != LOCAL_FILE_HEADER {
208:             bail!("corrupt zip entry {}", entry.name);
209:         }
210:         // The local header repeats the name and may carry a different extra field.
211:         let name_len = u16_at(&self.data, base + 26) as usize;
212:         let extra_len = u16_at(&self.data, base + 28) as usize;
213:         let start = base + 30 + name_len + extra_len;
214:         let end = start + entry.compressed_size as usize;
215:         if end > self.data.len() {
216:             bail!("zip entry {} runs past the end of the archive", entry.name);
217:         }
218:         let raw = &self.data[start..end];
219: 
220:         match entry.compression {
221:             0 => Ok(Cow::Borrowed(raw)),
222:             8 => {
223:                 let mut out = Vec::with_capacity(entry.uncompressed_size as usize);
224:                 DeflateDecoder::new(raw)
225:                     .read_to_end(&mut out)
226:                     .with_context(|| format!("inflating zip entry {}", entry.name))?;
227:                 Ok(Cow::Owned(out))
228:             }
229:             other => bail!("unsupported zip compression method {other}"),
230:         }
231:     }
