23: impl Curvature {
24:     /// Flip the local Y of every control point -- used when a panel is mirrored
25:     /// during edge-loop normalisation.
26:     pub fn flip_y(&mut self) {
27:         match self {
28:             Curvature::Circle { right, .. } => *right = !*right,
29:             Curvature::Quadratic { params } | Curvature::Cubic { params } => {
30:                 for p in params.iter_mut() {
31:                     p[1] = -p[1];
32:                 }
33:             }
34:         }
35:     }
36: }

67: pub fn edge_as_curve(vertices: &[V2], edge: &EdgeSpec, flip_y: bool) -> Curve {
68:     let start = vertices[edge.endpoints[0]];
69:     let end = vertices[edge.endpoints[1]];
70:     let sign = if flip_y { -1.0 } else { 1.0 };
71: 
72:     match &edge.curvature {
73:         None => Curve::line(start, end),
74:         Some(Curvature::Quadratic { params }) => {
75:             let c = rel_to_abs_2d(start, end, [params[0][0], sign * params[0][1]]);
76:             Curve::quad(start, c, end)
77:         }
78:         Some(Curvature::Cubic { params }) => {
79:             let c1 = rel_to_abs_2d(start, end, [params[0][0], sign * params[0][1]]);
80:             let c2 = rel_to_abs_2d(start, end, [params[1][0], sign * params[1][1]]);
81:             Curve::cubic(start, c1, c2, end)
82:         }
83:         Some(Curvature::Circle {
84:             radius,
85:             large_arc,
86:             right,
87:         }) => {
88:             let sweep = if flip_y { !*right } else { *right };
89:             Curve::Arc(crate::curve::Arc::new(
90:                 start,
91:                 [*radius, *radius],
92:                 0.0,
93:                 *large_arc,
94:                 sweep,
95:                 end,
96:             ))
97:         }
98:     }
99: }
