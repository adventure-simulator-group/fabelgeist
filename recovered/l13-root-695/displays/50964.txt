348: fn add_seam_bends(mesh: &mut GarmentMesh) {
349:     let mut roots: Vec<u32> = (0..mesh.positions.len() as u32).collect();
350:     fn root(roots: &[u32], mut i: u32) -> u32 {
351:         while roots[i as usize] != i {
352:             i = roots[i as usize];
353:         }
354:         i
355:     }
356:     for &[a, b] in &mesh.seams {
357:         let (a, b) = (root(&roots, a), root(&roots, b));
358:         roots[b as usize] = a;
359:     }
360:     let roots: Vec<_> = (0..roots.len() as u32).map(|i| root(&roots, i)).collect();
361:     let mut incident = std::collections::BTreeMap::<(u32, u32), Vec<[u32; 3]>>::new();
362:     for f in &mesh.triangles {
363:         for k in 0..3 {
364:             let (mut a, mut b, c) = (f[k], f[(k + 1) % 3], f[(k + 2) % 3]);
365:             if roots[a as usize] > roots[b as usize] {
366:                 std::mem::swap(&mut a, &mut b);
367:             }
368:             incident
369:                 .entry((roots[a as usize], roots[b as usize]))
370:                 .or_default()
371:                 .push([a, b, c]);
372:         }
373:     }
374:     for ((a, b), sides) in incident {
375:         if a == b || sides.len() != 2 || mesh.panel_of(sides[0][0]) == mesh.panel_of(sides[1][0]) {
376:             continue;
377:         }
378:         let side_points = |side: [u32; 3]| {
379:             let [a, b, c] = side.map(|i| mesh.positions[i as usize]);
380:             let edge = b - a;
381:             let length = edge.length();
382:             let x = (c - a).dot(edge) / length;
383:             let y = ((c - a).length_squared() - x * x).max(0.0).sqrt();
384:             (length, x, y)

387:         let (l1, x1, y1) = side_points(sides[1]);

415: /// Pair up the particles along two sewn edges.
416: ///
417: /// The two edges rarely have the same number of vertices -- they are different
418: /// lengths, and each was resampled independently -- so both are walked by
419: /// fraction of their length rather than by index. The finer of the two sets
420: /// the number of stitches, so no vertex on either edge is left unsewn.
421: fn seam_pairs(seam: &Seam, meshes: &[PanelMesh], offsets: &[u32]) -> anyhow::Result<Vec<[u32; 2]>> {
422:     let chain_a = chain(seam.a, meshes, offsets)?;
423:     let mut chain_b = chain(seam.b, meshes, offsets)?;
424:     if seam.reversed {
425:         chain_b.reverse();
426:     }
427:     if chain_a.len() < 2 || chain_b.len() < 2 {
428:         return Err(anyhow!("an edge has fewer than two vertices"));
429:     }
430: 
431:     let stitches = chain_a.len().max(chain_b.len());
432:     let mut pairs = Vec::with_capacity(stitches);
433:     for stitch in 0..stitches {
434:         let t = stitch as f32 / (stitches - 1) as f32;
435:         let a = chain_a[((t * (chain_a.len() - 1) as f32).round() as usize).min(chain_a.len() - 1)];
436:         let b = chain_b[((t * (chain_b.len() - 1) as f32).round() as usize).min(chain_b.len() - 1)];
437:         pairs.push([a.min(b), a.max(b)]);
438:     }
439:     Ok(pairs)
440: }
