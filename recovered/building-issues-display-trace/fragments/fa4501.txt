44:                 ));
45:             }
46:         }

50:     audit_structural_assemblies(plan, &mut issues)?;
51:

68:                 walk,
69:                 WallWalk::Round { centre, elevation_metres, .. }
70:                     if close_vec(*centre, tower.centre_metres())

72:             )
73:         });
74:         let access = plan.stairs.iter().any(|stair| {
75:             matches!(
76:                 stair,
77:                 Stair::Spiral { centre, base_height_metres, rise_metres, .. }
78:                     if close_vec(*centre, tower.centre_metres())
79:                         && close(*base_height_metres + *rise_metres, tower.wall_height_metres)
80:             )
81:         });
82:         if tower.battlement.is_some() && !deck {
83:             issues.push(issue(
84:                 "missing_tower_deck",
85:                 format!("battlemented tower {index} has no annular fighting deck"),
86:             ));
87:         }
88:         if tower.battlement.is_some() && !access {
89:             issues.push(issue(
90:                 "inaccessible_tower_deck",
91:                 format!("battlemented tower {index} has no stair reaching its deck"),
92:             ));
93:         }
94:     }

99:                 && close(run.base_height_metres, wall.height_metres)
100:                 && run.outward == wall.outward

102:             issues.push(issue(
103:                 "unprotected_curtain_wall",

107:     }

109:     audit_gatehouse_assemblies(plan, &mut issues);
