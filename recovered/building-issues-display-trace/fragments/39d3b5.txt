67:     fn for_buildings(buildings: &[GeneratedBuilding], output: &Path) -> Result<Self> {
68:         for building in buildings {

76:         }

78:             buildings: buildings.len(),
79:             boundaries: 0,
80:             doors: buildings

83:                 .collect::<std::result::Result<Vec<_>, _>>()?
84:                 .into_iter()
85:                 .sum(),
86:             windows: buildings
87:                 .iter()

89:                 .collect::<std::result::Result<Vec<_>, _>>()?
90:                 .into_iter()
91:                 .sum(),
92:             signs: BTreeMap::new(),
93:             output: output.to_owned(),
94:         })
95:     }
96: }

108: pub(super) fn setup(

110:     buildings: &[GeneratedBuilding],

112:     input: &Path,
113:     output: &Path,
114:     profile: &str,
115: ) -> Result<Option<Vec<BuildingReviewCamera>>> {
116:     if !is_profile(profile) {
117:         return Ok(None);
118:     }
119:     let path = input.with_extension("review.json");
120:     let bytes = std::fs::read(&path)?;
121:     let fixture: ReviewFixture = serde_json::from_slice(&bytes)?;
122:     let specs =
123:         super::selected_capture_views(profile, &[]).map_err(bevy::ecs::error::BevyError::from)?;
124:     assert_eq!(
125:         fixture.views.len(),
126:         specs.len() - 1,
127:         "fixture must cover every review view"
128:     );
129:     for (view, spec) in fixture.views.iter().zip(&specs[1..]) {
130:         assert_eq!(view.slug, spec.slug);
131:     }
132:     let mut requirements = ReviewRequirements::for_buildings(buildings, output)?;
133:     requirements.boundaries = boundaries;
134:     requirements.doors += boundaries;
135:     for (&id, sign) in &fixture.signs {
136:         let building = buildings
137:             .iter()
138:             .find(|b| b.placement.id == id)
139:             .ok_or_else(|| format!("review sign building {id} is absent"))?;
140:         assert!(
141:             building
142:                 .placement
143:                 .program
144:                 .usage
145:                 .and_then(adventuresim_building_generator::signs::shop_trade)
146:                 .is_some(),
147:             "sign requires a public shopfront"
148:         );
149:         let site = SignSite::for_plan(&building.plan)
150:             .ok_or_else(|| format!("review sign building {id} lacks structural support"))?;
151:         assert!(
152:             site.supports(&building.plan, sign.mount),
153:             "review sign must fit its specified mount"
154:         );
155:         requirements.signs.insert(
156:             id,
157:             ExpectedSign {
158:                 sign: sign.clone(),
159:                 site,
160:             },
161:         );
162:     }
163:     let cameras = fixture
164:         .views
165:         .iter()
166:         .map(|view| view.camera(buildings, &fixture.signs))
167:         .collect::<Result<Vec<_>>>()?;
168:     std::fs::write(output.join("input.review.json"), bytes)?;
169:     commands.insert_resource(requirements);
170:     commands.insert_resource(fixture);
171:     Ok(Some(cameras))
172: }
