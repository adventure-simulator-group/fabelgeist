1: /// Recomputes an existing roof graph under its declared pivot policy.
2: /// Changing pitch on roofs with child, parent, inset-wall or heating attachments
3: /// requires a topology change and is rejected. Unchanged-pitch requests are no-ops.
4: /// Any error leaves the plan unchanged.
5: pub fn set_roof_pitch(
6:     plan: &mut BuildingPlan,
7:     id: RoofAssemblyId,
8:     pitch_degrees: f32,
9: ) -> Result<(), RoofEditError> {
10:     if !(15.0..=75.0).contains(&pitch_degrees) {
11:         return Err(RoofEditError::PitchOutsideProjectRange);
12:     }
13:     let assembly_index = plan
14:         .roof_assemblies
15:         .iter()
16:         .position(|roof| roof.id == id)
17:         .ok_or(RoofEditError::MissingAssembly)?;
18:     let assembly = &plan.roof_assemblies[assembly_index];
19:     let old_pitch = assembly
20:         .faces
21:         .first()
22:         .map_or(pitch_degrees, |face| face.pitch_degrees);
23:     if (old_pitch - pitch_degrees).abs() < 0.0001 {
24:         return Ok(());
25:     }
26:     if !assembly.children.is_empty()
27:         || assembly.parent.is_some()
28:         || assembly
29:             .enclosure_faces
30:             .iter()
31:             .any(|face| !face.inset_walls.is_empty())
32:         || plan
33:             .domestic_heating
34:             .as_ref()
35:             .is_some_and(|h| h.roof.roof == id)
36:     {
37:         return Err(RoofEditError::TopologyEvent);
38:     }
39:     let old_tan = old_pitch.to_radians().tan();
40:     if old_tan.abs() <= 0.0001 {
41:         return Err(RoofEditError::TopologyEvent);
42:     }
43:     let factor = pitch_degrees.to_radians().tan() / old_tan;
44:     let min_y = assembly
45:         .faces
46:         .iter()
47:         .flat_map(|face| face.polygon.iter().map(|point| point.y))
48:         .fold(f32::INFINITY, f32::min);
49:     let max_y = assembly
50:         .faces
51:         .iter()
52:         .flat_map(|face| face.polygon.iter().map(|point| point.y))
53:         .fold(f32::NEG_INFINITY, f32::max);
54:     let pivot_policy = assembly.pivot_policy;
55:     let scale_y = |y: f32| match pivot_policy {
56:         RoofPivotPolicy::KeepEave | RoofPivotPolicy::KeepChildAttachment => {
57:             min_y + (y - min_y) * factor
58:         }
59:         RoofPivotPolicy::KeepRidge => max_y - (max_y - y) * factor,
60:     };
61:     let mut reconstructed = assembly.clone();
62:     let mut geometry = plan.resolved_geometry.clone();
63:     reconstruct_roof_pitch(
64:         &mut reconstructed,
65:         &mut geometry,
66:         plan,
67:         pitch_degrees,
68:         min_y,
69:         scale_y,
70:     )
71:     .map_err(|cause| RoofEditError::Construction {
72:         roof: id,
73:         cause: Box::new(cause),
74:     })?;
75:     plan.roof_assemblies[assembly_index] = reconstructed;
76:     plan.resolved_geometry = geometry;
77:     Ok(())
78: }

80: fn reconstruct_roof_pitch(
81:     assembly: &mut RoofAssembly,
82:     geometry: &mut ResolvedGeometry,
83:     plan: &BuildingPlan,
84:     pitch_degrees: f32,
85:     min_y: f32,
86:     scale_y: impl Fn(f32) -> f32,
87: ) -> Result<(), GenerationError> {
88:     for face in &mut assembly.faces {
89:         for point in &mut face.polygon {
90:             point.y = scale_y(point.y);
91:         }
92:         let bounds = roof_polygon_bounds(face.id, &face.polygon)?;
93:         face.plane = roof_plane(&face.polygon);
94:         face.pitch_degrees = pitch_degrees;
95:         if let Some(surface) = geometry
96:             .surfaces
97:             .iter_mut()
98:             .find(|surface| surface.id == face.drainage_catchment)
99:         {
100:             surface.bounds = bounds;
101:         }
102:         if let Some(catchment) = geometry
103:             .drainage_catchments
104:             .iter_mut()
105:             .find(|catchment| catchment.id == face.drainage_catchment)
106:         {
107:             let centre = face.polygon.iter().copied().sum::<Vec3>() / face.polygon.len() as f32;
108:             let low = face
109:                 .polygon
110:                 .iter()
111:                 .min_by(|a, b| a.y.total_cmp(&b.y))
112:                 .copied()
113:                 .ok_or(GenerationError::EmptyRoofFace { face: face.id })?;
114:             catchment.centre = centre;
115:             catchment.inner_elevation_metres = face
116:                 .polygon
117:                 .iter()
118:                 .map(|point| point.y)
119:                 .fold(f32::NEG_INFINITY, f32::max);
120:             catchment.outer_elevation_metres = low.y;
121:             if let Some(route) = geometry
122:                 .drainage_routes
123:                 .iter_mut()
124:                 .find(|route| route.id == catchment.outlet_route)
125:             {
126:                 route.inlet = centre;
127:                 route.outlet = low;
128:             }
129:         }
130:     }
131:     gable_enclosure::update_pitch(
132:         &mut assembly.enclosure_faces,
133:         &assembly.faces,
134:         assembly
135:             .source_piece_index
136:             .filter(|_| assembly.kind == RoofKind::Gable)
137:             .map(|index| plan.roofs[index]),
138:         &plan.wall_assemblies,
139:         min_y,
140:         &scale_y,
141:     );
142:     for edge in &mut assembly.edges {
143:         edge.start.y = scale_y(edge.start.y);
144:         edge.end.y = scale_y(edge.end.y);
145:     }
146:     refresh_edge_weathering(assembly, geometry)?;

150: fn refresh_edge_weathering(
151:     assembly: &RoofAssembly,
152:     geometry: &mut ResolvedGeometry,
153: ) -> Result<(), GenerationError> {
154:     for (edge_index, edge) in assembly.edges.iter().enumerate() {
155:         let delta = edge.end - edge.start;
156:         let plan_length = Vec2::new(delta.x, delta.z).length().max(0.05);
157:         let edge_pitch = delta.y.atan2(plan_length);
158:         let weather_id =
159:             ResolvedItemId((0x8_u64 << 60) | (assembly.id.0 << 16) | 0x5000 | edge_index as u64);
160:         if let Some(solid) = geometry
161:             .solids
162:             .iter_mut()
163:             .find(|solid| solid.id == weather_id)
164:         {
165:             let treated_plan_length = if edge.kind == RoofEdgeKind::Eave {
166:                 (plan_length - 0.36_f32.min(plan_length * 0.5)).max(0.05)
167:             } else {
168:                 plan_length
169:             };
170:             solid.centre = Position::<Architectural>::from_metres(
171:                 (edge.start + edge.end) * 0.5
172:                     + if edge.kind == RoofEdgeKind::Eave {
173:                         Vec3::NEG_Y * 0.06
174:                     } else {
175:                         Vec3::Y * 0.035
176:                     },
177:             )?;
178:             {
179:                 let mut native_geometry = solid.size.metres();
180:                 native_geometry.x = if edge.kind == RoofEdgeKind::Eave {
181:                     treated_plan_length
182:                 } else {
183:                     treated_plan_length / edge_pitch.cos().abs().max(0.01)
184:                 };
185:                 solid.size = CuboidDimensions::from_metres(native_geometry).map_err(|cause| {
186:                     crate::CollisionError {
187:                         source_id: solid.id,
188:                         cause,
189:                     }
190:                 })?;
191:             };
192:             solid.yaw_radians = Radians::new(delta.z.atan2(delta.x))?;
193:             solid.longfall_radians = Radians::new(if edge.kind == RoofEdgeKind::Eave {
194:                 0.012
195:             } else {
196:                 edge_pitch
197:             })?;
198:         }
199:         if let Some(flashing_id) = edge.flashing
200:             && let Some(flashing) = geometry
201:                 .solids
202:                 .iter_mut()
203:                 .find(|solid| solid.id == flashing_id)
204:         {
205:             flashing.centre = Position::<Architectural>::from_metres(
206:                 (edge.start + edge.end) * 0.5 + Vec3::Y * (flashing.size.metres().y * 0.5),
207:             )?;
208:             {
209:                 let mut native_geometry = flashing.size.metres();
210:                 native_geometry.x = delta.length().max(0.05);
211:                 flashing.size =
212:                     CuboidDimensions::from_metres(native_geometry).map_err(|cause| {
213:                         crate::CollisionError {
214:                             source_id: flashing.id,
215:                             cause,
216:                         }
217:                     })?;
218:             };
219:             flashing.yaw_radians = Radians::new(delta.z.atan2(delta.x))?;
220:             flashing.longfall_radians = Radians::new(if edge.kind == RoofEdgeKind::Valley {
221:                 edge_pitch
222:             } else {
223:                 0.0
224:             })?;
225:         }
226:     }
227:     Ok(())
228: }
