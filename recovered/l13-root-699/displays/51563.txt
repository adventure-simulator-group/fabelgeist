191: /// The centre of a set of points' bounding box.
192: ///
193: /// The box rather than the mean: a mean is pulled about by how finely a panel
194: /// happened to be meshed, so it moves when the resolution slider does -- and
195: /// a pivot that moves makes the same pose mean two different things on two
196: /// builds of the same pattern.
197: fn center(points: &[Vec3]) -> Vec3 {
198:     if points.is_empty() {
199:         return Vec3::default();
200:     }
201:     let mut lowest = points[0];
202:     let mut highest = points[0];
203:     for point in &points[1..] {
204:         lowest = lowest.min(*point);
205:         highest = highest.max(*point);
206:     }
207:     (lowest + highest) * 0.5
208: }

17:     pub fn new(positions: Vec<Vec3>, faces: Vec<[u32; 3]>, clearance: f32, inward: Vec3) -> Self {
18:         assert!(inward.is_finite() && inward.length() > 0.0);
19:         assert!(clearance.is_finite() && clearance >= 0.0);
20:         Self {

27:     /// Find the first finite plate seen from inside the layer. All triangles
28:     /// must face the given inward direction. This supports open and overlapping
29:     /// plates without extending their edges into infinite collision planes.
30:     fn correction(&self, point: Vec3) -> Option<(Vec3, Vec3)> {
31:         let bounds = self.surface.bvh.bounds();
32:         let reach = (point - bounds.center()).length() + bounds.extent().length() + self.clearance;
33:         let origin = point + self.inward * reach;
34:         let ray = fabelgeist_bvh::Ray::new(origin, self.inward * -1.0);
35:         let (index, distance) = self.surface.raycast(&ray, reach * 2.0)?;
36:         let closest = origin - self.inward * distance;
37:         let (a, b, c) = self.surface.triangle(index);
38:         let raw = (b - a).cross(c - a);
39:         if raw.length() <= 1e-10 {
40:             return None;
41:         }
42:         let normal = raw / raw.length();
43:         let alignment = normal.dot(self.inward);
44:         if alignment <= 1e-5 {
45:             return None;
46:         }
47:         let signed = (point - closest).dot(normal);
48:         if signed >= self.clearance {
49:             return None;
50:         }
51:         Some((point + normal * (self.clearance - signed), normal))
52:     }

54:     /// Maximum remaining displacement needed to satisfy an outer layer.
55:     pub fn residual(&self, positions: &[Vec3]) -> f32 {
56:         positions
57:             .iter()
58:             .filter_map(|&p| self.correction(p).map(|(q, _)| (q - p).length()))
59:             .fold(0.0, f32::max)
60:     }

62:     /// Project host particles, preserving pinned vertices and sliding velocity.
63:     pub fn project(
64:         &self,
65:         positions: &mut [Vec3],
66:         velocities: &mut [Vec3],
67:         masses: &[ParticleInverseMass],
68:         faces: &[[u32; 3]],
69:     ) -> bool {
70:         assert_eq!(positions.len(), velocities.len());
71:         assert_eq!(positions.len(), masses.len());
72:         let mut changed = false;
73:         for (i, point) in positions.iter_mut().enumerate() {
74:             if masses[i].mobility() == ParticleMobility::Prescribed {
75:                 continue;
76:             }
77:             if let Some((corrected, normal)) = self.correction(*point) {
78:                 *point = corrected;
79:                 let incoming = velocities[i].dot(normal).min(0.0);
80:                 velocities[i] -= normal * incoming;
81:                 changed = true;
82:             }
83:         }
84:         changed |= self.project_surface(positions, velocities, masses, faces);
85:         changed
86:     }
