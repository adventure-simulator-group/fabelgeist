433: /// WGSL for walking the hierarchy from inside another kernel.
434: ///
435: /// Generates `<prefix>_query_aabb(lower, upper)` and
436: /// `<prefix>_query_point(point, radius)`, both calling `callback(primitive)`
437: /// for every primitive whose *box* passes. The caller does the exact test --
438: /// the BVH only ever promises candidates.
439: pub fn traversal_source(config: &TraversalConfig) -> String {
440:     let TraversalConfig {
441:         callback,
442:         nodes,
443:         right_children,
444:         indices,
445:         prefix,
446:         stack_size,
447:     } = config;
448: 
449:     format!(
450:         r#"
451: fn {prefix}_query_aabb(query_lower: vec3<f32>, query_upper: vec3<f32>) {{
452:     var stack: array<u32, {stack_size}u>;
453:     var depth = 1u;
454:     stack[0] = 0u;
455: 
456:     // A full stack drops the deepest branch rather than writing out of
457:     // bounds, so an over-deep tree loses candidates instead of corrupting
458:     // memory next door.
459:     loop {{
460:         if (depth == 0u) {{
461:             break;
462:         }}
463:         depth = depth - 1u;
464:         let index = stack[depth];
465: 
466:         let lower = {nodes}[index * 2u];
467:         let upper = {nodes}[index * 2u + 1u];
468:         if (any(lower.xyz > query_upper) || any(upper.xyz < query_lower)) {{
469:             continue;
470:         }}
471: 
472:         if (bitcast<u32>(upper.w) != 0u) {{
473:             {callback}({indices}[bitcast<u32>(lower.w)]);
474:         }} else {{
475:             if (depth + 2u <= {stack_size}u) {{
476:                 stack[depth] = bitcast<u32>(lower.w);
477:                 stack[depth + 1u] = {right_children}[index];
478:                 depth = depth + 2u;
479:             }}
480:         }}
481:     }}
482: }}
483: 
484: fn {prefix}_query_point(point: vec3<f32>, radius: f32) {{
485:     let r = vec3<f32>(radius);
486:     {prefix}_query_aabb(point - r, point + r);
487: }}
488: "#
489:     )
490: }
