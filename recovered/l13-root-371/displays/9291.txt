15:     gltfs: &Assets<Gltf>,
16:     gltf_meshes: &Assets<GltfMesh>,
17:     gltf_nodes: &Assets<GltfNode>,

19:     meshes: &Assets<Mesh>,

22:     let gltf = gltfs.get(base_handle)?;
23: 
24:     for node_handle in &gltf.nodes {
25:         let node = gltf_nodes.get(node_handle)?;
26:         let (Some(mesh_handle), Some(skin_handle)) = (node.mesh.as_ref(), node.skin.as_ref())
27:         else {
28:             continue;
29:         };
30:         let gltf_mesh = gltf_meshes.get(mesh_handle)?;
31:         let skin = gltf_skins.get(skin_handle)?;
32:         let inverse_bindposes_asset = inverse_bindposes.get(&skin.inverse_bind_matrices)?;
33:         for joint in &skin.joints {

38:             let mesh = meshes.get(&primitive.mesh)?;

42:                 skin,
43:                 inverse_bindposes_asset,

54: fn build_runtime_body(
55:     mesh: &Mesh,
56:     gltf: &Gltf,
57:     skin: &GltfSkin,
58:     inverse_bindposes: &SkinnedMeshInverseBindposes,
59:     gltf_nodes: &Assets<GltfNode>,
60: ) -> anyhow::Result<CanonicalBody> {
61:     let positions = attribute_vec3(mesh, Mesh::ATTRIBUTE_POSITION, "positions")?;
62:     let normals = attribute_vec3(mesh, Mesh::ATTRIBUTE_NORMAL, "normals")?;
63:     let texcoords = body_surface_coordinates(&positions)?;
64:     let joint_indices = attribute_u16x4(mesh, Mesh::ATTRIBUTE_JOINT_INDEX, "joint indices")?;
65:     let joint_weights = attribute_f32x4(mesh, Mesh::ATTRIBUTE_JOINT_WEIGHT, "joint weights")?;
66:     let indices = mesh
67:         .indices()
68:         .context("runtime armor body has no index buffer")?;
69:     let indices = match indices {
70:         Indices::U16(indices) => indices.iter().map(|index| u32::from(*index)).collect(),
71:         Indices::U32(indices) => indices.clone(),
72:     };
73:     let (triangles, remainder) = indices.as_chunks::<3>();
74:     if !remainder.is_empty() {
75:         bail!("runtime armor body index buffer is not triangle-aligned");
76:     }
77:     let faces = triangles.to_vec();
78: 
79:     let joint_names: RigJointTable<RigJointName> = skin
80:         .joints
81:         .iter()
82:         .map(|joint| {
83:             gltf_nodes
84:                 .get(joint)
85:                 .map(|node| RigJointName::from(node.name.clone()))
86:                 .context("runtime body joint asset is missing")
87:         })
88:         .collect::<anyhow::Result<_>>()?;
89:     let global_joint_states: RigJointTable<[f32; 8]> = inverse_bindposes
90:         .iter()
91:         .map(|inverse_bindpose| {
92:             let (scale, rotation, position) =
93:                 inverse_bindpose.inverse().to_scale_rotation_translation();
94:             [
95:                 position.x, position.y, position.z, rotation.x, rotation.y, rotation.z, rotation.w,
96:                 scale.x,
97:             ]
98:         })
99:         .collect();
100:     anyhow::ensure!(
101:         joint_names.len() == global_joint_states.len(),
102:         "runtime body joint names and bind poses differ in count"
103:     );
104: 
105:     let body = RuntimeBody {
106:         domain: "runtime_body_cylindrical_v1".into(),
107:         faces: faces.clone(),
108:         positions,
109:         normals,
110:         texcoords,
111:         texcoord_faces: faces,
112:         joint_indices: PackedRigSkin::from_native(
113:             joint_indices
114:                 .into_iter()
115:                 .map(|joint| {
116:                     [
117:                         u32::from(joint[0]),
118:                         u32::from(joint[1]),
119:                         u32::from(joint[2]),
120:                         u32::from(joint[3]),
121:                         0,
122:                         0,
123:                         0,
124:                         0,
125:                     ]
126:                 })
127:                 .collect(),
128:             joint_names.joint_count(),
129:         )?,
130:         joint_weights: joint_weights
131:             .into_iter()
132:             .map(|weight| {
133:                 [
134:                     weight[0], weight[1], weight[2], weight[3], 0.0, 0.0, 0.0, 0.0,
135:                 ]
136:             })
137:             .collect(),
138:         joint_names,
139:         global_joint_states,
140:         device: Default::default(),
141:     };
142:     Ok(CanonicalBody {
143:         body,
144:         targets: mesh.get_morph_targets().unwrap_or_default().to_vec(),
145:         names: mesh.morph_target_names().unwrap_or_default().to_vec(),
146:         rig: FittingRig::new(gltf, skin, gltf_nodes)?,
147:     })
148: }

161: /// The canonical animation body contains geometry and skin without a UV atlas.
162: /// Runtime surfaces own a geometric UV domain; they do not claim correspondence
163: /// with the offline MHR atlas used by authored equipment attachment sockets.
164: fn body_surface_coordinates(positions: &[[f32; 3]]) -> anyhow::Result<Vec<[f32; 2]>> {
165:     let mut minimum = f32::INFINITY;
166:     let mut maximum = f32::NEG_INFINITY;
167:     for position in positions {
168:         if !position.iter().all(|value| value.is_finite()) {
169:             bail!("non-finite runtime body vertex");
170:         }
171:         minimum = minimum.min(position[1]);
172:         maximum = maximum.max(position[1]);
173:     }
174:     let height = maximum - minimum;
175:     if !height.is_finite() || height <= 0.0 {
176:         bail!("runtime body has no height");
177:     }
178:     Ok(positions
179:         .iter()
180:         .map(|[x, y, z]| {
181:             [
182:                 x.atan2(*z) / std::f32::consts::TAU + 0.5,
183:                 (y - minimum) / height,
184:             ]
185:         })
186:         .collect())
187: }
