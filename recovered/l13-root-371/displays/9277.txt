231:     visited: &mut std::collections::HashSet<Entity>,
232:     output: &mut Vec<PreviewTarget>,
233: ) {
234:     if !visited.insert(entity) {
235:         return;
236:     }
237:     let Ok((_, _, _, properties, _)) = items.get(entity) else {
238:         output.push(PreviewTarget::Occupied {
239:             entity,
240:             attachment: PreviewAttachment::Detached,
241:         });
242:         return;
243:     };
244:     let Some(equipment) = item_catalog::definition(&properties.id)
245:         .and_then(|definition| definition.equipment.as_ref())
246:     else {
247:         output.push(PreviewTarget::Occupied {
248:             entity,
249:             attachment: PreviewAttachment::Detached,
250:         });
251:         return;
252:     };
253:     let mut points: Vec<_> = equipment.attachment_points.iter().collect();
254:     points.sort_by_key(|point| (point.order, point.id.as_str()));
255:     for point in points
256:         .into_iter()
257:         .filter(|point| point.locations.is_empty() || point.locations.contains(&location))
258:     {
259:         for capacity_index in 0..point.capacity {
260:             let child =
261:                 items.iter().find_map(|(child, _, _, _, topology)| {
262:                     topology.occupancies.iter().any(|occupancy| {
263:                     matches!(
264:                         &occupancy.anchor,
265:                         TacticalEquipmentAnchor::ItemAttachment { parent, attachment_point_id }
266:                             if *parent == entity
267:                                 && attachment_point_id == &point.id
268:                                 && occupancy.capacity_index == capacity_index
269:                     )
270:                 }).then_some(child)
271:                 });
272:             if let Some(child) = child {
273:                 append_hud_preview(child, location, items, visited, output);
274:             } else {
275:                 output.push(PreviewTarget::EmptyAttachment { parent: entity });
276:             }
277:         }
278:     }
279:     output.push(PreviewTarget::Occupied {
280:         entity,
281:         attachment: PreviewAttachment::from_predicate(items.get(entity).is_ok_and(
282:             |(_, _, _, _, topology)| {
283:                 topology.occupancies.iter().any(|occupancy| {
284:                     matches!(
285:                         occupancy.anchor,
286:                         TacticalEquipmentAnchor::ItemAttachment { .. }
287:                     )
288:                 })
289:             },
290:         )),
291:     });
292: }
