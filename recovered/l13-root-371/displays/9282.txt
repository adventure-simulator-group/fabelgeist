54:         let items: std::collections::HashSet<_> = items.collect();
55:         let mut ready = 0;
56:         for (placeholder, visibility, runtime, resolved, failed) in &self.roots {
57:             if !items.contains(&placeholder.0) {
58:                 continue;
59:             }
60:             if failed {
61:                 return Err("Could not generate character equipment");

64:                 ready += 1;
65:             }
66:         }
67:         Ok(ready == items.len()

69:                 !items.contains(&part.item) || *visibility != Visibility::Hidden
70:             }))
71:     }

69: /// Joint attributes select Bevy's skinned pipeline even without `SkinnedMesh`.
70: /// Await wearer binding while hidden, and use a joint-free mesh when dropped.
71: pub(super) fn sync_render_bindings(
72:     mut commands: Commands,
73:     mut meshes: ResMut<Assets<Mesh>>,
74:     mut parts: Query<EquipmentRenderState>,
75:     items: Query<(Option<&ItemOf>, Option<&EquipSlot>, Has<TacticalSceneItem>)>,
76: ) {
77:     for (entity, part, current_mesh, skin, current_sources, mut visibility) in &mut parts {
78:         let worn = items.get(part.item).is_ok_and(|(owner, slot, scene)| {
79:             owner.is_some() && !scene && holding_side(slot).is_none()
80:         });
81:         if worn && skin.is_none() {
82:             visibility.set_if_neq(Visibility::Hidden);
83:             continue;
84:         }
85:         let mut sources = current_sources
86:             .cloned()
87:             .unwrap_or_else(|| EquipmentMeshSources {
88:                 bound: current_mesh.0.clone(),
89:                 unbound: None,
90:             });
91:         let mut sources_changed = current_sources.is_none();
92:         let desired = if skin.is_some() {
93:             sources.bound.clone()
94:         } else {
95:             if sources.unbound.is_none() {
96:                 let Some(source) = meshes.get(&sources.bound) else {
97:                     visibility.set_if_neq(Visibility::Hidden);
98:                     continue;
99:                 };
100:                 let mut unbound = source.clone();
101:                 unbound.remove_attribute(Mesh::ATTRIBUTE_JOINT_INDEX);
102:                 unbound.remove_attribute(Mesh::ATTRIBUTE_JOINT_WEIGHT);
103:                 center_unbound_mesh(&mut unbound, part.rigid_center);
104:                 sources.unbound = Some(meshes.add(unbound));
105:                 sources_changed = true;
106:             }
107:             sources
108:                 .unbound
109:                 .as_ref()
110:                 .expect("joint-free mesh prepared")
111:                 .clone()
112:         };
113:         if current_mesh.0 != desired {
114:             commands.entity(entity).insert(Mesh3d(desired));
115:         }
116:         if sources_changed {
117:             commands.entity(entity).insert(sources);
118:         }
119:         visibility.set_if_neq(Visibility::Inherited);
120:     }
121: }

123: /// Fitted geometry uses body coordinates. Rigid carried/world geometry uses
124: /// its own center, excluding unused body vertices left by clothing cutouts.
125: fn center_unbound_mesh(mesh: &mut Mesh, center: Vec3) {
126:     let Some(positions) = mesh
127:         .attribute(Mesh::ATTRIBUTE_POSITION)
128:         .and_then(bevy::mesh::VertexAttributeValues::as_float3)
129:     else {
130:         return;
131:     };
132:     if !center.is_finite() {
133:         return;
134:     }
135:     let centered: Vec<_> = positions
136:         .iter()
137:         .map(|position| (Vec3::from_array(*position) - center).to_array())
138:         .collect();
139:     mesh.insert_attribute(Mesh::ATTRIBUTE_POSITION, centered);
140: }
