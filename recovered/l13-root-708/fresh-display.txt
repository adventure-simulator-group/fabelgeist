205:         let mut without_wrong_level = vec![outline.clone()];
206:         let mut wrong_control = None;
207:         let tolerance = f64::from(self.limits.contact_tolerance_metres.metres());
208:         for indices in self.mesh.support_triangles() {
209:             let points = indices.map(|i| self.mesh.positions()[i as usize]);
210:             let triangle =
211:                 foundations::GroundTriangle::new(points).ok_or(FloorBearingIssue::Geometry)?;
212:             // Exact disjoint bounds cannot contribute floor area. Edge contact
213:             // remains a candidate for the existing clipping calculation.
214:             if !bounds.intersects(triangle.bounds()) {
215:                 continue;
216:             }
217:             let mut face = points.map(|p| p.xz().as_dvec2()).to_vec();
218:             if planar::signed_area(&face) < 0.0 {
219:                 face.reverse();
220:             }
221:             let intersection = intersection_with_outline(face.clone(), &outline);
222:             // Retaining-edge contact at another level has no bearing area.
223:             if planar::signed_area(&intersection) == 0.0 {
224:                 continue;
225:             }
226:             if let Some(point) = intersection.iter().find(|point| {
227:                 (triangle.height_f64(**point) - f64::from(bearing.elevation.metres())).abs()
228:                     > tolerance
229:             }) {
230:                 // Another retaining/contact level is ineligible coverage. Its
231:                 // uncovered edge sliver uses the existing perimeter allowance.
232:                 wrong_control = Some((*point, triangle.height_f64(*point)));
233:                 without_wrong_level = without_wrong_level
234:                     .into_iter()
235:                     .flat_map(|polygon| planar::subtract(polygon, &face))
236:                     .collect();
237:                 continue;
238:             }
239:             // Subtract original convex faces, not their rounded clipped outline.
240:             // Clipping may add tiny signed bends or redundant edges; treating
241:             // those fragments as convex cutters can inflate the missing union.
242:             // Duplicate original faces still cannot mask a hole.
243:             missing = missing
244:                 .into_iter()
245:                 .flat_map(|polygon| planar::subtract(polygon, &face))
246:                 .collect();
247:         }
248:         let absent = missing
249:             .iter()
250:             .map(|p| planar::signed_area(p).abs())
251:             .sum::<f64>();
252:         let wrong_area = planar::signed_area(&outline).abs()
253:             - without_wrong_level
254:                 .iter()
255:                 .map(|p| planar::signed_area(p).abs())
256:                 .sum::<f64>();
257:         Ok(FloorRegionCoverage {
258:             absent,
259:             wrong_area,
260:             wrong_control,
261:         })
262:     }
263: }

271:     for index in 0..outline.len() {

274:         polygon = planar::clip(polygon, |p| edge.perp_dot(p - a));

281: fn has_partial_shared_edge(first: &[DVec2], second: &[DVec2]) -> bool {
282:     let splits = |vertices: &[DVec2], edges: &[DVec2]| {
283:         vertices.iter().any(|point| {
284:             edges
285:                 .iter()
286:                 .zip(edges.iter().cycle().skip(1))
287:                 .take(edges.len())
288:                 .any(|(a, b)| {
289:                     let edge = *b - *a;
290:                     let offset = *point - *a;
291:                     edge.perp_dot(offset) == 0.0
292:                         && offset.dot(edge) > 0.0
293:                         && offset.dot(edge) < edge.length_squared()
294:                 })
295:         })
296:     };
297:     splits(first, second) || splits(second, first)
298: }
