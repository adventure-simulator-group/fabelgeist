132: fn tactical_iteration_app(seed: Seed) -> App {
133:     let mut app = App::new();
134:     app.add_plugins((
135:         MinimalPlugins,
136:         bevy::transform::TransformPlugin,
137:         AdventureSimulatorReplicationPlugin,
138:         AdventureSimulatorPhysicsPlugin {
139:             enable_simulation: true,
140:             enable_presentation_simulation: false,
141:         },
142:         CombatPlugin,
143:         TacticalEquipmentPlugin,
144:         BotPlugin,
145:     ))
146:     .insert_resource(TimeUpdateStrategy::ManualDuration(Duration::from_secs_f32(
147:         TACTICAL_TICK_SECONDS,
148:     )))
149:     .insert_resource(Time::<()>::default())
150:     .insert_resource(IterationClock { tick: 0 })
151:     .insert_resource(IterationLog::default())
152:     .insert_resource(crate::bot::CombatRandom::seeded(seed))
153:     .add_observer(crate::player_projection::on_player_added)
154:     .add_observer(record_attack_start)
155:     .add_observer(record_attack_committed_to_defense)
156:     .add_observer(record_attack_transformed_by_defense)
157:     .add_observer(record_continuation_decision)
158:     .add_observer(record_defense_resolution)
159:     .add_observer(record_resolved_attack)
160:     .add_systems(
161:         FixedPostUpdate,
162:         (
163:             restore_authoritative_movement_intent
164:                 .before(AdventureSimulatorPhysicsSet::ApplyCharacterMotor),
165:             (
166:                 trace_authoritative_quickstep_after_collision,
167:                 update_attack_facing_targets,
168:                 update_skeleton_locomotion,
169:                 update_character_motion_snapshots,
170:             )
171:                 .chain()
172:                 .after(AhoySystems::MoveCharacters),
173:         ),
174:     )
175:     .add_systems(
176:         Update,
177:         record_condition_changes.after(crate::combat::CombatSet::Condition),
178:     );
179:     app.finish();
180:     app.cleanup();
181:     app.world_mut().spawn((
182:         Name::new("Melee iteration ground"),
183:         RigidBody::Static,
184:         Collider::cuboid(40.0, 0.5, 40.0),
185:         Transform::from_xyz(0.0, -0.25, 0.0),
186:     ));
187:     app
188: }
