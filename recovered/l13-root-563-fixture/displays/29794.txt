11: fn spawn_weapon(world: &mut World, owner: Entity, build: &MeleeIterationBuild) {
12:     if let Some(weapon) = build.combatant.equipment.melee_weapon {
13:         let definition = adventuresim_core::item_catalog::definition(build.weapon_id)
14:             .expect("iteration roster validates its weapon");
15:         let authored = definition
16:             .equipment
17:             .as_ref()
18:             .expect("weapon equipment metadata");
19:         let skills = weapon.skills;
20:         world.spawn((
21:             ItemOf(owner),
22:             ItemProperties {
23:                 id: build.weapon_id.into(),
24:                 weight: definition.weight_kg,
25:             },
26:             TacticalEquipmentPhysical {
27:                 dimensions_m: Vec3::from_array(authored.physical.dimensions_m),
28:                 grip_to_tip_m: authored.physical.grip_to_tip_m,
29:                 striking_head_length_m: weapon.striking_head_length_m,
30:                 anchor_offset_m: Vec3::from_array(authored.physical.anchor_offset_m),
31:             },
32:             TacticalInventoryItemId(1),
33:             EquipSlot::HoldingRight,
34:             WeaponItem {
35:                 striking_material: authored
36:                     .striking_material
37:                     .expect("weapon striking material"),
38:                 skill_weights: [
39:                     skills.polearm,
40:                     skills.axe,
41:                     skills.bludgeon,
42:                     skills.sword,
43:                     skills.knife,
44:                     0.0,
45:                     0.0,
46:                     0.0,
47:                     skills.throw,
48:                 ],
49: 
50:                 prefers_stab: matches!(
51:                     weapon.preferred_melee_style,
52:                     adventuresim_core::combat_style::MeleeAttackStyle::Stab
53:                 ),
54:                 precision: weapon.precision,
55:                 reach: weapon.melee_reach,
56:                 grip_to_tip_m: authored.physical.grip_to_tip_m,
57:                 moment_of_inertia_kg_m2: match definition.kind {
58:                     adventuresim_core::item_catalog::ItemKind::Weapon {
59:                         moment_of_inertia_kg_m2,
60:                         ..
61:                     } => moment_of_inertia_kg_m2,
62:                     _ => unreachable!(),
63:                 },
64: 
65:                 melee: true,
66:                 ranged: false,
67:             },
68:         ));
69:     }
70: }

72: fn spawn_shield(world: &mut World, owner: Entity, build: &MeleeIterationBuild) {

74:         let definition = adventuresim_core::item_catalog::definition(shield_id)
75:             .expect("iteration roster validates its shield");
76:         world.spawn((
77:             ItemOf(owner),
78:             ItemProperties {

80:                 weight: definition.weight_kg,

82:             TacticalInventoryItemId(2),

84:             ShieldItem {

91: fn spawn_armor(world: &mut World, owner: Entity, build: &MeleeIterationBuild) {
92:     for (item_index, armor_id) in build.armor_ids.iter().enumerate() {

102:             .count();
103:         let placement = authored
104:             .placements
105:             .get(occurrence % authored.placements.len().max(1))
106:             .expect("iteration armor has an authored placement");
107:         let part = placement
108:             .protection
109:             .first()
